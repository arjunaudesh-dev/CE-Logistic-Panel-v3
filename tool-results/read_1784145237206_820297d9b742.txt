   416→   416→// Loose case-insensitive partial match between a row's supplier/receivedFrom

   418→   418→function looselyMatches(supplier: string | null | undefined, name: string): boolean {
   419→   419→  if (!supplier || !name) return false;
   420→   420→  const a = supplier.toLowerCase().trim();
   421→   421→  const b = name.toLowerCase().trim();
   422→   422→  if (!a || !b) return false;
   423→   423→  return a.includes(b) || b.includes(a);
   424→   424→}

   494→   494→// Generic project/vote filter applied to any row that has a supplier-like field.
   495→   495→// Pure helper (module-level) so the React Compiler can preserve manual memoization.
   496→   496→function applyProjVoteFilter<
   497→   497→  T extends { receivedFrom?: string | null; supplier?: string | null }
   498→   498→>(
   499→   499→  rows: T[],
   500→   500→  refField: "receivedFrom" | "supplier",
   501→   501→  projFilter: string,
   502→   502→  voteFilt: string,
   503→   503→  projectById: Map<string, Project>,
   504→   504→  projectByVote: Map<string, Project>
   505→   505→): T[] {
   506→   506→  let out = rows;
   507→   507→  if (projFilter !== "all") {
   508→   508→    const p = projectById.get(projFilter);
   509→   509→    if (p) {
   510→   510→      const matches = out.filter((r) =>
   511→   511→        looselyMatches(r[refField] ?? null, p.name)
   512→   512→      );
   513→   513→      if (matches.length > 0) out = matches;
   514→   514→    }
   515→   515→  }
   516→   516→  if (voteFilt !== "all") {
   517→   517→    const p = projectByVote.get(voteFilt);
   518→   518→    if (p) {
   519→   519→      const matches = out.filter((r) =>
   520→   520→        looselyMatches(r[refField] ?? null, p.name)
   521→   521→      );
   522→   522→      if (matches.length > 0) out = matches;
   523→   523→    }
   524→   524→  }
   525→   525→  return out;
   526→   526→}

   529→   529→// store category matches. Items not in stores are excluded when a type is set.
   530→   530→function applyTypeFilter<
   531→   531→  T extends { itemName?: string; name?: string }
   532→   532→>(
   533→   533→  rows: T[],
   534→   534→  nameField: "itemName" | "name",
   535→   535→  tFilter: string,
   536→   536→  storeByName: Map<string, StoreItem>
   537→   537→): T[] {
   538→   538→  if (tFilter === "all") return rows;
   539→   539→  return rows.filter((r) => {
   540→   540→    const name = r[nameField];
   541→   541→    const s = storeByName.get((name || "").toLowerCase().trim());
   542→   542→    return s?.category === tFilter;
   543→   543→  });
   544→   544→}

   549→   549→): number {
   550→   550→  const s = storeByName.get((itemName || "").toLowerCase().trim());
   551→   551→  return s?.unitCost ?? 0;

   562→   562→// ─────────────────────────────────────────────────────────
   563→   563→// Split-Order helpers — encode / decode Linked PO numbers in the
   564→   564→// PartSupplyOrder.supplier field (since the model has no dedicated column).
   565→   565→// Format: `<supplier> [PO: PO-001, PO-002]`
   566→   566→// ─────────────────────────────────────────────────────────
   567→   567→const SPLIT_PO_PREFIX = " [PO: ";
   568→   568→const SPLIT_PO_SUFFIX = "]";
   569→   569→
   570→   570→function formatSupplierWithPos(supplier: string, pos: string[]): string {
   571→   571→  const trimmedSupplier = supplier.trim();
   572→   572→  const cleanPos = pos.map((p) => p.trim()).filter(Boolean);
   573→   573→  if (cleanPos.length === 0) return trimmedSupplier;
   574→   574→  return `${trimmedSupplier}${SPLIT_PO_PREFIX}${cleanPos.join(", ")}${SPLIT_PO_SUFFIX}`;
   575→   575→}
   576→   576→
   577→   577→function parseSupplierWithPos(
   578→   578→  raw: string | null | undefined
   579→   579→): { supplier: string; pos: string[] } {
   580→   580→  if (!raw) return { supplier: "", pos: [] };
   581→   581→  const text = raw;
   582→   582→  const startIdx = text.lastIndexOf(SPLIT_PO_PREFIX);
   583→   583→  if (startIdx === -1 || !text.endsWith(SPLIT_PO_SUFFIX)) {
   584→   584→    return { supplier: text, pos: [] };
   585→   585→  }
   586→   586→  const endIdx = text.length - SPLIT_PO_SUFFIX.length;
   587→   587→  const poStr = text.slice(startIdx + SPLIT_PO_PREFIX.length, endIdx);
   588→   588→  const pos = poStr
   589→   589→    .split(",")
   590→   590→    .map((p) => p.trim())
   591→   591→    .filter(Boolean);
   592→   592→  const supplier = text.slice(0, startIdx).trim();
   593→   593→  return { supplier, pos };
   594→   594→}

   597→   597→// useAllProjectFiles — fetch all project files across every project.
   598→   598→// Used by the Collection Pipeline (Dashboard), the QCI tab (to find the
   599→   599→// matching file when a QCI is recorded) and the Send-to-Area-Store dialog
   600→   600→// (to update the file's collectionStatus).
   601→   601→// ─────────────────────────────────────────────────────────
   602→   602→function useAllProjectFiles(enabled: boolean = true): {
   603→   603→  data: ProjectFile[];
   604→   604→  loading: boolean;
   605→   605→  error: string | null;
   606→   606→  refresh: () => void;
   607→   607→} {
   608→   608→  const { data: projects, loading: pLoading, error: pError } =
   609→   609→    useFetch<Project[]>(enabled ? "/api/projects" : null);
   610→   610→  const projectIds = useMemo(
   611→   611→    () => (projects ?? []).map((p) => p.id),
   612→   612→    [projects]
   613→   613→  );
   614→   614→  const [files, setFiles] = useState<ProjectFile[]>([]);
   615→   615→  const [fetching, setFetching] = useState<boolean>(false);
   616→   616→  const [fetchError, setFetchError] = useState<string | null>(null);
   617→   617→  const [nonce, setNonce] = useState(0);
   618→   618→
   619→   619→  const refresh = useCallback(() => setNonce((n) => n + 1), []);
   620→   620→
   621→   621→  useEffect(() => {
   622→   622→    if (!enabled) return;
   623→   623→    if (pLoading || pError) return;
   624→   624→    if (projectIds.length === 0) return;
   625→   625→    let cancelled = false;
   626→   626→    (async () => {
   627→   627→      // Defer the loading-state setter so it isn't a synchronous setState
   628→   628→      // in the effect body (mirrors the `useFetch` pattern).
   629→   629→      await Promise.resolve();
   630→   630→      if (cancelled) return;
   631→   631→      setFetching(true);
   632→   632→      setFetchError(null);
   633→   633→      try {
   634→   634→        const results = await Promise.all(
   635→   635→          projectIds.map(async (pid) => {
   636→   636→            try {
   637→   637→              const res = await fetch(
   638→   638→                `/api/projects/${pid}/files?includeArchived=false`
   639→   639→              );
   640→   640→              if (!res.ok) return [] as ProjectFile[];
   641→   641→              const json = (await res.json()) as ProjectFile[];
   642→   642→              return Array.isArray(json) ? json : [];
   643→   643→            } catch {
   644→   644→              return [] as ProjectFile[];
   645→   645→            }
   646→   646→          })
   647→   647→        );
   648→   648→        if (cancelled) return;
   649→   649→        setFiles(results.flat());
   650→   650→        setFetching(false);
   651→   651→      } catch (e) {
   652→   652→        if (cancelled) return;
   653→   653→        setFetchError((e as Error).message || "Failed to load files");
   654→   654→        setFetching(false);
   655→   655→      }
   656→   656→    })();
   657→   657→    return () => {
   658→   658→      cancelled = true;
   659→   659→    };
   660→   660→  }, [projectIds, pLoading, pError, nonce, enabled]);
   661→   661→
   662→   662→  // Derive the final hook state without synchronous setState in the effect.
   663→   663→  if (!enabled) return { data: [], loading: false, error: null, refresh };
   664→   664→  if (pError) return { data: [], loading: false, error: pError, refresh };
   665→   665→  if (pLoading) return { data: [], loading: true, error: null, refresh };
   666→   666→  if (projectIds.length === 0)
   667→   667→    return { data: [], loading: false, error: null, refresh };
   668→   668→  return { data: files, loading: fetching, error: fetchError, refresh };
   669→   669→}
