    71→interface EstimateRow {
    72→  itemName: string;
    73→  qty: string;

    75→  /** Per-unit cost (renamed from "estimateCost" in earlier versions).
    76→   *  The CSV column header is `UnitCost`; the parser also accepts the legacy

   111→  const [form, setForm] = useState({

   114→    voteNumber: "",
   115→    voteAllocation: "",

   117→    actualSpent: "0",
   118→    startDate: "",
   119→    endDate: "", // EDC

   123→    financialMean: "Vote/Funds",
   124→    requirementGroup: "", // for Maintenance projects — groups sub-requirements
   125→    projectOfficer: "",
   126→    status: "Active",
   127→    description: "",

   130→    { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },

   133→  const selectedVote = votes?.find((v) => v.id === selectedVoteId);
   134→
   135→  // Tracks whether the user has manually edited the abbreviation field. When
   136→  // false, the abbreviation auto-suggests from the project name. Once the user
   137→  // touches the field, we stop auto-suggesting so their override sticks.
   138→  const [abbrevTouched, setAbbrevTouched] = useState(false);
   139→
   140→  // Suggest an abbreviation from a project name by taking the first letter of
   141→  // each significant word (skipping common stop-words / numbers / symbols).
   142→  // Example: "No.03 Slipway Hangar" → "NSH", "Engine Fire Suppression" → "EFS".
   143→  const suggestAbbreviation = (name: string): string => {
   144→    const stop = new Set(["of", "for", "and", "the", "in", "on", "at", "to", "no", "a", "an"]);
   145→    const tokens = name
   146→      .replace(/[^\p{L}\p{N}\s-]/gu, " ")
   147→      .split(/[\s-]+/)
   148→      .map((t) => t.trim())
   149→      .filter(Boolean);
   150→    const letters = tokens
   151→      .filter((t) => !stop.has(t.toLowerCase()))
   152→      .map((t) => t[0]!.toUpperCase());
   153→    return letters.slice(0, 5).join("");
   154→  };
   155→
   156→  const handleNameChange = (value: string) => {
   157→    setForm((prev) => ({
   158→      ...prev,
   159→      name: value,
   160→      // keep auto-suggesting the abbreviation until the user edits it manually
   161→      nameAbbreviation: abbrevTouched ? prev.nameAbbreviation : suggestAbbreviation(value),
   162→    }));
   163→  };
   164→
   165→  const handleAbbrevChange = (value: string) => {
   166→    setAbbrevTouched(true);
   167→    // Force uppercase, strip non-alphanumeric (keep letters + digits)
   168→    const cleaned = value.toUpperCase().replace(/[^A-Z0-9]/g, "").slice(0, 8);
   169→    setForm((prev) => ({ ...prev, nameAbbreviation: cleaned }));
   170→  };
   171→
   172→  const handleVoteSelect = (id: string) => {
   173→    setSelectedVoteId(id);
   174→    const v = votes?.find((vt) => vt.id === id);
   175→    if (v) {
   176→      setForm((prev) => ({
   177→        ...prev,
   178→        voteNumber: v.voteNumber,
   179→        voteAllocation: v.allocation ? String(v.allocation) : prev.voteAllocation,
   180→      }));
   181→    }
   182→  };
   183→
   184→  // Toggle the "Funds" checkbox. When checked, the project is funded via a non-vote
   185→  // fund (default Malima Fund) and the Vote Number becomes optional. When unchecked,
   186→  // the project is funded via Vote and the Vote Number is mandatory.
   187→  const handleFundsToggle = (checked: boolean) => {
   188→    setFundsChecked(checked);
   189→    setCustomVoteNumber(false);
   190→    setSelectedVoteId("");
   191→    if (checked) {
   192→      setForm((prev) => ({
   193→        ...prev,
   194→        financialMean: "Malima Fund",
   195→        voteNumber: "",
   196→        voteAllocation: "",
   197→      }));
   198→    } else {
   199→      setForm((prev) => ({
   200→        ...prev,
   201→        financialMean: "Vote/Funds",
   202→        voteNumber: "",
   203→      }));
   204→    }
   205→  };
   206→
   207→  const handleFundSelect = (fund: string) => {
   208→    setForm((prev) => ({ ...prev, financialMean: fund }));
   209→  };
   210→
   211→  const nonVotePlaceholder =
   212→    form.financialMean === "Contractual" ? "CONTRACTUAL" : "NON-NAVAL";
   213→  const nonVoteHint =
   214→    form.financialMean === "Contractual"
   215→      ? "Contractual funding — Vote Number can be 'CONTRACTUAL' or the ARO reference."
   216→      : form.financialMean === "Malima Fund"
   217→      ? "Malima Fund — Vote Number can be 'NON-NAVAL' or the Malima reference number."
   218→      : "Non-Public Fund — Vote Number can be 'NON-NAVAL' or the fund reference.";
   219→
   220→  const addRow = () =>
   221→    setEstimateRows([
   222→      ...estimateRows,
   223→      { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },
   224→    ]);
   225→  const removeRow = (idx: number) =>
   226→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
   227→
   228→  // Update a single field on a row. When Qty or UnitCost changes,
   229→  // recompute the row's TotalCost automatically (= Qty × UnitCost) so the
   230→  // editable grid always shows a live total. Manual edits to TotalCost itself
   231→  // (or other fields) leave the total as-is.
   232→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
   233→    setEstimateRows(
   234→      estimateRows.map((r, i) => {
   235→        if (i !== idx) return r;
   236→        const next = { ...r, [field]: value };
   237→        if (field === "qty" || field === "estimateCost") {
   238→          const q = Number(next.qty);
   239→          const c = Number(next.estimateCost);
   240→          next.totalCost =
   241→            !isNaN(q) && !isNaN(c) && q > 0 && c > 0 ? String(q * c) : "";
   242→        }
   243→        return next;
   244→      })
   245→    );
   246→
   247→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   248→  const validEstimateRows = estimateRows.filter(
   249→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   250→  );
   251→  const hasValidEstimate = validEstimateRows.length > 0;
   252→
   253→  const downloadTemplate = () => {
   254→    // Template now uses `UnitCost` as the column header (renamed from `EstimateCost`).
   255→    // The sample row demonstrates the relationship TotalCost = Qty × UnitCost (2 × 185000 = 370000).
   256→    const header = "ItemName,Qty,Unit,UnitCost,TotalCost";
   257→    const sample = "Centrifugal Pump 15kW,2,NOS,185000,370000";
   258→    const sample2 = ",,NOS,,";
   259→    const csv = `${header}\n${sample}\n${sample2}\n`;
   260→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   261→    const url = URL.createObjectURL(blob);
   262→    const a = document.createElement("a");
   263→    a.href = url;
   264→    a.download = "material-estimate-template.csv";
   265→    a.click();
   266→    URL.revokeObjectURL(url);
   267→    toast({
   268→      title: "Template downloaded",
   269→      description:
   270→        "Columns: ItemName, Qty, Unit, UnitCost (per unit), TotalCost (= Qty × UnitCost — leave blank to auto-compute).",
   271→    });
   272→  };
   273→
   274→  // Simple CSV parser — handles quoted fields with commas
   275→  const parseCsv = (text: string): string[][] => {
   276→    const rows: string[][] = [];
   277→    let cur: string[] = [];
   278→    let field = "";
   279→    let inQuotes = false;
   280→    for (let i = 0; i < text.length; i++) {
   281→      const ch = text[i];
   282→      if (inQuotes) {
   283→        if (ch === '"' && text[i + 1] === '"') {
   284→          field += '"';
   285→          i++;
   286→        } else if (ch === '"') {
   287→          inQuotes = false;
   288→        } else {
   289→          field += ch;
   290→        }
   291→      } else {
   292→        if (ch === '"') inQuotes = true;
   293→        else if (ch === ",") {
   294→          cur.push(field);
   295→          field = "";
   296→        } else if (ch === "\n" || ch === "\r") {
   297→          if (ch === "\r" && text[i + 1] === "\n") i++;
   298→          cur.push(field);
   299→          rows.push(cur);
   300→          cur = [];
   301→          field = "";
   302→        } else {
   303→          field += ch;
   304→        }
   305→      }
   306→    }
   307→    if (field.length > 0 || cur.length > 0) {
   308→      cur.push(field);
   309→      rows.push(cur);
   310→    }
   311→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   312→  };
   313→
   314→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   315→    const file = e.target.files?.[0];
   316→    if (!file) return;
   317→    if (!file.name.toLowerCase().endsWith(".csv")) {
   318→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   319→      e.target.value = "";
   320→      return;
   321→    }
   322→    try {
   323→      const text = await file.text();
   324→      const rows = parseCsv(text);
   325→      if (rows.length === 0) {
   326→        toast({ title: "CSV file is empty", variant: "destructive" });
   327→        e.target.value = "";
   328→        return;
   329→      }
   330→      const header = rows[0].map((c) => c.trim().toLowerCase());
   331→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   332→      const idxQty = header.findIndex((h) => h.includes("qty"));
   333→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   334→      // Find the per-unit cost column. Prefer `UnitCost` (new name); fall back to
   335→      // the legacy `EstimateCost` header for backward compatibility. Also matches
   336→      // any column with "cost"/"price" but without "total".
   337→      const idxCost = header.findIndex(
   338→        (h) =>
   339→          (h === "unitcost" || h.includes("unit cost") || h.includes("unitcost")) ||
   340→          ((h.includes("estimate") && (h.includes("cost") || h.includes("price"))) ||
   341→           ((h.includes("cost") || h.includes("price")) && !h.includes("total")))
   342→      );
   343→      const idxTotal = header.findIndex(
   344→        (h) => h.includes("total") && (h.includes("cost") || h.includes("price"))
   345→      );
   346→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   347→      const parsed: EstimateRow[] = dataRows
   348→        .map((r) => {
   349→          const itemName = idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim();
   350→          const qty = idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim();
   351→          const unit = (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS";
   352→          // Per-unit cost — read from the UnitCost (or legacy EstimateCost) column.
   353→          const estimateCost = idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim();
   354→          // TotalCost: prefer the CSV's TotalCost column when present & non-blank,
   355→          // otherwise compute it as Qty × UnitCost (per the task spec).
   356→          const totalRaw =
   357→            idxTotal >= 0 ? (r[idxTotal] || "").trim() : "";
   358→          let totalCost = "";
   359→          if (totalRaw && !isNaN(Number(totalRaw))) {
   360→            totalCost = totalRaw;
   361→          } else {
   362→            const q = Number(qty);
   363→            const c = Number(estimateCost);
   364→            if (!isNaN(q) && !isNaN(c) && q > 0 && c > 0) {
   365→              totalCost = String(q * c);
   366→            }
   367→          }
   368→          return { itemName, qty, unit, estimateCost, totalCost };
   369→        })
   370→        .filter((r) => r.itemName);
   371→      if (parsed.length === 0) {
   372→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   373→        e.target.value = "";
   374→        return;
   375→      }
   376→      setPreviewRows(parsed);
   377→    } catch {
   378→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   379→    }
   380→    e.target.value = "";
   381→  };
   382→
   383→  const confirmImport = () => {
   384→    if (!previewRows) return;
   385→    setEstimateRows(previewRows);
   386→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   387→    setPreviewRows(null);
   388→  };
   389→
   390→  const handleSaveAndProcess = async () => {
   391→    if (!form.name) {
   392→      toast({ title: "Project Name is required", variant: "destructive" });

   394→    }
   395→    // Vote Number is mandatory UNLESS the Funds checkbox is ticked (non-vote funding).

   398→        title: "Vote Number is required",
   399→        description: "Tick the Funds checkbox for non-vote funding (Malima/Non-Public/Contractual).",
   400→        variant: "destructive",
   401→      });
   402→      return;
   403→    }
   404→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   405→    if (validRows.length === 0) {
   406→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   407→      return;
   408→    }
   409→    setSaving(true);
   410→    // 1. Create project
   411→    const projRes = await apiRequest("/api/projects", "POST", form);
   412→    if (!projRes.ok) {
   413→      setSaving(false);
   414→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   415→      return;
   416→    }
   417→    const project = projRes.data as { id: string };
   418→    // 2. Export Estimate into DB
   419→    const estRes = await apiRequest("/api/estimates", "POST", {
   420→      projectId: project.id,
   421→      name: `${form.name} — Material Estimate`,
   422→      items: validRows.map((r) => ({
   423→        itemName: r.itemName,
   424→        qty: Number(r.qty) || 0,
   425→        unit: r.unit,
   426→        estimateCost: Number(r.estimateCost) || 0,
   427→      })),
   428→    });
   429→    setSaving(false);
   430→    if (estRes.ok) {
   431→      const est = estRes.data as { id: string };
   432→      setCreatedProjectId(project.id);
   433→      setCreatedEstimateId(est.id);
   434→      setStep(2);
   435→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   436→    } else {
   437→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   438→    }
   439→  };
   440→
   441→  const handleGoToProject = () => {
   442→    if (createdProjectId) {
   443→      onCreated({
   444→        id: createdProjectId,
   445→        name: form.name,
   446→        nameAbbreviation: form.nameAbbreviation || null,
   447→        voteNumber: form.voteNumber,
   448→        voteAllocation: Number(form.voteAllocation) || 0,
   449→        committed: Number(form.committed) || 0,
   450→        actualSpent: Number(form.actualSpent) || 0,
   451→        benefactor: form.benefactor || null,
   452→        reference: form.reference || null,
   453→        endDate: form.endDate || null,
   454→        startDate: form.startDate || null,
   455→        projectType: form.projectType,
   456→        financialMean: form.financialMean,
   457→        requirementGroup: form.projectType === "Maintenance" ? (form.requirementGroup.trim() || null) : null,
   458→      });
   459→      // reset
   460→      setStep(1);
   461→      setCreatedProjectId(null);
   462→      setCreatedEstimateId(null);
   463→      setFundsChecked(false);
   464→      setAbbrevTouched(false);
   465→      setForm({
   466→        name: "", nameAbbreviation: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   467→        startDate: "", endDate: "", benefactor: "", reference: "",
   468→        projectType: "New Construction", financialMean: "Vote/Funds",
   469→        requirementGroup: "",
   470→        projectOfficer: "", status: "Active", description: "",
   471→      });
   472→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" }]);
   473→      setCustomVoteNumber(false);
   474→      setSelectedVoteId("");
   475→    }
   476→  };
   477→
   478→  if (!open) return null;
   479→
   480→  return (
   481→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   482→      {/* Header */}
   483→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   484→        <div className="flex items-center gap-3">
   485→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   486→            <X className="h-5 w-5" />
   487→          </button>
   488→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   489→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   490→          </h2>
   491→        </div>
   492→        {/* Stepper */}
   493→        <div className="hidden sm:flex items-center gap-2 text-xs">
   494→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   495→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   496→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   497→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   498→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   499→        </div>
   500→      </div>
   501→
   502→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   503→        {step === 1 && (
   504→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   505→            {/* Project details */}
   506→            <section className="bg-card border border-border rounded-xl p-5">
   507→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   508→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   509→              </h3>
   510→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   511→                <div className="sm:col-span-2 space-y-1.5">
   512→                  <Label>Project Name *</Label>
   513→                  <Input value={form.name} onChange={(e) => handleNameChange(e.target.value)} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
