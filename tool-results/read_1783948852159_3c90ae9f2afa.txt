   138→function storeTypeColor(type: string): string {
   139→  switch (type) {
   140→    case "Zone":
   141→      return "bg-blue-500/15 text-blue-400 border-blue-500/30";
   142→    case "AreaStore":
   143→      return "bg-violet-500/15 text-violet-400 border-violet-500/30";
   144→    case "Ready-Use":
   145→      return "bg-emerald-500/15 text-emerald-400 border-emerald-500/30";
   146→    case "Balance":
   147→      return "bg-amber-500/15 text-amber-400 border-amber-500/30";

   149→      return "bg-pink-500/15 text-pink-400 border-pink-500/30";

   166→function activeBadgeClass(active: boolean): string {
   167→  return active
   168→    ? "bg-emerald-500/15 text-emerald-400 border-emerald-500/30"

   172→function formatDate(value: string | null | undefined): string {
   173→  if (!value) return "—";
   174→  const d = new Date(value);
   175→  if (isNaN(d.getTime())) return "—";
   176→  return d.toLocaleString("en-GB", {
   177→    day: "2-digit",
   178→    month: "short",
   179→    year: "numeric",
   180→  });
   181→}

   282→function UserProfilesTab() {
   283→  const { data: list, loading, error, refresh } = useFetch<UserProfile[]>(
   284→    "/api/settings/users"
   285→  );
   286→  const { toast } = useToast();
   287→
   288→  const [search, setSearch] = useState("");
   289→  const [roleFilter, setRoleFilter] = useState("all");
   290→  const [deptFilter, setDeptFilter] = useState("all");
   291→  const [activeFilter, setActiveFilter] = useState("all");
   292→
   293→  const [addOpen, setAddOpen] = useState(false);
   294→  const [editTarget, setEditTarget] = useState<UserProfile | null>(null);
   295→  const [deleteTarget, setDeleteTarget] = useState<UserProfile | null>(null);
   296→  const [deleting, setDeleting] = useState(false);
   297→
   298→  const filtered = useMemo(() => {
   299→    if (!list) return [];
   300→    const s = search.trim().toLowerCase();
   301→    return list.filter((u) => {
   302→      if (roleFilter !== "all" && u.role !== roleFilter) return false;
   303→      if (deptFilter !== "all" && (u.department ?? "—") !== deptFilter) return false;
   304→      if (activeFilter === "active" && !u.active) return false;
   305→      if (activeFilter === "inactive" && u.active) return false;
   306→      if (s) {
   307→        const hay =
   308→          `${u.username} ${u.displayName} ${u.email ?? ""} ${u.department ?? ""}`.toLowerCase();
   309→        if (!hay.includes(s)) return false;
   310→      }
   311→      return true;
   312→    });
   313→  }, [list, search, roleFilter, deptFilter, activeFilter]);
   314→
   315→  const filtersActive =
   316→    search !== "" ||
   317→    roleFilter !== "all" ||
   318→    deptFilter !== "all" ||
   319→    activeFilter !== "all";
   320→
   321→  function clearFilters() {
   322→    setSearch("");
   323→    setRoleFilter("all");
   324→    setDeptFilter("all");
   325→    setActiveFilter("all");
   326→  }
   327→
   328→  const activeCount = useMemo(
   329→    () => list?.filter((u) => u.active).length ?? 0,
   330→    [list]
   331→  );
   332→  const adminCount = useMemo(
   333→    () => list?.filter((u) => u.role === "Admin").length ?? 0,
   334→    [list]
   335→  );
   336→
   337→  async function handleDelete() {
   338→    if (!deleteTarget) return;
   339→    setDeleting(true);
   340→    const res = await apiRequest(
   341→      `/api/settings/users/${deleteTarget.id}`,
   342→      "DELETE"
   343→    );
   344→    setDeleting(false);
   345→    if (res.ok) {
   346→      toast({
   347→        title: "User deleted",
   348→        description: deleteTarget.displayName,
   349→      });
   350→      setDeleteTarget(null);
   351→      refresh();
   352→    } else {
   353→      toast({
   354→        title: "Failed to delete",
   355→        description: res.error,
   356→        variant: "destructive",
   357→      });
   358→    }
   359→  }
   360→
   361→  if (loading) return <LoadingState label="Loading users..." />;
   362→  if (error) return <ErrorState message={error} />;
   363→
   364→  return (
   365→    <div className="space-y-4">
   366→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   367→        <StatCard
   368→          label="Total Users"
   369→          value={list?.length ?? 0}
   370→          icon={Users}
   371→          accent="blue"
   372→          sub="All profiles"
   373→        />
   374→        <StatCard
   375→          label="Active"
   376→          value={activeCount}
   377→          icon={CheckCircle2}
   378→          accent="green"
   379→          sub="Can sign in"
   380→        />
   381→        <StatCard
   382→          label="Administrators"
   383→          value={adminCount}
   384→          icon={Shield}
   385→          accent="red"
   386→          sub="Full access"
   387→        />
   388→        <StatCard
   389→          label="Departments"
   390→          value={
   391→            new Set(
   392→              list?.map((u) => u.department).filter(Boolean) as string[]
   393→            ).size
   394→          }
   395→          icon={Building2}
   396→          accent="purple"
   397→          sub="Distinct units"
   398→        />
   399→      </div>
   400→
   401→      <div className="bg-card border border-border rounded-xl p-4 space-y-4">
   402→        <div className="flex flex-col lg:flex-row lg:items-end gap-3">
   403→          <div className="flex-1 min-w-0">
   404→            <Label className="text-xs text-muted-foreground mb-1.5 block">
   405→              Search (Username / Name / Email)
   406→            </Label>
   407→            <div className="relative">
   408→              <Search className="h-4 w-4 absolute left-3 top-1/2 -translate-y-1/2 text-muted-foreground" />
   409→              <Input
   410→                placeholder="Search users..."
   411→                value={search}
   412→                onChange={(e) => setSearch(e.target.value)}
   413→                className="pl-9 bg-background"
   414→              />
   415→            </div>
   416→          </div>
   417→          <div className="grid grid-cols-3 gap-2 lg:gap-3">
   418→            <Select value={roleFilter} onValueChange={setRoleFilter}>
   419→              <SelectTrigger className="bg-background w-full">
   420→                <SelectValue placeholder="Role" />
   421→              </SelectTrigger>
   422→              <SelectContent>
   423→                <SelectItem value="all">All Roles</SelectItem>
   424→                {USER_ROLES.map((r) => (
   425→                  <SelectItem key={r} value={r}>
   426→                    {r}
   427→                  </SelectItem>
   428→                ))}
   429→              </SelectContent>
   430→            </Select>
   431→            <Select value={deptFilter} onValueChange={setDeptFilter}>
   432→              <SelectTrigger className="bg-background w-full">
   433→                <SelectValue placeholder="Dept" />
   434→              </SelectTrigger>
   435→              <SelectContent>
   436→                <SelectItem value="all">All Departments</SelectItem>
   437→                {DEPARTMENTS.map((d) => (
   438→                  <SelectItem key={d} value={d}>
   439→                    {d}
   440→                  </SelectItem>
   441→                ))}
   442→              </SelectContent>
   443→            </Select>
   444→            <Select value={activeFilter} onValueChange={setActiveFilter}>
   445→              <SelectTrigger className="bg-background w-full">
   446→                <SelectValue placeholder="Status" />
   447→              </SelectTrigger>
   448→              <SelectContent>
   449→                <SelectItem value="all">All Status</SelectItem>
   450→                <SelectItem value="active">Active only</SelectItem>
   451→                <SelectItem value="inactive">Inactive only</SelectItem>
   452→              </SelectContent>
   453→            </Select>
   454→          </div>
   455→          <div className="flex items-center gap-2">
   456→            {filtersActive && (
   457→              <Button
   458→                type="button"
   459→                variant="ghost"
   460→                size="sm"
   461→                onClick={clearFilters}
   462→              >
   463→                <X className="h-4 w-4 mr-1" />
   464→                Clear
   465→              </Button>
   466→            )}
   467→            <Button type="button" onClick={() => setAddOpen(true)}>
   468→              <Plus className="h-4 w-4 mr-1" />
   469→              Add User
   470→            </Button>
   471→          </div>
   472→        </div>
   473→      </div>
   474→
   475→      <div className="bg-card border border-border rounded-xl overflow-hidden">
   476→        {filtered.length === 0 ? (
   477→          <EmptyState
   478→            icon={Users}
   479→            title={
   480→              filtersActive ? "No matching users" : "No users configured"
   481→            }
   482→            description={
   483→              filtersActive
   484→                ? "Try adjusting your search or filters."
   485→                : "Add your first user profile to get started."
   486→            }
   487→            action={
   488→              filtersActive ? (
   489→                <Button variant="outline" size="sm" onClick={clearFilters}>
   490→                  Clear Filters
   491→                </Button>
   492→              ) : (
   493→                <Button size="sm" onClick={() => setAddOpen(true)}>
   494→                  <Plus className="h-4 w-4 mr-1" />
   495→                  Add User
   496→                </Button>
   497→              )
   498→            }
   499→          />
   500→        ) : (
   501→          <ScrollArea className="max-h-[60vh]">
   502→            <Table>
   503→              <TableHeader className="sticky top-0 bg-card z-10">
   504→                <TableRow className="border-border hover:bg-transparent">
   505→                  <TableHead className="w-[40px]">Sr</TableHead>
   506→                  <TableHead>Username</TableHead>
   507→                  <TableHead>Display Name</TableHead>
   508→                  <TableHead>Email</TableHead>
   509→                  <TableHead>Role</TableHead>
   510→                  <TableHead>Department</TableHead>
   511→                  <TableHead>Status</TableHead>
   512→                  <TableHead>Created</TableHead>
   513→                  <TableHead className="text-right">Actions</TableHead>
   514→                </TableRow>
   515→              </TableHeader>
   516→              <TableBody>
   517→                {filtered.map((u, idx) => (

   522→                    <TableCell>
   523→                      <div className="flex items-center gap-2">
   524→                        <div className="h-7 w-7 rounded-full bg-primary/15 text-primary flex items-center justify-center text-xs font-semibold">
   525→                          {u.username.slice(0, 2).toUpperCase()}
   526→                        </div>
   527→                        <span className="font-medium text-foreground font-mono text-sm">
   528→                          {u.username}
   529→                        </span>
   530→                      </div>
   531→                    </TableCell>
   532→                    <TableCell className="font-medium text-foreground">
   533→                      {u.displayName}
   534→                    </TableCell>
   535→                    <TableCell>
   536→                      {u.email ? (
   537→                        <a
   538→                          href={`mailto:${u.email}`}
   539→                          className="inline-flex items-center gap-1 text-xs text-muted-foreground hover:text-primary"

   573→                            Inactive
   574→                          </>
   575→                        )}
   576→                      </CustomBadge>
   577→                    </TableCell>
   578→                    <TableCell className="text-xs text-muted-foreground">
   579→                      {formatDate(u.createdAt)}
   580→                    </TableCell>
   581→                    <TableCell className="text-right">
   582→                      <div className="flex items-center justify-end gap-1">
   583→                        <Button
   584→                          variant="ghost"
   585→                          size="sm"
   586→                          className="h-8 w-8 p-0"
   587→                          onClick={() => setEditTarget(u)}
   588→                          title="Edit user"
   589→                        >
   590→                          <Pencil className="h-4 w-4" />
   591→                        </Button>
   592→                        <Button
   593→                          variant="ghost"
   594→                          size="sm"
   595→                          className="h-8 w-8 p-0 text-red-400 hover:text-red-500 hover:bg-red-500/10"
   596→                          onClick={() => setDeleteTarget(u)}
   597→                          title="Delete user"
   598→                        >
   599→                          <Trash2 className="h-4 w-4" />

   601→                      </div>
   602→                    </TableCell>
   603→                  </TableRow>

   606→            </Table>

   608→        )}
   609→      </div>
   610→
   611→      {/* Add dialog */}

   618→          onSaved={() => {
   619→            setAddOpen(false);
   620→            refresh();
   621→          }}
   622→        />
   623→      )}
   624→
   625→      {/* Edit dialog (keyed remount ensures fresh form state) */}
   626→      {editTarget && (
   627→        <UserFormDialog
   628→          key={`edit-${editTarget.id}`}
   629→          open={!!editTarget}
   630→          onOpenChange={(o) => !o && setEditTarget(null)}
   631→          initial={editTarget}
