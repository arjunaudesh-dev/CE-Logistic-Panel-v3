   402→    changedBy: string;

   414→          comment: input.comment.trim() || `Bulk initiated → ${input.newPosition}`,

  1337→// ─── Update Position Dialog (dedicated edit popup with next-position + responsible person) ───
  1338→function UpdatePositionDialog({
  1339→  file,
  1340→  projectId,
  1341→  positionToResponsible,
  1342→  open,
  1343→  onOpenChange,
  1344→  onSaved,
  1345→}: {
  1346→  file: ProjectFile;
  1347→  projectId: string;
  1348→  positionToResponsible: Map<string, string>;
  1349→  open: boolean;
  1350→  onOpenChange: (o: boolean) => void;
  1351→  onSaved: () => void;
  1352→}) {
  1353→  const currentPosition = file.presentPosition;
  1354→  const [nextPosition, setNextPosition] = useState(currentPosition);
  1355→  const [responsiblePerson, setResponsiblePerson] = useState(file.responsiblePerson || "");
  1356→  const [fileNumber, setFileNumber] = useState(file.fileNumber || "");
  1357→  const [supplier, setSupplier] = useState(file.supplier || "");
  1358→  const [changedBy, setChangedBy] = useState("");
  1359→  const [comment, setComment] = useState("");
  1360→  const [saving, setSaving] = useState(false);
  1361→  const { toast } = useToast();
  1362→
  1363→  // When the user picks a different next position, prefill the responsible person
  1364→  // from the process-flow defaults — but only if the user hasn't manually overridden.
  1365→  const defaultForCurrent = positionToResponsible.get(currentPosition) || "";
  1366→  const defaultForNext = positionToResponsible.get(nextPosition) || "";
  1367→
  1368→  const handleNextPositionChange = (v: string) => {
  1369→    setNextPosition(v);
  1370→    // Auto-fill responsible person from the new position's default — only if user hasn't overridden
  1371→    const isUsingDefault = responsiblePerson === defaultForCurrent || responsiblePerson === "";
  1372→    if (isUsingDefault) {
  1373→      const newDefault = positionToResponsible.get(v) || "";
  1374→      setResponsiblePerson(newDefault);
  1375→    }
  1376→  };
  1377→
  1378→  const positionChanged = nextPosition !== currentPosition;
  1379→
  1380→  const submit = async () => {
  1381→    if (positionChanged && !changedBy.trim()) {
  1382→      toast({ title: "Changed By is required when updating position", variant: "destructive" });
  1383→      return;
  1384→    }
  1385→    setSaving(true);
  1386→    const body: Record<string, unknown> = {
  1387→      // Always allow file number + supplier updates
  1388→      fileNumber: fileNumber.trim() || null,
  1389→      supplier: supplier.trim() || null,
  1390→      // Position update
  1391→      presentPosition: nextPosition,
  1392→      responsiblePerson: responsiblePerson.trim() || null,
  1393→      changedBy: changedBy.trim() || "Officer",
  1394→      comment: comment.trim() || (positionChanged ? `Position changed: ${currentPosition} → ${nextPosition}` : "File metadata updated"),
  1395→    };
  1396→    const res = await apiRequest(`/api/projects/${projectId}/files/${file.id}`, "PUT", body);
  1397→    setSaving(false);
  1398→    if (res.ok) {
  1399→      toast({
  1400→        title: positionChanged ? "Position updated" : "File updated",
  1401→        description: positionChanged ? `${currentPosition} → ${nextPosition}` : undefined,
  1402→      });
  1403→      onSaved();
  1404→    } else {
  1405→      toast({ title: "Failed to update", description: res.error, variant: "destructive" });
  1406→    }
  1407→  };
  1408→
  1409→  return (
  1410→    <Dialog open={open} onOpenChange={onOpenChange}>
  1411→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
  1412→        <DialogHeader>
