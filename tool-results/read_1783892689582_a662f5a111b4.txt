   139→   139→function todayISO(): string {
   140→   140→  return new Date().toISOString().slice(0, 10);
   141→   141→}
   142→   142→function yesterdayISO(): string {
   143→   143→  return new Date(Date.now() - 86400000).toISOString().slice(0, 10);

   145→   145→function getLast7Days(): string[] {
   146→   146→  const days: string[] = [];
   147→   147→  const t = new Date();
   148→   148→  for (let i = 6; i >= 0; i--) {
   149→   149→    const d = new Date(t);
   150→   150→    d.setDate(d.getDate() - i);
   151→   151→    days.push(d.toISOString().slice(0, 10));
   152→   152→  }
   153→   153→  return days;
   154→   154→}
   155→   155→function formatDateLabel(dateStr: string): string {
   156→   156→  const d = new Date(dateStr + "T00:00:00");
   157→   157→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short" });
   158→   158→}
   159→   159→function formatFullDate(dateStr?: string | null): string {
   160→   160→  if (!dateStr) return "—";
   161→   161→  const d = new Date(dateStr + "T00:00:00");
   162→   162→  if (isNaN(d.getTime())) return dateStr;
   163→   163→  return d.toLocaleDateString("en-IN", {
   164→   164→    day: "2-digit",
   165→   165→    month: "short",
   166→   166→    year: "numeric",
   167→   167→  });
   168→   168→}
   169→   169→function formatDateTime(iso: string): string {
   170→   170→  const d = new Date(iso);
   171→   171→  if (isNaN(d.getTime())) return iso;
   172→   172→  return d.toLocaleString("en-IN", {
   173→   173→    day: "2-digit",
   174→   174→    month: "short",
   175→   175→    year: "numeric",
   176→   176→    hour: "2-digit",
   177→   177→    minute: "2-digit",
   178→   178→  });
   179→   179→}
   180→   180→function progressColor(pct: number): string {
   181→   181→  if (pct > 90) return "bg-emerald-500";
   182→   182→  if (pct >= 50) return "bg-amber-500";
   183→   183→  return "bg-red-500";
   184→   184→}
   185→   185→function statusColor(status: string): string {
   186→   186→  if (
   187→   187→    status === "Collected" ||
   188→   188→    (PART_SUPPLY_STATUSES as readonly string[]).includes(status) &&
   189→   189→      status === "Completed"
   190→   190→  ) {
   191→   191→    return "bg-emerald-500/15 text-emerald-400 border-emerald-500/30";
   192→   192→  }
   193→   193→  if (status === "In Progress" || status === "Active") {
   194→   194→    return "bg-blue-500/15 text-blue-400 border-blue-500/30";
   195→   195→  }
   196→   196→  if (status === "Pending") {
   197→   197→    return "bg-amber-500/15 text-amber-400 border-amber-500/30";
   198→   198→  }
   199→   199→  return "bg-slate-500/15 text-slate-300 border-slate-500/30";
   200→   200→}

  1677→  1677→function getISOWeek(dateStr: string): number {
  1678→  1678→  const d = new Date(dateStr + "T00:00:00");
  1679→  1679→  d.setHours(0, 0, 0, 0);
  1680→  1680→  // Thursday in current week decides the year
  1681→  1681→  d.setDate(d.getDate() + 3 - ((d.getDay() + 6) % 7));
  1682→  1682→  // January 4 is always in week 1
  1683→  1683→  const week1 = new Date(d.getFullYear(), 0, 4);
  1684→  1684→  return (
  1685→  1685→    1 +
  1686→  1686→    Math.round(
  1687→  1687→      ((d.getTime() - week1.getTime()) / 86400000 -
  1688→  1688→        3 +
  1689→  1689→        ((week1.getDay() + 6) % 7)) /
  1690→  1690→        7
  1691→  1691→    )
  1692→  1692→  );
  1693→  1693→}

  1695→  1695→function getWeekRange(dateStr: string): { key: string; label: string } {
  1696→  1696→  const d = new Date(dateStr + "T00:00:00");
  1697→  1697→  const dayOfWeek = (d.getDay() + 6) % 7; // 0 = Monday
  1698→  1698→  const monday = new Date(d);
  1699→  1699→  monday.setDate(d.getDate() - dayOfWeek);
  1700→  1700→  const sunday = new Date(monday);
  1701→  1701→  sunday.setDate(monday.getDate() + 6);
  1702→  1702→  const weekNum = getISOWeek(dateStr);
  1703→  1703→  const year = monday.getFullYear();
  1704→  1704→  const fmt = (dt: Date) =>
  1705→  1705→    dt.toLocaleDateString("en-IN", { day: "2-digit", month: "short" });
  1706→  1706→  return {
  1707→  1707→    key: `${year}-W${String(weekNum).padStart(2, "0")}`,
  1708→  1708→    label: `Week ${weekNum} · ${fmt(monday)} – ${fmt(sunday)}`,
  1709→  1709→  };
  1710→  1710→}
