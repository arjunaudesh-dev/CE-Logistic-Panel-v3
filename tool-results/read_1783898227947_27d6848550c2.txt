   165→// ────────────────────────────────────────────────────────────

   168→export function StoresModule() {
   169→  const [tab, setTab] = useState<"zones" | "stores">("zones");
   170→

   172→  const {
   173→    data: zones,
   174→    loading: zLoading,
   175→    error: zError,
   176→    refresh: refreshZones,
   177→  } = useFetch<StoreZone[]>("/api/stores/zones");
   178→  const {
   179→    data: items,
   180→    loading: iLoading,
   181→    error: iError,
   182→    refresh: refreshItems,
   183→  } = useFetch<StoreItem[]>("/api/stores");
   184→
   185→  // Stores-tab filter state
   186→  const [category, setCategory] = useState<string>("All");
   187→  const [search, setSearch] = useState("");
   188→  const [zoneFilter, setZoneFilter] = useState("all");
   189→  const [stockFilter, setStockFilter] = useState<StockFilter>("all");
   190→  const [sortKey, setSortKey] = useState<SortKey>("description");
   191→  const [sortDir, setSortDir] = useState<"asc" | "desc">("asc");
   192→  const [page, setPage] = useState(1);
   193→
   194→  // Dialogs
   195→  const [formOpen, setFormOpen] = useState(false);
   196→  const [editing, setEditing] = useState<StoreItem | null>(null);
   197→  const [bulkOpen, setBulkOpen] = useState(false);
   198→  const [viewing, setViewing] = useState<StoreItem | null>(null);
   199→  const [earmarkOpen, setEarmarkOpen] = useState(false);
   200→  const [earmarkListOpen, setEarmarkListOpen] = useState(false);
   201→  const [findCodeOpen, setFindCodeOpen] = useState(false);
   202→
   203→  const { toast } = useToast();
   204→
   205→  // Derived: filtered + sorted list
   206→  const filtered = useMemo(() => {
   207→    if (!items) return [];
   208→    let list = items.slice();
   209→    if (category !== "All") list = list.filter((i) => i.category === category);
   210→    if (zoneFilter !== "all") list = list.filter((i) => i.zoneName === zoneFilter);
   211→    const q = search.trim().toLowerCase();
   212→    if (q) list = list.filter((i) => i.description.toLowerCase().includes(q));
   213→    list = list.filter((i) => {
   214→      if (stockFilter === "out") return i.qty <= 0;
   215→      if (stockFilter === "low")
   216→        return i.minStock > 0 && i.qty > 0 && i.qty <= i.minStock;
   217→      if (stockFilter === "in") return i.minStock === 0 || i.qty > i.minStock;
   218→      if (stockFilter === "over") return i.maxStock > 0 && i.qty >= i.maxStock;
   219→      return true;
   220→    });
   221→    list.sort((a, b) => {
   222→      let cmp = 0;
   223→      if (sortKey === "description") cmp = a.description.localeCompare(b.description);
   224→      else if (sortKey === "qty") cmp = a.qty - b.qty;
   225→      else if (sortKey === "cost") cmp = a.totalCost - b.totalCost;
   226→      else if (sortKey === "category") cmp = a.category.localeCompare(b.category);
   227→      return sortDir === "asc" ? cmp : -cmp;
   228→    });
   229→    return list;
   230→  }, [items, category, zoneFilter, search, stockFilter, sortKey, sortDir]);
   231→
   232→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   233→  const safePage = Math.min(page, totalPages);
   234→  const pageItems = filtered.slice(
   235→    (safePage - 1) * PAGE_SIZE,
   236→    safePage * PAGE_SIZE
   237→  );
   238→
   239→  // Stats reflect the currently filtered set
   240→  const stats = useMemo(() => {
   241→    const net = filtered.reduce((s, i) => s + i.totalCost, 0);
   242→    const task = filtered.length;
   243→    const avail = filtered.filter((i) => i.qty > 0).length;
   244→    const team = filtered.filter((i) => i.minStock > 0 && i.qty <= i.minStock).length;
   245→    return { net, task, avail, team };
   246→  }, [filtered]);
   247→
   248→  // Wrap filter setters so any change also resets pagination to page 1
   249→  const setCategoryR = (c: string) => {
   250→    setCategory(c);
   251→    setPage(1);
   252→  };
   253→  const setSearchR = (s: string) => {
   254→    setSearch(s);
   255→    setPage(1);
   256→  };
   257→  const setZoneFilterR = (z: string) => {
   258→    setZoneFilter(z);
   259→    setPage(1);
   260→  };
   261→  const setStockFilterR = (s: StockFilter) => {
   262→    setStockFilter(s);
   263→    setPage(1);
   264→  };
   265→  const setSortKeyR = (k: SortKey) => {
   266→    setSortKey(k);
   267→    setPage(1);
   268→  };
   269→  const setSortDirR = (d: "asc" | "desc") => {
   270→    setSortDir(d);
   271→    setPage(1);
   272→  };
   273→  const toggleSortR = (k: SortKey) => {
   274→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   275→    else {
   276→      setSortKey(k);
   277→      setSortDir("asc");
   278→    }
   279→    setPage(1);
   280→  };
   281→
   282→  // Handlers
   283→  const handleZoneClick = (zoneName: string) => {
   284→    setZoneFilter(zoneName);
   285→    setCategory("All");
   286→    setStockFilter("all");
   287→    setSearch("");
   288→    setPage(1);
   289→    setTab("stores");
   290→    toast({
   291→      title: `Browsing ${zoneName}`,
   292→      description: "Switched to Stores tab with zone filter applied.",
   293→    });
   294→  };
   295→
   296→  const openAdd = () => {
   297→    setEditing(null);
   298→    setFormOpen(true);
   299→  };
   300→
   301→  const openEdit = (item: StoreItem) => {
   302→    setEditing(item);
   303→    setFormOpen(true);
   304→  };
   305→
   306→  const handleSaved = () => {
   307→    setFormOpen(false);
   308→    setEditing(null);
   309→    refreshItems();
   310→    refreshZones();
   311→    toast({
   312→      title: editing ? "Item updated" : "Item added",
   313→      description: editing
   314→        ? "Store item has been updated successfully."
   315→        : "New store item has been added to the store.",
   316→    });
   317→  };
   318→
   319→  const toggleSort = (k: SortKey) => {
   320→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   321→    else {
   322→      setSortKey(k);
   323→      setSortDir("asc");
   324→    }
   325→  };
   326→
   327→  return (
   328→    <div>
   329→      <PageHeader
   330→        title="Store Management"
   331→        description="Zones, inventory & stock-level monitoring across the dockyard"
   332→        icon={Warehouse}
   333→        actions={
   334→          <div className="flex flex-wrap items-center gap-2 justify-end">
   335→            <Button
   336→              variant="outline"
   337→              size="sm"
   338→              onClick={() => setFindCodeOpen(true)}
   339→              className="gap-1.5"
   340→              title="Find item codes using AI"
   341→            >
   342→              <ScanSearch className="h-4 w-4" /> Find Item Codes (AI)
   343→            </Button>
   344→            <Button
   345→              variant="outline"
   346→              size="sm"
   347→              onClick={() => setEarmarkListOpen(true)}
   348→              className="gap-1.5"
   349→              title="View all earmarked items"
   350→            >
   351→              <List className="h-4 w-4" /> View Earmarked
   352→            </Button>
   353→            <Button
   354→              variant="outline"
   355→              size="sm"
   356→              onClick={() => setEarmarkOpen(true)}
   357→              className="gap-1.5"
   358→              title="Earmark an item for a project without a PR"
   359→            >
   360→              <BookmarkPlus className="h-4 w-4" /> Earmark Item
   361→            </Button>
   362→            <Button
   363→              variant="outline"
   364→              size="sm"
   365→              onClick={() => setBulkOpen(true)}
   366→              className="gap-1.5"
   367→            >
   368→              <Upload className="h-4 w-4" /> Bulk Upload
   369→            </Button>
   370→            <Button onClick={openAdd} size="sm" className="gap-1.5">
   371→              <Plus className="h-4 w-4" /> Add Item
   372→            </Button>
   373→          </div>
   374→        }
   375→      />
   376→
   377→      <Tabs
   378→        value={tab}
   379→        onValueChange={(v) => setTab(v as "zones" | "stores")}
   380→        className="w-full"
   381→      >
   382→        <TabsList>
   383→          <TabsTrigger value="zones" className="gap-1.5">
   384→            <Boxes className="h-4 w-4" /> Zones
   385→          </TabsTrigger>
   386→          <TabsTrigger value="stores" className="gap-1.5">
   387→            <Package className="h-4 w-4" /> Stores
   388→          </TabsTrigger>
   389→        </TabsList>
   390→
   391→        {/* ─── Zones tab ─── */}
   392→        <TabsContent value="zones" className="mt-4">
   393→          <ZonesTab
   394→            zones={zones}
   395→            loading={zLoading}
   396→            error={zError}
   397→            onZoneClick={handleZoneClick}
   398→          />
   399→        </TabsContent>
   400→
   401→        {/* ─── Stores tab ─── */}
   402→        <TabsContent value="stores" className="mt-4">
   403→          <StoresTab
   404→            items={items}
   405→            loading={iLoading}
   406→            error={iError}
   407→            zones={zones}
   408→            stats={stats}
   409→            category={category}
   410→            setCategory={setCategoryR}
   411→            search={search}
   412→            setSearch={setSearchR}
   413→            zoneFilter={zoneFilter}
   414→            setZoneFilter={setZoneFilterR}
   415→            stockFilter={stockFilter}
   416→            setStockFilter={setStockFilterR}
   417→            sortKey={sortKey}
   418→            sortDir={sortDir}
   419→            setSortKey={setSortKeyR}
   420→            setSortDir={setSortDirR}
   421→            toggleSort={toggleSortR}
   422→            filtered={filtered}
   423→            pageItems={pageItems}
   424→            safePage={safePage}
   425→            totalPages={totalPages}
   426→            setPage={setPage}
   427→            onEdit={openEdit}

   432→      </Tabs>
   433→
   434→      {/* Add / Edit form */}
   435→      <ItemFormDialog

   437→        open={formOpen}

   465→        }}

   468→      {/* Earmark item dialog */}

   495→// ────────────────────────────────────────────────────────────
   496→// Zones tab — dashboard view with scaled tiles
   497→// ────────────────────────────────────────────────────────────
   498→interface ZonesTabProps {
   499→  zones: StoreZone[] | null;
   500→  loading: boolean;
   501→  error: string | null;
   502→  onZoneClick: (zoneName: string) => void;
   503→}
   504→
   505→function ZonesTab({ zones, loading, error, onZoneClick }: ZonesTabProps) {
   506→  if (loading) return <LoadingState label="Loading zones..." />;
   507→  if (error) return <ErrorState message={error} />;
   508→  if (!zones || zones.length === 0)
   509→    return (
   510→      <EmptyState
   511→        icon={Boxes}
   512→        title="No zones configured"
   513→        description="Store zones will appear here as tiles once data is seeded."
   514→      />
   515→    );
   516→
   517→  const totalValue = zones.reduce((s, z) => s + z.totalValue, 0);
   518→  const totalItems = zones.reduce((s, z) => s + z.itemCount, 0);
   519→  const totalLow = zones.reduce((s, z) => s + z.lowStockCount, 0);
   520→
   521→  return (
   522→    <div>
   523→      {/* Mini summary strip */}
   524→      <div className="grid grid-cols-3 gap-3 mb-5">
   525→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   526→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   527→            Total Stock Value
   528→          </p>
   529→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   530→            {formatCurrency(totalValue)}
   531→          </p>
   532→        </div>
   533→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   534→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   535→            Total Items
   536→          </p>
   537→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   538→            {formatNumber(totalItems)}
   539→          </p>
   540→        </div>
   541→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   542→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   543→            Low Stock Items
   544→          </p>
   545→          <p
   546→            className={cn(
   547→              "text-base sm:text-xl font-semibold mt-1",
   548→              totalLow > 0 ? "text-amber-400" : "text-emerald-400"
   549→            )}
   550→          >
   551→            {formatNumber(totalLow)}
   552→          </p>
   553→        </div>
   554→      </div>
   555→
   556→      {/* Tiles grid */}
   557→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   558→        {zones.map((z) => {
   559→          const status =
   560→            z.lowStockCount >= 3
   561→              ? "critical"
   562→              : z.lowStockCount > 0
   563→                ? "attention"
   564→                : "healthy";
   565→          const statusClasses =
   566→            status === "critical"
   567→              ? "border-red-500/40 hover:border-red-500/70"
   568→              : status === "attention"
   569→                ? "border-amber-500/40 hover:border-amber-500/70"
   570→                : "border-emerald-500/30 hover:border-emerald-500/60";
   571→          const minHeight = 120 + Math.min(z.totalValue / 100000, 80);
   572→
   573→          return (
   574→            <button
   575→              key={z.id}
   576→              onClick={() => onZoneClick(z.name)}
   577→              style={{ minHeight }}
   578→              className={cn(
   579→                "group bg-card border rounded-xl p-4 sm:p-5 text-left transition-all flex flex-col justify-between",
   580→                "hover:shadow-lg hover:shadow-black/20 hover:-translate-y-0.5",
   581→                statusClasses
   582→              )}
   583→            >
   584→              {/* Header */}
   585→              <div className="flex items-start justify-between gap-2 mb-3">
   586→                <div className="min-w-0">
   587→                  <h3 className="font-semibold text-foreground group-hover:text-primary transition-colors truncate text-sm sm:text-base">
   588→                    {z.name}
   589→                  </h3>
   590→                  <p className="text-[10px] sm:text-xs text-muted-foreground mt-0.5">
   591→                    {z.itemCount} items tracked
   592→                  </p>
   593→                </div>
   594→                <CustomBadge
   595→                  className={cn(
   596→                    "border-0 shrink-0",
   597→                    status === "critical" &&
   598→                      "bg-red-500/15 text-red-400",
   599→                    status === "attention" &&
   600→                      "bg-amber-500/15 text-amber-400",
   601→                    status === "healthy" &&
   602→                      "bg-emerald-500/15 text-emerald-400"
   603→                  )}
   604→                >
   605→                  {z.type}
   606→                </CustomBadge>
   607→              </div>
   608→
   609→              {/* Value */}
   610→              <div className="mb-3">
   611→                <p className="text-[10px] uppercase tracking-wide text-muted-foreground">
   612→                  Stock Value
   613→                </p>
   614→                <p className="text-lg sm:text-2xl font-bold text-foreground">
   615→                  {formatCurrency(z.totalValue)}
   616→                </p>
   617→              </div>
   618→
   619→              {/* Status chips */}
   620→              <div className="flex flex-wrap gap-1.5 mt-auto">
   621→                <ZoneChip

  1153→  // Initialise from `editing` — parent remounts this component via `key`

  1155→  const [form, setForm] = useState<ItemForm>(() =>
  1156→    editing

  1168→          ilmsCode: editing.ilmsCode ?? "",
  1169→          zoneName: editing.zoneName,
  1170→        }
  1171→      : DEFAULT_FORM
  1172→  );
  1173→  const [saving, setSaving] = useState(false);
  1174→  const { toast } = useToast();
  1175→
  1176→  const qtyNum = Number(form.qty) || 0;
  1177→  const unitCostNum = Number(form.unitCost) || 0;
  1178→  const fileCost = qtyNum * unitCostNum;
  1179→
  1180→  const set = (k: keyof ItemForm, v: string) =>
  1181→    setForm((prev) => ({ ...prev, [k]: v }));

  1183→  const submit = async () => {

  1185→      toast({
  1186→        title: "Description is required",
  1187→        variant: "destructive",
  1188→      });
  1189→      return;
  1190→    }
  1191→    setSaving(true);
  1192→    const payload = {

  1195→      unit: form.unit || "NOS",

  1211→      onSaved();
  1212→    } else {
  1213→      toast({
  1214→        title: editing ? "Failed to update item" : "Failed to add item",

  1468→// Simple RFC-4180-aware CSV parser
  1469→function parseCSV(text: string): string[][] {
  1470→  const rows: string[][] = [];
  1471→  let row: string[] = [];
  1472→  let cell = "";
  1473→  let inQuotes = false;
  1474→  for (let i = 0; i < text.length; i++) {
  1475→    const c = text[i];
  1476→    if (inQuotes) {
  1477→      if (c === '"') {
  1478→        if (text[i + 1] === '"') {
  1479→          cell += '"';
  1480→          i++;
  1481→        } else {
  1482→          inQuotes = false;
  1483→        }
  1484→      } else {
  1485→        cell += c;
  1486→      }
  1487→    } else {
  1488→      if (c === '"') {
  1489→        inQuotes = true;
  1490→      } else if (c === ",") {
  1491→        row.push(cell);
  1492→        cell = "";
  1493→      } else if (c === "\n") {
  1494→        row.push(cell);
  1495→        rows.push(row);
  1496→        row = [];
  1497→        cell = "";
  1498→      } else if (c === "\r") {
  1499→        // skip CR
  1500→      } else {
  1501→        cell += c;
  1502→      }
  1503→    }
  1504→  }
  1505→  if (cell !== "" || row.length > 0) {
  1506→    row.push(cell);
  1507→    rows.push(row);
  1508→  }
  1509→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1510→}
