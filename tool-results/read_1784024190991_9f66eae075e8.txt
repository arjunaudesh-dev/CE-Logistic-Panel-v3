    80→
    81→export function AddProjectWizard({ open, onClose, onCreated }: Props) {
    82→  const [step, setStep] = useState<1 | 2>(1);
    83→  const [saving, setSaving] = useState(false);
    84→  const [createdProjectId, setCreatedProjectId] = useState<string | null>(null);
    85→  const [createdEstimateId, setCreatedEstimateId] = useState<string | null>(null);
    86→  const { toast } = useToast();
    87→  const fileInputRef = useRef<HTMLInputElement>(null);
    88→  const [previewRows, setPreviewRows] = useState<EstimateRow[] | null>(null);
    89→  const [customVoteNumber, setCustomVoteNumber] = useState(false);
    90→  const [selectedVoteId, setSelectedVoteId] = useState("");
    91→  const { data: votes } = useFetch<Vote[]>("/api/votes");
    92→
    93→  const [form, setForm] = useState({
    94→    name: "",
    95→    voteNumber: "",
    96→    voteAllocation: "",
    97→    committed: "0",
    98→    actualSpent: "0",
    99→    startDate: "",
   100→    endDate: "", // EDC
   101→    benefactor: "", // replaces Department
   102→    reference: "", // manual text — how vote allocated
   103→    projectType: "New Construction",
   104→    financialMean: "Vote/Funds",
   105→    projectOfficer: "",
   106→    status: "Active",
   107→    description: "",
   108→  });
   109→  const [estimateRows, setEstimateRows] = useState<EstimateRow[]>([
   110→    { itemName: "", qty: "", unit: "NOS", estimateCost: "" },
   111→  ]);
   112→
   113→  const selectedVote = votes?.find((v) => v.id === selectedVoteId);
   114→
   115→  const handleVoteSelect = (id: string) => {
   116→    setSelectedVoteId(id);
   117→    const v = votes?.find((vt) => vt.id === id);
   118→    if (v) {
   119→      setForm((prev) => ({
   120→        ...prev,
   121→        voteNumber: v.voteNumber,
   122→        voteAllocation: v.allocation ? String(v.allocation) : prev.voteAllocation,
   123→      }));
   124→    }
   125→  };
   126→
   127→  const handleFinancialMeanChange = (v: string) => {
   128→    setForm((prev) => ({ ...prev, financialMean: v, voteNumber: "" }));
   129→    setCustomVoteNumber(false);
   130→    setSelectedVoteId("");
   131→  };
   132→
   133→  const isVoteFunds = form.financialMean === "Vote/Funds";
   134→  const isNonVote = !isVoteFunds;
   135→  const nonVotePlaceholder = form.financialMean === "Contractual" ? "CONTRACTUAL" : "NON-NAVAL";
   136→  const nonVoteHint =
   137→    form.financialMean === "Contractual"
   138→      ? "Contractual funding — Vote Number should be 'CONTRACTUAL' or the ARO reference."
   139→      : form.financialMean === "Malima Fund"
   140→      ? "Malima Fund — Vote Number can be 'NON-NAVAL' or the Malima reference number."
   141→      : "Non-Public Fund — Vote Number can be 'NON-NAVAL' or the fund reference.";
   142→
   143→  const addRow = () =>
   144→    setEstimateRows([...estimateRows, { itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
   145→  const removeRow = (idx: number) =>
   146→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
   147→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
   148→    setEstimateRows(estimateRows.map((r, i) => (i === idx ? { ...r, [field]: value } : r)));
   149→
   150→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   151→  const validEstimateRows = estimateRows.filter(
   152→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   153→  );
   154→  const hasValidEstimate = validEstimateRows.length > 0;
   155→
   156→  const downloadTemplate = () => {
   157→    const header = "ItemName,Qty,Unit,EstimateCost";
   158→    const sample = "Centrifugal Pump 15kW,2,NOS,185000";
   159→    const csv = `${header}\n${sample}\n`;
   160→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   161→    const url = URL.createObjectURL(blob);
   162→    const a = document.createElement("a");
   163→    a.href = url;
   164→    a.download = "material-estimate-template.csv";
   165→    a.click();
   166→    URL.revokeObjectURL(url);
   167→    toast({ title: "Template downloaded", description: "Fill in ItemName, Qty, Unit, EstimateCost" });
   168→  };
   169→
   170→  // Simple CSV parser — handles quoted fields with commas
   171→  const parseCsv = (text: string): string[][] => {
   172→    const rows: string[][] = [];
   173→    let cur: string[] = [];
   174→    let field = "";
   175→    let inQuotes = false;
   176→    for (let i = 0; i < text.length; i++) {
   177→      const ch = text[i];
   178→      if (inQuotes) {
   179→        if (ch === '"' && text[i + 1] === '"') {
   180→          field += '"';
   181→          i++;
   182→        } else if (ch === '"') {
   183→          inQuotes = false;
   184→        } else {
   185→          field += ch;
   186→        }
   187→      } else {
   188→        if (ch === '"') inQuotes = true;
   189→        else if (ch === ",") {
   190→          cur.push(field);
   191→          field = "";
   192→        } else if (ch === "\n" || ch === "\r") {
   193→          if (ch === "\r" && text[i + 1] === "\n") i++;
   194→          cur.push(field);
   195→          rows.push(cur);
   196→          cur = [];
   197→          field = "";
   198→        } else {
   199→          field += ch;
   200→        }
   201→      }
   202→    }
   203→    if (field.length > 0 || cur.length > 0) {
   204→      cur.push(field);
   205→      rows.push(cur);
   206→    }
   207→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   208→  };
   209→
   210→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   211→    const file = e.target.files?.[0];
   212→    if (!file) return;
   213→    if (!file.name.toLowerCase().endsWith(".csv")) {
   214→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   215→      e.target.value = "";
   216→      return;
   217→    }
   218→    try {
   219→      const text = await file.text();
   220→      const rows = parseCsv(text);
   221→      if (rows.length === 0) {
   222→        toast({ title: "CSV file is empty", variant: "destructive" });
   223→        e.target.value = "";
   224→        return;
   225→      }
   226→      const header = rows[0].map((c) => c.trim().toLowerCase());
   227→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   228→      const idxQty = header.findIndex((h) => h.includes("qty"));
   229→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   230→      const idxCost = header.findIndex((h) => h.includes("cost") || h.includes("price"));
   231→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   232→      const parsed: EstimateRow[] = dataRows
   233→        .map((r) => ({
   234→          itemName: idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim(),
   235→          qty: idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim(),
   236→          unit: (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS",
   237→          estimateCost: idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim(),
   238→        }))
   239→        .filter((r) => r.itemName);
   240→      if (parsed.length === 0) {
   241→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   242→        e.target.value = "";
   243→        return;
   244→      }
   245→      setPreviewRows(parsed);
   246→    } catch {
   247→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   248→    }
   249→    e.target.value = "";
   250→  };
   251→
   252→  const confirmImport = () => {
   253→    if (!previewRows) return;
   254→    setEstimateRows(previewRows);
   255→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   256→    setPreviewRows(null);
   257→  };
   258→
   259→  const handleSaveAndProcess = async () => {
   260→    if (!form.name || !form.voteNumber) {
   261→      toast({ title: "Project Name and Vote Number are required", variant: "destructive" });
   262→      return;
   263→    }
   264→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   265→    if (validRows.length === 0) {
   266→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   267→      return;
   268→    }
   269→    setSaving(true);
   270→    // 1. Create project
   271→    const projRes = await apiRequest("/api/projects", "POST", form);
   272→    if (!projRes.ok) {
   273→      setSaving(false);
   274→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   275→      return;
   276→    }
   277→    const project = projRes.data as { id: string };
   278→    // 2. Export Estimate into DB
   279→    const estRes = await apiRequest("/api/estimates", "POST", {
   280→      projectId: project.id,
   281→      name: `${form.name} — Material Estimate`,
   282→      items: validRows.map((r) => ({
   283→        itemName: r.itemName,
   284→        qty: Number(r.qty) || 0,
   285→        unit: r.unit,
   286→        estimateCost: Number(r.estimateCost) || 0,
   287→      })),
   288→    });
   289→    setSaving(false);
   290→    if (estRes.ok) {
   291→      const est = estRes.data as { id: string };
   292→      setCreatedProjectId(project.id);
   293→      setCreatedEstimateId(est.id);
   294→      setStep(2);
   295→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   296→    } else {
   297→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   298→    }
   299→  };
   300→
   301→  const handleGoToProject = () => {
   302→    if (createdProjectId) {
   303→      onCreated({
   304→        id: createdProjectId,
   305→        name: form.name,
   306→        voteNumber: form.voteNumber,
   307→        voteAllocation: Number(form.voteAllocation) || 0,
   308→        committed: Number(form.committed) || 0,
   309→        actualSpent: Number(form.actualSpent) || 0,
   310→        benefactor: form.benefactor || null,
   311→        reference: form.reference || null,
   312→        endDate: form.endDate || null,
   313→        startDate: form.startDate || null,
   314→        projectType: form.projectType,
   315→        financialMean: form.financialMean,
   316→      });
   317→      // reset
   318→      setStep(1);
   319→      setCreatedProjectId(null);
   320→      setCreatedEstimateId(null);
   321→      setForm({
   322→        name: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   323→        startDate: "", endDate: "", benefactor: "", reference: "",
   324→        projectType: "New Construction", financialMean: "Vote/Funds",
   325→        projectOfficer: "", status: "Active", description: "",
   326→      });
   327→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
   328→      setCustomVoteNumber(false);
   329→      setSelectedVoteId("");
   330→    }
   331→  };
   332→
   333→  if (!open) return null;
   334→
   335→  return (
   336→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   337→      {/* Header */}
   338→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   339→        <div className="flex items-center gap-3">
   340→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   341→            <X className="h-5 w-5" />
   342→          </button>
   343→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   344→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   345→          </h2>
   346→        </div>
   347→        {/* Stepper */}
   348→        <div className="hidden sm:flex items-center gap-2 text-xs">
   349→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   350→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   351→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   352→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   353→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   354→        </div>
   355→      </div>
   356→
   357→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   358→        {step === 1 && (
   359→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   360→            {/* Project details */}
   361→            <section className="bg-card border border-border rounded-xl p-5">
   362→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   363→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   364→              </h3>
   365→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   366→                <div className="sm:col-span-2 space-y-1.5">
   367→                  <Label>Project Name *</Label>
   368→                  <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
   369→                </div>
   370→
   371→                {/* Vote Number — conditional UI based on Financial Mean */}
   372→                <div className="space-y-1.5">
   373→                  <Label className="flex items-center gap-1.5">
   374→                    <Vote className="h-3.5 w-3.5 text-primary" />
   375→                    Vote Number *
   376→                  </Label>
   377→                  {isVoteFunds && !customVoteNumber ? (
   378→                    <>
   379→                      <Select value={selectedVoteId} onValueChange={handleVoteSelect}>
   380→                        <SelectTrigger className="bg-background border-border">
   381→                          <SelectValue placeholder="Select a vote from the list…" />
   382→                        </SelectTrigger>
   383→                        <SelectContent>
   384→                          {votes && votes.length > 0 ? (
   385→                            votes.map((v) => (
   386→                              <SelectItem key={v.id} value={v.id}>
   387→                                <span className="font-mono">{v.voteNumber}</span> — {v.description} ({v.voteType})
   388→                              </SelectItem>
   389→                            ))
   390→                          ) : (
   391→                            <SelectItem value="_none" disabled>No votes available</SelectItem>
   392→                          )}
   393→                        </SelectContent>
   394→                      </Select>
   395→                      {selectedVote ? (
   396→                        <div className="flex items-center gap-2 pt-0.5 flex-wrap">
   397→                          <CustomBadge className={cn("border", voteTypeColor(selectedVote.voteType))}>
   398→                            {selectedVote.voteType}
   399→                          </CustomBadge>
   400→                          {selectedVote.allocation > 0 && (
   401→                            <span className="text-[11px] text-muted-foreground">
   402→                              Allocation: {formatCurrency(selectedVote.allocation)}
   403→                            </span>
   404→                          )}
   405→                          {selectedVote.financialYear && (
   406→                            <span className="text-[11px] text-muted-foreground">
   407→                              · FY: {selectedVote.financialYear}
   408→                            </span>
   409→                          )}
   410→                          {!selectedVote.active && (
   411→                            <CustomBadge className="border-0 bg-amber-500/15 text-amber-400">Inactive</CustomBadge>
   412→                          )}
   413→                        </div>
   414→                      ) : (
   415→                        <p className="text-[11px] text-muted-foreground flex items-start gap-1">
   416→                          <Info className="h-3 w-3 mt-0.5 shrink-0" />
   417→                          Capital votes fund one-time construction. Recurring votes fund annual maintenance.
   418→                        </p>
   419→                      )}
   420→                      <button
   421→                        type="button"
   422→                        onClick={() => {
   423→                          setCustomVoteNumber(true);
   424→                          setSelectedVoteId("");
   425→                          setForm((prev) => ({ ...prev, voteNumber: "" }));
   426→                        }}
   427→                        className="text-[11px] text-primary hover:underline flex items-center gap-1 self-start"
   428→                      >
   429→                        <Pencil className="h-3 w-3" /> Enter custom vote number
   430→                      </button>
   431→                    </>
   432→                  ) : isVoteFunds && customVoteNumber ? (
   433→                    <>
   434→                      <Input
   435→                        value={form.voteNumber}
   436→                        onChange={(e) => setForm({ ...form, voteNumber: e.target.value })}
   437→                        placeholder="e.g. 223-1-01-2104"
   438→                        className="bg-background border-border"
   439→                      />
   440→                      <button
   441→                        type="button"
   442→                        onClick={() => {
   443→                          setCustomVoteNumber(false);
   444→                          setForm((prev) => ({ ...prev, voteNumber: "" }));
   445→                        }}
   446→                        className="text-[11px] text-primary hover:underline flex items-center gap-1 self-start"
   447→                      >
   448→                        <ChevronLeft className="h-3 w-3" /> Back to vote list
   449→                      </button>
   450→                    </>
   451→                  ) : (
   452→                    <>
   453→                      <Input
   454→                        value={form.voteNumber}
   455→                        onChange={(e) => setForm({ ...form, voteNumber: e.target.value })}
   456→                        placeholder={nonVotePlaceholder}
   457→                        className="bg-background border-border"
   458→                      />
   459→                      <p className="text-[11px] text-amber-400 flex items-start gap-1">
   460→                        <Landmark className="h-3 w-3 mt-0.5 shrink-0" />
   461→                        {nonVoteHint}
   462→                      </p>
   463→                    </>
   464→                  )}
   465→                </div>
   466→
   467→                {/* Project Type */}
   468→                <div className="space-y-1.5">
   469→                  <Label>Project Type *</Label>
   470→                  <Select value={form.projectType} onValueChange={(v) => setForm({ ...form, projectType: v })}>
   471→                    <SelectTrigger className="bg-background border-border">
   472→                      <SelectValue />
   473→                    </SelectTrigger>
   474→                    <SelectContent>
   475→                      {PROJECT_TYPES.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   476→                    </SelectContent>
   477→                  </Select>
   478→                  {form.projectType && (
   479→                    <div className="pt-0.5">
   480→                      <CustomBadge className={cn("border", projectTypeColor(form.projectType))}>
   481→                        {form.projectType}
   482→                      </CustomBadge>
   483→                    </div>
   484→                  )}
   485→                </div>
   486→
   487→                {/* Financial Mean */}
   488→                <div className="space-y-1.5">
   489→                  <Label>Financial Mean *</Label>
   490→                  <Select value={form.financialMean} onValueChange={handleFinancialMeanChange}>
   491→                    <SelectTrigger className="bg-background border-border">
   492→                      <SelectValue />
   493→                    </SelectTrigger>
   494→                    <SelectContent>
   495→                      {FINANCIAL_MEANS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   496→                    </SelectContent>
   497→                  </Select>
   498→                  {form.financialMean && (
   499→                    <div className="pt-0.5">
   500→                      <CustomBadge className={cn("border", financialMeanColor(form.financialMean))}>
   501→                        {form.financialMean}
   502→                      </CustomBadge>
   503→                    </div>
   504→                  )}
   505→                </div>
   506→
   507→                {/* Reference — manual text entry (how vote allocated) */}
   508→                <div className="space-y-1.5">
   509→                  <Label>Reference (how vote allocated)</Label>
   510→                  <Input
   511→                    value={form.reference}
   512→                    onChange={(e) => setForm({ ...form, reference: e.target.value })}
   513→                    placeholder="e.g. Vote 2104 - Allocation 2025"
   514→                    className="bg-background border-border"
   515→                  />
   516→                </div>
   517→
   518→                <div className="space-y-1.5">
   519→                  <Label>Vote Allocation (Rs.)</Label>
   520→                  <Input type="number" value={form.voteAllocation} onChange={(e) => setForm({ ...form, voteAllocation: e.target.value })} placeholder="0" className="bg-background border-border" />
   521→                </div>
   522→                <div className="space-y-1.5">
   523→                  <Label>Benefactor</Label>
   524→                  <Input value={form.benefactor} onChange={(e) => setForm({ ...form, benefactor: e.target.value })} placeholder="e.g. Engineering Dept" className="bg-background border-border" />
   525→                </div>
   526→                <div className="space-y-1.5">
   527→                  <Label>Start Date</Label>
