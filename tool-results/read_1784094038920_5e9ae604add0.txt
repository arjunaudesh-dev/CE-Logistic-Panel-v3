   121→
   122→export function PresentPositionDialog({ project, open, onOpenChange, onChanged }: Props) {
   123→  const { data: files, loading, refresh } = useFetch<ProjectFile[]>(
   124→    `/api/projects/${project.id}/files`
   125→  );
   126→  const { data: estimates, refresh: refreshEst } = useFetch<Estimate[]>(
   127→    `/api/estimates?projectId=${project.id}`
   128→  );
   129→  // Fetch process flows once to support the "Update Position" popup's default responsible person lookup
   130→  const { data: processFlows } = useFetch<ProcessFlow[]>(`/api/process-flows`);
   131→  const [activeTab, setActiveTab] = useState<"raised" | "estimate">("raised");
   132→  const [search, setSearch] = useState("");
   133→  const [positionFilter, setPositionFilter] = useState("all");
   134→  const [fileNumberFilter, setFileNumberFilter] = useState<FileNumberFilter>("all");
   135→  const [addOpen, setAddOpen] = useState(false);
   136→  const [updateFile, setUpdateFile] = useState<ProjectFile | null>(null);
   137→  const [logFile, setLogFile] = useState<ProjectFile | null>(null);
   138→  const [allocOpen, setAllocOpen] = useState(false);
   139→  const [activeEstimateId, setActiveEstimateId] = useState<string | null>(null);
   140→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
   141→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
   142→  // Group-by-File (default ON) — merges items sharing the same PR/fileGroupKey into a single row
   143→  const [groupByFile, setGroupByFile] = useState(true);
   144→  // Tracks which group keys are expanded (showing their individual items)
   145→  const [expandedGroups, setExpandedGroups] = useState<Set<string>>(new Set());
   146→  const { toast } = useToast();
   147→
   148→  const toggleGroup = (key: string) => {
   149→    setExpandedGroups((prev) => {
   150→      const next = new Set(prev);
   151→      if (next.has(key)) next.delete(key);
   152→      else next.add(key);
   153→      return next;
   154→    });
   155→  };
   156→
   157→  // Build a lookup of positionName -> defaultResponsiblePerson across all flows.
   158→  // Multiple flows may define the same positionName; we keep the first non-empty match.
   159→  const positionToResponsible = useMemo(() => {
   160→    const map = new Map<string, string>();
   161→    processFlows?.forEach((flow) => {
   162→      flow.steps.forEach((step) => {
   163→        if (step.positionName && step.defaultResponsiblePerson && !map.has(step.positionName)) {
   164→          map.set(step.positionName, step.defaultResponsiblePerson);
   165→        }
   166→      });
   167→    });
   168→    return map;
   169→  }, [processFlows]);
   170→
   171→  // Position summary
   172→  const positionSummary = useMemo(() => {
   173→    const map: Record<string, { count: number; cost: number }> = {};
   174→    files?.forEach((f) => {
   175→      if (!map[f.presentPosition]) map[f.presentPosition] = { count: 0, cost: 0 };
   176→      map[f.presentPosition].count++;
   177→      map[f.presentPosition].cost += f.fileCost;
   178→    });
   179→    return Object.entries(map).sort((a, b) => b[1].count - a[1].count);
   180→  }, [files]);
   181→
   182→  const totalFileCost = files?.reduce((s, f) => s + f.fileCost, 0) ?? 0;
   183→
   184→  // pending estimate items (not yet initiated) — "items to be raised PR"
   185→  const pendingEstimateItems = useMemo(() => {
   186→    const list: Estimate["items"][number][] = [];
   187→    estimates?.forEach((e) => {
   188→      e.items.forEach((it) => {
   189→        if (it.status === "Pending") list.push(it);
   190→      });
   191→    });
   192→    return list;
   193→  }, [estimates]);
   194→  const initiatedCount = useMemo(
   195→    () =>
   196→      estimates?.reduce(
   197→        (s, e) => s + e.items.filter((i) => i.status === "Initiated").length,
   198→        0
   199→      ) ?? 0,
   200→    [estimates]
   201→  );
   202→
   203→  // File-number counts for the filter toggle
   204→  const fileNumberCounts = useMemo(() => {
   205→    const all = files?.length || 0;
   206→    const withFn = files?.filter((f) => f.fileNumber && f.fileNumber.trim()).length || 0;
   207→    return { all, with: withFn, without: all - withFn };
   208→  }, [files]);
   209→
   210→  const filteredFiles = useMemo(() => {
   211→    let list = files || [];
   212→    if (search) {
   213→      const q = search.toLowerCase();
   214→      list = list.filter(
   215→        (f) =>
   216→          f.prNumber.toLowerCase().includes(q) ||
   217→          f.itemCode.toLowerCase().includes(q) ||
   218→          f.itemName.toLowerCase().includes(q) ||
   219→          (f.supplier || "").toLowerCase().includes(q) ||
   220→          (f.fileNumber || "").toLowerCase().includes(q)
   221→      );
   222→    }
   223→    if (positionFilter !== "all") {
   224→      list = list.filter((f) => f.presentPosition === positionFilter);
   225→    }
   226→    if (fileNumberFilter === "with") {
   227→      list = list.filter((f) => f.fileNumber && f.fileNumber.trim());
   228→    } else if (fileNumberFilter === "without") {
   229→      list = list.filter((f) => !f.fileNumber || !f.fileNumber.trim());
   230→    }
   231→    list = [...list].sort((a, b) => {
   232→      let cmp = 0;
   233→      if (sortKey === "fileCost") cmp = a.fileCost - b.fileCost;
   234→      else if (sortKey === "prNumber") cmp = a.prNumber.localeCompare(b.prNumber);
   235→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   236→      return sortDir === "asc" ? cmp : -cmp;
   237→    });
   238→    return list;
   239→  }, [files, search, positionFilter, fileNumberFilter, sortKey, sortDir]);
   240→
   241→  const toggleSort = (key: typeof sortKey) => {
   242→    if (sortKey === key) setSortDir(sortDir === "asc" ? "desc" : "asc");
   243→    else {
   244→      setSortKey(key);
   245→      setSortDir("desc");
   246→    }
   247→  };
   248→
   249→  // Group files by fileGroupKey (falls back to prNumber). Items sharing the same
   250→  // key collapse into a single "merged" row in the register when groupByFile is ON.
   251→  interface FileGroup {
   252→    key: string;
   253→    items: ProjectFile[];
   254→    totalCost: number;
   255→    totalQty: number;
   256→    commonSupplier: string | null;
   257→    commonCategory: string | null;
   258→    commonPosition: string | null;
   259→    commonFileNumber: string | null;
   260→    commonUnit: string | null;
   261→    commonResponsiblePerson: string | null;
   262→    firstCreatedAt: string;
   263→  }
   264→
   265→  const groupedView = useMemo<{ groups: FileGroup[]; singles: ProjectFile[] }>(() => {
   266→    if (!groupByFile) return { groups: [], singles: filteredFiles };
   267→    const map = new Map<string, ProjectFile[]>();
   268→    for (const f of filteredFiles) {
   269→      const k = (f.fileGroupKey && f.fileGroupKey.trim()) || f.prNumber;
   270→      if (!map.has(k)) map.set(k, []);
   271→      map.get(k)!.push(f);
   272→    }
   273→    const groups: FileGroup[] = [];
   274→    const singles: ProjectFile[] = [];
   275→    for (const [key, items] of map) {
   276→      if (items.length >= 2) {
   277→        const totalCost = items.reduce((s, f) => s + f.fileCost, 0);
   278→        const totalQty = items.reduce((s, f) => s + f.qty, 0);
   279→        const suppliers = new Set(items.map((f) => f.supplier).filter(Boolean));
   280→        const categories = new Set(items.map((f) => f.fileCategory));
   281→        const positions = new Set(items.map((f) => f.presentPosition));
   282→        const fileNumbers = new Set(items.map((f) => f.fileNumber).filter(Boolean));
   283→        const units = new Set(items.map((f) => f.unit));
   284→        const responsible = new Set(items.map((f) => f.responsiblePerson).filter(Boolean));
   285→        const firstCreatedAt = items
   286→          .map((f) => f.createdAt)
   287→          .sort((a, b) => new Date(b).getTime() - new Date(a).getTime())[0];
   288→        groups.push({
   289→          key,
   290→          items,
   291→          totalCost,
   292→          totalQty,
   293→          commonSupplier: suppliers.size === 1 ? Array.from(suppliers)[0]! : null,
   294→          commonCategory: categories.size === 1 ? items[0]!.fileCategory : null,
   295→          commonPosition: positions.size === 1 ? items[0]!.presentPosition : null,
   296→          commonFileNumber: fileNumbers.size === 1 ? Array.from(fileNumbers)[0]! : null,
   297→          commonUnit: units.size === 1 ? items[0]!.unit : null,
   298→          commonResponsiblePerson: responsible.size === 1 ? Array.from(responsible)[0]! : null,
   299→          firstCreatedAt,
   300→        });
   301→      } else {
   302→        singles.push(items[0]!);
   303→      }
   304→    }
   305→    return { groups, singles };
   306→  }, [filteredFiles, groupByFile]);
   307→
   308→  const handleDelete = async (file: ProjectFile) => {
   309→    if (!confirm(`Delete file ${file.prNumber} (${file.itemName})?`)) return;
   310→    const res = await apiRequest(`/api/projects/${project.id}/files/${file.id}`, "DELETE");
   311→    if (res.ok) {
   312→      toast({ title: "File deleted" });
   313→      refresh();
   314→      onChanged();
   315→    } else {
   316→      toast({ title: "Delete failed", description: res.error, variant: "destructive" });
   317→    }
   318→  };
   319→
   320→  const exportExcel = () => {
   321→    const rows = filteredFiles.map((f) => ({
   322→      "PR Number": f.prNumber,
   323→      "File Number": f.fileNumber || "",
   324→      "Item Code": f.itemCode,
   325→      "Item Name": f.itemName,
   326→      Unit: f.unit,
   327→      Qty: f.qty,
   328→      "Unit Price": f.unitPrice,
   329→      "File Cost": f.fileCost,
   330→      Supplier: f.supplier || "",
   331→      "Present Position": f.presentPosition,
   332→      "Responsible Person": f.responsiblePerson || "",
   333→      Category: f.fileCategory,
   334→    }));
   335→    const headers = Object.keys(rows[0] || { "PR Number": "" });
   336→    const csv = [
   337→      headers.join(","),
   338→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h] ?? ""}"`).join(",")),
   339→    ].join("\n");
   340→    const blob = new Blob([csv], { type: "text/csv" });
   341→    const url = URL.createObjectURL(blob);
   342→    const a = document.createElement("a");
   343→    a.href = url;
   344→    a.download = `${project.name}-files.csv`;
   345→    a.click();
   346→    URL.revokeObjectURL(url);
   347→  };
   348→
   349→  // pick the first processed estimate (or any) for the "PR from Estimate" tab
   350→  const estimateForTab = estimates?.find((e) => e.status === "Processed") || estimates?.[0] || null;
   351→
   352→  // Single file row renderer — shared between the ungrouped view and the
   353→  // "singles" list of the grouped view. Lives inside the component so it can
   354→  // access setLogFile / setUpdateFile / handleDelete / project.id / refresh.
   355→  const renderSingleFileRow = (f: ProjectFile) => {
   356→    const hasFileNumber = !!(f.fileNumber && f.fileNumber.trim());
   357→    return (
   358→      <TableRow key={f.id} className="border-border cursor-pointer hover:bg-secondary/40" onClick={() => setLogFile(f)}>
   359→        <TableCell className="font-mono text-xs text-foreground">{f.prNumber}</TableCell>
   360→        <TableCell>
   361→          {hasFileNumber ? (
   362→            <span className="inline-flex items-center gap-1 font-mono text-xs text-emerald-400">
   363→              <Hash className="h-3 w-3" />
   364→              {f.fileNumber}
   365→            </span>
   366→          ) : (
   367→            <span className="text-muted-foreground">—</span>
   368→          )}
   369→        </TableCell>
   370→        <TableCell className="font-mono text-xs">{f.itemCode}</TableCell>
   371→        <TableCell className="text-foreground">{f.itemName}</TableCell>
   372→        <TableCell className="text-right">{f.qty} {f.unit}</TableCell>
   373→        <TableCell className="text-right">{formatCurrency(f.unitPrice)}</TableCell>
   374→        <TableCell className="text-right font-medium text-foreground">{formatCurrency(f.fileCost)}</TableCell>
   375→        <TableCell className="text-xs">{f.supplier || "—"}</TableCell>
   376→        <TableCell>
   377→          <div className="flex flex-col gap-0.5">
   378→            <PositionBadge position={f.presentPosition} />
   379→            {f.responsiblePerson && (
   380→              <span className="inline-flex items-center gap-1 text-[10px] text-muted-foreground">
   381→                <User className="h-2.5 w-2.5" />
   382→                {f.responsiblePerson}
   383→              </span>
   384→            )}
   385→          </div>
   386→        </TableCell>
   387→        <TableCell><CategoryBadge category={f.fileCategory} /></TableCell>
   388→        <TableCell className="text-right">
   389→          <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   390→            <QuickUpdatePopover
   391→              files={[f]}
   392→              projectId={project.id}
   393→              onSaved={() => { refresh(); onChanged(); }}
   394→            />
   395→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setUpdateFile(f)} title="Edit File (full dialog)"><Pencil className="h-3.5 w-3.5" /></Button>
   396→            <Button variant="ghost" size="icon" className="h-7 w-7 text-red-400 hover:text-red-500" onClick={() => handleDelete(f)} title="Delete file"><Trash2 className="h-3.5 w-3.5" /></Button>
   397→          </div>
   398→        </TableCell>
   399→      </TableRow>
   400→    );
   401→  };
   402→
   403→  if (!open) return null;
   404→
   405→  return (
   406→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   407→      {/* Header */}
   408→      <div className="border-b border-border bg-card shrink-0">
   409→        <div className="px-4 sm:px-6 py-3 flex items-center justify-between gap-4">
   410→          <div className="flex items-center gap-3 min-w-0">
   411→            <button onClick={() => onOpenChange(false)} className="text-muted-foreground hover:text-foreground shrink-0">
   412→              <X className="h-5 w-5" />
   413→            </button>
   414→            <div className="min-w-0">
   415→              <h2 className="font-semibold text-foreground text-sm sm:text-base truncate">{project.name}</h2>
   416→              <p className="text-xs text-muted-foreground">
   417→                {project.voteNumber}
   418→                {project.benefactor && ` · Benefactor: ${project.benefactor}`}
   419→                {project.reference && ` · Ref: ${project.reference}`}
   420→                {project.endDate && ` · EDC: ${new Date(project.endDate).toLocaleDateString("en-GB")}`}
   421→              </p>
   422→            </div>
   423→          </div>
   424→          <Button variant="outline" size="sm" onClick={() => setAllocOpen(true)} className="gap-1.5 shrink-0">
   425→            <History className="h-4 w-4" /> Allocations
   426→          </Button>
   427→        </div>
   428→        {/* Financial summary bar */}
   429→        <div className="px-4 sm:px-6 pb-3 grid grid-cols-2 sm:grid-cols-4 gap-2">
   430→          <FinBox label="Vote Allocation" value={formatCurrency(project.voteAllocation)} tone="text-blue-400" />
   431→          <FinBox label="Committed" value={formatCurrency(project.committed)} tone="text-amber-400" />
   432→          <FinBox label="Actual Spent" value={formatCurrency(project.actualSpent)} tone="text-red-400" />
   433→          <FinBox label="File Cost (Total)" value={formatCurrency(totalFileCost)} tone="text-emerald-400" />
   434→        </div>
   435→        {/* Tabs */}
   436→        <div className="px-4 sm:px-6 flex items-center gap-1 border-t border-border">
   437→          <TabButton
   438→            active={activeTab === "raised"}
   439→            onClick={() => setActiveTab("raised")}
   440→            icon={<FileText className="h-3.5 w-3.5" />}
   441→            label="PR Raised Files"
   442→            count={files?.length || 0}
   443→          />
   444→          <TabButton
   445→            active={activeTab === "estimate"}
   446→            onClick={() => setActiveTab("estimate")}
   447→            icon={<Sparkles className="h-3.5 w-3.5" />}
   448→            label="PR from Estimate"
   449→            count={pendingEstimateItems.length}
   450→            tone="amber"
   451→          />
   452→        </div>
   453→      </div>
   454→
   455→      {/* Body */}
   456→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   457→        <div className="p-4 sm:p-6 space-y-4">
   458→          {/* Position summary (always visible) */}
   459→          <div>
   460→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   461→              Present Position Summary
   462→            </h4>
   463→            <div className="flex flex-wrap gap-2">
   464→              {positionSummary.length === 0 ? (
   465→                <p className="text-sm text-muted-foreground">No files yet.</p>
   466→              ) : (
   467→                positionSummary.map(([pos, info]) => (
   468→                  <div key={pos} className={cn("flex items-center gap-2 rounded-lg border px-3 py-1.5", positionColor(pos))}>
   469→                    <span className="text-sm font-medium">{pos}</span>
   470→                    <span className="text-xs opacity-80">{info.count} file{info.count > 1 ? "s" : ""} · {formatCurrency(info.cost)}</span>
   471→                  </div>
   472→                ))
   473→              )}
   474→            </div>
   475→          </div>
   476→
   477→          {activeTab === "raised" ? (
   478→            <>
   479→              {/* PR raised summary chips */}
   480→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   481→                <MiniStat label="PR Raised (Files)" value={String(files?.length || 0)} sub={formatCurrency(totalFileCost)} tone="text-blue-400" />
   482→                <MiniStat label="Initiated from Estimate" value={String(initiatedCount)} sub="forwarded PRs" tone="text-violet-400" />
   483→                <MiniStat label="Items to Raise PR" value={String(pendingEstimateItems.length)} sub="from estimate" tone="text-amber-400" />
   484→              </div>
   485→
   486→              {/* Toolbar */}
   487→              <div className="flex flex-wrap items-center gap-2">
   488→                <div className="relative flex-1 min-w-[200px]">
   489→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   490→                  <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search PR, file no, item, supplier..." className="pl-9 bg-background border-border" />
   491→                </div>
   492→                <Select value={positionFilter} onValueChange={setPositionFilter}>
   493→                  <SelectTrigger className="w-[180px] bg-background border-border"><SelectValue placeholder="Position" /></SelectTrigger>
   494→                  <SelectContent>
   495→                    <SelectItem value="all">All Positions</SelectItem>
   496→                    {PRESENT_POSITIONS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
   497→                  </SelectContent>
   498→                </Select>
   499→                {/* File Number filter toggle */}
   500→                <div className="inline-flex rounded-md border border-border overflow-hidden bg-background">
   501→                  <FilterPill
   502→                    label={`All (${fileNumberCounts.all})`}
   503→                    active={fileNumberFilter === "all"}
   504→                    onClick={() => setFileNumberFilter("all")}
   505→                  />
   506→                  <FilterPill

   863→  const isEdit = !!file;
   864→  const [saving, setSaving] = useState(false);
   865→  const [form, setForm] = useState({
   866→    prNumber: file?.prNumber || "",
   867→    itemCode: file?.itemCode || "",
   868→    itemName: file?.itemName || "",
   869→    unit: file?.unit || "NOS",
   870→    qty: String(file?.qty ?? ""),

   877→    changedBy: "",
   878→    comment: "",
   879→  });
   880→  const { toast } = useToast();
   881→
   882→  const submit = async () => {
   883→    if (!form.prNumber || !form.itemCode || !form.itemName) {
   884→      toast({ title: "PR Number, Item Code and Item Name are required", variant: "destructive" });
   885→      return;

   893→      toast({ title: isEdit ? "File updated" : "File added" });

   896→      toast({ title: "Failed", description: res.error, variant: "destructive" });

   899→
   900→  return (
   901→    <Dialog open={open} onOpenChange={onOpenChange}>
   902→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">

   913→          <div className="sm:col-span-2"><Field label="Item Name *"><Input value={form.itemName} onChange={(e) => setForm({ ...form, itemName: e.target.value })} className="bg-background border-border" /></Field></div>

  1159→// ─── Quick Update Position Popover (inline, lightweight) ───
  1160→// Used on every file row + every merged-group row. For groups, prompts the

  1162→function QuickUpdatePopover({
  1163→  files,
  1164→  projectId,
  1165→  onSaved,
  1166→  isGroup,
  1167→}: {
  1168→  files: ProjectFile[];
  1169→  projectId: string;
  1170→  onSaved: () => void;
  1171→  isGroup?: boolean;
  1172→}) {
  1173→  const [open, setOpen] = useState(false);
  1174→  const [newPosition, setNewPosition] = useState<string>(files[0]?.presentPosition || "PR Raised");
  1175→  const [comment, setComment] = useState("");
  1176→  const [changedBy, setChangedBy] = useState("Officer");

  1178→  const { toast } = useToast();

  1180→  const currentPosition = files[0]?.presentPosition || "";
  1181→  const positionChanged = newPosition !== currentPosition;
  1182→  const isMultiFile = isGroup && files.length > 1;
  1183→
  1184→  const submit = async () => {
  1185→    if (!newPosition) {
  1186→      toast({ title: "Select a new position", variant: "destructive" });

  1202→          changedBy: changedBy.trim() || "Officer",

  1240→        className="w-80 bg-popover text-popover-foreground border-border"
  1241→        align="end"
  1242→        onClick={(e) => e.stopPropagation()}
  1243→      >
  1244→        <div className="space-y-3">
  1245→          <div className="flex items-center gap-2">
  1246→            <RefreshCw className="h-4 w-4 text-primary" />

  1255→              {files[0]?.prNumber} · {files[0]?.itemName}
