   187→/** Strip everything except digits from a phone string. */
   188→function digits(num: string | null | undefined): string {
   189→  if (!num) return "";
   190→  return num.replace(/[^\d]/g, "");
   191→}
   192→
   193→/** Build a tel: link (digits only, keep simple). */
   194→function telLink(num: string | null | undefined): string | null {
   195→  const d = digits(num);
   196→  return d ? `tel:${d}` : null;
   197→}
   198→
   199→/**
   200→ * Build a wa.me link.
   201→ * Sri Lankan numbers starting with 0 are converted to +94.
   202→ * Otherwise, return the digits as-is.
   203→ */
   204→function waMeLink(num: string | null | undefined): string | null {
   205→  const d = digits(num);
   206→  if (!d) return null;
   207→  if (d.startsWith("0")) {
   208→    return `https://wa.me/94${d.slice(1)}`;
   209→  }
   210→  return `https://wa.me/${d}`;
   211→}
   212→
   213→/** Bar color for a rating score. */
   214→function ratingBarColor(score: number): string {
   215→  if (score >= 4.5) return "#10b981"; // emerald
   216→  if (score >= 3.5) return "#3b82f6"; // blue
   217→  if (score >= 2.5) return "#f59e0b"; // amber
   218→  return "#ef4444"; // red
   219→}
   220→
   221→/** Color for an individual rating category progress bar (matches ratingColor thresholds). */
   222→function ratingProgressClass(score: number): string {
   223→  if (score >= 4.5) return "bg-emerald-500";
   224→  if (score >= 3.5) return "bg-blue-500";
   225→  if (score >= 2.5) return "bg-amber-500";
   226→  return "bg-red-500";
   227→}
   228→
   229→function formatDate(value: string | null | undefined): string {
   230→  if (!value) return "—";
   231→  const d = new Date(value);
   232→  if (isNaN(d.getTime())) return "—";
   233→  return d.toLocaleString("en-GB", {
   234→    day: "2-digit",
   235→    month: "short",
   236→    year: "numeric",
   237→    hour: "2-digit",
   238→    minute: "2-digit",
   239→  });
   240→}
