   369→// Loose case-insensitive partial match between a row's supplier/receivedFrom
   370→// and a project name. Used for project / vote filters on the dashboard.
   371→function looselyMatches(supplier: string | null | undefined, name: string): boolean {
   372→  if (!supplier || !name) return false;
   373→  const a = supplier.toLowerCase().trim();
   374→  const b = name.toLowerCase().trim();
   375→  if (!a || !b) return false;
   376→  return a.includes(b) || b.includes(a);
   377→}

   415→/**
   416→ * Suggest an area store based on the item's store category.
   417→ * Falls back to "General" when no mapping matches.
   418→ *   BMS / Plumbing      → BMS
   419→ *   Metal / Aluminium   → Metal
   420→ *   Paint               → Paint
   421→ *   Lubricant Oil       → Oil & Lubricant
   422→ *   Tools               → General
   423→ *   (default)           → General
   424→ */
   425→function suggestStoreForCategory(category: string | null | undefined): string {
   426→  if (!category) return "General";
   427→  const cat = category.trim();
   428→  for (const [store, cats] of Object.entries(AREA_STORE_CATEGORIES)) {
   429→    if (cats.includes(cat)) return store;
   430→  }
   431→  return "General";

   447→// Generic project/vote filter applied to any row that has a supplier-like field.
   448→// Pure helper (module-level) so the React Compiler can preserve manual memoization.
   449→function applyProjVoteFilter<
   450→  T extends { receivedFrom?: string | null; supplier?: string | null }
   451→>(
   452→  rows: T[],
   453→  refField: "receivedFrom" | "supplier",
   454→  projFilter: string,
   455→  voteFilt: string,
   456→  projectById: Map<string, Project>,
   457→  projectByVote: Map<string, Project>
   458→): T[] {
   459→  let out = rows;
   460→  if (projFilter !== "all") {
   461→    const p = projectById.get(projFilter);
   462→    if (p) {
   463→      const matches = out.filter((r) =>
   464→        looselyMatches(r[refField] ?? null, p.name)
   465→      );
   466→      if (matches.length > 0) out = matches;
   467→    }
   468→  }
   469→  if (voteFilt !== "all") {
   470→    const p = projectByVote.get(voteFilt);
   471→    if (p) {
   472→      const matches = out.filter((r) =>
   473→        looselyMatches(r[refField] ?? null, p.name)
   474→      );
   475→      if (matches.length > 0) out = matches;
   476→    }
   477→  }
   478→  return out;
   479→}

   481→// Type filter — looks up the item in stores by name; only includes rows whose
   482→// store category matches. Items not in stores are excluded when a type is set.
   483→function applyTypeFilter<
   484→  T extends { itemName?: string; name?: string }
   485→>(
   486→  rows: T[],
   487→  nameField: "itemName" | "name",
   488→  tFilter: string,
   489→  storeByName: Map<string, StoreItem>
   490→): T[] {
   491→  if (tFilter === "all") return rows;
   492→  return rows.filter((r) => {
   493→    const name = r[nameField];
   494→    const s = storeByName.get((name || "").toLowerCase().trim());
   495→    return s?.category === tFilter;
   496→  });
   497→}

  1675→  // driverName repurposed to hold the "Provost Sailor Checked" name (saved
  1676→  // into the PartSupplyEntry.driverName column on the server).
  1677→  const [driverName, setDriverName] = useState("");
  1678→  const [qcCheckedBy, setQcCheckedBy] = useState("");
  1679→  // Dimension inputs — quantity is auto-calculated from H × W × D.
  1680→  const [dimH, setDimH] = useState("");
  1681→  const [dimW, setDimW] = useState("");
  1682→  const [dimD, setDimD] = useState("");

  1684→  const [submitting, setSubmitting] = useState(false);
  1685→  const { toast } = useToast();

  1687→  // Fetch stakeholders (only when dialog open) for the Provost Sailor and
  1688→  // QC Checked-By dropdowns.

  1690→    open ? "/api/stakeholders" : null
  1691→  );

  1695→  const provostSailors = useMemo(() => {
  1696→    return (stakeholders ?? []).filter((s) => {
  1697→      const cat = (s.category || "").toLowerCase();
  1698→      const dept = (s.department || "").toLowerCase();
  1699→      return cat.includes("sailor") || dept.includes("provost");
  1700→    });
  1701→  }, [stakeholders]);
  1702→
  1703→  // CCED Sailors for QC — category === "CCED Sailor" (preferred) but fall
  1704→  // back to anyone with category/appointment containing "CCED" so the
  1705→  // dropdown isn't empty when the category hasn't been seeded yet.
  1706→  const ccedSailors = useMemo(() => {
  1707→    return (stakeholders ?? []).filter((s) => {
  1708→      const cat = (s.category || "").toLowerCase();
  1709→      const appt = (s.appointment || "").toLowerCase();

  1712→        cat.includes("cced") ||
  1713→        appt.includes("cced")
  1714→      );
  1715→    });
  1716→  }, [stakeholders]);
  1717→
  1718→  // Auto-calculate Quantity from dimensions: H × W × D in the selected
  1719→  // unit (cubic meters or cubic inches).
  1720→  const computedQty = useMemo(() => {
  1721→    const h = Number(dimH);
  1722→    const w = Number(dimW);
  1723→    const d = Number(dimD);

  1725→    if (h <= 0 || w <= 0 || d <= 0) return null;
  1726→    const product = h * w * d;
  1727→    // Round to 4 decimal places to keep storage clean.
  1728→    return Math.round(product * 10000) / 10000;
  1729→  }, [dimH, dimW, dimD]);

  1732→  // Using a useEffect here would be flagged by react-hooks rules; instead
  1733→  // we derive the displayed qty directly from computedQty when dimensions
  1734→  // are present, and fall back to the manual `qty` string otherwise.
  1735→  const effectiveQty = computedQty != null ? String(computedQty) : qty;
  1736→
  1737→  async function submit() {

  1741→      return;
  1742→    }

  1744→    const res = await apiRequest(
  1745→      `/api/collection/part-supply/${order.id}/entries`,

  1748→        date,
  1749→        vehicleNumber: vehicleNumber.trim() || undefined,
  1750→        qty: q,
  1751→        // CCED Sailor name → stored in the dedicated qcCheckedBy column.
  1752→        qcCheckedBy: qcCheckedBy.trim() || undefined,
  1753→        // sailorCount field removed from the UI; preserve server contract
  1754→        // by sending 0 (model default).
  1755→        sailorCount: 0,
  1756→        // Provost Sailor name → stored in driverName column (repurposed).
  1757→        driverName: driverName.trim() || undefined,
  1758→        // driverLicense / driverPhone no longer captured in the UI.
  1759→        driverLicense: undefined,
  1760→        driverPhone: undefined,
  1761→      }
  1762→    );
  1763→    setSubmitting(false);
  1764→    if (!res.ok) {
  1765→      toast({ title: res.error || "Failed", variant: "destructive" });
  1766→      return;
  1767→    }
  1768→    setVehicleNumber("");
  1769→    setQty("");
  1770→    setQcCheckedBy("");
  1771→    setDriverName("");
  1772→    setDimH("");
  1773→    setDimW("");
  1774→    setDimD("");
  1775→    setDimUnit(DIMENSION_UNITS[0]);
  1776→    onCreated();
  1777→  }
  1778→
  1779→  return (
  1780→    <Dialog open={open} onOpenChange={onOpenChange}>
  1781→      <DialogContent className="sm:max-w-lg max-h-[92vh] overflow-y-auto scrollbar-thin">
