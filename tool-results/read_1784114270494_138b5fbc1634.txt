   132→  // Fetch process flows once to support the "Update Position" popup's default responsible person lookup

   134→  const [activeTab, setActiveTab] = useState<"raised" | "estimate">("raised");
   135→  const [search, setSearch] = useState("");
   136→  const [positionFilter, setPositionFilter] = useState("all");
   137→  const [fileNumberFilter, setFileNumberFilter] = useState<FileNumberFilter>("all");
   138→  const [addOpen, setAddOpen] = useState(false);
   139→  const [updateFile, setUpdateFile] = useState<ProjectFile | null>(null);
   140→  const [logFile, setLogFile] = useState<ProjectFile | null>(null);
   141→  const [allocOpen, setAllocOpen] = useState(false);
   142→  const [activeEstimateId, setActiveEstimateId] = useState<string | null>(null);
   143→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
   144→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");

   150→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());

   152→  const [bulkInitiateOpen, setBulkInitiateOpen] = useState(false);
   153→  const { toast } = useToast();

   155→  const toggleGroup = (key: string) => {
   156→    setExpandedGroups((prev) => {
   157→      const next = new Set(prev);
   158→      if (next.has(key)) next.delete(key);
   159→      else next.add(key);
   160→      return next;
   161→    });
   162→  };

   164→  // ─── Multi-select helpers ───

   167→  // temporal-dead-zone when the useMemo callbacks evaluate on first render.
   168→  const toggleSelect = (id: string) => {
   169→    setSelectedIds((prev) => {
   170→      const next = new Set(prev);
   171→      if (next.has(id)) next.delete(id);
   172→      else next.add(id);
   173→      return next;
   174→    });
   175→  };
   176→  const selectMany = (ids: string[], on: boolean) => {
   177→    setSelectedIds((prev) => {
   178→      const next = new Set(prev);
   179→      if (on) ids.forEach((id) => next.add(id));
   180→      else ids.forEach((id) => next.delete(id));
   181→      return next;
   182→    });
   183→  };
   184→  const clearSelection = () => setSelectedIds(new Set());
   185→
   186→  // Build a lookup of positionName -> defaultResponsiblePerson across all flows.
   187→  // Multiple flows may define the same positionName; we keep the first non-empty match.
   188→  const positionToResponsible = useMemo(() => {
   189→    const map = new Map<string, string>();
   190→    processFlows?.forEach((flow) => {
   191→      flow.steps.forEach((step) => {
   192→        if (step.positionName && step.defaultResponsiblePerson && !map.has(step.positionName)) {
   193→          map.set(step.positionName, step.defaultResponsiblePerson);
   194→        }
   195→      });
   196→    });
   197→    return map;
   198→  }, [processFlows]);
   199→
   200→  // Position summary
   201→  const positionSummary = useMemo(() => {
   202→    const map: Record<string, { count: number; cost: number }> = {};
   203→    files?.forEach((f) => {
   204→      if (!map[f.presentPosition]) map[f.presentPosition] = { count: 0, cost: 0 };
   205→      map[f.presentPosition].count++;
   206→      map[f.presentPosition].cost += f.fileCost;
   207→    });
   208→    return Object.entries(map).sort((a, b) => b[1].count - a[1].count);
   209→  }, [files]);
   210→
   211→  const totalFileCost = files?.reduce((s, f) => s + f.fileCost, 0) ?? 0;
   212→
   213→  // pending estimate items (not yet initiated) — "items to be raised PR"
   214→  const pendingEstimateItems = useMemo(() => {
   215→    const list: Estimate["items"][number][] = [];
   216→    estimates?.forEach((e) => {
   217→      e.items.forEach((it) => {
   218→        if (it.status === "Pending") list.push(it);
   219→      });
   220→    });
   221→    return list;
   222→  }, [estimates]);
   223→  const initiatedCount = useMemo(
   224→    () =>
   225→      estimates?.reduce(
   226→        (s, e) => s + e.items.filter((i) => i.status === "Initiated").length,
   227→        0
   228→      ) ?? 0,
   229→    [estimates]
   230→  );
   231→
   232→  // File-number counts for the filter toggle
   233→  const fileNumberCounts = useMemo(() => {
   234→    const all = files?.length || 0;
   235→    const withFn = files?.filter((f) => f.fileNumber && f.fileNumber.trim()).length || 0;
   236→    return { all, with: withFn, without: all - withFn };
   237→  }, [files]);
   238→
   239→  const filteredFiles = useMemo(() => {
   240→    let list = files || [];
   241→    if (search) {
   242→      const q = search.toLowerCase();
   243→      list = list.filter(
   244→        (f) =>
   245→          f.prNumber.toLowerCase().includes(q) ||
   246→          f.itemCode.toLowerCase().includes(q) ||
   247→          f.itemName.toLowerCase().includes(q) ||
   248→          (f.supplier || "").toLowerCase().includes(q) ||
   249→          (f.fileNumber || "").toLowerCase().includes(q)
   250→      );
   251→    }
   252→    if (positionFilter !== "all") {
   253→      list = list.filter((f) => f.presentPosition === positionFilter);
   254→    }
   255→    if (fileNumberFilter === "with") {
   256→      list = list.filter((f) => f.fileNumber && f.fileNumber.trim());
   257→    } else if (fileNumberFilter === "without") {
   258→      list = list.filter((f) => !f.fileNumber || !f.fileNumber.trim());
   259→    }
   260→    list = [...list].sort((a, b) => {
   261→      let cmp = 0;
   262→      if (sortKey === "fileCost") cmp = a.fileCost - b.fileCost;
   263→      else if (sortKey === "prNumber") cmp = a.prNumber.localeCompare(b.prNumber);
   264→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   265→      return sortDir === "asc" ? cmp : -cmp;
   266→    });
   267→    return list;
   268→  }, [files, search, positionFilter, fileNumberFilter, sortKey, sortDir]);
   269→
   270→  const toggleSort = (key: typeof sortKey) => {
   271→    if (sortKey === key) setSortDir(sortDir === "asc" ? "desc" : "asc");
   272→    else {
   273→      setSortKey(key);
   274→      setSortDir("desc");
   275→    }
   276→  };
   277→
   278→  // Group files by fileGroupKey (falls back to prNumber). Items sharing the same
   279→  // key collapse into a single "merged" row in the register when groupByFile is ON.
   280→  interface FileGroup {
   281→    key: string;
   282→    items: ProjectFile[];
   283→    totalCost: number;
   284→    totalQty: number;
   285→    commonSupplier: string | null;
   286→    commonCategory: string | null;
   287→    commonPosition: string | null;
   288→    commonFileNumber: string | null;
   289→    commonUnit: string | null;
   290→    commonResponsiblePerson: string | null;
   291→    firstCreatedAt: string;
   292→  }
   293→
   294→  const groupedView = useMemo<{ groups: FileGroup[]; singles: ProjectFile[] }>(() => {
   295→    if (!groupByFile) return { groups: [], singles: filteredFiles };
   296→    const map = new Map<string, ProjectFile[]>();
   297→    for (const f of filteredFiles) {
   298→      const k = (f.fileGroupKey && f.fileGroupKey.trim()) || f.prNumber;
   299→      if (!map.has(k)) map.set(k, []);
   300→      map.get(k)!.push(f);
   301→    }
   302→    const groups: FileGroup[] = [];
   303→    const singles: ProjectFile[] = [];
   304→    for (const [key, items] of map) {
   305→      if (items.length >= 2) {
   306→        const totalCost = items.reduce((s, f) => s + f.fileCost, 0);
   307→        const totalQty = items.reduce((s, f) => s + f.qty, 0);
   308→        const suppliers = new Set(items.map((f) => f.supplier).filter(Boolean));
   309→        const categories = new Set(items.map((f) => f.fileCategory));
   310→        const positions = new Set(items.map((f) => f.presentPosition));
   311→        const fileNumbers = new Set(items.map((f) => f.fileNumber).filter(Boolean));
   312→        const units = new Set(items.map((f) => f.unit));
   313→        const responsible = new Set(items.map((f) => f.responsiblePerson).filter(Boolean));
   314→        const firstCreatedAt = items
   315→          .map((f) => f.createdAt)
   316→          .sort((a, b) => new Date(b).getTime() - new Date(a).getTime())[0];
   317→        groups.push({
   318→          key,
   319→          items,
   320→          totalCost,
   321→          totalQty,
   322→          commonSupplier: suppliers.size === 1 ? Array.from(suppliers)[0]! : null,
   323→          commonCategory: categories.size === 1 ? items[0]!.fileCategory : null,
   324→          commonPosition: positions.size === 1 ? items[0]!.presentPosition : null,
   325→          commonFileNumber: fileNumbers.size === 1 ? Array.from(fileNumbers)[0]! : null,
   326→          commonUnit: units.size === 1 ? items[0]!.unit : null,
   327→          commonResponsiblePerson: responsible.size === 1 ? Array.from(responsible)[0]! : null,
   328→          firstCreatedAt,
   329→        });
   330→      } else {
   331→        singles.push(items[0]!);
   332→      }
   333→    }
   334→    return { groups, singles };
   335→  }, [filteredFiles, groupByFile]);
   336→
   337→  // ─── Multi-select derived state (depends on filteredFiles) ───
   338→  const allVisibleIds = useMemo(
   339→    () => filteredFiles.map((f) => f.id),
   340→    [filteredFiles]
   341→  );
   342→  const visibleSelectedCount = useMemo(
   343→    () => allVisibleIds.filter((id) => selectedIds.has(id)).length,
   344→    [allVisibleIds, selectedIds]
   345→  );
   346→  const allVisibleSelected =
   347→    allVisibleIds.length > 0 && visibleSelectedCount === allVisibleIds.length;
   348→  const someVisibleSelected =
   349→    visibleSelectedCount > 0 && visibleSelectedCount < allVisibleIds.length;
   350→  const toggleSelectAllVisible = () =>
   351→    selectMany(allVisibleIds, !allVisibleSelected);
   352→
   353→  // Resolve the "next position" suggestion — picks the next index after the most
   354→  // common current position among the selected files (used as a default in the
   355→  // Bulk Initiate dialog).
   356→  const suggestedNextPosition = useMemo(() => {
   357→    if (selectedIds.size === 0) return "GS Approval";
   358→    const counts = new Map<string, number>();
   359→    filteredFiles.forEach((f) => {
   360→      if (selectedIds.has(f.id)) {
   361→        counts.set(f.presentPosition, (counts.get(f.presentPosition) || 0) + 1);
   362→      }
   363→    });
   364→    let top = "";
   365→    let topN = 0;
   366→    counts.forEach((n, pos) => {
   367→      if (n > topN) {
   368→        topN = n;
   369→        top = pos;
   370→      }
   371→    });
   372→    const idx = PRESENT_POSITIONS.indexOf(top as (typeof PRESENT_POSITIONS)[number]);
   373→    if (idx >= 0 && idx < PRESENT_POSITIONS.length - 1) {
   374→      return PRESENT_POSITIONS[idx + 1]!;
   375→    }
   376→    return top || "GS Approval";
   377→  }, [filteredFiles, selectedIds]);
   378→
   379→  // Bulk Initiate: updates the present position of every selected file via
   380→  // PUT /api/projects/{projectId}/files/{fileId}. Reports per-file success/failure
   381→  // counts via toast and refreshes the file register + summary afterwards.
   382→  const handleBulkInitiate = async (input: {
   383→    newPosition: string;
   384→    changedBy: string;
   385→    comment: string;
   386→  }) => {
   387→    const selected = filteredFiles.filter((f) => selectedIds.has(f.id));
   388→    if (selected.length === 0) {
   389→      toast({ title: "No files selected", variant: "destructive" });
   390→      return;
   391→    }
   392→    const results = await Promise.all(
   393→      selected.map((f) =>
   394→        apiRequest(`/api/projects/${project.id}/files/${f.id}`, "PUT", {
   395→          presentPosition: input.newPosition,
   396→          comment: input.comment.trim() || `Bulk initiated → ${input.newPosition}`,
   397→          changedBy: input.changedBy.trim() || "Officer",
   398→        })
   399→      )
   400→    );
   401→    const failed = results.filter((r) => !r.ok);
   402→    if (failed.length === 0) {
   403→      toast({
   404→        title: `Initiated ${selected.length} file(s) → ${input.newPosition}`,
   405→        description: "Position updated for all selected files",
   406→      });
   407→    } else {
   408→      toast({
   409→        title: `${failed.length} of ${selected.length} update(s) failed`,
   410→        description: failed[0]?.error,
   411→        variant: "destructive",
   412→      });
   413→    }
   414→    clearSelection();
   415→    setBulkInitiateOpen(false);
   416→    refresh();
   417→    onChanged();
   418→  };
   419→
   420→  const handleDelete = async (file: ProjectFile) => {
   421→    if (!confirm(`Delete file ${file.prNumber} (${file.itemName})?`)) return;
   422→    const res = await apiRequest(`/api/projects/${project.id}/files/${file.id}`, "DELETE");
   423→    if (res.ok) {
   424→      toast({ title: "File deleted" });
   425→      refresh();
   426→      onChanged();
   427→    } else {
   428→      toast({ title: "Delete failed", description: res.error, variant: "destructive" });
   429→    }
   430→  };
   431→
   432→  const exportExcel = () => {
   433→    const rows = filteredFiles.map((f) => ({
   434→      "PR Number": f.prNumber,
   435→      "File Number": f.fileNumber || "",
   436→      "Item Code": f.itemCode,
   437→      "Item Name": f.itemName,
   438→      Unit: f.unit,
   439→      Qty: f.qty,
   440→      "Unit Price": f.unitPrice,
   441→      "File Cost": f.fileCost,
   442→      Supplier: f.supplier || "",
   443→      "Present Position": f.presentPosition,
   444→      "Responsible Person": f.responsiblePerson || "",
   445→      Category: f.fileCategory,
   446→    }));
   447→    const headers = Object.keys(rows[0] || { "PR Number": "" });
   448→    const csv = [
   449→      headers.join(","),
   450→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h] ?? ""}"`).join(",")),
   451→    ].join("\n");
   452→    const blob = new Blob([csv], { type: "text/csv" });
   453→    const url = URL.createObjectURL(blob);
   454→    const a = document.createElement("a");
   455→    a.href = url;
   456→    a.download = `${project.name}-files.csv`;
   457→    a.click();
   458→    URL.revokeObjectURL(url);
   459→  };
   460→
   461→  // pick the first processed estimate (or any) for the "PR from Estimate" tab
   462→  const estimateForTab = estimates?.find((e) => e.status === "Processed") || estimates?.[0] || null;
   463→
   464→  // Single file row renderer — shared between the ungrouped view and the
   465→  // "singles" list of the grouped view. Lives inside the component so it can
   466→  // access setLogFile / setUpdateFile / handleDelete / project.id / refresh.
   467→  const renderSingleFileRow = (f: ProjectFile) => {

   469→    const isSelected = selectedIds.has(f.id);
   470→    const itemCost = (f.unitPrice || 0) * (f.qty || 0);
   471→    return (
   472→      <TableRow
   473→        key={f.id}
   474→        className={cn(
   475→          "border-border cursor-pointer hover:bg-secondary/40",
   476→          isSelected && "bg-primary/[0.06]"
   477→        )}
   478→        onClick={() => setLogFile(f)}
   479→      >
   480→        <TableCell className="w-10" onClick={(e) => e.stopPropagation()}>
   481→          <Checkbox
   482→            checked={isSelected}
   483→            onCheckedChange={() => toggleSelect(f.id)}
   484→            aria-label={`Select ${f.prNumber}`}
   485→          />
   486→        </TableCell>
   487→        <TableCell className="font-mono text-xs text-foreground">{f.prNumber}</TableCell>
   488→        <TableCell>
   489→          {hasFileNumber ? (
   490→            <span className="inline-flex items-center gap-1 font-mono text-xs text-emerald-400">
   491→              <Hash className="h-3 w-3" />
   492→              {f.fileNumber}
   493→            </span>
   494→          ) : (
   495→            <span className="text-muted-foreground">—</span>
   496→          )}
   497→        </TableCell>
   498→        <TableCell className="font-mono text-xs">{f.itemCode}</TableCell>
   499→        <TableCell className="text-foreground">{f.itemName}</TableCell>
   500→        <TableCell className="text-right">{f.qty} {f.unit}</TableCell>
   501→        <TableCell className="text-right">{formatCurrency(f.unitPrice)}</TableCell>
   502→        <TableCell className="text-right text-foreground/90">{formatCurrency(itemCost)}</TableCell>
   503→        <TableCell className="text-right font-medium text-foreground">{formatCurrency(f.fileCost)}</TableCell>
   504→        <TableCell className="text-xs">{f.supplier || "—"}</TableCell>
   505→        <TableCell>
   506→          <div className="flex flex-col gap-0.5">
   507→            <PositionBadge position={f.presentPosition} />
   508→            {f.responsiblePerson && (
   509→              <span className="inline-flex items-center gap-1 text-[10px] text-muted-foreground">
   510→                <User className="h-2.5 w-2.5" />
   511→                {f.responsiblePerson}
   512→              </span>
   513→            )}
   514→          </div>
   515→        </TableCell>
   516→        <TableCell><CategoryBadge category={f.fileCategory} /></TableCell>
   517→        <TableCell className="text-right">
   518→          <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   519→            <QuickUpdatePopover
   520→              files={[f]}
   521→              projectId={project.id}
   522→              onSaved={() => { refresh(); onChanged(); }}
   523→            />
   524→            <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setUpdateFile(f)} title="Edit File (full dialog)"><Pencil className="h-3.5 w-3.5" /></Button>
   525→            <Button variant="ghost" size="icon" className="h-7 w-7 text-red-400 hover:text-red-500" onClick={() => handleDelete(f)} title="Delete file"><Trash2 className="h-3.5 w-3.5" /></Button>
   526→          </div>
   527→        </TableCell>
   528→      </TableRow>
   529→    );
   530→  };
   531→
   532→  if (!open) return null;
   533→
   534→  return (
   535→    <div className="fixed inset-0 z-50 bg-background flex flex-col">

  1224→        title: positionChanged ? "Position updated" : "File updated",
  1225→        description: positionChanged ? `${currentPosition} → ${nextPosition}` : undefined,

  1257→                  {currentPosition}

  1376→function QuickUpdatePopover({
  1377→  files,
  1378→  projectId,
  1379→  onSaved,
  1380→  isGroup,
  1381→}: {
  1382→  files: ProjectFile[];
  1383→  projectId: string;
  1384→  onSaved: () => void;
  1385→  isGroup?: boolean;
  1386→}) {
  1387→  const [open, setOpen] = useState(false);
  1388→  const [newPosition, setNewPosition] = useState<string>(files[0]?.presentPosition || "PR Raised");
  1389→  const [comment, setComment] = useState("");

  1393→
  1394→  const currentPosition = files[0]?.presentPosition || "";
  1395→  const positionChanged = newPosition !== currentPosition;
  1396→  const isMultiFile = isGroup && files.length > 1;
  1397→

  1399→    if (!newPosition) {
  1400→      toast({ title: "Select a new position", variant: "destructive" });
  1401→      return;
  1402→    }
  1403→    if (isMultiFile) {
  1404→      const ok = window.confirm(`Update position for all ${files.length} items in this file?`);
  1405→      if (!ok) return;
  1406→    }

  1408→    const defaultComment = positionChanged
  1409→      ? `Position changed: ${currentPosition} → ${newPosition}`
  1410→      : "Position updated";
  1411→    const results = await Promise.all(
  1412→      files.map((f) =>
  1413→        apiRequest(`/api/projects/${projectId}/files/${f.id}`, "PUT", {
  1414→          presentPosition: newPosition,
  1415→          comment: comment.trim() || defaultComment,
  1416→          changedBy: changedBy.trim() || "Officer",
  1417→        })
  1418→      )

  1424→        title: isMultiFile ? `Position updated for ${files.length} items` : "Position updated",
  1425→        description: positionChanged ? `${currentPosition} → ${newPosition}` : undefined,
  1426→      });
  1427→      setOpen(false);
  1428→      setComment("");
  1429→      setChangedBy("Officer");
  1430→      onSaved();
  1431→    } else {
  1432→      toast({
  1433→        title: `${failed.length} of ${files.length} update(s) failed`,
  1434→        description: failed[0]?.error,
  1435→        variant: "destructive",
  1436→      });
  1437→    }
  1438→  };
  1439→

  1525→// ─── Bulk Initiate Dialog (multi-select → pick target position → bulk PUT) ───
  1526→function BulkInitiateDialog({
  1527→  open,
  1528→  onOpenChange,
  1529→  selectedCount,
  1530→  defaultPosition,
  1531→  onConfirm,
  1532→}: {
  1533→  open: boolean;
  1534→  onOpenChange: (o: boolean) => void;
  1535→  selectedCount: number;
  1536→  defaultPosition: string;
  1537→  onConfirm: (input: { newPosition: string; changedBy: string; comment: string }) => Promise<void>;
  1538→}) {
  1539→  const [newPosition, setNewPosition] = useState(defaultPosition);
  1540→  const [changedBy, setChangedBy] = useState("Officer");
  1541→  const [comment, setComment] = useState("");
  1542→  const [saving, setSaving] = useState(false);
  1543→  const { toast } = useToast();
  1544→
  1545→  // The parent uses a `key` based on the selected count + suggested position to
  1546→  // remount this dialog whenever the user opens it with a fresh set of selections,
  1547→  // so `useState(defaultPosition)` always picks up the latest suggested position.
  1548→
  1549→  const submit = async () => {
  1550→    if (!newPosition) {
  1551→      toast({ title: "Select a new position", variant: "destructive" });
  1552→      return;
  1553→    }
  1554→    if (selectedCount === 0) {
  1555→      toast({ title: "No files selected", variant: "destructive" });
  1556→      return;
  1557→    }
  1558→    setSaving(true);
  1559→    await onConfirm({ newPosition, changedBy, comment });
  1560→    setSaving(false);
  1561→    // Reset form after submit (parent closes the dialog)
  1562→    setComment("");
  1563→    setChangedBy("Officer");
  1564→  };
  1565→
  1566→  return (
  1567→    <Dialog open={open} onOpenChange={(o) => { if (!saving) onOpenChange(o); }}>
  1568→      <DialogContent className="max-w-md bg-card border-border">
  1569→        <DialogHeader>
  1570→          <DialogTitle className="text-foreground flex items-center gap-2">
  1571→            <CheckSquare className="h-5 w-5 text-primary" />
  1572→            Initiate Selected Files
  1573→          </DialogTitle>
  1574→          <DialogDescription className="text-muted-foreground">
  1575→            Move <span className="font-medium text-foreground">{selectedCount}</span> selected file{selectedCount === 1 ? "" : "s"} to a new present position. A log entry is created for each file.
  1576→          </DialogDescription>
  1577→        </DialogHeader>
  1578→
  1579→        <div className="space-y-4 py-2">
  1580→          <div className="space-y-1.5">
  1581→            <Label className="text-xs text-muted-foreground">New Position</Label>
  1582→            <Select value={newPosition} onValueChange={setNewPosition}>
  1583→              <SelectTrigger className="bg-background border-border">
  1584→                <SelectValue />
  1585→              </SelectTrigger>
  1586→              <SelectContent>
  1587→                {PRESENT_POSITIONS.map((p) => (
  1588→                  <SelectItem key={p} value={p}>{p}</SelectItem>
  1589→                ))}
  1590→              </SelectContent>
  1591→            </Select>
  1592→            {defaultPosition && (
  1593→              <p className="text-[11px] text-muted-foreground">
  1594→                Suggested next position: <span className="text-foreground">{defaultPosition}</span>
  1595→              </p>
  1596→            )}
  1597→          </div>
  1598→
  1599→          <div className="space-y-1.5">
  1600→            <Label className="text-xs text-muted-foreground">Changed By</Label>
  1601→            <Input
  1602→              value={changedBy}
  1603→              onChange={(e) => setChangedBy(e.target.value)}
  1604→              placeholder="Your name / rank"
  1605→              className="bg-background border-border"
  1606→            />
  1607→          </div>
  1608→
  1609→          <div className="space-y-1.5">
  1610→            <Label className="text-xs text-muted-foreground">Comment (optional)</Label>
  1611→            <Input
  1612→              value={comment}
  1613→              onChange={(e) => setComment(e.target.value)}
  1614→              placeholder={`Bulk initiated → ${newPosition}`}
  1615→              className="bg-background border-border"
  1616→            />
  1617→          </div>
  1618→        </div>
  1619→
  1620→        <DialogFooter>
  1621→          <Button variant="outline" onClick={() => onOpenChange(false)} disabled={saving}>Cancel</Button>
  1622→          <Button onClick={submit} disabled={saving || selectedCount === 0} className="gap-1.5">
  1623→            {saving ? <Loader2 className="h-4 w-4 animate-spin" /> : <CheckSquare className="h-4 w-4" />}
  1624→            Initiate {selectedCount} File{selectedCount === 1 ? "" : "s"}
  1625→          </Button>
  1626→        </DialogFooter>
  1627→      </DialogContent>
  1628→    </Dialog>
  1629→  );

  1633→function AllocationHistoryDialog({ project, open, onOpenChange, onChanged }: { project: Project; open: boolean; onOpenChange: (o: boolean) => void; onChanged: () => void }) {
  1634→  const { data: allocations, refresh } = useFetch<Allocation[]>(`/api/projects/${project.id}/allocations`);
  1635→  const [saving, setSaving] = useState(false);

  1640→  const committed = allocations?.filter((a) => a.type === "Commitment").reduce((s, a) => s + a.amount, 0) ?? 0;
  1641→  const actual = allocations?.filter((a) => a.type === "Actual").reduce((s, a) => s + a.amount, 0) ?? 0;

  1652→      toast({ title: "Allocation added" });
