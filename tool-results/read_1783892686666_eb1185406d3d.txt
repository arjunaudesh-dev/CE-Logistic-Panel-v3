   110→// ────────────────────────────────────────────────────────────
   111→// Main component
   112→// ────────────────────────────────────────────────────────────
   113→export function StoresModule() {
   114→  const [tab, setTab] = useState<"zones" | "stores">("zones");
   115→
   116→  // Shared fetches
   117→  const {
   118→    data: zones,
   119→    loading: zLoading,
   120→    error: zError,
   121→    refresh: refreshZones,
   122→  } = useFetch<StoreZone[]>("/api/stores/zones");
   123→  const {
   124→    data: items,
   125→    loading: iLoading,
   126→    error: iError,
   127→    refresh: refreshItems,
   128→  } = useFetch<StoreItem[]>("/api/stores");
   129→
   130→  // Stores-tab filter state
   131→  const [category, setCategory] = useState<string>("All");
   132→  const [search, setSearch] = useState("");
   133→  const [zoneFilter, setZoneFilter] = useState("all");
   134→  const [stockFilter, setStockFilter] = useState<StockFilter>("all");
   135→  const [sortKey, setSortKey] = useState<SortKey>("description");
   136→  const [sortDir, setSortDir] = useState<"asc" | "desc">("asc");
   137→  const [page, setPage] = useState(1);
   138→
   139→  // Dialogs
   140→  const [formOpen, setFormOpen] = useState(false);
   141→  const [editing, setEditing] = useState<StoreItem | null>(null);
   142→  const [bulkOpen, setBulkOpen] = useState(false);
   143→  const [viewing, setViewing] = useState<StoreItem | null>(null);
   144→
   145→  const { toast } = useToast();
   146→
   147→  // Derived: filtered + sorted list
   148→  const filtered = useMemo(() => {
   149→    if (!items) return [];
   150→    let list = items.slice();
   151→    if (category !== "All") list = list.filter((i) => i.category === category);
   152→    if (zoneFilter !== "all") list = list.filter((i) => i.zoneName === zoneFilter);
   153→    const q = search.trim().toLowerCase();
   154→    if (q) list = list.filter((i) => i.description.toLowerCase().includes(q));
   155→    list = list.filter((i) => {
   156→      if (stockFilter === "out") return i.qty <= 0;
   157→      if (stockFilter === "low")
   158→        return i.minStock > 0 && i.qty > 0 && i.qty <= i.minStock;
   159→      if (stockFilter === "in") return i.minStock === 0 || i.qty > i.minStock;
   160→      if (stockFilter === "over") return i.maxStock > 0 && i.qty >= i.maxStock;
   161→      return true;
   162→    });
   163→    list.sort((a, b) => {
   164→      let cmp = 0;
   165→      if (sortKey === "description") cmp = a.description.localeCompare(b.description);
   166→      else if (sortKey === "qty") cmp = a.qty - b.qty;
   167→      else if (sortKey === "cost") cmp = a.totalCost - b.totalCost;
   168→      else if (sortKey === "category") cmp = a.category.localeCompare(b.category);
   169→      return sortDir === "asc" ? cmp : -cmp;
   170→    });
   171→    return list;
   172→  }, [items, category, zoneFilter, search, stockFilter, sortKey, sortDir]);
   173→
   174→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   175→  const safePage = Math.min(page, totalPages);
   176→  const pageItems = filtered.slice(
   177→    (safePage - 1) * PAGE_SIZE,
   178→    safePage * PAGE_SIZE
   179→  );
   180→
   181→  // Stats reflect the currently filtered set
   182→  const stats = useMemo(() => {
   183→    const net = filtered.reduce((s, i) => s + i.totalCost, 0);
   184→    const task = filtered.length;
   185→    const avail = filtered.filter((i) => i.qty > 0).length;
   186→    const team = filtered.filter((i) => i.minStock > 0 && i.qty <= i.minStock).length;
   187→    return { net, task, avail, team };
   188→  }, [filtered]);
   189→
   190→  // Wrap filter setters so any change also resets pagination to page 1
   191→  const setCategoryR = (c: string) => {
   192→    setCategory(c);
   193→    setPage(1);
   194→  };
   195→  const setSearchR = (s: string) => {
   196→    setSearch(s);
   197→    setPage(1);
   198→  };
   199→  const setZoneFilterR = (z: string) => {
   200→    setZoneFilter(z);
   201→    setPage(1);
   202→  };
   203→  const setStockFilterR = (s: StockFilter) => {
   204→    setStockFilter(s);
   205→    setPage(1);
   206→  };
   207→  const setSortKeyR = (k: SortKey) => {
   208→    setSortKey(k);
   209→    setPage(1);
   210→  };
   211→  const setSortDirR = (d: "asc" | "desc") => {
   212→    setSortDir(d);
   213→    setPage(1);
   214→  };
   215→  const toggleSortR = (k: SortKey) => {
   216→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   217→    else {
   218→      setSortKey(k);
   219→      setSortDir("asc");
   220→    }
   221→    setPage(1);
   222→  };
   223→
   224→  // Handlers
   225→  const handleZoneClick = (zoneName: string) => {
   226→    setZoneFilter(zoneName);
   227→    setCategory("All");
   228→    setStockFilter("all");
   229→    setSearch("");
   230→    setPage(1);
   231→    setTab("stores");
   232→    toast({
   233→      title: `Browsing ${zoneName}`,
   234→      description: "Switched to Stores tab with zone filter applied.",
   235→    });
   236→  };
   237→
   238→  const openAdd = () => {
   239→    setEditing(null);
   240→    setFormOpen(true);
   241→  };
   242→
   243→  const openEdit = (item: StoreItem) => {
   244→    setEditing(item);
   245→    setFormOpen(true);
   246→  };
   247→
   248→  const handleSaved = () => {
   249→    setFormOpen(false);
   250→    setEditing(null);
   251→    refreshItems();
   252→    refreshZones();
   253→    toast({
   254→      title: editing ? "Item updated" : "Item added",
   255→      description: editing
   256→        ? "Store item has been updated successfully."
   257→        : "New store item has been added to the store.",
   258→    });
   259→  };
   260→
   261→  const toggleSort = (k: SortKey) => {
   262→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   263→    else {
   264→      setSortKey(k);
   265→      setSortDir("asc");
   266→    }
   267→  };
   268→
   269→  return (
   270→    <div>
   271→      <PageHeader
   272→        title="Store Management"
   273→        description="Zones, inventory & stock-level monitoring across the dockyard"
   274→        icon={Warehouse}
   275→        actions={
   276→          <>
   277→            <Button
   278→              variant="outline"
   279→              size="sm"
   280→              onClick={() => setBulkOpen(true)}
   281→              className="gap-1.5"
   282→            >
   283→              <Upload className="h-4 w-4" /> Bulk Upload
   284→            </Button>
   285→            <Button onClick={openAdd} size="sm" className="gap-1.5">
   286→              <Plus className="h-4 w-4" /> Add Item
   287→            </Button>
   288→          </>
   289→        }
   290→      />
   291→
   292→      <Tabs
   293→        value={tab}
   294→        onValueChange={(v) => setTab(v as "zones" | "stores")}
   295→        className="w-full"
   296→      >
   297→        <TabsList>
   298→          <TabsTrigger value="zones" className="gap-1.5">
   299→            <Boxes className="h-4 w-4" /> Zones
   300→          </TabsTrigger>
   301→          <TabsTrigger value="stores" className="gap-1.5">
   302→            <Package className="h-4 w-4" /> Stores
   303→          </TabsTrigger>
   304→        </TabsList>
   305→
   306→        {/* ─── Zones tab ─── */}
   307→        <TabsContent value="zones" className="mt-4">
   308→          <ZonesTab
   309→            zones={zones}
   310→            loading={zLoading}
   311→            error={zError}
   312→            onZoneClick={handleZoneClick}
   313→          />
   314→        </TabsContent>
   315→
   316→        {/* ─── Stores tab ─── */}
   317→        <TabsContent value="stores" className="mt-4">
   318→          <StoresTab
   319→            items={items}
   320→            loading={iLoading}
   321→            error={iError}
   322→            zones={zones}
   323→            stats={stats}
   324→            category={category}
   325→            setCategory={setCategoryR}
   326→            search={search}
   327→            setSearch={setSearchR}
   328→            zoneFilter={zoneFilter}
   329→            setZoneFilter={setZoneFilterR}
   330→            stockFilter={stockFilter}
   331→            setStockFilter={setStockFilterR}
   332→            sortKey={sortKey}
   333→            sortDir={sortDir}
   334→            setSortKey={setSortKeyR}
   335→            setSortDir={setSortDirR}
   336→            toggleSort={toggleSortR}
   337→            filtered={filtered}
   338→            pageItems={pageItems}
   339→            safePage={safePage}
   340→            totalPages={totalPages}
   341→            setPage={setPage}
   342→            onEdit={openEdit}
   343→            onView={(it) => setViewing(it)}
   344→            onAdd={openAdd}
   345→          />
   346→        </TabsContent>
   347→      </Tabs>
   348→
   349→      {/* Add / Edit form */}
   350→      <ItemFormDialog
   351→        key={editing ? `edit-${editing.id}` : "new"}
   352→        open={formOpen}
   353→        onOpenChange={(o) => {
   354→          setFormOpen(o);
   355→          if (!o) setEditing(null);
   356→        }}
   357→        editing={editing}
   358→        zones={zones}
   359→        onSaved={handleSaved}
   360→      />
   361→
   362→      {/* Bulk upload */}
   363→      <BulkUploadDialog
   364→        open={bulkOpen}
   365→        onOpenChange={setBulkOpen}
   366→        zones={zones}
   367→        onDone={() => {
   368→          refreshItems();
   369→          refreshZones();
   370→        }}
   371→      />
   372→
   373→      {/* View dialog */}
   374→      <ViewItemDialog
   375→        item={viewing}
   376→        onOpenChange={(o) => !o && setViewing(null)}
   377→        onEdit={(it) => {
   378→          setViewing(null);
   379→          openEdit(it);
   380→        }}
   381→      />
   382→    </div>
   383→  );
   384→}

   386→// ────────────────────────────────────────────────────────────
   387→// Zones tab — dashboard view with scaled tiles

   389→interface ZonesTabProps {
   390→  zones: StoreZone[] | null;
   391→  loading: boolean;
   392→  error: string | null;
   393→  onZoneClick: (zoneName: string) => void;

   395→
   396→function ZonesTab({ zones, loading, error, onZoneClick }: ZonesTabProps) {
   397→  if (loading) return <LoadingState label="Loading zones..." />;
   398→  if (error) return <ErrorState message={error} />;
   399→  if (!zones || zones.length === 0)
   400→    return (
   401→      <EmptyState
   402→        icon={Boxes}
   403→        title="No zones configured"
   404→        description="Store zones will appear here as tiles once data is seeded."
   405→      />
   406→    );
   407→
   408→  const totalValue = zones.reduce((s, z) => s + z.totalValue, 0);
   409→  const totalItems = zones.reduce((s, z) => s + z.itemCount, 0);
   410→  const totalLow = zones.reduce((s, z) => s + z.lowStockCount, 0);
   411→
   412→  return (
   413→    <div>
   414→      {/* Mini summary strip */}
   415→      <div className="grid grid-cols-3 gap-3 mb-5">
   416→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   417→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   418→            Total Stock Value
   419→          </p>
   420→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   421→            {formatCurrency(totalValue)}
   422→          </p>
   423→        </div>
   424→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   425→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   426→            Total Items
   427→          </p>
   428→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   429→            {formatNumber(totalItems)}
   430→          </p>
   431→        </div>
   432→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   433→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   434→            Low Stock Items
   435→          </p>
   436→          <p
   437→            className={cn(
   438→              "text-base sm:text-xl font-semibold mt-1",
   439→              totalLow > 0 ? "text-amber-400" : "text-emerald-400"
   440→            )}
   441→          >
   442→            {formatNumber(totalLow)}
   443→          </p>
   444→        </div>
   445→      </div>
   446→
   447→      {/* Tiles grid */}
   448→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   449→        {zones.map((z) => {
   450→          const status =
   451→            z.lowStockCount >= 3
   452→              ? "critical"
   453→              : z.lowStockCount > 0
   454→                ? "attention"
   455→                : "healthy";
   456→          const statusClasses =
   457→            status === "critical"
   458→              ? "border-red-500/40 hover:border-red-500/70"
   459→              : status === "attention"
   460→                ? "border-amber-500/40 hover:border-amber-500/70"
   461→                : "border-emerald-500/30 hover:border-emerald-500/60";
   462→          const minHeight = 120 + Math.min(z.totalValue / 100000, 80);
   463→
   464→          return (
   465→            <button
   466→              key={z.id}
   467→              onClick={() => onZoneClick(z.name)}
   468→              style={{ minHeight }}
   469→              className={cn(
   470→                "group bg-card border rounded-xl p-4 sm:p-5 text-left transition-all flex flex-col justify-between",
   471→                "hover:shadow-lg hover:shadow-black/20 hover:-translate-y-0.5",
   472→                statusClasses
   473→              )}
   474→            >
   475→              {/* Header */}
   476→              <div className="flex items-start justify-between gap-2 mb-3">
   477→                <div className="min-w-0">
   478→                  <h3 className="font-semibold text-foreground group-hover:text-primary transition-colors truncate text-sm sm:text-base">
   479→                    {z.name}
   480→                  </h3>
   481→                  <p className="text-[10px] sm:text-xs text-muted-foreground mt-0.5">
   482→                    {z.itemCount} items tracked
   483→                  </p>
   484→                </div>
   485→                <CustomBadge
   486→                  className={cn(
   487→                    "border-0 shrink-0",
   488→                    status === "critical" &&
   489→                      "bg-red-500/15 text-red-400",
   490→                    status === "attention" &&
   491→                      "bg-amber-500/15 text-amber-400",
   492→                    status === "healthy" &&
   493→                      "bg-emerald-500/15 text-emerald-400"
   494→                  )}
   495→                >
   496→                  {z.type}
   497→                </CustomBadge>
   498→              </div>
   499→
   500→              {/* Value */}
   501→              <div className="mb-3">
   502→                <p className="text-[10px] uppercase tracking-wide text-muted-foreground">
   503→                  Stock Value
   504→                </p>
   505→                <p className="text-lg sm:text-2xl font-bold text-foreground">
   506→                  {formatCurrency(z.totalValue)}
   507→                </p>
   508→              </div>
   509→
   510→              {/* Status chips */}
   511→              <div className="flex flex-wrap gap-1.5 mt-auto">
   512→                <ZoneChip
   513→                  label="Running out"
   514→                  value={z.lowStockCount}
   515→                  tone={z.lowStockCount > 0 ? "red" : "muted"}
   516→                />
   517→                <ZoneChip
   518→                  label="Received"
   519→                  value={z.recentReceipts}
   520→                  tone={z.recentReceipts > 0 ? "green" : "muted"}
   521→                />
   522→                <ZoneChip
   523→                  label="To collect"
   524→                  value={z.pendingCollection}
   525→                  tone={z.pendingCollection > 0 ? "blue" : "muted"}
   526→                />
   527→                <ZoneChip
   528→                  label="Non-moving"
   529→                  value={z.nonMovingCount}
   530→                  tone={z.nonMovingCount > 0 ? "amber" : "muted"}
   531→                />
   532→              </div>
   533→            </button>
   534→          );
   535→        })}
   536→      </div>
   537→
   538→      <p className="text-xs text-muted-foreground mt-4 text-center">
   539→        Click a zone tile to drill into its detailed inventory.
   540→      </p>
   541→    </div>
   542→  );
   543→}
   544→
   545→function ZoneChip({
   546→  label,
   547→  value,
   548→  tone,
   549→}: {
   550→  label: string;
   551→  value: number;
   552→  tone: "red" | "amber" | "green" | "blue" | "muted";
   553→}) {
   554→  const toneClasses = {
   555→    red: "bg-red-500/10 text-red-400 border-red-500/30",
   556→    amber: "bg-amber-500/10 text-amber-400 border-amber-500/30",
   557→    green: "bg-emerald-500/10 text-emerald-400 border-emerald-500/30",
   558→    blue: "bg-blue-500/10 text-blue-400 border-blue-500/30",
   559→    muted: "bg-secondary/60 text-muted-foreground border-border",
   560→  };
   561→  return (
   562→    <span
   563→      className={cn(
   564→        "inline-flex items-center gap-1 rounded-md border px-1.5 py-0.5 text-[10px] sm:text-xs font-medium",
   565→        toneClasses[tone]
   566→      )}
   567→    >
   568→      <span className="opacity-80">{label}</span>
   569→      <span className="font-semibold">{value}</span>
   570→    </span>
   571→  );
   572→}

   606→function StoresTab(props: StoresTabProps) {
   607→  const {
   608→    items,
   609→    loading,
   610→    error,
   611→    zones,

  1359→// Simple RFC-4180-aware CSV parser
  1360→function parseCSV(text: string): string[][] {
  1361→  const rows: string[][] = [];
  1362→  let row: string[] = [];
  1363→  let cell = "";
  1364→  let inQuotes = false;
  1365→  for (let i = 0; i < text.length; i++) {
  1366→    const c = text[i];
  1367→    if (inQuotes) {
  1368→      if (c === '"') {
  1369→        if (text[i + 1] === '"') {
  1370→          cell += '"';
  1371→          i++;
  1372→        } else {
  1373→          inQuotes = false;
  1374→        }
  1375→      } else {
  1376→        cell += c;
  1377→      }
  1378→    } else {
  1379→      if (c === '"') {
  1380→        inQuotes = true;
  1381→      } else if (c === ",") {
  1382→        row.push(cell);
  1383→        cell = "";
  1384→      } else if (c === "\n") {
  1385→        row.push(cell);
  1386→        rows.push(row);
  1387→        row = [];
  1388→        cell = "";
  1389→      } else if (c === "\r") {
  1390→        // skip CR
  1391→      } else {
  1392→        cell += c;
  1393→      }
  1394→    }
  1395→  }
  1396→  if (cell !== "" || row.length > 0) {
  1397→    row.push(cell);
  1398→    rows.push(row);
  1399→  }
  1400→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1401→}
