   161→   161→      return true;
   162→   162→    });

   165→   165→      if (sortKey === "description") cmp = a.description.localeCompare(b.description);
   166→   166→      else if (sortKey === "qty") cmp = a.qty - b.qty;
   167→   167→      else if (sortKey === "cost") cmp = a.totalCost - b.totalCost;
   168→   168→      else if (sortKey === "category") cmp = a.category.localeCompare(b.category);
   169→   169→      return sortDir === "asc" ? cmp : -cmp;
   170→   170→    });
   171→   171→    return list;
   172→   172→  }, [items, category, zoneFilter, search, stockFilter, sortKey, sortDir]);

   175→   175→  const safePage = Math.min(page, totalPages);

   177→   177→    (safePage - 1) * PAGE_SIZE,
   178→   178→    safePage * PAGE_SIZE

   187→   187→    return { net, task, avail, team };

   197→   197→    setPage(1);
   198→   198→  };
   199→   199→  const setZoneFilterR = (z: string) => {
   200→   200→    setZoneFilter(z);
   201→   201→    setPage(1);
   202→   202→  };
   203→   203→  const setStockFilterR = (s: StockFilter) => {
   204→   204→    setStockFilter(s);
   205→   205→    setPage(1);
   206→   206→  };
   207→   207→  const setSortKeyR = (k: SortKey) => {
   208→   208→    setSortKey(k);
   209→   209→    setPage(1);
   210→   210→  };
   211→   211→  const setSortDirR = (d: "asc" | "desc") => {
   212→   212→    setSortDir(d);
   213→   213→    setPage(1);
   214→   214→  };
   215→   215→  const toggleSortR = (k: SortKey) => {
   216→   216→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   217→   217→    else {
   218→   218→      setSortKey(k);
   219→   219→      setSortDir("asc");
   220→   220→    }
   221→   221→    setPage(1);
   222→   222→  };
   223→   223→
   224→   224→  // Handlers
   225→   225→  const handleZoneClick = (zoneName: string) => {
   226→   226→    setZoneFilter(zoneName);
   227→   227→    setCategory("All");
   228→   228→    setStockFilter("all");
   229→   229→    setSearch("");
   230→   230→    setPage(1);
   231→   231→    setTab("stores");
   232→   232→    toast({
   233→   233→      title: `Browsing ${zoneName}`,
   234→   234→      description: "Switched to Stores tab with zone filter applied.",
   235→   235→    });
   236→   236→  };
   237→   237→
   238→   238→  const openAdd = () => {
   239→   239→    setEditing(null);
   240→   240→    setFormOpen(true);
   241→   241→  };
   242→   242→
   243→   243→  const openEdit = (item: StoreItem) => {
   244→   244→    setEditing(item);
   245→   245→    setFormOpen(true);
   246→   246→  };
   247→   247→
   248→   248→  const handleSaved = () => {
   249→   249→    setFormOpen(false);
   250→   250→    setEditing(null);
   251→   251→    refreshItems();
   252→   252→    refreshZones();
   253→   253→    toast({
   254→   254→      title: editing ? "Item updated" : "Item added",
   255→   255→      description: editing
   256→   256→        ? "Store item has been updated successfully."
   257→   257→        : "New store item has been added to the store.",
   258→   258→    });
   259→   259→  };
   260→   260→
   261→   261→  const toggleSort = (k: SortKey) => {
   262→   262→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   263→   263→    else {
   264→   264→      setSortKey(k);
   265→   265→      setSortDir("asc");
   266→   266→    }
   267→   267→  };
   268→   268→
   269→   269→  return (

   274→   274→        icon={Warehouse}
   275→   275→        actions={

   286→   286→              <Plus className="h-4 w-4" /> Add Item

   330→   330→            stockFilter={stockFilter}
   331→   331→            setStockFilter={setStockFilterR}
   332→   332→            sortKey={sortKey}
   333→   333→            sortDir={sortDir}
   334→   334→            setSortKey={setSortKeyR}
   335→   335→            setSortDir={setSortDirR}
   336→   336→            toggleSort={toggleSortR}
   337→   337→            filtered={filtered}
   338→   338→            pageItems={pageItems}
   339→   339→            safePage={safePage}
   340→   340→            totalPages={totalPages}

   356→   356→        }}
   357→   357→        editing={editing}
   358→   358→        zones={zones}
   359→   359→        onSaved={handleSaved}

   361→   361→
   362→   362→      {/* Bulk upload */}

   369→   369→          refreshZones();

   386→   386→// ────────────────────────────────────────────────────────────
   387→   387→// Zones tab — dashboard view with scaled tiles
   388→   388→// ────────────────────────────────────────────────────────────
   389→   389→interface ZonesTabProps {
   390→   390→  zones: StoreZone[] | null;
   391→   391→  loading: boolean;
   392→   392→  error: string | null;
   393→   393→  onZoneClick: (zoneName: string) => void;

   395→   395→
   396→   396→function ZonesTab({ zones, loading, error, onZoneClick }: ZonesTabProps) {
   397→   397→  if (loading) return <LoadingState label="Loading zones..." />;
   398→   398→  if (error) return <ErrorState message={error} />;
   399→   399→  if (!zones || zones.length === 0)
   400→   400→    return (
   401→   401→      <EmptyState
   402→   402→        icon={Boxes}
   403→   403→        title="No zones configured"
   404→   404→        description="Store zones will appear here as tiles once data is seeded."
   405→   405→      />
   406→   406→    );
   407→   407→
   408→   408→  const totalValue = zones.reduce((s, z) => s + z.totalValue, 0);
   409→   409→  const totalItems = zones.reduce((s, z) => s + z.itemCount, 0);
   410→   410→  const totalLow = zones.reduce((s, z) => s + z.lowStockCount, 0);
   411→   411→
   412→   412→  return (
   413→   413→    <div>
   414→   414→      {/* Mini summary strip */}
   415→   415→      <div className="grid grid-cols-3 gap-3 mb-5">
   416→   416→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   417→   417→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   418→   418→            Total Stock Value
   419→   419→          </p>
   420→   420→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   421→   421→            {formatCurrency(totalValue)}
   422→   422→          </p>
   423→   423→        </div>
   424→   424→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   425→   425→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   426→   426→            Total Items
   427→   427→          </p>
   428→   428→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   429→   429→            {formatNumber(totalItems)}
   430→   430→          </p>
   431→   431→        </div>
   432→   432→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   433→   433→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   434→   434→            Low Stock Items
   435→   435→          </p>
   436→   436→          <p
   437→   437→            className={cn(
   438→   438→              "text-base sm:text-xl font-semibold mt-1",
   439→   439→              totalLow > 0 ? "text-amber-400" : "text-emerald-400"
   440→   440→            )}
   441→   441→          >
   442→   442→            {formatNumber(totalLow)}
   443→   443→          </p>
   444→   444→        </div>
   445→   445→      </div>
   446→   446→
   447→   447→      {/* Tiles grid */}
   448→   448→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   449→   449→        {zones.map((z) => {
   450→   450→          const status =
   451→   451→            z.lowStockCount >= 3
   452→   452→              ? "critical"
   453→   453→              : z.lowStockCount > 0
   454→   454→                ? "attention"
   455→   455→                : "healthy";
   456→   456→          const statusClasses =
   457→   457→            status === "critical"
   458→   458→              ? "border-red-500/40 hover:border-red-500/70"
   459→   459→              : status === "attention"
   460→   460→                ? "border-amber-500/40 hover:border-amber-500/70"
   461→   461→                : "border-emerald-500/30 hover:border-emerald-500/60";
   462→   462→          const minHeight = 120 + Math.min(z.totalValue / 100000, 80);
   463→   463→
   464→   464→          return (
   465→   465→            <button
   466→   466→              key={z.id}
   467→   467→              onClick={() => onZoneClick(z.name)}
   468→   468→              style={{ minHeight }}
   469→   469→              className={cn(
   470→   470→                "group bg-card border rounded-xl p-4 sm:p-5 text-left transition-all flex flex-col justify-between",
   471→   471→                "hover:shadow-lg hover:shadow-black/20 hover:-translate-y-0.5",
   472→   472→                statusClasses
   473→   473→              )}
   474→   474→            >
   475→   475→              {/* Header */}
   476→   476→              <div className="flex items-start justify-between gap-2 mb-3">
   477→   477→                <div className="min-w-0">
   478→   478→                  <h3 className="font-semibold text-foreground group-hover:text-primary transition-colors truncate text-sm sm:text-base">
   479→   479→                    {z.name}
   480→   480→                  </h3>
   481→   481→                  <p className="text-[10px] sm:text-xs text-muted-foreground mt-0.5">
   482→   482→                    {z.itemCount} items tracked
   483→   483→                  </p>
   484→   484→                </div>
   485→   485→                <CustomBadge
   486→   486→                  className={cn(
   487→   487→                    "border-0 shrink-0",
   488→   488→                    status === "critical" &&
   489→   489→                      "bg-red-500/15 text-red-400",
   490→   490→                    status === "attention" &&
   491→   491→                      "bg-amber-500/15 text-amber-400",
   492→   492→                    status === "healthy" &&
   493→   493→                      "bg-emerald-500/15 text-emerald-400"
   494→   494→                  )}
   495→   495→                >
   496→   496→                  {z.type}
   497→   497→                </CustomBadge>
   498→   498→              </div>

  1359→  1359→// Simple RFC-4180-aware CSV parser
  1360→  1360→function parseCSV(text: string): string[][] {
  1361→  1361→  const rows: string[][] = [];
  1362→  1362→  let row: string[] = [];
  1363→  1363→  let cell = "";
  1364→  1364→  let inQuotes = false;
  1365→  1365→  for (let i = 0; i < text.length; i++) {
  1366→  1366→    const c = text[i];
  1367→  1367→    if (inQuotes) {
  1368→  1368→      if (c === '"') {
  1369→  1369→        if (text[i + 1] === '"') {
  1370→  1370→          cell += '"';
  1371→  1371→          i++;
  1372→  1372→        } else {
  1373→  1373→          inQuotes = false;
  1374→  1374→        }
  1375→  1375→      } else {
  1376→  1376→        cell += c;
  1377→  1377→      }
  1378→  1378→    } else {
  1379→  1379→      if (c === '"') {
  1380→  1380→        inQuotes = true;
  1381→  1381→      } else if (c === ",") {
  1382→  1382→        row.push(cell);
  1383→  1383→        cell = "";
  1384→  1384→      } else if (c === "\n") {
  1385→  1385→        row.push(cell);
  1386→  1386→        rows.push(row);
  1387→  1387→        row = [];
  1388→  1388→        cell = "";
  1389→  1389→      } else if (c === "\r") {
  1390→  1390→        // skip CR
  1391→  1391→      } else {
  1392→  1392→        cell += c;
  1393→  1393→      }
  1394→  1394→    }
  1395→  1395→  }
  1396→  1396→  if (cell !== "" || row.length > 0) {
  1397→  1397→    row.push(cell);
  1398→  1398→    rows.push(row);
  1399→  1399→  }
  1400→  1400→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1401→  1401→}
