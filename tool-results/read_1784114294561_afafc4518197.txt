    70→interface EstimateRow {
    71→  itemName: string;
    72→  qty: string;
    73→  unit: string;
    74→  /** Per-unit cost (renamed from "estimateCost" in earlier versions).
    75→   *  The CSV column header is `UnitCost`; the parser also accepts the legacy
    76→   *  `EstimateCost` header for backward compatibility. The API field name stays

    79→  /** Total cost for the row — from CSV's TotalCost column when present,

    85→}

    95→export function AddProjectWizard({ open, onClose, onCreated }: Props) {
    96→  const [step, setStep] = useState<1 | 2>(1);
    97→  const [saving, setSaving] = useState(false);
    98→  const [createdProjectId, setCreatedProjectId] = useState<string | null>(null);
    99→  const [createdEstimateId, setCreatedEstimateId] = useState<string | null>(null);
   100→  const { toast } = useToast();
   101→  const fileInputRef = useRef<HTMLInputElement>(null);
   102→  const [previewRows, setPreviewRows] = useState<EstimateRow[] | null>(null);
   103→  const [customVoteNumber, setCustomVoteNumber] = useState(false);
   104→  const [selectedVoteId, setSelectedVoteId] = useState("");
   105→  // "Funds" checkbox state — when ticked, the project is funded via a non-vote fund
   106→  // (Malima/Non-Public/Contractual) and the Vote Number is NOT mandatory.
   107→  const [fundsChecked, setFundsChecked] = useState(false);
   108→  const { data: votes } = useFetch<Vote[]>("/api/votes");
   109→
   110→  const [form, setForm] = useState({
   111→    name: "",
   112→    voteNumber: "",
   113→    voteAllocation: "",
   114→    committed: "0",
   115→    actualSpent: "0",
   116→    startDate: "",
   117→    endDate: "", // EDC
   118→    benefactor: "", // replaces Department
   119→    reference: "", // manual text — how vote allocated
   120→    projectType: "New Construction",
   121→    financialMean: "Vote/Funds",
   122→    projectOfficer: "",
   123→    status: "Active",
   124→    description: "",
   125→  });
   126→  const [estimateRows, setEstimateRows] = useState<EstimateRow[]>([
   127→    { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },
   128→  ]);
   129→
   130→  const selectedVote = votes?.find((v) => v.id === selectedVoteId);
   131→
   132→  const handleVoteSelect = (id: string) => {
   133→    setSelectedVoteId(id);
   134→    const v = votes?.find((vt) => vt.id === id);
   135→    if (v) {
   136→      setForm((prev) => ({
   137→        ...prev,
   138→        voteNumber: v.voteNumber,
   139→        voteAllocation: v.allocation ? String(v.allocation) : prev.voteAllocation,
   140→      }));
   141→    }
   142→  };
   143→
   144→  // Toggle the "Funds" checkbox. When checked, the project is funded via a non-vote
   145→  // fund (default Malima Fund) and the Vote Number becomes optional. When unchecked,
   146→  // the project is funded via Vote and the Vote Number is mandatory.
   147→  const handleFundsToggle = (checked: boolean) => {
   148→    setFundsChecked(checked);
   149→    setCustomVoteNumber(false);
   150→    setSelectedVoteId("");
   151→    if (checked) {
   152→      setForm((prev) => ({
   153→        ...prev,
   154→        financialMean: "Malima Fund",
   155→        voteNumber: "",
   156→        voteAllocation: "",
   157→      }));
   158→    } else {
   159→      setForm((prev) => ({
   160→        ...prev,
   161→        financialMean: "Vote/Funds",
   162→        voteNumber: "",
   163→      }));
   164→    }
   165→  };
   166→
   167→  const handleFundSelect = (fund: string) => {
   168→    setForm((prev) => ({ ...prev, financialMean: fund }));
   169→  };
   170→
   171→  const nonVotePlaceholder =
   172→    form.financialMean === "Contractual" ? "CONTRACTUAL" : "NON-NAVAL";
   173→  const nonVoteHint =
   174→    form.financialMean === "Contractual"
   175→      ? "Contractual funding — Vote Number can be 'CONTRACTUAL' or the ARO reference."
   176→      : form.financialMean === "Malima Fund"
   177→      ? "Malima Fund — Vote Number can be 'NON-NAVAL' or the Malima reference number."
   178→      : "Non-Public Fund — Vote Number can be 'NON-NAVAL' or the fund reference.";
   179→
   180→  const addRow = () =>
   181→    setEstimateRows([
   182→      ...estimateRows,
   183→      { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },
   184→    ]);
   185→  const removeRow = (idx: number) =>
   186→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
   187→
   188→  // Update a single field on a row. When Qty or UnitCost changes,
   189→  // recompute the row's TotalCost automatically (= Qty × UnitCost) so the
   190→  // editable grid always shows a live total. Manual edits to TotalCost itself
   191→  // (or other fields) leave the total as-is.
   192→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
   193→    setEstimateRows(
   194→      estimateRows.map((r, i) => {
   195→        if (i !== idx) return r;
   196→        const next = { ...r, [field]: value };
   197→        if (field === "qty" || field === "estimateCost") {
   198→          const q = Number(next.qty);
   199→          const c = Number(next.estimateCost);
   200→          next.totalCost =
   201→            !isNaN(q) && !isNaN(c) && q > 0 && c > 0 ? String(q * c) : "";
   202→        }
   203→        return next;
   204→      })
   205→    );
   206→
   207→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   208→  const validEstimateRows = estimateRows.filter(
   209→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   210→  );
   211→  const hasValidEstimate = validEstimateRows.length > 0;
   212→
   213→  const downloadTemplate = () => {
   214→    // Template now uses `UnitCost` as the column header (renamed from `EstimateCost`).
   215→    // The sample row demonstrates the relationship TotalCost = Qty × UnitCost (2 × 185000 = 370000).
   216→    const header = "ItemName,Qty,Unit,UnitCost,TotalCost";
   217→    const sample = "Centrifugal Pump 15kW,2,NOS,185000,370000";
   218→    const sample2 = ",,NOS,,";
   219→    const csv = `${header}\n${sample}\n${sample2}\n`;
   220→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   221→    const url = URL.createObjectURL(blob);
   222→    const a = document.createElement("a");
   223→    a.href = url;
   224→    a.download = "material-estimate-template.csv";
   225→    a.click();
   226→    URL.revokeObjectURL(url);
   227→    toast({
   228→      title: "Template downloaded",
   229→      description:
   230→        "Columns: ItemName, Qty, Unit, UnitCost (per unit), TotalCost (= Qty × UnitCost — leave blank to auto-compute).",
   231→    });
   232→  };
   233→
   234→  // Simple CSV parser — handles quoted fields with commas
   235→  const parseCsv = (text: string): string[][] => {
   236→    const rows: string[][] = [];
   237→    let cur: string[] = [];
   238→    let field = "";
   239→    let inQuotes = false;
   240→    for (let i = 0; i < text.length; i++) {
   241→      const ch = text[i];
   242→      if (inQuotes) {
   243→        if (ch === '"' && text[i + 1] === '"') {
   244→          field += '"';
   245→          i++;
   246→        } else if (ch === '"') {
   247→          inQuotes = false;
   248→        } else {
   249→          field += ch;
   250→        }
   251→      } else {
   252→        if (ch === '"') inQuotes = true;
   253→        else if (ch === ",") {
   254→          cur.push(field);
   255→          field = "";
   256→        } else if (ch === "\n" || ch === "\r") {
   257→          if (ch === "\r" && text[i + 1] === "\n") i++;
   258→          cur.push(field);
   259→          rows.push(cur);
   260→          cur = [];
   261→          field = "";
   262→        } else {
   263→          field += ch;
   264→        }
   265→      }
   266→    }
   267→    if (field.length > 0 || cur.length > 0) {
   268→      cur.push(field);
   269→      rows.push(cur);
   270→    }
   271→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   272→  };
   273→
   274→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   275→    const file = e.target.files?.[0];
   276→    if (!file) return;
   277→    if (!file.name.toLowerCase().endsWith(".csv")) {
   278→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   279→      e.target.value = "";
   280→      return;
   281→    }
   282→    try {
   283→      const text = await file.text();
   284→      const rows = parseCsv(text);
   285→      if (rows.length === 0) {
   286→        toast({ title: "CSV file is empty", variant: "destructive" });
   287→        e.target.value = "";
   288→        return;
   289→      }
   290→      const header = rows[0].map((c) => c.trim().toLowerCase());
   291→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   292→      const idxQty = header.findIndex((h) => h.includes("qty"));
   293→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   294→      // Find the per-unit cost column. Prefer `UnitCost` (new name); fall back to
   295→      // the legacy `EstimateCost` header for backward compatibility. Also matches
   296→      // any column with "cost"/"price" but without "total".
   297→      const idxCost = header.findIndex(
   298→        (h) =>
   299→          (h === "unitcost" || h.includes("unit cost") || h.includes("unitcost")) ||
   300→          ((h.includes("estimate") && (h.includes("cost") || h.includes("price"))) ||
   301→           ((h.includes("cost") || h.includes("price")) && !h.includes("total")))
   302→      );
   303→      const idxTotal = header.findIndex(
   304→        (h) => h.includes("total") && (h.includes("cost") || h.includes("price"))
   305→      );
   306→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   307→      const parsed: EstimateRow[] = dataRows
   308→        .map((r) => {
   309→          const itemName = idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim();
   310→          const qty = idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim();
   311→          const unit = (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS";
   312→          // Per-unit cost — read from the UnitCost (or legacy EstimateCost) column.
   313→          const estimateCost = idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim();
   314→          // TotalCost: prefer the CSV's TotalCost column when present & non-blank,
   315→          // otherwise compute it as Qty × UnitCost (per the task spec).
   316→          const totalRaw =
   317→            idxTotal >= 0 ? (r[idxTotal] || "").trim() : "";
   318→          let totalCost = "";
   319→          if (totalRaw && !isNaN(Number(totalRaw))) {
   320→            totalCost = totalRaw;
   321→          } else {
   322→            const q = Number(qty);
   323→            const c = Number(estimateCost);
   324→            if (!isNaN(q) && !isNaN(c) && q > 0 && c > 0) {
   325→              totalCost = String(q * c);
   326→            }
   327→          }
   328→          return { itemName, qty, unit, estimateCost, totalCost };
   329→        })
   330→        .filter((r) => r.itemName);
   331→      if (parsed.length === 0) {
   332→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   333→        e.target.value = "";
   334→        return;
   335→      }
   336→      setPreviewRows(parsed);
   337→    } catch {
   338→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   339→    }
   340→    e.target.value = "";
   341→  };
   342→
   343→  const confirmImport = () => {
   344→    if (!previewRows) return;
   345→    setEstimateRows(previewRows);
   346→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   347→    setPreviewRows(null);
   348→  };
   349→
   350→  const handleSaveAndProcess = async () => {
   351→    if (!form.name) {
   352→      toast({ title: "Project Name is required", variant: "destructive" });
   353→      return;
   354→    }
   355→    // Vote Number is mandatory UNLESS the Funds checkbox is ticked (non-vote funding).
   356→    if (!fundsChecked && !form.voteNumber.trim()) {
   357→      toast({
   358→        title: "Vote Number is required",
   359→        description: "Tick the Funds checkbox for non-vote funding (Malima/Non-Public/Contractual).",
   360→        variant: "destructive",
   361→      });
   362→      return;
   363→    }
   364→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   365→    if (validRows.length === 0) {
   366→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   367→      return;
   368→    }
   369→    setSaving(true);
   370→    // 1. Create project
   371→    const projRes = await apiRequest("/api/projects", "POST", form);
   372→    if (!projRes.ok) {
   373→      setSaving(false);
   374→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   375→      return;
   376→    }
   377→    const project = projRes.data as { id: string };
   378→    // 2. Export Estimate into DB
   379→    const estRes = await apiRequest("/api/estimates", "POST", {
   380→      projectId: project.id,
   381→      name: `${form.name} — Material Estimate`,
   382→      items: validRows.map((r) => ({
   383→        itemName: r.itemName,
   384→        qty: Number(r.qty) || 0,
   385→        unit: r.unit,
   386→        estimateCost: Number(r.estimateCost) || 0,
   387→      })),
   388→    });
   389→    setSaving(false);
   390→    if (estRes.ok) {
   391→      const est = estRes.data as { id: string };
   392→      setCreatedProjectId(project.id);
   393→      setCreatedEstimateId(est.id);
   394→      setStep(2);
   395→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   396→    } else {
   397→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   398→    }
   399→  };
   400→
   401→  const handleGoToProject = () => {
   402→    if (createdProjectId) {
   403→      onCreated({
   404→        id: createdProjectId,
   405→        name: form.name,
   406→        voteNumber: form.voteNumber,
   407→        voteAllocation: Number(form.voteAllocation) || 0,
   408→        committed: Number(form.committed) || 0,
   409→        actualSpent: Number(form.actualSpent) || 0,
   410→        benefactor: form.benefactor || null,
   411→        reference: form.reference || null,
   412→        endDate: form.endDate || null,
   413→        startDate: form.startDate || null,
   414→        projectType: form.projectType,
   415→        financialMean: form.financialMean,
   416→      });
   417→      // reset
   418→      setStep(1);
   419→      setCreatedProjectId(null);
   420→      setCreatedEstimateId(null);
   421→      setFundsChecked(false);
   422→      setForm({
   423→        name: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   424→        startDate: "", endDate: "", benefactor: "", reference: "",
   425→        projectType: "New Construction", financialMean: "Vote/Funds",
   426→        projectOfficer: "", status: "Active", description: "",
   427→      });
   428→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" }]);
   429→      setCustomVoteNumber(false);
   430→      setSelectedVoteId("");
   431→    }
   432→  };
   433→
   434→  if (!open) return null;
   435→
   436→  return (
   437→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   438→      {/* Header */}
   439→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   440→        <div className="flex items-center gap-3">
   441→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   442→            <X className="h-5 w-5" />
   443→          </button>
   444→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   445→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   446→          </h2>
   447→        </div>
   448→        {/* Stepper */}
   449→        <div className="hidden sm:flex items-center gap-2 text-xs">
   450→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   451→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   452→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   453→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   454→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   455→        </div>
   456→      </div>
   457→
   458→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   459→        {step === 1 && (
   460→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   461→            {/* Project details */}
   462→            <section className="bg-card border border-border rounded-xl p-5">
   463→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   464→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   465→              </h3>
   466→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   467→                <div className="sm:col-span-2 space-y-1.5">
   468→                  <Label>Project Name *</Label>
   469→                  <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
   470→                </div>
   471→
   472→                {/* Vote Number — conditional UI based on Funds checkbox state */}
   473→                <div className="space-y-1.5">
   474→                  <Label className="flex items-center gap-1.5">
   475→                    <Vote className="h-3.5 w-3.5 text-primary" />
   476→                    Vote Number{!fundsChecked && " *"}
   477→                    {fundsChecked && (
   478→                      <span className="text-[10px] font-normal text-muted-foreground ml-1">(optional)</span>
   479→                    )}
   480→                  </Label>
   481→                  {!fundsChecked && !customVoteNumber ? (
   482→                    <>
   483→                      <Select value={selectedVoteId} onValueChange={handleVoteSelect}>
   484→                        <SelectTrigger className="bg-background border-border">
   485→                          <SelectValue placeholder="Select a vote from the list…" />
   486→                        </SelectTrigger>
   487→                        <SelectContent>
   488→                          {votes && votes.length > 0 ? (
   489→                            votes.map((v) => (
   490→                              <SelectItem key={v.id} value={v.id}>
   491→                                <span className="font-mono">{v.voteNumber}</span> — {v.description} ({v.voteType})
   492→                              </SelectItem>
   493→                            ))
   494→                          ) : (
   495→                            <SelectItem value="_none" disabled>No votes available</SelectItem>
   496→                          )}
   497→                        </SelectContent>
   498→                      </Select>
   499→                      {selectedVote ? (
   500→                        <div className="flex items-center gap-2 pt-0.5 flex-wrap">
   501→                          <CustomBadge className={cn("border", voteTypeColor(selectedVote.voteType))}>
   502→                            {selectedVote.voteType}
   503→                          </CustomBadge>
   504→                          {selectedVote.allocation > 0 && (
   505→                            <span className="text-[11px] text-muted-foreground">
   506→                              Allocation: {formatCurrency(selectedVote.allocation)}
   507→                            </span>
   508→                          )}
   509→                          {selectedVote.financialYear && (
   510→                            <span className="text-[11px] text-muted-foreground">
   511→                              · FY: {selectedVote.financialYear}
   512→                            </span>
   513→                          )}
   514→                          {!selectedVote.active && (
   515→                            <CustomBadge className="border-0 bg-amber-500/15 text-amber-400">Inactive</CustomBadge>
   516→                          )}
   517→                        </div>
