    93→
    94→export function PresentPositionDialog({ project, open, onOpenChange, onChanged }: Props) {
    95→  const { data: files, loading, refresh } = useFetch<ProjectFile[]>(
    96→    `/api/projects/${project.id}/files`
    97→  );
    98→  const { data: estimates, refresh: refreshEst } = useFetch<Estimate[]>(
    99→    `/api/estimates?projectId=${project.id}`
   100→  );
   101→  const [activeTab, setActiveTab] = useState<"raised" | "estimate">("raised");
   102→  const [search, setSearch] = useState("");
   103→  const [positionFilter, setPositionFilter] = useState("all");
   104→  const [addOpen, setAddOpen] = useState(false);
   105→  const [editFile, setEditFile] = useState<ProjectFile | null>(null);
   106→  const [logFile, setLogFile] = useState<ProjectFile | null>(null);
   107→  const [allocOpen, setAllocOpen] = useState(false);
   108→  const [activeEstimateId, setActiveEstimateId] = useState<string | null>(null);
   109→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
   110→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
   111→  const { toast } = useToast();
   112→
   113→  // Position summary
   114→  const positionSummary = useMemo(() => {
   115→    const map: Record<string, { count: number; cost: number }> = {};
   116→    files?.forEach((f) => {
   117→      if (!map[f.presentPosition]) map[f.presentPosition] = { count: 0, cost: 0 };
   118→      map[f.presentPosition].count++;
   119→      map[f.presentPosition].cost += f.fileCost;
   120→    });
   121→    return Object.entries(map).sort((a, b) => b[1].count - a[1].count);
   122→  }, [files]);
   123→
   124→  const totalFileCost = files?.reduce((s, f) => s + f.fileCost, 0) ?? 0;
   125→
   126→  // pending estimate items (not yet initiated) — "items to be raised PR"
   127→  const pendingEstimateItems = useMemo(() => {
   128→    const list: Estimate["items"][number][] = [];
   129→    estimates?.forEach((e) => {
   130→      e.items.forEach((it) => {
   131→        if (it.status === "Pending") list.push(it);
   132→      });
   133→    });
   134→    return list;
   135→  }, [estimates]);
   136→  const pendingCost = 0; // approx cost fetched per-estimate in the table
   137→  const initiatedCount = useMemo(
   138→    () =>
   139→      estimates?.reduce(
   140→        (s, e) => s + e.items.filter((i) => i.status === "Initiated").length,
   141→        0
   142→      ) ?? 0,
   143→    [estimates]
   144→  );
   145→
   146→  const filteredFiles = useMemo(() => {
   147→    let list = files || [];
   148→    if (search) {
   149→      const q = search.toLowerCase();
   150→      list = list.filter(
   151→        (f) =>
   152→          f.prNumber.toLowerCase().includes(q) ||
   153→          f.itemCode.toLowerCase().includes(q) ||
   154→          f.itemName.toLowerCase().includes(q) ||
   155→          (f.supplier || "").toLowerCase().includes(q)
   156→      );
   157→    }
   158→    if (positionFilter !== "all") {
   159→      list = list.filter((f) => f.presentPosition === positionFilter);
   160→    }
   161→    list = [...list].sort((a, b) => {
   162→      let cmp = 0;
   163→      if (sortKey === "fileCost") cmp = a.fileCost - b.fileCost;
   164→      else if (sortKey === "prNumber") cmp = a.prNumber.localeCompare(b.prNumber);
   165→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   166→      return sortDir === "asc" ? cmp : -cmp;
   167→    });
   168→    return list;
   169→  }, [files, search, positionFilter, sortKey, sortDir]);
   170→
   171→  const toggleSort = (key: typeof sortKey) => {
   172→    if (sortKey === key) setSortDir(sortDir === "asc" ? "desc" : "asc");
   173→    else {
   174→      setSortKey(key);
   175→      setSortDir("desc");
   176→    }
   177→  };
   178→
   179→  const handleDelete = async (file: ProjectFile) => {
   180→    if (!confirm(`Delete file ${file.prNumber} (${file.itemName})?`)) return;
   181→    const res = await apiRequest(`/api/projects/${project.id}/files/${file.id}`, "DELETE");
   182→    if (res.ok) {
   183→      toast({ title: "File deleted" });
   184→      refresh();
   185→      onChanged();
   186→    } else {
   187→      toast({ title: "Delete failed", description: res.error, variant: "destructive" });
   188→    }
   189→  };
   190→
   191→  const exportExcel = () => {
   192→    const rows = filteredFiles.map((f) => ({
   193→      "PR Number": f.prNumber,
   194→      "Item Code": f.itemCode,
   195→      "Item Name": f.itemName,
   196→      Unit: f.unit,
   197→      Qty: f.qty,
   198→      "Unit Price": f.unitPrice,
   199→      "File Cost": f.fileCost,
   200→      Supplier: f.supplier || "",
   201→      "Present Position": f.presentPosition,
   202→      Category: f.fileCategory,
   203→    }));
   204→    const headers = Object.keys(rows[0] || {});
   205→    const csv = [
   206→      headers.join(","),
   207→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h]}"`).join(",")),
   208→    ].join("\n");
   209→    const blob = new Blob([csv], { type: "text/csv" });
   210→    const url = URL.createObjectURL(blob);
   211→    const a = document.createElement("a");
   212→    a.href = url;
   213→    a.download = `${project.name}-files.csv`;
   214→    a.click();
   215→    URL.revokeObjectURL(url);
   216→  };
   217→
   218→  // pick the first processed estimate (or any) for the "PR from Estimate" tab
   219→  const estimateForTab = estimates?.find((e) => e.status === "Processed") || estimates?.[0] || null;
   220→
   221→  if (!open) return null;
   222→
   223→  return (
   224→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   225→      {/* Header */}
   226→      <div className="border-b border-border bg-card shrink-0">
   227→        <div className="px-4 sm:px-6 py-3 flex items-center justify-between gap-4">
   228→          <div className="flex items-center gap-3 min-w-0">
   229→            <button onClick={() => onOpenChange(false)} className="text-muted-foreground hover:text-foreground shrink-0">
   230→              <X className="h-5 w-5" />
   231→            </button>
   232→            <div className="min-w-0">
   233→              <h2 className="font-semibold text-foreground text-sm sm:text-base truncate">{project.name}</h2>
   234→              <p className="text-xs text-muted-foreground">
   235→                {project.voteNumber}
   236→                {project.benefactor && ` · Benefactor: ${project.benefactor}`}
   237→                {project.reference && ` · Ref: ${project.reference}`}
   238→                {project.endDate && ` · EDC: ${new Date(project.endDate).toLocaleDateString("en-GB")}`}
   239→              </p>
   240→            </div>
   241→          </div>
   242→          <Button variant="outline" size="sm" onClick={() => setAllocOpen(true)} className="gap-1.5 shrink-0">
   243→            <History className="h-4 w-4" /> Allocations
   244→          </Button>
   245→        </div>
   246→        {/* Financial summary bar */}
   247→        <div className="px-4 sm:px-6 pb-3 grid grid-cols-2 sm:grid-cols-4 gap-2">
   248→          <FinBox label="Vote Allocation" value={formatCurrency(project.voteAllocation)} tone="text-blue-400" />
   249→          <FinBox label="Committed" value={formatCurrency(project.committed)} tone="text-amber-400" />
   250→          <FinBox label="Actual Spent" value={formatCurrency(project.actualSpent)} tone="text-red-400" />
   251→          <FinBox label="File Cost (Total)" value={formatCurrency(totalFileCost)} tone="text-emerald-400" />
   252→        </div>
   253→        {/* Tabs */}
   254→        <div className="px-4 sm:px-6 flex items-center gap-1 border-t border-border">
   255→          <TabButton
   256→            active={activeTab === "raised"}
   257→            onClick={() => setActiveTab("raised")}
   258→            icon={<FileText className="h-3.5 w-3.5" />}
   259→            label="PR Raised Files"
   260→            count={files?.length || 0}
   261→          />
   262→          <TabButton
   263→            active={activeTab === "estimate"}
   264→            onClick={() => setActiveTab("estimate")}
   265→            icon={<Sparkles className="h-3.5 w-3.5" />}
   266→            label="PR from Estimate"
   267→            count={pendingEstimateItems.length}
   268→            tone="amber"
   269→          />
   270→        </div>
   271→      </div>
   272→
   273→      {/* Body */}
   274→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   275→        <div className="p-4 sm:p-6 space-y-4">
   276→          {/* Position summary (always visible) */}
   277→          <div>
   278→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   279→              Present Position Summary
   280→            </h4>
   281→            <div className="flex flex-wrap gap-2">
   282→              {positionSummary.length === 0 ? (
   283→                <p className="text-sm text-muted-foreground">No files yet.</p>
   284→              ) : (
   285→                positionSummary.map(([pos, info]) => (
   286→                  <div key={pos} className={cn("flex items-center gap-2 rounded-lg border px-3 py-1.5", positionColor(pos))}>
   287→                    <span className="text-sm font-medium">{pos}</span>
   288→                    <span className="text-xs opacity-80">{info.count} file{info.count > 1 ? "s" : ""} · {formatCurrency(info.cost)}</span>
   289→                  </div>
   290→                ))
   291→              )}
   292→            </div>
   293→          </div>
   294→
   295→          {activeTab === "raised" ? (
   296→            <>
   297→              {/* PR raised summary chips */}
   298→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   299→                <MiniStat label="PR Raised (Files)" value={String(files?.length || 0)} sub={formatCurrency(totalFileCost)} tone="text-blue-400" />
   300→                <MiniStat label="Initiated from Estimate" value={String(initiatedCount)} sub="forwarded PRs" tone="text-violet-400" />
   301→                <MiniStat label="Items to Raise PR" value={String(pendingEstimateItems.length)} sub="from estimate" tone="text-amber-400" />
   302→              </div>
   303→
   304→              {/* Toolbar */}
   305→              <div className="flex flex-wrap items-center gap-2">
   306→                <div className="relative flex-1 min-w-[200px]">
   307→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   308→                  <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search PR, item, supplier..." className="pl-9 bg-background border-border" />
   309→                </div>
   310→                <Select value={positionFilter} onValueChange={setPositionFilter}>
   311→                  <SelectTrigger className="w-[180px] bg-background border-border"><SelectValue placeholder="Position" /></SelectTrigger>
   312→                  <SelectContent>
   313→                    <SelectItem value="all">All Positions</SelectItem>
   314→                    {PRESENT_POSITIONS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
   315→                  </SelectContent>
   316→                </Select>
   317→                <Button variant="outline" size="sm" onClick={exportExcel} className="gap-1.5"><Download className="h-4 w-4" /> Export</Button>
   318→                <Button size="sm" onClick={() => setAddOpen(true)} className="gap-1.5"><Plus className="h-4 w-4" /> Add File</Button>
   319→              </div>
   320→
   321→              {/* File register table */}
   322→              <div className="border border-border rounded-lg overflow-hidden">
   323→                <div className="overflow-x-auto">
   324→                  <Table>
   325→                    <TableHeader>
   326→                      <TableRow className="hover:bg-transparent border-border">
   327→                        <SortTh label="PR Number" onClick={() => toggleSort("prNumber")} active={sortKey === "prNumber"} dir={sortDir} />
   328→                        <TableHead className="text-muted-foreground">Item Code</TableHead>
   329→                        <TableHead className="text-muted-foreground">Item Name</TableHead>
   330→                        <TableHead className="text-muted-foreground text-right">Qty</TableHead>
   331→                        <TableHead className="text-muted-foreground text-right">Unit Price</TableHead>
   332→                        <SortTh label="File Cost" onClick={() => toggleSort("fileCost")} active={sortKey === "fileCost"} dir={sortDir} align="right" />
   333→                        <TableHead className="text-muted-foreground">Supplier</TableHead>
   334→                        <TableHead className="text-muted-foreground">Position</TableHead>
   335→                        <TableHead className="text-muted-foreground">Category</TableHead>
   336→                        <TableHead className="text-muted-foreground text-right">Actions</TableHead>
   337→                      </TableRow>
   338→                    </TableHeader>
   339→                    <TableBody>
   340→                      {loading ? (
   341→                        <TableRow><TableCell colSpan={10} className="text-center py-8 text-muted-foreground"><Loader2 className="h-5 w-5 animate-spin mx-auto" /></TableCell></TableRow>
   342→                      ) : filteredFiles.length === 0 ? (
   343→                        <TableRow><TableCell colSpan={10} className="text-center py-8 text-muted-foreground">No files found. Click "Add File" to create one.</TableCell></TableRow>
   344→                      ) : (
   345→                        filteredFiles.map((f) => (
   346→                          <TableRow key={f.id} className="border-border cursor-pointer hover:bg-secondary/40" onClick={() => setLogFile(f)}>
   347→                            <TableCell className="font-mono text-xs text-foreground">{f.prNumber}</TableCell>
   348→                            <TableCell className="font-mono text-xs">{f.itemCode}</TableCell>
   349→                            <TableCell className="text-foreground">{f.itemName}</TableCell>
   350→                            <TableCell className="text-right">{f.qty} {f.unit}</TableCell>
   351→                            <TableCell className="text-right">{formatCurrency(f.unitPrice)}</TableCell>
   352→                            <TableCell className="text-right font-medium text-foreground">{formatCurrency(f.fileCost)}</TableCell>
   353→                            <TableCell className="text-xs">{f.supplier || "—"}</TableCell>
   354→                            <TableCell><PositionBadge position={f.presentPosition} /></TableCell>
   355→                            <TableCell><CategoryBadge category={f.fileCategory} /></TableCell>
   356→                            <TableCell className="text-right">
   357→                              <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   358→                                <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setEditFile(f)}><Pencil className="h-3.5 w-3.5" /></Button>
   359→                                <Button variant="ghost" size="icon" className="h-7 w-7 text-red-400 hover:text-red-500" onClick={() => handleDelete(f)}><Trash2 className="h-3.5 w-3.5" /></Button>
   360→                              </div>
   361→                            </TableCell>
   362→                          </TableRow>
   363→                        ))
   364→                      )}
   365→                    </TableBody>
   366→                  </Table>
   367→                </div>
   368→              </div>
   369→              <p className="text-xs text-muted-foreground">{filteredFiles.length} file(s) · Click any row to view full log</p>
   370→            </>
   371→          ) : (
   372→            /* PR from Estimate tab */
   373→            <>
   374→              <div className="bg-secondary/40 rounded-lg p-3 flex items-center gap-2 flex-wrap">
   375→                <FileSignature className="h-4 w-4 text-amber-400" />
   376→                <p className="text-sm text-foreground">
   377→                  Items pending PR initiation from processed estimates. Click <span className="font-medium text-primary">Initiate File</span> to raise a PR — it will be forwarded to the "PR Raised Files" tab.
   378→                </p>
   379→              </div>
   380→              {estimates && estimates.length > 0 ? (
   381→                <div className="space-y-4">
   382→                  {/* estimate selector */}
   383→                  {estimates.length > 1 && (
   384→                    <div className="flex items-center gap-2 flex-wrap">
   385→                      <span className="text-xs text-muted-foreground">Estimate:</span>
   386→                      {estimates.map((e) => (
   387→                        <button
   388→                          key={e.id}
   389→                          onClick={() => setActiveEstimateId(e.id)}
   390→                          className={cn(
   391→                            "px-2.5 py-1 rounded-md text-xs border transition-colors",
   392→                            (activeEstimateId || estimateForTab?.id) === e.id
   393→                              ? "border-primary bg-primary/10 text-primary"
   394→                              : "border-border text-muted-foreground hover:text-foreground"
   395→                          )}
   396→                        >
   397→                          {e.name} <CustomBadge className={cn("border-0 ml-1", estimateStatusColor(e.status))}>{e.status}</CustomBadge>
   398→                        </button>
   399→                      ))}
   400→                    </div>
   401→                  )}
   402→                  <EstimateItemsTable
   403→                    estimateId={activeEstimateId || estimateForTab?.id || estimates[0].id}
   404→                    showProcessButton
   405→                    onChanged={() => { refreshEst(); refresh(); onChanged(); }}
   406→                  />
   407→                </div>
   408→              ) : (
   409→                <EmptyState
   410→                  icon={Sparkles}
   411→                  title="No estimates for this project"
   412→                  description="Create a new project via the Add Project wizard to generate a material estimate, or add files directly in the PR Raised Files tab."
   413→                />
   414→              )}
   415→            </>
   416→          )}
   417→        </div>
   418→      </div>
   419→
   420→      {/* Add File */}
   421→      <FileFormDialog projectId={project.id} open={addOpen} onOpenChange={setAddOpen} onSaved={() => { setAddOpen(false); refresh(); onChanged(); }} />
   422→      {/* Edit File */}
   423→      {editFile && (
   424→        <FileFormDialog projectId={project.id} file={editFile} open={!!editFile} onOpenChange={(o) => !o && setEditFile(null)} onSaved={() => { setEditFile(null); refresh(); onChanged(); }} />
   425→      )}
   426→      {/* File Log */}
   427→      <FileLogDialog file={logFile} open={!!logFile} onOpenChange={(o) => !o && setLogFile(null)} />
   428→      {/* Allocation History */}
   429→      <AllocationHistoryDialog project={project} open={allocOpen} onOpenChange={setAllocOpen} onChanged={onChanged} />
   430→    </div>
   431→  );
   432→}

   434→function TabButton({ active, onClick, icon, label, count, tone }: { active: boolean; onClick: () => void; icon: React.ReactNode; label: string; count: number; tone?: "amber" }) {
   435→  return (
   436→    <button
   437→      onClick={onClick}
   438→      className={cn(
   439→        "flex items-center gap-2 px-4 py-2.5 text-sm font-medium border-b-2 transition-colors -mb-px",
   440→        active
   441→          ? "border-primary text-primary"
   442→          : "border-transparent text-muted-foreground hover:text-foreground"
   443→      )}
   444→    >
   445→      {icon}
   446→      {label}
   447→      <span className={cn(
   448→        "px-1.5 py-0.5 rounded text-[10px] font-semibold",
   449→        active ? "bg-primary/15 text-primary" : "bg-secondary text-muted-foreground",
   450→        tone === "amber" && count > 0 && "bg-amber-500/15 text-amber-400"
   451→      )}>
   452→        {count}
   453→      </span>
   454→    </button>
   455→  );
   456→}
   457→
   458→function FinBox({ label, value, tone }: { label: string; value: string; tone: string }) {
   459→  return (
   460→    <div className="bg-secondary/40 rounded-lg p-3">
   461→      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</p>
   462→      <p className={`text-sm font-semibold ${tone}`}>{value}</p>
   463→    </div>
   464→  );
   465→}
   466→
   467→function MiniStat({ label, value, sub, tone }: { label: string; value: string; sub: string; tone: string }) {
   468→  return (
   469→    <div className="bg-secondary/40 rounded-lg p-3">
   470→      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</p>
   471→      <p className={`text-lg font-semibold ${tone}`}>{value}</p>
   472→      <p className="text-[10px] text-muted-foreground">{sub}</p>
   473→    </div>
   474→  );
   475→}
   476→
   477→function SortTh({ label, onClick, active, dir, align = "left" }: { label: string; onClick: () => void; active: boolean; dir: "asc" | "desc"; align?: "left" | "right" }) {
   478→  return (
   479→    <TableHead className={align === "right" ? "text-muted-foreground text-right" : "text-muted-foreground"}>
   480→      <button onClick={onClick} className="inline-flex items-center gap-1 hover:text-foreground">
   481→        {label}
   482→        {active && (dir === "asc" ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />)}
   483→      </button>
   484→    </TableHead>

   489→function FileFormDialog({ projectId, file, open, onOpenChange, onSaved }: { projectId: string; file?: ProjectFile; open: boolean; onOpenChange: (o: boolean) => void; onSaved: () => void }) {
   490→  const isEdit = !!file;
   491→  const [saving, setSaving] = useState(false);
   492→  const [form, setForm] = useState({
   493→    prNumber: file?.prNumber || "",
   494→    itemCode: file?.itemCode || "",
   495→    itemName: file?.itemName || "",
   496→    unit: file?.unit || "NOS",
   497→    qty: String(file?.qty ?? ""),
   498→    unitPrice: String(file?.unitPrice ?? ""),
   499→    fileCost: String(file?.fileCost ?? ""),
   500→    supplier: file?.supplier || "",
   501→    presentPosition: file?.presentPosition || "PR Raised",
   502→    fileCategory: file?.fileCategory || "General",
   503→    changedBy: "",
   504→    comment: "",
   505→  });
   506→  const { toast } = useToast();
   507→
   508→  const submit = async () => {
   509→    if (!form.prNumber || !form.itemCode || !form.itemName) {
   510→      toast({ title: "PR Number, Item Code and Item Name are required", variant: "destructive" });
   511→      return;
   512→    }
   513→    setSaving(true);
   514→    const url = isEdit ? `/api/projects/${projectId}/files/${file!.id}` : `/api/projects/${projectId}/files`;
   515→    const method = isEdit ? "PUT" : "POST";
   516→    const res = await apiRequest(url, method, form);
   517→    setSaving(false);
   518→    if (res.ok) {
   519→      toast({ title: isEdit ? "File updated" : "File added" });
   520→      onSaved();
   521→    } else {
   522→      toast({ title: "Failed", description: res.error, variant: "destructive" });
   523→    }
   524→  };
   525→
   526→  return (
   527→    <Dialog open={open} onOpenChange={onOpenChange}>
   528→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
   529→        <DialogHeader>
   530→          <DialogTitle className="text-foreground flex items-center gap-2">
   531→            <FileText className="h-5 w-5 text-primary" />
   532→            {isEdit ? `Edit File — ${file!.prNumber}` : "Add Processing File"}
   533→          </DialogTitle>
   534→        </DialogHeader>
   535→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 py-2">
   536→          <Field label="PR Number *"><Input value={form.prNumber} onChange={(e) => setForm({ ...form, prNumber: e.target.value })} className="bg-background border-border" /></Field>
   537→          <Field label="Item Code *"><Input value={form.itemCode} onChange={(e) => setForm({ ...form, itemCode: e.target.value })} className="bg-background border-border" /></Field>
   538→          <div className="sm:col-span-2"><Field label="Item Name *"><Input value={form.itemName} onChange={(e) => setForm({ ...form, itemName: e.target.value })} className="bg-background border-border" /></Field></div>
   539→          <Field label="Unit"><Input value={form.unit} onChange={(e) => setForm({ ...form, unit: e.target.value })} className="bg-background border-border" /></Field>
   540→          <Field label="Qty"><Input type="number" value={form.qty} onChange={(e) => setForm({ ...form, qty: e.target.value })} className="bg-background border-border" /></Field>
   541→          <Field label="Unit Price (Rs.)"><Input type="number" value={form.unitPrice} onChange={(e) => setForm({ ...form, unitPrice: e.target.value })} className="bg-background border-border" /></Field>
   542→          <Field label="File Cost (Rs.) — auto if empty"><Input type="number" value={form.fileCost} onChange={(e) => setForm({ ...form, fileCost: e.target.value })} placeholder="auto" className="bg-background border-border" /></Field>
   543→          <Field label="Supplier"><Input value={form.supplier} onChange={(e) => setForm({ ...form, supplier: e.target.value })} className="bg-background border-border" /></Field>
   544→          <Field label="Present Position">
   545→            <Select value={form.presentPosition} onValueChange={(v) => setForm({ ...form, presentPosition: v })}>
   546→              <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   547→              <SelectContent>{PRESENT_POSITIONS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}</SelectContent>
   548→            </Select>
   549→          </Field>
   550→          <Field label="File Category">
   551→            <Select value={form.fileCategory} onValueChange={(v) => setForm({ ...form, fileCategory: v })}>
   552→              <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   553→              <SelectContent>{FILE_CATEGORIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}</SelectContent>
   554→            </Select>
   555→          </Field>
   556→          {isEdit && (
   557→            <>
   558→              <Field label="Changed By"><Input value={form.changedBy} onChange={(e) => setForm({ ...form, changedBy: e.target.value })} placeholder="Your name" className="bg-background border-border" /></Field>
   559→              <div className="sm:col-span-2"><Field label="Comment (for log)"><Input value={form.comment} onChange={(e) => setForm({ ...form, comment: e.target.value })} placeholder="Optional note" className="bg-background border-border" /></Field></div>
   560→            </>
   561→          )}
   562→        </div>
   563→        <DialogFooter>
   564→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   565→          <Button onClick={submit} disabled={saving} className="gap-1.5">{saving && <Loader2 className="h-4 w-4 animate-spin" />}{isEdit ? "Update File" : "Add File"}</Button>
   566→        </DialogFooter>
   567→      </DialogContent>
   568→    </Dialog>
   569→  );
   570→}

   581→// ─── Allocation History ───
   582→function AllocationHistoryDialog({ project, open, onOpenChange, onChanged }: { project: Project; open: boolean; onOpenChange: (o: boolean) => void; onChanged: () => void }) {
   583→  const { data: allocations, refresh } = useFetch<Allocation[]>(`/api/projects/${project.id}/allocations`);
   584→  const [saving, setSaving] = useState(false);
   585→  const [form, setForm] = useState({ date: new Date().toISOString().slice(0, 10), description: "", amount: "", type: "Commitment", authorizedBy: "" });
   586→  const { toast } = useToast();
   587→
   588→  const totalAllocated = allocations?.reduce((s, a) => s + a.amount, 0) ?? 0;
   589→  const committed = allocations?.filter((a) => a.type === "Commitment").reduce((s, a) => s + a.amount, 0) ?? 0;
   590→  const actual = allocations?.filter((a) => a.type === "Actual").reduce((s, a) => s + a.amount, 0) ?? 0;
   591→
   592→  const submit = async () => {
   593→    if (!form.description || !form.amount) {
   594→      toast({ title: "Description and amount required", variant: "destructive" });
   595→      return;
   596→    }
   597→    setSaving(true);
   598→    const res = await apiRequest(`/api/projects/${project.id}/allocations`, "POST", form);
   599→    setSaving(false);
   600→    if (res.ok) {
   601→      toast({ title: "Allocation added" });
   602→      setForm({ date: new Date().toISOString().slice(0, 10), description: "", amount: "", type: "Commitment", authorizedBy: "" });
   603→      refresh();
   604→      onChanged();
   605→    } else {
   606→      toast({ title: "Failed", description: res.error, variant: "destructive" });
   607→    }
   608→  };
   609→
   610→  return (
   611→    <Dialog open={open} onOpenChange={onOpenChange}>
   612→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
   613→        <DialogHeader>
   614→          <DialogTitle className="text-foreground flex items-center gap-2">
   615→            <History className="h-5 w-5 text-primary" />
   616→            Allocation History — {project.name}

   621→          <FinBox label="Committed" value={formatCurrency(committed)} tone="text-amber-400" />
   622→          <FinBox label="Actual" value={formatCurrency(actual)} tone="text-red-400" />
   623→        </div>
   624→        <div className="bg-secondary/40 rounded-lg p-4 mb-4">
   625→          <h4 className="text-sm font-semibold text-foreground mb-3">Add Allocation</h4>
   626→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3">
