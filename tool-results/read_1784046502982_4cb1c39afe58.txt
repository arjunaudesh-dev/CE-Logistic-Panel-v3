    64→  id: string;
    65→  name: string;
    66→  voteNumber: string;

    90→}
    91→

    97→}
    98→

   101→  name: string;

   103→}
   104→

   106→  project: Project;
   107→  open: boolean;

   110→}
   111→
   112→type FileNumberFilter = "all" | "with" | "without";
   113→
   114→export function PresentPositionDialog({ project, open, onOpenChange, onChanged }: Props) {
   115→  const { data: files, loading, refresh } = useFetch<ProjectFile[]>(
   116→    `/api/projects/${project.id}/files`
   117→  );
   118→  const { data: estimates, refresh: refreshEst } = useFetch<Estimate[]>(
   119→    `/api/estimates?projectId=${project.id}`
   120→  );
   121→  // Fetch process flows once to support the "Update Position" popup's default responsible person lookup
   122→  const { data: processFlows } = useFetch<ProcessFlow[]>(`/api/process-flows`);
   123→  const [activeTab, setActiveTab] = useState<"raised" | "estimate">("raised");
   124→  const [search, setSearch] = useState("");
   125→  const [positionFilter, setPositionFilter] = useState("all");
   126→  const [fileNumberFilter, setFileNumberFilter] = useState<FileNumberFilter>("all");
   127→  const [addOpen, setAddOpen] = useState(false);
   128→  const [updateFile, setUpdateFile] = useState<ProjectFile | null>(null);
   129→  const [logFile, setLogFile] = useState<ProjectFile | null>(null);
   130→  const [allocOpen, setAllocOpen] = useState(false);
   131→  const [activeEstimateId, setActiveEstimateId] = useState<string | null>(null);
   132→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
   133→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
   134→  const { toast } = useToast();
   135→
   136→  // Build a lookup of positionName -> defaultResponsiblePerson across all flows.
   137→  // Multiple flows may define the same positionName; we keep the first non-empty match.
   138→  const positionToResponsible = useMemo(() => {
   139→    const map = new Map<string, string>();
   140→    processFlows?.forEach((flow) => {
   141→      flow.steps.forEach((step) => {
   142→        if (step.positionName && step.defaultResponsiblePerson && !map.has(step.positionName)) {
   143→          map.set(step.positionName, step.defaultResponsiblePerson);
   144→        }
   145→      });
   146→    });
   147→    return map;
   148→  }, [processFlows]);
   149→
   150→  // Position summary
   151→  const positionSummary = useMemo(() => {
   152→    const map: Record<string, { count: number; cost: number }> = {};
   153→    files?.forEach((f) => {
   154→      if (!map[f.presentPosition]) map[f.presentPosition] = { count: 0, cost: 0 };
   155→      map[f.presentPosition].count++;
   156→      map[f.presentPosition].cost += f.fileCost;
   157→    });
   158→    return Object.entries(map).sort((a, b) => b[1].count - a[1].count);
   159→  }, [files]);
   160→
   161→  const totalFileCost = files?.reduce((s, f) => s + f.fileCost, 0) ?? 0;
   162→
   163→  // pending estimate items (not yet initiated) — "items to be raised PR"
   164→  const pendingEstimateItems = useMemo(() => {
   165→    const list: Estimate["items"][number][] = [];
   166→    estimates?.forEach((e) => {
   167→      e.items.forEach((it) => {
   168→        if (it.status === "Pending") list.push(it);
   169→      });
   170→    });
   171→    return list;
   172→  }, [estimates]);
   173→  const initiatedCount = useMemo(
   174→    () =>
   175→      estimates?.reduce(
   176→        (s, e) => s + e.items.filter((i) => i.status === "Initiated").length,
   177→        0
   178→      ) ?? 0,
   179→    [estimates]
   180→  );
   181→
   182→  // File-number counts for the filter toggle
   183→  const fileNumberCounts = useMemo(() => {
   184→    const all = files?.length || 0;
   185→    const withFn = files?.filter((f) => f.fileNumber && f.fileNumber.trim()).length || 0;
   186→    return { all, with: withFn, without: all - withFn };
   187→  }, [files]);
   188→
   189→  const filteredFiles = useMemo(() => {
   190→    let list = files || [];
   191→    if (search) {
   192→      const q = search.toLowerCase();
   193→      list = list.filter(
   194→        (f) =>
   195→          f.prNumber.toLowerCase().includes(q) ||
   196→          f.itemCode.toLowerCase().includes(q) ||
   197→          f.itemName.toLowerCase().includes(q) ||
   198→          (f.supplier || "").toLowerCase().includes(q) ||
   199→          (f.fileNumber || "").toLowerCase().includes(q)
   200→      );
   201→    }
   202→    if (positionFilter !== "all") {
   203→      list = list.filter((f) => f.presentPosition === positionFilter);
   204→    }
   205→    if (fileNumberFilter === "with") {
   206→      list = list.filter((f) => f.fileNumber && f.fileNumber.trim());
   207→    } else if (fileNumberFilter === "without") {
   208→      list = list.filter((f) => !f.fileNumber || !f.fileNumber.trim());
   209→    }
   210→    list = [...list].sort((a, b) => {
   211→      let cmp = 0;
   212→      if (sortKey === "fileCost") cmp = a.fileCost - b.fileCost;
   213→      else if (sortKey === "prNumber") cmp = a.prNumber.localeCompare(b.prNumber);
   214→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   215→      return sortDir === "asc" ? cmp : -cmp;
   216→    });
   217→    return list;
   218→  }, [files, search, positionFilter, fileNumberFilter, sortKey, sortDir]);
   219→
   220→  const toggleSort = (key: typeof sortKey) => {
   221→    if (sortKey === key) setSortDir(sortDir === "asc" ? "desc" : "asc");
   222→    else {
   223→      setSortKey(key);
   224→      setSortDir("desc");
   225→    }
   226→  };
   227→
   228→  const handleDelete = async (file: ProjectFile) => {
   229→    if (!confirm(`Delete file ${file.prNumber} (${file.itemName})?`)) return;
   230→    const res = await apiRequest(`/api/projects/${project.id}/files/${file.id}`, "DELETE");
   231→    if (res.ok) {
   232→      toast({ title: "File deleted" });
   233→      refresh();
   234→      onChanged();
   235→    } else {
   236→      toast({ title: "Delete failed", description: res.error, variant: "destructive" });
   237→    }
   238→  };
   239→
   240→  const exportExcel = () => {
   241→    const rows = filteredFiles.map((f) => ({
   242→      "PR Number": f.prNumber,
   243→      "File Number": f.fileNumber || "",
   244→      "Item Code": f.itemCode,
   245→      "Item Name": f.itemName,
   246→      Unit: f.unit,
   247→      Qty: f.qty,
   248→      "Unit Price": f.unitPrice,
   249→      "File Cost": f.fileCost,
   250→      Supplier: f.supplier || "",
   251→      "Present Position": f.presentPosition,
   252→      "Responsible Person": f.responsiblePerson || "",
   253→      Category: f.fileCategory,
   254→    }));
   255→    const headers = Object.keys(rows[0] || { "PR Number": "" });
   256→    const csv = [
   257→      headers.join(","),
   258→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h] ?? ""}"`).join(",")),
   259→    ].join("\n");
   260→    const blob = new Blob([csv], { type: "text/csv" });
   261→    const url = URL.createObjectURL(blob);
   262→    const a = document.createElement("a");
   263→    a.href = url;
   264→    a.download = `${project.name}-files.csv`;
   265→    a.click();
   266→    URL.revokeObjectURL(url);
   267→  };
   268→
   269→  // pick the first processed estimate (or any) for the "PR from Estimate" tab
   270→  const estimateForTab = estimates?.find((e) => e.status === "Processed") || estimates?.[0] || null;
   271→
   272→  if (!open) return null;
   273→
   274→  return (
   275→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   276→      {/* Header */}
   277→      <div className="border-b border-border bg-card shrink-0">
   278→        <div className="px-4 sm:px-6 py-3 flex items-center justify-between gap-4">
   279→          <div className="flex items-center gap-3 min-w-0">
   280→            <button onClick={() => onOpenChange(false)} className="text-muted-foreground hover:text-foreground shrink-0">
   281→              <X className="h-5 w-5" />
   282→            </button>
   283→            <div className="min-w-0">
   284→              <h2 className="font-semibold text-foreground text-sm sm:text-base truncate">{project.name}</h2>
   285→              <p className="text-xs text-muted-foreground">
   286→                {project.voteNumber}
   287→                {project.benefactor && ` · Benefactor: ${project.benefactor}`}
   288→                {project.reference && ` · Ref: ${project.reference}`}
   289→                {project.endDate && ` · EDC: ${new Date(project.endDate).toLocaleDateString("en-GB")}`}
   290→              </p>
   291→            </div>
   292→          </div>
   293→          <Button variant="outline" size="sm" onClick={() => setAllocOpen(true)} className="gap-1.5 shrink-0">
   294→            <History className="h-4 w-4" /> Allocations
   295→          </Button>
   296→        </div>
   297→        {/* Financial summary bar */}
   298→        <div className="px-4 sm:px-6 pb-3 grid grid-cols-2 sm:grid-cols-4 gap-2">
   299→          <FinBox label="Vote Allocation" value={formatCurrency(project.voteAllocation)} tone="text-blue-400" />
   300→          <FinBox label="Committed" value={formatCurrency(project.committed)} tone="text-amber-400" />
   301→          <FinBox label="Actual Spent" value={formatCurrency(project.actualSpent)} tone="text-red-400" />
   302→          <FinBox label="File Cost (Total)" value={formatCurrency(totalFileCost)} tone="text-emerald-400" />
   303→        </div>
   304→        {/* Tabs */}
   305→        <div className="px-4 sm:px-6 flex items-center gap-1 border-t border-border">
   306→          <TabButton
   307→            active={activeTab === "raised"}
   308→            onClick={() => setActiveTab("raised")}
   309→            icon={<FileText className="h-3.5 w-3.5" />}
   310→            label="PR Raised Files"
   311→            count={files?.length || 0}
   312→          />
   313→          <TabButton
   314→            active={activeTab === "estimate"}
   315→            onClick={() => setActiveTab("estimate")}
   316→            icon={<Sparkles className="h-3.5 w-3.5" />}
   317→            label="PR from Estimate"
   318→            count={pendingEstimateItems.length}
   319→            tone="amber"
   320→          />
   321→        </div>
   322→      </div>
   323→
   324→      {/* Body */}
   325→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   326→        <div className="p-4 sm:p-6 space-y-4">
   327→          {/* Position summary (always visible) */}
   328→          <div>
   329→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   330→              Present Position Summary
   331→            </h4>
   332→            <div className="flex flex-wrap gap-2">
   333→              {positionSummary.length === 0 ? (
   334→                <p className="text-sm text-muted-foreground">No files yet.</p>
   335→              ) : (
   336→                positionSummary.map(([pos, info]) => (
   337→                  <div key={pos} className={cn("flex items-center gap-2 rounded-lg border px-3 py-1.5", positionColor(pos))}>
   338→                    <span className="text-sm font-medium">{pos}</span>
   339→                    <span className="text-xs opacity-80">{info.count} file{info.count > 1 ? "s" : ""} · {formatCurrency(info.cost)}</span>
   340→                  </div>
   341→                ))
   342→              )}
   343→            </div>
   344→          </div>
   345→
   346→          {activeTab === "raised" ? (
   347→            <>
   348→              {/* PR raised summary chips */}
   349→              <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   350→                <MiniStat label="PR Raised (Files)" value={String(files?.length || 0)} sub={formatCurrency(totalFileCost)} tone="text-blue-400" />
   351→                <MiniStat label="Initiated from Estimate" value={String(initiatedCount)} sub="forwarded PRs" tone="text-violet-400" />
   352→                <MiniStat label="Items to Raise PR" value={String(pendingEstimateItems.length)} sub="from estimate" tone="text-amber-400" />
   353→              </div>
   354→
   355→              {/* Toolbar */}
   356→              <div className="flex flex-wrap items-center gap-2">
   357→                <div className="relative flex-1 min-w-[200px]">
   358→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   359→                  <Input value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search PR, file no, item, supplier..." className="pl-9 bg-background border-border" />
   360→                </div>
   361→                <Select value={positionFilter} onValueChange={setPositionFilter}>
   362→                  <SelectTrigger className="w-[180px] bg-background border-border"><SelectValue placeholder="Position" /></SelectTrigger>
   363→                  <SelectContent>
   364→                    <SelectItem value="all">All Positions</SelectItem>
   365→                    {PRESENT_POSITIONS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
   366→                  </SelectContent>
   367→                </Select>
   368→                {/* File Number filter toggle */}
   369→                <div className="inline-flex rounded-md border border-border overflow-hidden bg-background">
   370→                  <FilterPill
   371→                    label={`All (${fileNumberCounts.all})`}
   372→                    active={fileNumberFilter === "all"}
   373→                    onClick={() => setFileNumberFilter("all")}
   374→                  />
   375→                  <FilterPill
   376→                    label={`With File No (${fileNumberCounts.with})`}
   377→                    active={fileNumberFilter === "with"}
   378→                    onClick={() => setFileNumberFilter("with")}
   379→                  />
   380→                  <FilterPill
   381→                    label={`Pending (${fileNumberCounts.without})`}
   382→                    active={fileNumberFilter === "without"}
   383→                    onClick={() => setFileNumberFilter("without")}
   384→                  />
   385→                </div>
   386→                <Button variant="outline" size="sm" onClick={exportExcel} className="gap-1.5"><Download className="h-4 w-4" /> Export</Button>
   387→                <Button size="sm" onClick={() => setAddOpen(true)} className="gap-1.5"><Plus className="h-4 w-4" /> Add File</Button>
   388→              </div>
   389→
   390→              {/* File register table */}
   391→              <div className="border border-border rounded-lg overflow-hidden">
   392→                <div className="overflow-x-auto">
   393→                  <Table>
   394→                    <TableHeader>
   395→                      <TableRow className="hover:bg-transparent border-border">
   396→                        <SortTh label="PR Number" onClick={() => toggleSort("prNumber")} active={sortKey === "prNumber"} dir={sortDir} />
   397→                        <TableHead className="text-muted-foreground">File Number</TableHead>
   398→                        <TableHead className="text-muted-foreground">Item Code</TableHead>
   399→                        <TableHead className="text-muted-foreground">Item Name</TableHead>
   400→                        <TableHead className="text-muted-foreground text-right">Qty</TableHead>
   401→                        <TableHead className="text-muted-foreground text-right">Unit Price</TableHead>
   402→                        <SortTh label="File Cost" onClick={() => toggleSort("fileCost")} active={sortKey === "fileCost"} dir={sortDir} align="right" />
   403→                        <TableHead className="text-muted-foreground">Supplier</TableHead>
   404→                        <TableHead className="text-muted-foreground">Position</TableHead>
   405→                        <TableHead className="text-muted-foreground">Category</TableHead>
   406→                        <TableHead className="text-muted-foreground text-right">Actions</TableHead>
   407→                      </TableRow>
   408→                    </TableHeader>
   409→                    <TableBody>
   410→                      {loading ? (
   411→                        <TableRow><TableCell colSpan={11} className="text-center py-8 text-muted-foreground"><Loader2 className="h-5 w-5 animate-spin mx-auto" /></TableCell></TableRow>
   412→                      ) : filteredFiles.length === 0 ? (
   413→                        <TableRow><TableCell colSpan={11} className="text-center py-8 text-muted-foreground">No files found. Click &quot;Add File&quot; to create one.</TableCell></TableRow>
   414→                      ) : (
   415→                        filteredFiles.map((f) => {
   416→                          const hasFileNumber = !!(f.fileNumber && f.fileNumber.trim());
   417→                          return (
   418→                            <TableRow key={f.id} className="border-border cursor-pointer hover:bg-secondary/40" onClick={() => setLogFile(f)}>
   419→                              <TableCell className="font-mono text-xs text-foreground">{f.prNumber}</TableCell>
   420→                              <TableCell>
   421→                                {hasFileNumber ? (
   422→                                  <span className="inline-flex items-center gap-1 font-mono text-xs text-emerald-400">
   423→                                    <Hash className="h-3 w-3" />
   424→                                    {f.fileNumber}
   425→                                  </span>
   426→                                ) : (
   427→                                  <button
   428→                                    onClick={(e) => { e.stopPropagation(); setUpdateFile(f); }}
   429→                                    className="inline-flex items-center gap-1 text-xs text-amber-400 hover:text-amber-300 underline-offset-2 hover:underline"
   430→                                  >
   431→                                    <Hash className="h-3 w-3" /> Assign File No
   432→                                  </button>
   433→                                )}
   434→                              </TableCell>
   435→                              <TableCell className="font-mono text-xs">{f.itemCode}</TableCell>
   436→                              <TableCell className="text-foreground">{f.itemName}</TableCell>
   437→                              <TableCell className="text-right">{f.qty} {f.unit}</TableCell>
   438→                              <TableCell className="text-right">{formatCurrency(f.unitPrice)}</TableCell>
   439→                              <TableCell className="text-right font-medium text-foreground">{formatCurrency(f.fileCost)}</TableCell>
   440→                              <TableCell className="text-xs">{f.supplier || "—"}</TableCell>
   441→                              <TableCell>
   442→                                <div className="flex flex-col gap-0.5">
   443→                                  <PositionBadge position={f.presentPosition} />
   444→                                  {f.responsiblePerson && (
   445→                                    <span className="inline-flex items-center gap-1 text-[10px] text-muted-foreground">
   446→                                      <User className="h-2.5 w-2.5" />
   447→                                      {f.responsiblePerson}
   448→                                    </span>
   449→                                  )}
   450→                                </div>
   451→                              </TableCell>
   452→                              <TableCell><CategoryBadge category={f.fileCategory} /></TableCell>
   453→                              <TableCell className="text-right">
   454→                                <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   455→                                  <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setUpdateFile(f)} title="Update Position"><Pencil className="h-3.5 w-3.5" /></Button>
   456→                                  <Button variant="ghost" size="icon" className="h-7 w-7 text-red-400 hover:text-red-500" onClick={() => handleDelete(f)}><Trash2 className="h-3.5 w-3.5" /></Button>
   457→                                </div>
   458→                              </TableCell>
   459→                            </TableRow>
   460→                          );
   461→                        })
   462→                      )}
   463→                    </TableBody>
   464→                  </Table>
   465→                </div>
   466→              </div>
   467→              <p className="text-xs text-muted-foreground">{filteredFiles.length} file(s) · Click any row to view full log · Click the pencil to update position &amp; assign file number</p>
   468→            </>
   469→          ) : (
   470→            /* PR from Estimate tab */
   471→            <>
   472→              <div className="bg-secondary/40 rounded-lg p-3 flex items-center gap-2 flex-wrap">
   473→                <FileSignature className="h-4 w-4 text-amber-400" />
   474→                <p className="text-sm text-foreground">
   475→                  Items pending PR initiation from processed estimates. Click <span className="font-medium text-primary">Initiate File</span> to raise a PR — it will be forwarded to the &quot;PR Raised Files&quot; tab.
   476→                </p>
   477→              </div>
   478→              {estimates && estimates.length > 0 ? (
   479→                <div className="space-y-4">
   480→                  {/* estimate selector */}
   481→                  {estimates.length > 1 && (
   482→                    <div className="flex items-center gap-2 flex-wrap">
   483→                      <span className="text-xs text-muted-foreground">Estimate:</span>
   484→                      {estimates.map((e) => (
   485→                        <button
   486→                          key={e.id}
   487→                          onClick={() => setActiveEstimateId(e.id)}
   488→                          className={cn(
   489→                            "px-2.5 py-1 rounded-md text-xs border transition-colors",
   490→                            (activeEstimateId || estimateForTab?.id) === e.id
   491→                              ? "border-primary bg-primary/10 text-primary"

   500→                  <EstimateItemsTable
   501→                    estimateId={activeEstimateId || estimateForTab?.id || estimates[0].id}
   502→                    showProcessButton
   503→                    onChanged={() => { refreshEst(); refresh(); onChanged(); }}

   509→                  title="No estimates for this project"

   511→                />
   512→              )}
   513→            </>
   514→          )}
   515→        </div>
   516→      </div>
   517→
   518→      {/* Add File */}
   519→      <FileFormDialog projectId={project.id} open={addOpen} onOpenChange={setAddOpen} onSaved={() => { setAddOpen(false); refresh(); onChanged(); }} />
   520→      {/* Update Position popup (replaces simple Edit) */}
   521→      {updateFile && (
   522→        <UpdatePositionDialog
   523→          file={updateFile}
   524→          projectId={project.id}
   525→          positionToResponsible={positionToResponsible}
   526→          open={!!updateFile}

   533→      {/* Allocation History */}
   534→      <AllocationHistoryDialog project={project} open={allocOpen} onOpenChange={setAllocOpen} onChanged={onChanged} />
   535→    </div>

   695→// ─── Update Position Dialog (dedicated edit popup with next-position + responsible person) ───
   696→function UpdatePositionDialog({
   697→  file,
   698→  projectId,
   699→  positionToResponsible,
   700→  open,
   701→  onOpenChange,
   702→  onSaved,
   703→}: {
   704→  file: ProjectFile;
   705→  projectId: string;
   706→  positionToResponsible: Map<string, string>;
   707→  open: boolean;
   708→  onOpenChange: (o: boolean) => void;
   709→  onSaved: () => void;
   710→}) {
   711→  const currentPosition = file.presentPosition;
   712→  const [nextPosition, setNextPosition] = useState(currentPosition);
   713→  const [responsiblePerson, setResponsiblePerson] = useState(file.responsiblePerson || "");
   714→  const [fileNumber, setFileNumber] = useState(file.fileNumber || "");
   715→  const [supplier, setSupplier] = useState(file.supplier || "");
   716→  const [changedBy, setChangedBy] = useState("");
   717→  const [comment, setComment] = useState("");
   718→  const [saving, setSaving] = useState(false);
   719→  const { toast } = useToast();
   720→
   721→  // When the user picks a different next position, prefill the responsible person
   722→  // from the process-flow defaults — but only if the user hasn't manually overridden.
   723→  const defaultForCurrent = positionToResponsible.get(currentPosition) || "";
   724→  const defaultForNext = positionToResponsible.get(nextPosition) || "";
   725→
   726→  const handleNextPositionChange = (v: string) => {
   727→    setNextPosition(v);
   728→    // Auto-fill responsible person from the new position's default — only if user hasn't overridden
   729→    const isUsingDefault = responsiblePerson === defaultForCurrent || responsiblePerson === "";
   730→    if (isUsingDefault) {
   731→      const newDefault = positionToResponsible.get(v) || "";
   732→      setResponsiblePerson(newDefault);
   733→    }
   734→  };
   735→
   736→  const positionChanged = nextPosition !== currentPosition;
   737→
   738→  const submit = async () => {
   739→    if (positionChanged && !changedBy.trim()) {
   740→      toast({ title: "Changed By is required when updating position", variant: "destructive" });
   741→      return;
   742→    }
   743→    setSaving(true);
   744→    const body: Record<string, unknown> = {
   745→      // Always allow file number + supplier updates
   746→      fileNumber: fileNumber.trim() || null,
   747→      supplier: supplier.trim() || null,
   748→      // Position update
   749→      presentPosition: nextPosition,
   750→      responsiblePerson: responsiblePerson.trim() || null,
   751→      changedBy: changedBy.trim() || "Officer",
   752→      comment: comment.trim() || (positionChanged ? `Position changed: ${currentPosition} → ${nextPosition}` : "File metadata updated"),
   753→    };
   754→    const res = await apiRequest(`/api/projects/${projectId}/files/${file.id}`, "PUT", body);
   755→    setSaving(false);
   756→    if (res.ok) {
   757→      toast({
   758→        title: positionChanged ? "Position updated" : "File updated",
   759→        description: positionChanged ? `${currentPosition} → ${nextPosition}` : undefined,
   760→      });
   761→      onSaved();
   762→    } else {
   763→      toast({ title: "Failed to update", description: res.error, variant: "destructive" });
   764→    }
   765→  };
   766→
   767→  return (
   768→    <Dialog open={open} onOpenChange={onOpenChange}>
   769→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
   770→        <DialogHeader>
   771→          <DialogTitle className="text-foreground flex items-center gap-2">
   772→            <Pencil className="h-5 w-5 text-primary" />
   773→            Update Position — {file.prNumber}
   774→          </DialogTitle>
   775→          <DialogDescription className="text-muted-foreground">
   776→            {file.itemName} · {file.itemCode}
   777→          </DialogDescription>
   778→        </DialogHeader>
   779→
   780→        <div className="space-y-5">
   781→          {/* Current → Next position display */}
   782→          <div className="bg-secondary/40 rounded-lg p-4">
   783→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-3 flex items-center gap-1.5">
   784→              <ArrowRight className="h-3.5 w-3.5" /> Position Transition
   785→            </h4>
   786→            <div className="grid grid-cols-1 sm:grid-cols-[1fr_auto_1fr] gap-3 items-center">
   787→              {/* Current */}
   788→              <div className="space-y-1.5">
   789→                <Label className="text-xs text-muted-foreground">Current Position</Label>
   790→                <div className={cn("rounded-lg border px-3 py-2 text-sm font-medium", positionColor(currentPosition))}>
   791→                  {currentPosition}
   792→                </div>
   793→                <p className="text-[11px] text-muted-foreground">
   794→                  Default officer: <span className="text-foreground">{defaultForCurrent || "—"}</span>
   795→                </p>
   796→              </div>
   797→              {/* Arrow */}
   798→              <div className="flex items-center justify-center">
   799→                <ArrowRight className="h-5 w-5 text-primary rotate-90 sm:rotate-0" />
   800→              </div>
   801→              {/* Next */}
   802→              <div className="space-y-1.5">
   803→                <Label className="text-xs text-muted-foreground">Next Position *</Label>
   804→                <Select value={nextPosition} onValueChange={handleNextPositionChange}>
   805→                  <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   806→                  <SelectContent>
   807→                    {PRESENT_POSITIONS.map((p) => (
   808→                      <SelectItem key={p} value={p}>{p}</SelectItem>
   809→                    ))}
   810→                  </SelectContent>
   811→                </Select>
   812→                <p className="text-[11px] text-muted-foreground">
   813→                  Default officer: <span className="text-foreground">{defaultForNext || "—"}</span>
   814→                </p>
   815→              </div>
   816→            </div>
   817→            {!positionChanged && (
   818→              <p className="text-[11px] text-amber-400 mt-3 flex items-center gap-1">
   819→                <ArrowRight className="h-3 w-3" />
   820→                Pick a different next position to log a position change.
   821→              </p>
   822→            )}
   823→          </div>
   824→
   825→          {/* Responsible Person (auto-filled, editable) */}
   826→          <div className="bg-secondary/40 rounded-lg p-4 space-y-3">
   827→            <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide flex items-center gap-1.5">
   828→              <UserCheck className="h-3.5 w-3.5" /> Responsible Person
   829→            </h4>
   830→            <Field label="Responsible Person (editable — defaults from Process Flow)">
   831→              <Input
   832→                value={responsiblePerson}
   833→                onChange={(e) => setResponsiblePerson(e.target.value)}
   834→                placeholder={defaultForNext || "e.g. Cdr. R. Perera"}
   835→                className="bg-background border-border"
   836→              />
   837→            </Field>
   838→            <p className="text-[11px] text-muted-foreground flex items-start gap-1">
   839→              <User className="h-3 w-3 mt-0.5 shrink-0" />
   840→              Auto-filled from the Process Flow defaults for the selected next position. Override if a different person is responsible.
   841→            </p>
   842→          </div>
   843→
   844→          {/* File Number + Supplier */}
   845→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   846→            <Field label="File Number (received after initial RL)">
   847→              <Input
   848→                value={fileNumber}
   849→                onChange={(e) => setFileNumber(e.target.value)}
   850→                placeholder="e.g. FILE/2025/001"
   851→                className="bg-background border-border"
   852→              />
   853→            </Field>
   854→            <Field label="Supplier">
   855→              <Input
   856→                value={supplier}
   857→                onChange={(e) => setSupplier(e.target.value)}
   858→                placeholder="Supplier name"
   859→                className="bg-background border-border"
   860→              />
   861→            </Field>
   862→          </div>
   863→
   864→          {/* Change log fields */}

   937→    <Dialog open={open} onOpenChange={onOpenChange}>
   938→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">

   946→          <FinBox label="Total Allocated" value={formatCurrency(totalAllocated)} tone="text-blue-400" />
   947→          <FinBox label="Committed" value={formatCurrency(committed)} tone="text-amber-400" />
