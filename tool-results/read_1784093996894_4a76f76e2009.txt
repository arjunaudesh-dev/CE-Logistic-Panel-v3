   257→function formatDate(value: string | null | undefined): string {
   258→  if (!value) return "—";
   259→  const d = new Date(value);
   260→  if (isNaN(d.getTime())) return "—";
   261→  return d.toLocaleString("en-GB", {
   262→    day: "2-digit",
   263→    month: "short",
   264→    year: "numeric",
   265→  });
   266→}

   488→    null

   492→    if (!list) return [];

   495→      if (roleFilter !== "all" && u.role !== roleFilter) return false;
   496→      if (deptFilter !== "all" && (u.department ?? "—") !== deptFilter) return false;

   501→          `${u.username} ${u.displayName} ${u.email ?? ""} ${u.department ?? ""}`.toLowerCase();
