   130→const PIE_STATUS_COLORS: Record<string, string> = COLLECTION_STATUSES.reduce(
   131→  (acc, status, i) => {

   139→function todayISO(): string {
   140→  return new Date().toISOString().slice(0, 10);
   141→}
   142→function yesterdayISO(): string {
   143→  return new Date(Date.now() - 86400000).toISOString().slice(0, 10);
   144→}
   145→function getLast7Days(): string[] {
   146→  const days: string[] = [];
   147→  const t = new Date();
   148→  for (let i = 6; i >= 0; i--) {
   149→    const d = new Date(t);
   150→    d.setDate(d.getDate() - i);
   151→    days.push(d.toISOString().slice(0, 10));
   152→  }
   153→  return days;
   154→}
   155→function formatDateLabel(dateStr: string): string {
   156→  const d = new Date(dateStr + "T00:00:00");
   157→  return d.toLocaleDateString("en-IN", { day: "2-digit", month: "short" });
   158→}
   159→function formatFullDate(dateStr?: string | null): string {
   160→  if (!dateStr) return "—";
   161→  const d = new Date(dateStr + "T00:00:00");
   162→  if (isNaN(d.getTime())) return dateStr;
   163→  return d.toLocaleDateString("en-IN", {
   164→    day: "2-digit",
   165→    month: "short",
   166→    year: "numeric",
   167→  });
   168→}
   169→function formatDateTime(iso: string): string {
   170→  const d = new Date(iso);
   171→  if (isNaN(d.getTime())) return iso;
   172→  return d.toLocaleString("en-IN", {
   173→    day: "2-digit",
   174→    month: "short",
   175→    year: "numeric",
   176→    hour: "2-digit",
   177→    minute: "2-digit",
   178→  });
   179→}
   180→function progressColor(pct: number): string {
   181→  if (pct > 90) return "bg-emerald-500";
   182→  if (pct >= 50) return "bg-amber-500";
   183→  return "bg-red-500";
   184→}
   185→function statusColor(status: string): string {
   186→  if (
   187→    status === "Collected" ||
   188→    (PART_SUPPLY_STATUSES as readonly string[]).includes(status) &&
   189→      status === "Completed"
   190→  ) {
   191→    return "bg-emerald-500/15 text-emerald-400 border-emerald-500/30";
   192→  }
   193→  if (status === "In Progress" || status === "Active") {
   194→    return "bg-blue-500/15 text-blue-400 border-blue-500/30";
   195→  }
   196→  if (status === "Pending") {
   197→    return "bg-amber-500/15 text-amber-400 border-amber-500/30";
   198→  }
   199→  return "bg-slate-500/15 text-slate-300 border-slate-500/30";
   200→}

   242→function DashboardTab() {
   243→  const { data: stats } = useFetch<Stats>("/api/stats");
   244→  const {
   245→    data: items,
   246→    loading: itemsLoading,
   247→    error: itemsError,
   248→  } = useFetch<CollectionItem[]>("/api/collection/items");
   249→  const { data: partSupply } = useFetch<PartSupplyOrder[]>(
   250→    "/api/collection/part-supply"
   251→  );
   252→  const { data: daily } = useFetch<DailyCollection[]>(
   253→    "/api/collection/daily"
   254→  );
   255→
   256→  // Stat figures
   257→  const pendingItems = (items ?? []).filter(
   258→    (i) => i.status === "Pending" || i.status === "In Progress"
   259→  );
   260→  const pendingValue = pendingItems.reduce(
   261→    (s, i) => s + (i.expectedQty || 0),
   262→    0
   263→  );
   264→  const activePartSupply =
   265→    stats?.collection?.partSupplyActive ??
   266→    (partSupply ?? []).filter((o) => o.status === "Active").length;
   267→  const collectedItems = (items ?? []).filter(
   268→    (i) => i.status === "Collected"
   269→  );
   270→  const todayStr = todayISO();
   271→  const todayDailyCount = (daily ?? []).filter(
   272→    (d) => d.date === todayStr
   273→  ).length;
   274→
   275→  // Trend chart — last 7 days
   276→  const trendData = useMemo(() => {
   277→    const days = getLast7Days();
   278→    const map = new Map<string, number>();
   279→    (daily ?? []).forEach((d) => {
   280→      map.set(d.date, (map.get(d.date) || 0) + (d.qty || 0));
   281→    });
   282→    return days.map((date) => ({
   283→      name: formatDateLabel(date),
   284→      qty: map.get(date) || 0,
   285→    }));
   286→  }, [daily]);
   287→
   288→  // Status breakdown
   289→  const statusBreakdown = useMemo(() => {
   290→    const counts: Record<string, number> = {};
   291→    (items ?? []).forEach((i) => {
   292→      counts[i.status] = (counts[i.status] || 0) + 1;
   293→    });
   294→    // ensure stable order using COLLECTION_STATUSES
   295→    return COLLECTION_STATUSES.filter((s) => counts[s] > 0).map((name) => ({
   296→      name,
   297→      value: counts[name],
   298→    }));
   299→  }, [items]);
   300→
   301→  const recentDaily = (daily ?? []).slice(0, 5);
   302→
   303→  if (itemsLoading) return <LoadingState label="Loading dashboard..." />;
   304→  if (itemsError) return <ErrorState message={itemsError} />;
   305→
   306→  return (
   307→    <div className="space-y-5">
   308→      {/* Stat cards */}
   309→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   310→        <StatCard
   311→          label="Items to be Received"
   312→          value={pendingItems.length}
   313→          icon={PackageX}
   314→          accent="blue"
   315→          sub={`${formatNumber(pendingValue)} qty expected`}
   316→        />
   317→        <StatCard
   318→          label="Part Supply Items"
   319→          value={activePartSupply}
   320→          icon={Layers}
   321→          accent="amber"
   322→          sub="Active part-supply orders"
   323→        />
   324→        <StatCard
   325→          label="Collected Items"
   326→          value={collectedItems.length}
   327→          icon={PackageCheck}
   328→          accent="green"
   329→          sub={`${todayDailyCount} collected today`}
   330→        />
   331→      </div>
   332→
   333→      {/* Charts */}
   334→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   335→        <div className="lg:col-span-2 bg-card border border-border rounded-xl p-5">
   336→          <h3 className="text-sm font-semibold text-foreground mb-4">
   337→            Collection Trend — Last 7 Days
   338→          </h3>
   339→          {trendData.length > 0 ? (
   340→            <ResponsiveContainer width="100%" height={280}>
   341→              <LineChart
   342→                data={trendData}
   343→                margin={{ top: 5, right: 16, left: 0, bottom: 5 }}
   344→              >
   345→                <CartesianGrid
   346→                  strokeDasharray="3 3"
   347→                  stroke="#2d3548"
   348→                  vertical={false}
   349→                />
   350→                <XAxis
   351→                  dataKey="name"
   352→                  tick={{ fill: "#94a3b8", fontSize: 11 }}
   353→                  axisLine={{ stroke: "#2d3548" }}
   354→                  tickLine={{ stroke: "#2d3548" }}
   355→                />
   356→                <YAxis
   357→                  tick={{ fill: "#94a3b8", fontSize: 11 }}
   358→                  axisLine={{ stroke: "#2d3548" }}
   359→                  tickLine={{ stroke: "#2d3548" }}
   360→                />
   361→                <Tooltip
   362→                  contentStyle={{
   363→                    background: "#1a1f2e",
   364→                    border: "1px solid #2d3548",
   365→                    borderRadius: 8,
   366→                    fontSize: 12,
   367→                  }}
   368→                />
   369→                <Line
   370→                  type="monotone"
   371→                  dataKey="qty"
   372→                  name="Qty Collected"
   373→                  stroke="#3b82f6"
   374→                  strokeWidth={2}
   375→                  dot={{ fill: "#3b82f6", r: 3 }}
   376→                  activeDot={{ r: 5 }}
   377→                />
   378→              </LineChart>
   379→            </ResponsiveContainer>
   380→          ) : (
   381→            <p className="text-sm text-muted-foreground py-10 text-center">
   382→              No collection data
   383→            </p>
   384→          )}
   385→        </div>
   386→
   387→        <div className="bg-card border border-border rounded-xl p-5">
   388→          <h3 className="text-sm font-semibold text-foreground mb-4">
   389→            Items by Status
   390→          </h3>
   391→          {statusBreakdown.length > 0 ? (
   392→            <ResponsiveContainer width="100%" height={280}>
   393→              <PieChart>
   394→                <Pie
   395→                  data={statusBreakdown}
   396→                  dataKey="value"
   397→                  nameKey="name"
   398→                  cx="50%"
   399→                  cy="50%"
   400→                  outerRadius={90}
   401→                  innerRadius={45}
   402→                  paddingAngle={2}
   403→                >
   404→                  {statusBreakdown.map((entry, i) => (
   405→                    <Cell
   406→                      key={i}
   407→                      fill={
   408→                        PIE_STATUS_COLORS[entry.name] || DEFAULT_PIE_COLOR
   409→                      }
   410→                    />
   411→                  ))}
   412→                </Pie>
   413→                <Tooltip
   414→                  contentStyle={{
   415→                    background: "#1a1f2e",
   416→                    border: "1px solid #2d3548",
   417→                    borderRadius: 8,
   418→                    fontSize: 12,
   419→                  }}
   420→                />
   421→                <Legend wrapperStyle={{ fontSize: 10 }} />
   422→              </PieChart>
   423→            </ResponsiveContainer>
   424→          ) : (
   425→            <p className="text-sm text-muted-foreground py-10 text-center">
   426→              No items

   429→        </div>
   430→      </div>
   431→
   432→      {/* Recent activity */}
   433→      <div className="bg-card border border-border rounded-xl p-5">
   434→        <div className="flex items-center justify-between mb-4">
   435→          <h3 className="text-sm font-semibold text-foreground">
   436→            Recent Collection Activity
   437→          </h3>
   438→          <CustomBadge className="bg-secondary border-border text-muted-foreground">
   439→            Last 5
   440→          </CustomBadge>
   441→        </div>
   442→        {recentDaily.length === 0 ? (
   443→          <EmptyState
   444→            icon={CalendarDays}
   445→            title="No recent collections"
   446→            description="Daily collections will appear here once recorded."
   447→          />
   448→        ) : (
   449→          <ul className="space-y-3">
   450→            {recentDaily.map((d) => (
   451→              <li
   452→                key={d.id}
   453→                className="flex items-start gap-3 pb-3 border-b border-border last:border-0 last:pb-0"
   454→              >
   455→                <div className="h-9 w-9 rounded-lg bg-emerald-500/10 text-emerald-400 flex items-center justify-center shrink-0">
   456→                  <PackageCheck className="h-4 w-4" />
   457→                </div>
   458→                <div className="min-w-0 flex-1">
   459→                  <p className="text-sm font-medium text-foreground truncate">
   460→                    {d.itemName}
   461→                  </p>
   462→                  <p className="text-xs text-muted-foreground">
   463→                    {formatNumber(d.qty)} qty · {d.receivedFrom || "—"} ·{" "}
   464→                    {formatFullDate(d.date)}
   465→                  </p>
   466→                </div>
   467→                <span className="text-xs text-muted-foreground shrink-0 hidden sm:inline">
   468→                  {formatDateTime(d.recordedAt)}
   469→                </span>
   470→              </li>
   471→            ))}
   472→          </ul>
   473→        )}
   474→      </div>
   475→    </div>
   476→  );
   477→}

  1677→function getISOWeek(dateStr: string): number {
  1678→  const d = new Date(dateStr + "T00:00:00");
  1679→  d.setHours(0, 0, 0, 0);
  1680→  // Thursday in current week decides the year
  1681→  d.setDate(d.getDate() + 3 - ((d.getDay() + 6) % 7));
  1682→  // January 4 is always in week 1
  1683→  const week1 = new Date(d.getFullYear(), 0, 4);
  1684→  return (
  1685→    1 +
  1686→    Math.round(
  1687→      ((d.getTime() - week1.getTime()) / 86400000 -
  1688→        3 +
  1689→        ((week1.getDay() + 6) % 7)) /
  1690→        7
  1691→    )
  1692→  );
  1693→}
  1694→
  1695→function getWeekRange(dateStr: string): { key: string; label: string } {
  1696→  const d = new Date(dateStr + "T00:00:00");
  1697→  const dayOfWeek = (d.getDay() + 6) % 7; // 0 = Monday
  1698→  const monday = new Date(d);
  1699→  monday.setDate(d.getDate() - dayOfWeek);
  1700→  const sunday = new Date(monday);
  1701→  sunday.setDate(monday.getDate() + 6);
  1702→  const weekNum = getISOWeek(dateStr);
  1703→  const year = monday.getFullYear();
  1704→  const fmt = (dt: Date) =>
  1705→    dt.toLocaleDateString("en-IN", { day: "2-digit", month: "short" });
  1706→  return {
  1707→    key: `${year}-W${String(weekNum).padStart(2, "0")}`,
  1708→    label: `Week ${weekNum} · ${fmt(monday)} – ${fmt(sunday)}`,
  1709→  };
  1710→}

  1712→function groupByPeriod(
  1713→  rows: DailyCollection[]
  1714→): [string, DailyCollection[]][] {
  1715→  const map = new Map<string, DailyCollection[]>();
  1716→  const labels = new Map<string, string>();
  1717→  rows.forEach((r) => {
  1718→    const { key, label } = getWeekRange(r.date);
  1719→    if (!map.has(key)) {
  1720→      map.set(key, []);
  1721→      labels.set(key, label);
  1722→    }
  1723→    map.get(key)!.push(r);
  1724→  });
  1725→  return Array.from(map.entries())
  1726→    .sort(([a], [b]) => b.localeCompare(a))
  1727→    .map(
  1728→      ([key, val]) => [labels.get(key) || key, val] as [string, DailyCollection[]]
  1729→    );
  1730→}

  1734→): [string, DailyCollection[]][] {

  1738→    if (!map.has(k)) map.set(k, []);
  1739→    map.get(k)!.push(r);
  1740→  });
  1741→  return Array.from(map.entries()).sort(([a], [b]) => a.localeCompare(b));
  1742→}
