   199→/** Strip everything except digits from a phone string. */
   200→function digits(num: string | null | undefined): string {
   201→  if (!num) return "";
   202→  return num.replace(/[^\d]/g, "");
   203→}
   204→
   205→/** Build a tel: link (digits only, keep simple). */
   206→function telLink(num: string | null | undefined): string | null {
   207→  const d = digits(num);
   208→  return d ? `tel:${d}` : null;
   209→}
   210→
   211→/**
   212→ * Build a wa.me link.
   213→ * Sri Lankan numbers starting with 0 are converted to +94.
   214→ * Otherwise, return the digits as-is.
   215→ */
   216→function waMeLink(num: string | null | undefined): string | null {
   217→  const d = digits(num);
   218→  if (!d) return null;
   219→  if (d.startsWith("0")) {
   220→    return `https://wa.me/94${d.slice(1)}`;
   221→  }
   222→  return `https://wa.me/${d}`;
   223→}
   224→
   225→/** Bar color for a rating score. */
   226→function ratingBarColor(score: number): string {
   227→  if (score >= 4.5) return "#10b981"; // emerald
   228→  if (score >= 3.5) return "#3b82f6"; // blue
   229→  if (score >= 2.5) return "#f59e0b"; // amber
   230→  return "#ef4444"; // red
   231→}
   232→
   233→/** Color for an individual rating category progress bar (matches ratingColor thresholds). */
   234→function ratingProgressClass(score: number): string {
   235→  if (score >= 4.5) return "bg-emerald-500";
   236→  if (score >= 3.5) return "bg-blue-500";
   237→  if (score >= 2.5) return "bg-amber-500";
   238→  return "bg-red-500";
   239→}

   241→function formatDate(value: string | null | undefined): string {
   242→  if (!value) return "—";
   243→  const d = new Date(value);
   244→  if (isNaN(d.getTime())) return "—";
   245→  return d.toLocaleString("en-GB", {
   246→    day: "2-digit",
   247→    month: "short",
   248→    year: "numeric",
   249→    hour: "2-digit",
   250→    minute: "2-digit",
   251→  });
   252→}
   253→
   254→/**
   255→ * Resolve a stakeholder's category, falling back to department + role when the
   256→ * legacy "Officer" / "Sailor" / null value is present (pre-categorisation data).
   257→ * Returns one of STAKEHOLDER_CATEGORIES.
   258→ */
   259→function resolveCategory(p: {
   260→  category?: string | null;
   261→  department?: string | null;
   262→  role?: string | null;
   263→}): string {
   264→  const cat = p.category;
   265→  if (cat && (STAKEHOLDER_CATEGORIES as readonly string[]).includes(cat)) {
   266→    return cat;
   267→  }
   268→  // Legacy fallback
   269→  const dept = p.department ?? "CSLOG";
   270→  const role = p.role ?? "Officer";
   271→  if (dept === "CE") return role === "Sailor" ? "CCED Sailor" : "CCED Officer";
   272→  return role === "Sailor" ? "CSLOG Sailor" : "CSLOG Officer";
   273→}
   274→
   275→/** Whether a category is a sailor category (for table column display). */
   276→function isSailorCategory(cat: string): boolean {
   277→  return cat === "CCED Sailor" || cat === "CSLOG Sailor";
   278→}

   998→      ? initial.category
   999→      : initial?.department === "CE"
  1000→        ? initial?.role === "Sailor"
  1001→          ? "CCED Sailor"
  1002→          : "CCED Officer"
  1003→        : initial?.role === "Sailor"

  1012→  );
