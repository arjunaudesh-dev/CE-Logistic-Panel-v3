    14→} from "@/components/ui/select";

    19→  DialogTitle,
    20→  DialogFooter,
    21→} from "@/components/ui/dialog";
    22→import { CustomBadge } from "@/components/erp/badges";
    23→import { useToast } from "@/hooks/use-toast";
    24→import {
    25→  PROJECT_STATUSES,
    26→  PROJECT_TYPES,
    27→  FINANCIAL_MEANS,
    28→  projectTypeColor,
    29→  financialMeanColor,
    30→} from "@/lib/constants";
    31→import { EstimateItemsTable } from "./estimate-items";
    32→import {
    33→  Loader2,

    46→  Info,
    47→} from "lucide-react";

    49→
    50→interface EstimateRow {
    51→  itemName: string;
    52→  qty: string;
    53→  unit: string;
    54→  estimateCost: string;
    55→}
    56→
    57→interface Props {
    58→  open: boolean;
    59→  onClose: () => void;
    60→  onCreated: (project: { id: string; name: string; voteNumber: string; voteAllocation: number; committed: number; actualSpent: number; benefactor: string | null; reference: string | null; endDate: string | null; startDate: string | null; projectType: string; financialMean: string | null }) => void;
    61→}
    62→
    63→const UNITS = ["NOS", "KG", "MTR", "LTR", "SHT", "SET", "BAG", "PR"];
    64→
    65→export function AddProjectWizard({ open, onClose, onCreated }: Props) {
    66→  const [step, setStep] = useState<1 | 2>(1);
    67→  const [saving, setSaving] = useState(false);
    68→  const [createdProjectId, setCreatedProjectId] = useState<string | null>(null);
    69→  const [createdEstimateId, setCreatedEstimateId] = useState<string | null>(null);
    70→  const { toast } = useToast();
    71→  const fileInputRef = useRef<HTMLInputElement>(null);
    72→  const [previewRows, setPreviewRows] = useState<EstimateRow[] | null>(null);
    73→
    74→  const [form, setForm] = useState({
    75→    name: "",
    76→    voteNumber: "",
    77→    voteAllocation: "",
    78→    committed: "0",
    79→    actualSpent: "0",
    80→    startDate: "",
    81→    endDate: "", // EDC
    82→    benefactor: "", // replaces Department
    83→    reference: "", // manual text — how vote allocated
    84→    projectType: "New Construction",
    85→    financialMean: "Vote",
    86→    projectOfficer: "",
    87→    status: "Active",
    88→    description: "",
    89→  });
    90→  const [estimateRows, setEstimateRows] = useState<EstimateRow[]>([
    91→    { itemName: "", qty: "", unit: "NOS", estimateCost: "" },
    92→  ]);
    93→
    94→  const addRow = () =>
    95→    setEstimateRows([...estimateRows, { itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
    96→  const removeRow = (idx: number) =>
    97→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
    98→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
    99→    setEstimateRows(estimateRows.map((r, i) => (i === idx ? { ...r, [field]: value } : r)));
   100→
   101→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   102→  const validEstimateRows = estimateRows.filter(
   103→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   104→  );
   105→  const hasValidEstimate = validEstimateRows.length > 0;
   106→
   107→  const isNonVote = form.financialMean && form.financialMean !== "Vote";
   108→
   109→  const downloadTemplate = () => {
   110→    const header = "ItemName,Qty,Unit,EstimateCost";
   111→    const sample = "Centrifugal Pump 15kW,2,NOS,185000";
   112→    const csv = `${header}\n${sample}\n`;
   113→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   114→    const url = URL.createObjectURL(blob);
   115→    const a = document.createElement("a");
   116→    a.href = url;
   117→    a.download = "material-estimate-template.csv";
   118→    a.click();
   119→    URL.revokeObjectURL(url);
   120→    toast({ title: "Template downloaded", description: "Fill in ItemName, Qty, Unit, EstimateCost" });
   121→  };
   122→
   123→  // Simple CSV parser — handles quoted fields with commas
   124→  const parseCsv = (text: string): string[][] => {
   125→    const rows: string[][] = [];
   126→    let cur: string[] = [];
   127→    let field = "";
   128→    let inQuotes = false;
   129→    for (let i = 0; i < text.length; i++) {
   130→      const ch = text[i];
   131→      if (inQuotes) {
   132→        if (ch === '"' && text[i + 1] === '"') {
   133→          field += '"';
   134→          i++;
   135→        } else if (ch === '"') {
   136→          inQuotes = false;
   137→        } else {
   138→          field += ch;
   139→        }
   140→      } else {
   141→        if (ch === '"') inQuotes = true;
   142→        else if (ch === ",") {
   143→          cur.push(field);
   144→          field = "";
   145→        } else if (ch === "\n" || ch === "\r") {
   146→          if (ch === "\r" && text[i + 1] === "\n") i++;
   147→          cur.push(field);
   148→          rows.push(cur);
   149→          cur = [];
   150→          field = "";
   151→        } else {
   152→          field += ch;
   153→        }
   154→      }
   155→    }
   156→    if (field.length > 0 || cur.length > 0) {
   157→      cur.push(field);
   158→      rows.push(cur);
   159→    }
   160→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   161→  };
   162→
   163→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   164→    const file = e.target.files?.[0];
   165→    if (!file) return;
   166→    if (!file.name.toLowerCase().endsWith(".csv")) {
   167→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   168→      e.target.value = "";
   169→      return;
   170→    }
   171→    try {
   172→      const text = await file.text();
   173→      const rows = parseCsv(text);
   174→      if (rows.length === 0) {
   175→        toast({ title: "CSV file is empty", variant: "destructive" });
   176→        e.target.value = "";
   177→        return;
   178→      }
   179→      const header = rows[0].map((c) => c.trim().toLowerCase());
   180→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   181→      const idxQty = header.findIndex((h) => h.includes("qty"));
   182→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   183→      const idxCost = header.findIndex((h) => h.includes("cost") || h.includes("price"));
   184→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   185→      const parsed: EstimateRow[] = dataRows
   186→        .map((r) => ({
   187→          itemName: idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim(),
   188→          qty: idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim(),
   189→          unit: (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS",
   190→          estimateCost: idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim(),
   191→        }))
   192→        .filter((r) => r.itemName);
   193→      if (parsed.length === 0) {
   194→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   195→        e.target.value = "";
   196→        return;
   197→      }
   198→      setPreviewRows(parsed);
   199→    } catch {
   200→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   201→    }
   202→    e.target.value = "";
   203→  };
   204→
   205→  const confirmImport = () => {
   206→    if (!previewRows) return;
   207→    setEstimateRows(previewRows);
   208→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   209→    setPreviewRows(null);
   210→  };
   211→
   212→  const handleSaveAndProcess = async () => {
   213→    if (!form.name || !form.voteNumber) {
   214→      toast({ title: "Project Name and Vote Number are required", variant: "destructive" });
   215→      return;
   216→    }
   217→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   218→    if (validRows.length === 0) {
   219→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   220→      return;
   221→    }
   222→    setSaving(true);
   223→    // 1. Create project
   224→    const projRes = await apiRequest("/api/projects", "POST", form);
   225→    if (!projRes.ok) {
   226→      setSaving(false);
   227→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   228→      return;
   229→    }
   230→    const project = projRes.data as { id: string };
   231→    // 2. Export Estimate into DB
   232→    const estRes = await apiRequest("/api/estimates", "POST", {
   233→      projectId: project.id,
   234→      name: `${form.name} — Material Estimate`,
   235→      items: validRows.map((r) => ({
   236→        itemName: r.itemName,
   237→        qty: Number(r.qty) || 0,
   238→        unit: r.unit,
   239→        estimateCost: Number(r.estimateCost) || 0,
   240→      })),
   241→    });
   242→    setSaving(false);
   243→    if (estRes.ok) {
   244→      const est = estRes.data as { id: string };
   245→      setCreatedProjectId(project.id);
   246→      setCreatedEstimateId(est.id);
   247→      setStep(2);
   248→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   249→    } else {
   250→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   251→    }
   252→  };
   253→
   254→  const handleGoToProject = () => {
   255→    if (createdProjectId) {
   256→      onCreated({
   257→        id: createdProjectId,
   258→        name: form.name,
   259→        voteNumber: form.voteNumber,
   260→        voteAllocation: Number(form.voteAllocation) || 0,
   261→        committed: Number(form.committed) || 0,
   262→        actualSpent: Number(form.actualSpent) || 0,
   263→        benefactor: form.benefactor || null,
   264→        reference: form.reference || null,
   265→        endDate: form.endDate || null,
   266→        startDate: form.startDate || null,
   267→        projectType: form.projectType,
   268→        financialMean: form.financialMean,
   269→      });
   270→      // reset
   271→      setStep(1);
   272→      setCreatedProjectId(null);
   273→      setCreatedEstimateId(null);
   274→      setForm({
   275→        name: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   276→        startDate: "", endDate: "", benefactor: "", reference: "",
   277→        projectType: "New Construction", financialMean: "Vote",
   278→        projectOfficer: "", status: "Active", description: "",
   279→      });
   280→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
   281→    }
   282→  };
   283→
   284→  if (!open) return null;
   285→
   286→  return (
   287→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   288→      {/* Header */}
   289→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   290→        <div className="flex items-center gap-3">
   291→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   292→            <X className="h-5 w-5" />
   293→          </button>
   294→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   295→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   296→          </h2>
   297→        </div>
   298→        {/* Stepper */}
   299→        <div className="hidden sm:flex items-center gap-2 text-xs">
   300→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   301→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   302→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   303→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   304→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   305→        </div>
   306→      </div>
   307→
   308→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   309→        {step === 1 && (
   310→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   311→            {/* Project details */}
   312→            <section className="bg-card border border-border rounded-xl p-5">
   313→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   314→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   315→              </h3>
   316→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   317→                <div className="sm:col-span-2 space-y-1.5">
   318→                  <Label>Project Name *</Label>
   319→                  <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
   320→                </div>
   321→                <div className="space-y-1.5">
   322→                  <Label>Vote Number *</Label>
   323→                  <Input
   324→                    value={form.voteNumber}
   325→                    onChange={(e) => setForm({ ...form, voteNumber: e.target.value })}
   326→                    placeholder={isNonVote ? "NON-NAVAL or CONTRACTUAL" : "e.g. V-2025-001"}
   327→                    className="bg-background border-border"
   328→                  />
   329→                  {isNonVote && (
   330→                    <p className="text-[11px] text-amber-400 flex items-start gap-1">
   331→                      <Info className="h-3 w-3 mt-0.5 shrink-0" />
   332→                      Non-Vote funding — Vote Number can be &apos;NON-NAVAL&apos; or &apos;CONTRACTUAL&apos;
   333→                    </p>
   334→                  )}
   335→                </div>
   336→
   337→                {/* Project Type */}
   338→                <div className="space-y-1.5">
   339→                  <Label>Project Type *</Label>
   340→                  <Select value={form.projectType} onValueChange={(v) => setForm({ ...form, projectType: v })}>
   341→                    <SelectTrigger className="bg-background border-border">
   342→                      <SelectValue />
   343→                    </SelectTrigger>
   344→                    <SelectContent>
   345→                      {PROJECT_TYPES.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   346→                    </SelectContent>
   347→                  </Select>
   348→                  {form.projectType && (
   349→                    <div className="pt-0.5">
   350→                      <CustomBadge className={cn("border", projectTypeColor(form.projectType))}>
   351→                        {form.projectType}
   352→                      </CustomBadge>
   353→                    </div>
   354→                  )}
   355→                </div>
   356→
   357→                {/* Financial Mean */}
   358→                <div className="space-y-1.5">
   359→                  <Label>Financial Mean *</Label>
   360→                  <Select value={form.financialMean} onValueChange={(v) => setForm({ ...form, financialMean: v })}>
   361→                    <SelectTrigger className="bg-background border-border">
   362→                      <SelectValue />
   363→                    </SelectTrigger>
   364→                    <SelectContent>
   365→                      {FINANCIAL_MEANS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   366→                    </SelectContent>
   367→                  </Select>
   368→                  {form.financialMean && (
   369→                    <div className="pt-0.5">
   370→                      <CustomBadge className={cn("border", financialMeanColor(form.financialMean))}>
   371→                        {form.financialMean}
   372→                      </CustomBadge>
   373→                    </div>
   374→                  )}
   375→                </div>
   376→
   377→                {/* Reference — manual text entry (how vote allocated) */}
   378→                <div className="space-y-1.5">
   379→                  <Label>Reference (how vote allocated)</Label>
   380→                  <Input
   381→                    value={form.reference}
   382→                    onChange={(e) => setForm({ ...form, reference: e.target.value })}
   383→                    placeholder="e.g. Vote 2104 - Allocation 2025"
   384→                    className="bg-background border-border"
   385→                  />
   386→                </div>
   387→
   388→                <div className="space-y-1.5">
   389→                  <Label>Vote Allocation (Rs.)</Label>
   390→                  <Input type="number" value={form.voteAllocation} onChange={(e) => setForm({ ...form, voteAllocation: e.target.value })} placeholder="0" className="bg-background border-border" />
   391→                </div>
   392→                <div className="space-y-1.5">
   393→                  <Label>Benefactor</Label>
   394→                  <Input value={form.benefactor} onChange={(e) => setForm({ ...form, benefactor: e.target.value })} placeholder="e.g. Engineering Dept" className="bg-background border-border" />
   395→                </div>
   396→                <div className="space-y-1.5">
   397→                  <Label>Start Date</Label>
   398→                  <Input type="date" value={form.startDate} onChange={(e) => setForm({ ...form, startDate: e.target.value })} className="bg-background border-border" />
   399→                </div>
   400→                <div className="space-y-1.5">
   401→                  <Label>EDC (Estimated Date of Completion)</Label>
   402→                  <Input type="date" value={form.endDate} onChange={(e) => setForm({ ...form, endDate: e.target.value })} className="bg-background border-border" />
   403→                </div>
   404→                <div className="space-y-1.5">
   405→                  <Label>Project Officer</Label>
   406→                  <Input value={form.projectOfficer} onChange={(e) => setForm({ ...form, projectOfficer: e.target.value })} placeholder="e.g. Cdr. R. Perera" className="bg-background border-border" />
   407→                </div>
   408→                <div className="space-y-1.5">
   409→                  <Label>Status</Label>
   410→                  <Select value={form.status} onValueChange={(v) => setForm({ ...form, status: v })}>
   411→                    <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   412→                    <SelectContent>
   413→                      {PROJECT_STATUSES.map((s) => <SelectItem key={s} value={s}>{s}</SelectItem>)}
   414→                    </SelectContent>
   415→                  </Select>
   416→                </div>
   417→                <div className="sm:col-span-2 lg:col-span-3 space-y-1.5">
   418→                  <Label>Description</Label>
   419→                  <Textarea value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} rows={2} placeholder="Brief description..." className="bg-background border-border" />
   420→                </div>
   421→              </div>
   422→            </section>
   423→
   424→            {/* Material Estimate — MANDATORY */}
   425→            <section className="bg-card border border-border rounded-xl p-5">
   426→              <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   427→                <div className="min-w-0">
   428→                  <h3 className="text-sm font-semibold text-foreground flex items-center gap-2">
   429→                    <FileSpreadsheet className="h-4 w-4 text-primary" />
   430→                    Material Estimate
   431→                    <span className="text-[11px] font-normal text-amber-400 flex items-center gap-1">
   432→                      <AlertCircle className="h-3 w-3" />
   433→                      Mandatory — required to create project material list
   434→                    </span>
   435→                  </h3>
   436→                </div>
   437→                <div className="flex items-center gap-2 flex-wrap">
   438→                  <Button onClick={downloadTemplate} size="sm" variant="outline" className="gap-1.5">
   439→                    <Download className="h-4 w-4" /> Download Template
   440→                  </Button>
   441→                  <Button
   442→                    onClick={() => fileInputRef.current?.click()}
   443→                    size="sm"
   444→                    variant="outline"
   445→                    className="gap-1.5"
   446→                  >
   447→                    <Upload className="h-4 w-4" /> Upload Estimate
   448→                  </Button>
   449→                  <input
   450→                    ref={fileInputRef}
   451→                    type="file"
   452→                    accept=".csv"
   453→                    className="hidden"
   454→                    onChange={handleFileUpload}
   455→                  />
   456→                  <Button onClick={addRow} size="sm" variant="outline" className="gap-1.5">
   457→                    <Plus className="h-4 w-4" /> Add Row
   458→                  </Button>
   459→                </div>
   460→              </div>
   461→
   462→              <div className="border border-border rounded-lg overflow-hidden">
   463→                <div className="grid grid-cols-[1fr_80px_90px_120px_40px] gap-2 px-3 py-2 bg-secondary/40 text-xs text-muted-foreground uppercase">
   464→                  <span>Item Name</span>
   465→                  <span className="text-right">Qty</span>
   466→                  <span>Deno</span>
   467→                  <span className="text-right">Est. Cost (Rs.)</span>
   468→                  <span></span>
   469→                </div>
   470→                <div className="divide-y divide-border">
   471→                  {estimateRows.map((row, idx) => (
   472→                    <div key={idx} className="grid grid-cols-[1fr_80px_90px_120px_40px] gap-2 px-3 py-2 items-center">
   473→                      <Input value={row.itemName} onChange={(e) => updateRow(idx, "itemName", e.target.value)} placeholder="e.g. Centrifugal Pump 15kW" className="bg-background border-border h-8 text-sm" />
   474→                      <Input type="number" value={row.qty} onChange={(e) => updateRow(idx, "qty", e.target.value)} placeholder="0" className="bg-background border-border h-8 text-sm text-right" />
   475→                      <Select value={row.unit} onValueChange={(v) => updateRow(idx, "unit", v)}>
   476→                        <SelectTrigger className="bg-background border-border h-8 text-sm"><SelectValue /></SelectTrigger>
   477→                        <SelectContent>
   478→                          {UNITS.map((u) => <SelectItem key={u} value={u}>{u}</SelectItem>)}
   479→                        </SelectContent>
   480→                      </Select>
   481→                      <Input type="number" value={row.estimateCost} onChange={(e) => updateRow(idx, "estimateCost", e.target.value)} placeholder="0" className="bg-background border-border h-8 text-sm text-right" />
   482→                      <button onClick={() => removeRow(idx)} className="text-muted-foreground hover:text-red-400 flex items-center justify-center">
   483→                        <Trash2 className="h-4 w-4" />
   484→                      </button>
   485→                    </div>
   486→                  ))}
   487→                </div>
   488→              </div>
   489→              <p className="text-xs text-muted-foreground mt-2 flex items-center gap-1">
   490→                <Info className="h-3 w-3" />
   491→                Tip: Estimate cost is per-unit. After saving, Step 2 will process these items with AI to match RC list &amp; inventory.
   492→                {!hasValidEstimate && (
   493→                  <span className="text-amber-400 ml-2 font-medium">· Add estimate items to continue</span>
   494→                )}
   495→              </p>
   496→            </section>
   497→
   498→            {/* Actions */}
   499→            <div className="flex items-center justify-between gap-3 sticky bottom-0 bg-background/80 backdrop-blur py-3 -mx-4 sm:-mx-6 px-4 sm:px-6 border-t border-border">
   500→              <Button variant="outline" onClick={onClose}>Cancel</Button>
   501→              <div className="flex items-center gap-2">
   502→                {!hasValidEstimate && (
   503→                  <span className="text-xs text-amber-400 hidden sm:flex items-center gap-1">

   507→                )}
   508→                <Button

   516→                </Button>

   520→        )}
   521→
   522→        {step === 2 && createdEstimateId && (

   558→            </DialogTitle>
   559→          </DialogHeader>

   561→            Review the parsed CSV rows below. Confirming will <span className="font-medium text-foreground">replace</span> the current estimate rows.

   571→              <div className="divide-y divide-border max-h-[50vh] overflow-y-auto scrollbar-thin">
   572→                {previewRows.map((r, i) => (
   573→                  <div key={i} className="grid grid-cols-[1fr_70px_80px_110px] gap-2 px-3 py-1.5 text-sm items-center">
   574→                    <span className="text-foreground truncate">{r.itemName || <span className="text-muted-foreground italic">—</span>}</span>
   575→                    <span className="text-right">{r.qty || "—"}</span>
   576→                    <span className="text-xs">{r.unit || "NOS"}</span>

   578→                  </div>
   579→                ))}
   580→              </div>
   581→            </div>
   582→          )}
   583→          <DialogFooter>
   584→            <Button variant="outline" onClick={() => setPreviewRows(null)}>Cancel</Button>
   585→            <Button onClick={confirmImport} disabled={!previewRows || previewRows.length === 0} className="gap-1.5">
   586→              <Check className="h-4 w-4" /> Confirm Import ({previewRows?.length || 0} rows)
   587→            </Button>
   588→          </DialogFooter>
   589→        </DialogContent>
   590→      </Dialog>
   591→    </div>
   592→  );
   593→}
