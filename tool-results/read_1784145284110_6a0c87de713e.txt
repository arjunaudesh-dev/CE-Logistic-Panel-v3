   132→export function EstimateItemsTable({ estimateId, showProcessButton = false, onChanged }: Props) {
   133→  const { data: estimate, loading, refresh } = useFetch<EstimateWithProject>(
   134→    `/api/estimates/${estimateId}`
   135→  );
   136→  const { data: prefixes } = useFetch<NumberPrefix[]>("/api/number-prefixes");
   137→  const { data: stakeholders } = useFetch<Stakeholder[]>("/api/stakeholders");
   138→  const [processing, setProcessing] = useState(false);
   139→  const [initiateItem, setInitiateItem] = useState<EstimateItem | null>(null);
   140→  // Multi-select for bulk "Initiate Selected" — tracks selected EstimateItem IDs.
   141→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   142→  const [bulkInitiateOpen, setBulkInitiateOpen] = useState(false);
   143→  const { toast } = useToast();
   144→
   145→  // Sorting state — shared across all 3 sections (single preference)
   146→  const [sortKey, setSortKey] = useState<SortKey>("srNo");
   147→  const [sortDir, setSortDir] = useState<SortDir>("asc");
   148→  // Group-by-file toggle — applies within each section
   149→  const [groupByFile, setGroupByFile] = useState(false);
   150→
   151→  const handleSort = (key: SortKey) => {
   152→    if (key === sortKey) {
   153→      setSortDir((prev) => (prev === "asc" ? "desc" : "asc"));
   154→    } else {
   155→      setSortKey(key);
   156→      setSortDir("asc");
   157→    }
   158→  };
   159→
   160→  const items = estimate?.items || [];
   161→  const pendingItems = items.filter((i) => i.status === "Pending");
   162→  const initiatedItems = items.filter((i) => i.status === "Initiated");
   163→
   164→  // Determine if the project context is contractual — infer from project type OR financial mean.
   165→  const projectType = estimate?.project?.projectType;
   166→  const financialMean = estimate?.project?.financialMean;
   167→  const isProjectContractual =
   168→    projectType === "Contractual Job" || financialMean === "Contractual";
   169→
   170→  // Split items into 3 groups according to type
   171→  const rcItems = items.filter((i) => i.isRC);
   172→  const contractualItems = items.filter((i) => !i.isRC && isProjectContractual);
   173→  const nonRateItems = items.filter((i) => !i.isRC && !isProjectContractual);
   174→
   175→  // ─── Multi-select helpers ───
   176→  const toggleSelect = (id: string) => {
   177→    setSelectedIds((prev) => {
   178→      const next = new Set(prev);
   179→      if (next.has(id)) next.delete(id);
   180→      else next.add(id);
   181→      return next;
   182→    });
   183→  };
   184→  const selectMany = (ids: string[], on: boolean) => {
   185→    setSelectedIds((prev) => {
   186→      const next = new Set(prev);
   187→      if (on) ids.forEach((id) => next.add(id));
   188→      else ids.forEach((id) => next.delete(id));
   189→      return next;
   190→    });
   191→  };
   192→  const clearSelection = () => setSelectedIds(new Set());
   193→  // All pending item IDs (only Pending items are eligible for PR initiation).
   194→  // Computed inline (no useMemo) since pendingItems is itself derived from items.
   195→  const allPendingIds = pendingItems.map((i) => i.id);
   196→  const visibleSelectedCount = allPendingIds.filter((id) => selectedIds.has(id)).length;
   197→  const allPendingSelected =
   198→    allPendingIds.length > 0 && visibleSelectedCount === allPendingIds.length;
   199→  const somePendingSelected =
   200→    visibleSelectedCount > 0 && visibleSelectedCount < allPendingIds.length;
   201→  const toggleSelectAllPending = () =>
   202→    selectMany(allPendingIds, !allPendingSelected);
   203→
   204→  // Officers + sailors for the BulkInitiatePRDialog selects (sorted by seniority)
   205→  const officers = useMemo(
   206→    () =>
   207→      (stakeholders || [])
   208→        .filter((s) => (s.category || "").toLowerCase().includes("officer"))
   209→        .sort((a, b) => {
   210→          const sa = a.seniority ?? 99;
   211→          const sb = b.seniority ?? 99;
   212→          if (sa !== sb) return sa - sb;
   213→          return (a.name || "").localeCompare(b.name || "");
   214→        }),
   215→    [stakeholders]
   216→  );
   217→  const sailors = useMemo(
   218→    () =>
   219→      (stakeholders || [])
   220→        .filter((s) => (s.category || "").toLowerCase().includes("sailor"))
   221→        .sort((a, b) => (a.name || "").localeCompare(b.name || "")),
   222→    [stakeholders]
   223→  );
   224→
   225→  // PR prefix auto-suggest — find the "PR" doc type in the number prefixes
   226→  const prPrefix = useMemo(
   227→    () => (prefixes || []).find((p) => p.docType === "PR"),
   228→    [prefixes]
   229→  );
   230→  const suggestedPrNumber = useMemo(() => {
   231→    if (!prPrefix) return "";
   232→    const num = String(prPrefix.nextNumber).padStart(3, "0");
   233→    return `${prPrefix.prefix}${num}${prPrefix.suffix || ""}`;
   234→  }, [prPrefix]);
   235→
   236→  // Bulk initiate: creates a PR for every selected estimate item via
   237→  // POST /api/estimates/{estimateId}/items/{itemId}. The same PR number, file
   238→  // number, authorized-by and created-by are reused across all items so they
   239→  // share a fileGroupKey (and thus collapse into a single grouped row in the
   240→  // PR Raised Files window).
   241→  const handleBulkInitiate = async (input: {
   242→    prNumber: string;
   243→    fileNumber: string;
   244→    fileGroupKey: string;
   245→    authorizedById: string;
   246→    createdById: string;
   247→  }) => {
   248→    const selected = pendingItems.filter((it) => selectedIds.has(it.id));
   249→    if (selected.length === 0) {
   250→      toast({ title: "No items selected", variant: "destructive" });
   251→      return;
   252→    }
   253→    const officer = officers.find((o) => o.id === input.authorizedById);
   254→    const sailor = sailors.find((s) => s.id === input.createdById);
   255→    const authorizedBy = officer
   256→      ? `${officer.name}${officer.appointment ? ` (${officer.appointment})` : ""}`
   257→      : input.authorizedById;
   258→    const createdBy = sailor
   259→      ? `${sailor.name}${sailor.appointment ? ` (${sailor.appointment})` : ""}`
   260→      : input.createdById;
   261→    const effectiveGroupKey = input.fileGroupKey.trim() || input.prNumber;
   262→    const results = await Promise.all(
   263→      selected.map((it) =>
   264→        apiRequest(`/api/estimates/${estimateId}/items/${it.id}`, "POST", {
   265→          prNumber: input.prNumber,
   266→          fileNumber: input.fileNumber,
   267→          fileGroupKey: effectiveGroupKey,
   268→          createdBy,
   269→          authorizedBy,
   270→          responsiblePerson: authorizedBy,
   271→        })
   272→      )
   273→    );
   274→    const failed = results.filter((r) => !r.ok);
   275→    if (failed.length === 0) {
   276→      toast({
   277→        title: `Initiated ${selected.length} file${selected.length === 1 ? "" : "s"} — ${input.prNumber}`,
   278→        description: "PR raised successfully. Item statuses updated below.",
   279→      });
   280→    } else {
   281→      toast({
   282→        title: `${failed.length} of ${selected.length} initiate(s) failed`,
   283→        description: failed[0]?.error,
   284→        variant: "destructive",
   285→      });
   286→    }
   287→    clearSelection();
   288→    setBulkInitiateOpen(false);
   289→    refresh();
   290→    onChanged?.();
   291→  };
   292→
   293→  const handleProcess = async () => {
   294→    setProcessing(true);
   295→    const res = await apiRequest(`/api/estimates/${estimateId}/process`, "POST");
   296→    setProcessing(false);
   297→    if (res.ok) {
   298→      toast({ title: "Estimate processed with AI", description: `${items.length} items matched to RC list & inventory` });
   299→      refresh();
   300→      onChanged?.();
   301→    } else {
   302→      toast({ title: "AI processing failed", description: res.error, variant: "destructive" });
   303→    }
   304→  };
   305→
   306→  if (loading) return <LoadingState label="Loading estimate..." />;
   307→
   308→  return (
   309→    <div className="space-y-3">
   310→      {showProcessButton && (
   311→        <div className="flex items-center justify-between gap-3 flex-wrap">
   312→          <div className="flex items-center gap-2">
   313→            <CustomBadge className={cn("border-0", estimateStatusColor(estimate?.status || "Draft"))}>
   314→              {estimate?.status || "Draft"}
   315→            </CustomBadge>
   316→            <span className="text-xs text-muted-foreground">
   317→              {items.length} items · {pendingItems.length} pending · {initiatedItems.length} initiated
   318→            </span>
   319→          </div>
   320→          {estimate?.status !== "Processed" && (
   321→            <Button onClick={handleProcess} disabled={processing} size="sm" className="gap-1.5">
   322→              {processing ? <Loader2 className="h-4 w-4 animate-spin" /> : <Sparkles className="h-4 w-4" />}
   323→              Process with AI
   324→            </Button>
   325→          )}
   326→        </div>
   327→      )}
   328→
   329→      {/* summary tiles — Total Cost now correctly sums (Unit Cost × Qty) per row,
   330→          where Unit Cost = rcRate for RC items and approxCost for non-RC items */}
   331→      {items.length > 0 && (
   332→        <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
   333→          <SummaryTile
   334→            label="Total Cost"
   335→            value={formatCurrency(items.reduce((s, i) => s + (i.isRC ? i.rcRate : i.approxCost) * (i.qty || 0), 0))}
   336→            tone="text-blue-400"
   337→            icon={<PackageSearch className="h-4 w-4" />}
   338→          />
   339→          <SummaryTile
   340→            label="RC Items"
   341→            value={String(rcItems.length)}
   342→            tone="text-emerald-400"
   343→            icon={<CheckCircle2 className="h-4 w-4" />}
   344→          />
   345→          <SummaryTile
   346→            label="Stock Available"
   347→            value={String(items.filter((i) => i.availableStock > 0).length)}
   348→            tone="text-amber-400"
   349→            icon={<AlertCircle className="h-4 w-4" />}
   350→          />
   351→          <SummaryTile
   352→            label="Initiated PRs"
   353→            value={String(initiatedItems.length)}
   354→            tone="text-violet-400"
   355→            icon={<FileSignature className="h-4 w-4" />}
   356→          />
   357→        </div>
   358→      )}
   359→
   360→      {/* Selection bar — appears when at least one pending item is selected */}
   361→      {selectedIds.size > 0 && (
   362→        <div className="flex items-center gap-3 flex-wrap rounded-lg border border-primary/40 bg-primary/[0.06] px-4 py-2.5">
   363→          <div className="flex items-center gap-2 text-sm text-foreground">
   364→            <ListChecks className="h-4 w-4 text-primary" />
   365→            <span className="font-medium">{selectedIds.size} item{selectedIds.size > 1 ? "s" : ""} selected</span>
   366→            <span className="text-xs text-muted-foreground">· ready to initiate</span>
   367→          </div>
   368→          <div className="flex items-center gap-2 ml-auto">
   369→            <Button variant="ghost" size="sm" onClick={clearSelection} className="h-8 text-xs gap-1.5">
   370→              <X className="h-3.5 w-3.5" /> Clear
   371→            </Button>
   372→            <Button
   373→              size="sm"
   374→              onClick={() => setBulkInitiateOpen(true)}
   375→              className="gap-1.5"
   376→              title="Raise a single PR covering all selected estimate items"
   377→            >
   378→              <CheckSquare className="h-4 w-4" /> Initiate Selected
   379→            </Button>
   380→          </div>
   381→        </div>
   382→      )}
   383→
   384→      {/* Group-by-file toggle — only enabled when at least one item has been initiated */}
   385→      {items.length > 0 && (
   386→        <div className="flex items-center justify-between gap-3 flex-wrap">
   387→          <div className="flex items-center gap-2">
   388→            {pendingItems.length > 0 && (
   389→              <Checkbox
   390→                checked={
   391→                  allPendingSelected
   392→                    ? true
   393→                    : somePendingSelected
   394→                    ? "indeterminate"
   395→                    : false
   396→                }
   397→                onCheckedChange={toggleSelectAllPending}
   398→                aria-label="Select all pending items"
   399→                className="mr-1"
   400→              />
   401→            )}
   402→            <p className="text-[11px] text-muted-foreground flex items-center gap-1">
   403→              <GroupIcon className="h-3 w-3" />
   404→              Sortable headers · {initiatedItems.length > 0
   405→                ? "Toggle to group multiple items sharing the same PR/File number"
   406→                : "Initiate files to enable the group-by-file view"}
   407→            </p>
   408→          </div>
   409→          <Button
   410→            size="sm"
   411→            variant={groupByFile ? "default" : "outline"}
   412→            onClick={() => setGroupByFile((v) => !v)}
   413→            disabled={initiatedItems.length === 0}
   414→            className="gap-1.5"
   415→            title={initiatedItems.length === 0 ? "No initiated items to group" : "Toggle group-by-file view"}
   416→          >
   417→            <GroupIcon className="h-4 w-4" />
   418→            {groupByFile ? "Ungroup View" : "Group by File"}
   419→          </Button>
   420→        </div>
   421→      )}
   422→
   423→      {items.length === 0 ? (
   424→        <EmptyState
   425→          icon={PackageSearch}
   426→          title="No estimate items"
   427→          description="Add material estimate items in Step 1, then process with AI."
   428→        />
   429→      ) : (
   430→        <div className="space-y-4">
   431→          {rcItems.length > 0 && (
   432→            <ItemsSection
   433→              title="RC (Rate Contract) Items"
   434→              icon={<ShieldCheck className="h-4 w-4" />}
   435→              variant="rc"
   436→              description="Items matched to an active Rate Contract — procurement can proceed at the locked RC rate."
   437→              items={rcItems}
   438→              onInitiate={(it) => setInitiateItem(it)}
   439→              sortKey={sortKey}
   440→              sortDir={sortDir}
   441→              onSort={handleSort}
   442→              groupByFile={groupByFile}
   443→              selectedIds={selectedIds}
   444→              onToggleSelect={toggleSelect}
   445→            />
   446→          )}
   447→          {nonRateItems.length > 0 && (
   448→            <ItemsSection
   449→              title="Non-Rate Items (Tender/Quotation Required)"
   450→              icon={<FileText className="h-4 w-4" />}
   451→              variant="non-rate"
   452→              description="These items are not on an active Rate Contract — they require a tender or quotation process."
   453→              items={nonRateItems}
   454→              onInitiate={(it) => setInitiateItem(it)}
   455→              sortKey={sortKey}
   456→              sortDir={sortDir}
   457→              onSort={handleSort}
   458→              groupByFile={groupByFile}
   459→              selectedIds={selectedIds}
   460→              onToggleSelect={toggleSelect}
   461→            />
   462→          )}
   463→          {contractualItems.length > 0 && (
   464→            <ItemsSection
   465→              title="Contractual Items (ARO Flow)"
   466→              icon={<ScrollText className="h-4 w-4" />}
   467→              variant="contractual"
   468→              description="These items follow the ARO (Annual Requirement Order) contractual process flow."
   469→              items={contractualItems}
   470→              onInitiate={(it) => setInitiateItem(it)}
   471→              sortKey={sortKey}
   472→              sortDir={sortDir}
   473→              onSort={handleSort}
   474→              groupByFile={groupByFile}
   475→              selectedIds={selectedIds}
   476→              onToggleSelect={toggleSelect}
   477→            />
   478→          )}
   479→        </div>
   480→      )}
   481→
   482→      {initiateItem && (
   483→        <InitiateFileDialog
   484→          key={initiateItem.id}
   485→          item={initiateItem}
   486→          open
   487→          onOpenChange={(o) => !o && setInitiateItem(null)}
   488→          onInitiated={() => {
   489→            setInitiateItem(null);
   490→            refresh();
   491→            onChanged?.();
   492→          }}
   493→          projectAbbreviation={estimate?.project?.nameAbbreviation}
   494→        />
   495→      )}
   496→
   497→      {/* Bulk Initiate PR dialog — creates a single PR covering all selected estimate items.
   498→          The `key` forces a fresh mount whenever the user opens the dialog with a
   499→          different set of selections so the suggested PR number refreshes. */}
   500→      {bulkInitiateOpen && (
   501→        <BulkInitiatePRDialog
   502→          key={`bulk-${selectedIds.size}-${suggestedPrNumber}`}
   503→          open={bulkInitiateOpen}
   504→          onOpenChange={setBulkInitiateOpen}
   505→          selectedCount={selectedIds.size}
   506→          suggestedPrNumber={suggestedPrNumber}
   507→          officers={officers}
   508→          sailors={sailors}
   509→          onConfirm={handleBulkInitiate}
   510→        />
   511→      )}
   512→    </div>
   513→  );
   514→}

   516→/** Sortable column header with chevron indicator. */
   517→function SortHeader({
   518→  label,
   519→  sortKey,
   520→  currentSortKey,
   521→  currentSortDir,
   522→  onSort,
   523→  className,
   524→  align,
   525→}: {
   526→  label: string;
   527→  sortKey: SortKey;
   528→  currentSortKey: SortKey;
   529→  currentSortDir: SortDir;
   530→  onSort: (key: SortKey) => void;
   531→  className?: string;
   532→  align?: "left" | "right";
   533→}) {
   534→  const active = sortKey === currentSortKey;
   535→  return (
   536→    <TableHead className={cn("text-muted-foreground", className)}>
   537→      <button
   538→        type="button"
   539→        onClick={() => onSort(sortKey)}
   540→        className={cn(
   541→          "inline-flex items-center gap-1 hover:text-foreground transition-colors",
   542→          align === "right" && "flex-row-reverse",
   543→          active && "text-foreground"
   544→        )}
   545→      >
   546→        {label}
   547→        {active ? (
   548→          currentSortDir === "asc" ? (
   549→            <ChevronUp className="h-3 w-3" />
   550→          ) : (
   551→            <ChevronDown className="h-3 w-3" />
   552→          )
   553→        ) : (
   554→          <ChevronsUpDown className="h-3 w-3 opacity-40" />
   555→        )}
   556→      </button>
   557→    </TableHead>
   558→  );
   559→}

   561→function ItemsSection({
   562→  title,
   563→  icon,
   564→  variant,
   565→  description,
   566→  items,
   567→  onInitiate,
   568→  sortKey,
   569→  sortDir,
   570→  onSort,
   571→  groupByFile,
   572→  selectedIds,
   573→  onToggleSelect,
   574→}: {
   575→  title: string;
   576→  icon: React.ReactNode;
   577→  variant: SectionVariant;
   578→  description: string;
   579→  items: EstimateItem[];
   580→  onInitiate: (item: EstimateItem) => void;
   581→  sortKey: SortKey;
   582→  sortDir: SortDir;
   583→  onSort: (key: SortKey) => void;
   584→  groupByFile: boolean;
   585→  selectedIds: Set<string>;
   586→  onToggleSelect: (id: string) => void;
   587→}) {
   588→  const theme = SECTION_THEME[variant];
   589→  const showRcCols = variant === "rc";
   590→  const DescIcon = variant === "rc" ? CheckCircle2 : AlertCircle;
   591→
   592→  // Per-item unit cost — approxCost is already per-unit (RC rate or estimateCost).
   593→  // No "divided by qty" here.
   594→  const unitCostOf = (it: EstimateItem) => (it.isRC ? it.rcRate : it.approxCost) || 0;
   595→  const totalCostOf = (it: EstimateItem) => unitCostOf(it) * (it.qty || 0);
   596→
   597→  // Sort items — sort is applied within each section
   598→  const sortedItems = useMemo(() => {
   599→    const arr = [...items];
   600→    arr.sort((a, b) => {
   601→      let cmp = 0;
   602→      switch (sortKey) {
   603→        case "srNo":
   604→          cmp = a.srNo - b.srNo;
   605→          break;
   606→        case "itemName":
   607→          cmp = (a.itemName || "").localeCompare(b.itemName || "");
   608→          break;
   609→        case "itemCode":
   610→          cmp = (a.itemCode || "").localeCompare(b.itemCode || "");
   611→          break;
   612→        case "qty":
   613→          cmp = a.qty - b.qty;
   614→          break;
   615→        case "unitCost":
   616→          cmp = unitCostOf(a) - unitCostOf(b);
   617→          break;
   618→        case "totalCost":
   619→          cmp = totalCostOf(a) - totalCostOf(b);
   620→          break;
   621→      }
   622→      return sortDir === "asc" ? cmp : -cmp;
   623→    });
   624→    return arr;
   625→  }, [items, sortKey, sortDir]);
   626→
   627→  // Group items by prNumber (or fileNumber) when groupByFile is enabled.
   628→  // Items without a prNumber/fileNumber remain as individual rows.
   629→  // Groups of 2+ items become expandable group cards; single-initiated items
   630→  // render as individual rows so the user still sees the PR badge inline.

   632→    if (!groupByFile) {
   633→      return { groups: [] as { key: string; items: EstimateItem[] }[], individuals: sortedItems };

   638→      const key = it.prNumber || it.fileNumber;
   639→      if (key) {
   640→        const existing = groupsMap.get(key);
   641→        if (existing) existing.push(it);
   642→        else groupsMap.set(key, [it]);
   643→      } else {
   644→        individuals.push(it);
   645→      }
   646→    }
   647→    const realGroups: { key: string; items: EstimateItem[] }[] = [];
   648→    for (const [key, groupItems] of groupsMap) {
   649→      if (groupItems.length >= 2) {
   650→        realGroups.push({ key, items: groupItems });
   651→      } else {
   652→        individuals.push(groupItems[0]);
   653→      }
   654→    }
   655→    return { groups: realGroups, individuals };
   656→  }, [sortedItems, groupByFile]);
   657→
   658→  const totalCost = items.reduce((s, i) => s + totalCostOf(i), 0);
   659→  const groupedItemCount = groupedView.groups.reduce((s, g) => s + g.items.length, 0);
   660→
   661→  return (
   662→    <div className="border border-border rounded-lg overflow-hidden">
   663→      {/* Colored header bar — icon + title + count + total cost */}
   664→      <div className={cn("flex items-center justify-between gap-3 px-4 py-2.5 border-b border-border flex-wrap", theme.bar)}>
   665→        <div className="flex items-center gap-2 min-w-0">
   666→          <span className="shrink-0">{icon}</span>
   667→          <h4 className={cn("text-sm font-semibold", theme.title)}>{title}</h4>
   668→          <span className="text-[11px] px-1.5 py-0.5 rounded bg-background/40 font-mono">
   669→            {items.length}
   670→          </span>
   671→        </div>
   672→        <div className="text-xs">
   673→          <span className="opacity-80">Total Cost: </span>
   674→          <span className={cn("font-semibold", theme.summary)}>{formatCurrency(totalCost)}</span>
   675→        </div>
   676→      </div>
   677→
   678→      {/* Variant description strip */}
   679→      <div className={cn("px-4 py-1.5 border-b border-border flex items-center gap-1.5 text-[11px]", theme.desc)}>
   680→        <DescIcon className="h-3 w-3 shrink-0" />
   681→        {description}
   682→      </div>
   683→
   684→      {/* Group cards — only when groupByFile is enabled and there are multi-item groups */}
   685→      {groupByFile && groupedView.groups.length > 0 && (
   686→        <div className="p-3 space-y-2 bg-secondary/10 border-b border-border">
   687→          <p className="text-[11px] uppercase tracking-wide text-muted-foreground flex items-center gap-1">
   688→            <Files className="h-3 w-3" />
   689→            {groupedView.groups.length} grouped file(s) · {groupedItemCount} items
   690→          </p>
   691→          {groupedView.groups.map((g) => (
   692→            <GroupCard
   693→              key={g.key}
   694→              groupKey={g.key}
   695→              items={g.items}
   696→              variant={variant}
   697→              onInitiate={onInitiate}
   698→            />
   699→          ))}
   700→        </div>
   701→      )}
   702→
   703→      {/* Individual items table — show ALL items (no pagination limit) in a scrollable container */}
   704→      {groupedView.individuals.length > 0 ? (
   705→        <div className="max-h-[60vh] overflow-auto scrollbar-thin">
   706→          <Table>
   707→            <TableHeader className="sticky top-0 z-10 bg-secondary/95 backdrop-blur">
   708→              <TableRow className="hover:bg-transparent border-border">
   709→                <TableHead className="text-muted-foreground w-10">Select</TableHead>
   710→                <SortHeader
   711→                  label="Sr"
   712→                  sortKey="srNo"
   713→                  currentSortKey={sortKey}
   714→                  currentSortDir={sortDir}
   715→                  onSort={onSort}
   716→                  className="w-12"
   717→                />
   718→                <SortHeader
   719→                  label="Item Name"
   720→                  sortKey="itemName"
   721→                  currentSortKey={sortKey}
   722→                  currentSortDir={sortDir}
   723→                  onSort={onSort}
   724→                />
   725→                <SortHeader
   726→                  label="Item Code"
   727→                  sortKey="itemCode"
   728→                  currentSortKey={sortKey}
   729→                  currentSortDir={sortDir}
   730→                  onSort={onSort}
   731→                />
   732→                <SortHeader
   733→                  label="Qty"
   734→                  sortKey="qty"
   735→                  currentSortKey={sortKey}
   736→                  currentSortDir={sortDir}
   737→                  onSort={onSort}
   738→                  className="text-right"
   739→                  align="right"
   740→                />
   741→                <TableHead className="text-muted-foreground">Deno</TableHead>
   742→                <SortHeader

   744→                  sortKey="unitCost"
   745→                  currentSortKey={sortKey}

   747→                  onSort={onSort}

   749→                  align="right"
   750→                />
   751→                <SortHeader
   752→                  label="Total Cost"
   753→                  sortKey="totalCost"
   754→                  currentSortKey={sortKey}
   755→                  currentSortDir={sortDir}
   756→                  onSort={onSort}
   757→                  className="text-right"
   758→                  align="right"
   759→                />
   760→                {showRcCols && <TableHead className="text-muted-foreground">RC No.</TableHead>}
