    71→interface EstimateRow {
    72→  itemName: string;
    73→  qty: string;
    74→  unit: string;
    75→  /** Per-unit cost (renamed from "estimateCost" in earlier versions).
    76→   *  The CSV column header is `UnitCost`; the parser also accepts the legacy

    96→export function AddProjectWizard({ open, onClose, onCreated }: Props) {
    97→  const [step, setStep] = useState<1 | 2>(1);

   103→  const [previewRows, setPreviewRows] = useState<EstimateRow[] | null>(null);
   104→  const [customVoteNumber, setCustomVoteNumber] = useState(false);
   105→  const [selectedVoteId, setSelectedVoteId] = useState("");
   106→  // "Funds" checkbox state — when ticked, the project is funded via a non-vote fund
   107→  // (Malima/Non-Public/Contractual) and the Vote Number is NOT mandatory.
   108→  const [fundsChecked, setFundsChecked] = useState(false);
   109→  const { data: votes } = useFetch<Vote[]>("/api/votes");
   110→
   111→  const [form, setForm] = useState({
   112→    name: "",
   113→    nameAbbreviation: "",
   114→    voteNumber: "",
   115→    voteAllocation: "",
   116→    committed: "0",
   117→    actualSpent: "0",
   118→    startDate: "",
   119→    endDate: "", // EDC
   120→    benefactor: "", // replaces Department
   121→    reference: "", // manual text — how vote allocated
   122→    projectType: "New Construction",
   123→    financialMean: "Vote/Funds",
   124→    projectOfficer: "",
   125→    status: "Active",
   126→    description: "",
   127→  });
   128→  const [estimateRows, setEstimateRows] = useState<EstimateRow[]>([
   129→    { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },
   130→  ]);
   131→
   132→  const selectedVote = votes?.find((v) => v.id === selectedVoteId);
   133→
   134→  // Tracks whether the user has manually edited the abbreviation field. When
   135→  // false, the abbreviation auto-suggests from the project name. Once the user
   136→  // touches the field, we stop auto-suggesting so their override sticks.
   137→  const [abbrevTouched, setAbbrevTouched] = useState(false);
   138→
   139→  // Suggest an abbreviation from a project name by taking the first letter of
   140→  // each significant word (skipping common stop-words / numbers / symbols).
   141→  // Example: "No.03 Slipway Hangar" → "NSH", "Engine Fire Suppression" → "EFS".
   142→  const suggestAbbreviation = (name: string): string => {
   143→    const stop = new Set(["of", "for", "and", "the", "in", "on", "at", "to", "no", "a", "an"]);
   144→    const tokens = name
   145→      .replace(/[^\p{L}\p{N}\s-]/gu, " ")
   146→      .split(/[\s-]+/)
   147→      .map((t) => t.trim())
   148→      .filter(Boolean);
   149→    const letters = tokens
   150→      .filter((t) => !stop.has(t.toLowerCase()))
   151→      .map((t) => t[0]!.toUpperCase());
   152→    return letters.slice(0, 5).join("");
   153→  };
   154→
   155→  const handleNameChange = (value: string) => {
   156→    setForm((prev) => ({
   157→      ...prev,
   158→      name: value,
   159→      // keep auto-suggesting the abbreviation until the user edits it manually
   160→      nameAbbreviation: abbrevTouched ? prev.nameAbbreviation : suggestAbbreviation(value),
   161→    }));
   162→  };
   163→
   164→  const handleAbbrevChange = (value: string) => {
   165→    setAbbrevTouched(true);
   166→    // Force uppercase, strip non-alphanumeric (keep letters + digits)
   167→    const cleaned = value.toUpperCase().replace(/[^A-Z0-9]/g, "").slice(0, 8);
   168→    setForm((prev) => ({ ...prev, nameAbbreviation: cleaned }));
   169→  };
   170→
   171→  const handleVoteSelect = (id: string) => {
   172→    setSelectedVoteId(id);
   173→    const v = votes?.find((vt) => vt.id === id);
   174→    if (v) {
   175→      setForm((prev) => ({
   176→        ...prev,
   177→        voteNumber: v.voteNumber,
   178→        voteAllocation: v.allocation ? String(v.allocation) : prev.voteAllocation,
   179→      }));
   180→    }
   181→  };
   182→
   183→  // Toggle the "Funds" checkbox. When checked, the project is funded via a non-vote
   184→  // fund (default Malima Fund) and the Vote Number becomes optional. When unchecked,
   185→  // the project is funded via Vote and the Vote Number is mandatory.
   186→  const handleFundsToggle = (checked: boolean) => {
   187→    setFundsChecked(checked);
   188→    setCustomVoteNumber(false);
   189→    setSelectedVoteId("");
   190→    if (checked) {
   191→      setForm((prev) => ({
   192→        ...prev,
   193→        financialMean: "Malima Fund",
   194→        voteNumber: "",
   195→        voteAllocation: "",
   196→      }));
   197→    } else {
   198→      setForm((prev) => ({
   199→        ...prev,
   200→        financialMean: "Vote/Funds",
   201→        voteNumber: "",
   202→      }));
   203→    }
   204→  };
   205→
   206→  const handleFundSelect = (fund: string) => {
   207→    setForm((prev) => ({ ...prev, financialMean: fund }));
   208→  };
   209→
   210→  const nonVotePlaceholder =
   211→    form.financialMean === "Contractual" ? "CONTRACTUAL" : "NON-NAVAL";
   212→  const nonVoteHint =
   213→    form.financialMean === "Contractual"
   214→      ? "Contractual funding — Vote Number can be 'CONTRACTUAL' or the ARO reference."
   215→      : form.financialMean === "Malima Fund"
   216→      ? "Malima Fund — Vote Number can be 'NON-NAVAL' or the Malima reference number."
   217→      : "Non-Public Fund — Vote Number can be 'NON-NAVAL' or the fund reference.";
   218→
   219→  const addRow = () =>
   220→    setEstimateRows([
   221→      ...estimateRows,
   222→      { itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" },
   223→    ]);
   224→  const removeRow = (idx: number) =>
   225→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
   226→
   227→  // Update a single field on a row. When Qty or UnitCost changes,
   228→  // recompute the row's TotalCost automatically (= Qty × UnitCost) so the
   229→  // editable grid always shows a live total. Manual edits to TotalCost itself
   230→  // (or other fields) leave the total as-is.
   231→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
   232→    setEstimateRows(
   233→      estimateRows.map((r, i) => {
   234→        if (i !== idx) return r;
   235→        const next = { ...r, [field]: value };
   236→        if (field === "qty" || field === "estimateCost") {
   237→          const q = Number(next.qty);
   238→          const c = Number(next.estimateCost);
   239→          next.totalCost =
   240→            !isNaN(q) && !isNaN(c) && q > 0 && c > 0 ? String(q * c) : "";
   241→        }
   242→        return next;
   243→      })
   244→    );
   245→
   246→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   247→  const validEstimateRows = estimateRows.filter(
   248→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   249→  );
   250→  const hasValidEstimate = validEstimateRows.length > 0;
   251→
   252→  const downloadTemplate = () => {
   253→    // Template now uses `UnitCost` as the column header (renamed from `EstimateCost`).
   254→    // The sample row demonstrates the relationship TotalCost = Qty × UnitCost (2 × 185000 = 370000).
   255→    const header = "ItemName,Qty,Unit,UnitCost,TotalCost";
   256→    const sample = "Centrifugal Pump 15kW,2,NOS,185000,370000";
   257→    const sample2 = ",,NOS,,";
   258→    const csv = `${header}\n${sample}\n${sample2}\n`;
   259→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   260→    const url = URL.createObjectURL(blob);
   261→    const a = document.createElement("a");
   262→    a.href = url;
   263→    a.download = "material-estimate-template.csv";
   264→    a.click();
   265→    URL.revokeObjectURL(url);
   266→    toast({
   267→      title: "Template downloaded",
   268→      description:
   269→        "Columns: ItemName, Qty, Unit, UnitCost (per unit), TotalCost (= Qty × UnitCost — leave blank to auto-compute).",
   270→    });
   271→  };
   272→
   273→  // Simple CSV parser — handles quoted fields with commas
   274→  const parseCsv = (text: string): string[][] => {
   275→    const rows: string[][] = [];
   276→    let cur: string[] = [];
   277→    let field = "";
   278→    let inQuotes = false;
   279→    for (let i = 0; i < text.length; i++) {
   280→      const ch = text[i];
   281→      if (inQuotes) {
   282→        if (ch === '"' && text[i + 1] === '"') {
   283→          field += '"';
   284→          i++;
   285→        } else if (ch === '"') {
   286→          inQuotes = false;
   287→        } else {
   288→          field += ch;
   289→        }
   290→      } else {
   291→        if (ch === '"') inQuotes = true;
   292→        else if (ch === ",") {
   293→          cur.push(field);
   294→          field = "";
   295→        } else if (ch === "\n" || ch === "\r") {
   296→          if (ch === "\r" && text[i + 1] === "\n") i++;
   297→          cur.push(field);
   298→          rows.push(cur);
   299→          cur = [];
   300→          field = "";
   301→        } else {
   302→          field += ch;
   303→        }
   304→      }
   305→    }
   306→    if (field.length > 0 || cur.length > 0) {
   307→      cur.push(field);
   308→      rows.push(cur);
   309→    }
   310→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   311→  };
   312→
   313→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   314→    const file = e.target.files?.[0];
   315→    if (!file) return;
   316→    if (!file.name.toLowerCase().endsWith(".csv")) {
   317→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   318→      e.target.value = "";
   319→      return;
   320→    }
   321→    try {
   322→      const text = await file.text();
   323→      const rows = parseCsv(text);
   324→      if (rows.length === 0) {
   325→        toast({ title: "CSV file is empty", variant: "destructive" });
   326→        e.target.value = "";
   327→        return;
   328→      }
   329→      const header = rows[0].map((c) => c.trim().toLowerCase());
   330→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   331→      const idxQty = header.findIndex((h) => h.includes("qty"));
   332→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   333→      // Find the per-unit cost column. Prefer `UnitCost` (new name); fall back to
   334→      // the legacy `EstimateCost` header for backward compatibility. Also matches
   335→      // any column with "cost"/"price" but without "total".
   336→      const idxCost = header.findIndex(
   337→        (h) =>
   338→          (h === "unitcost" || h.includes("unit cost") || h.includes("unitcost")) ||
   339→          ((h.includes("estimate") && (h.includes("cost") || h.includes("price"))) ||
   340→           ((h.includes("cost") || h.includes("price")) && !h.includes("total")))
   341→      );
   342→      const idxTotal = header.findIndex(
   343→        (h) => h.includes("total") && (h.includes("cost") || h.includes("price"))
   344→      );
   345→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   346→      const parsed: EstimateRow[] = dataRows
   347→        .map((r) => {
   348→          const itemName = idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim();
   349→          const qty = idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim();
   350→          const unit = (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS";
   351→          // Per-unit cost — read from the UnitCost (or legacy EstimateCost) column.
   352→          const estimateCost = idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim();
   353→          // TotalCost: prefer the CSV's TotalCost column when present & non-blank,
   354→          // otherwise compute it as Qty × UnitCost (per the task spec).
   355→          const totalRaw =
   356→            idxTotal >= 0 ? (r[idxTotal] || "").trim() : "";
   357→          let totalCost = "";
   358→          if (totalRaw && !isNaN(Number(totalRaw))) {
   359→            totalCost = totalRaw;
   360→          } else {
   361→            const q = Number(qty);
   362→            const c = Number(estimateCost);
   363→            if (!isNaN(q) && !isNaN(c) && q > 0 && c > 0) {
   364→              totalCost = String(q * c);
   365→            }
   366→          }
   367→          return { itemName, qty, unit, estimateCost, totalCost };
   368→        })
   369→        .filter((r) => r.itemName);
   370→      if (parsed.length === 0) {
   371→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   372→        e.target.value = "";
   373→        return;
   374→      }
   375→      setPreviewRows(parsed);
   376→    } catch {
   377→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   378→    }
   379→    e.target.value = "";
   380→  };
   381→
   382→  const confirmImport = () => {
   383→    if (!previewRows) return;
   384→    setEstimateRows(previewRows);
   385→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   386→    setPreviewRows(null);
   387→  };
   388→
   389→  const handleSaveAndProcess = async () => {
   390→    if (!form.name) {
   391→      toast({ title: "Project Name is required", variant: "destructive" });
   392→      return;
   393→    }
   394→    // Vote Number is mandatory UNLESS the Funds checkbox is ticked (non-vote funding).
   395→    if (!fundsChecked && !form.voteNumber.trim()) {
   396→      toast({
   397→        title: "Vote Number is required",
   398→        description: "Tick the Funds checkbox for non-vote funding (Malima/Non-Public/Contractual).",
   399→        variant: "destructive",
   400→      });
   401→      return;
   402→    }
   403→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   404→    if (validRows.length === 0) {
   405→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   406→      return;
   407→    }
   408→    setSaving(true);
   409→    // 1. Create project
   410→    const projRes = await apiRequest("/api/projects", "POST", form);
   411→    if (!projRes.ok) {
   412→      setSaving(false);
   413→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   414→      return;
   415→    }
   416→    const project = projRes.data as { id: string };
   417→    // 2. Export Estimate into DB
   418→    const estRes = await apiRequest("/api/estimates", "POST", {
   419→      projectId: project.id,
   420→      name: `${form.name} — Material Estimate`,
   421→      items: validRows.map((r) => ({
   422→        itemName: r.itemName,
   423→        qty: Number(r.qty) || 0,
   424→        unit: r.unit,
   425→        estimateCost: Number(r.estimateCost) || 0,
   426→      })),
   427→    });
   428→    setSaving(false);
   429→    if (estRes.ok) {
   430→      const est = estRes.data as { id: string };
   431→      setCreatedProjectId(project.id);
   432→      setCreatedEstimateId(est.id);
   433→      setStep(2);
   434→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   435→    } else {
   436→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   437→    }
   438→  };
   439→
   440→  const handleGoToProject = () => {
   441→    if (createdProjectId) {
   442→      onCreated({
   443→        id: createdProjectId,
   444→        name: form.name,
   445→        nameAbbreviation: form.nameAbbreviation || null,
   446→        voteNumber: form.voteNumber,
   447→        voteAllocation: Number(form.voteAllocation) || 0,
   448→        committed: Number(form.committed) || 0,
   449→        actualSpent: Number(form.actualSpent) || 0,
   450→        benefactor: form.benefactor || null,
   451→        reference: form.reference || null,
   452→        endDate: form.endDate || null,
   453→        startDate: form.startDate || null,
   454→        projectType: form.projectType,
   455→        financialMean: form.financialMean,
   456→      });
   457→      // reset
   458→      setStep(1);
   459→      setCreatedProjectId(null);
   460→      setCreatedEstimateId(null);
   461→      setFundsChecked(false);
   462→      setAbbrevTouched(false);
   463→      setForm({
   464→        name: "", nameAbbreviation: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   465→        startDate: "", endDate: "", benefactor: "", reference: "",
   466→        projectType: "New Construction", financialMean: "Vote/Funds",
   467→        projectOfficer: "", status: "Active", description: "",
   468→      });
   469→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "", totalCost: "" }]);
   470→      setCustomVoteNumber(false);
   471→      setSelectedVoteId("");
   472→    }
   473→  };
   474→
   475→  if (!open) return null;
   476→
   477→  return (
   478→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   479→      {/* Header */}
   480→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   481→        <div className="flex items-center gap-3">
   482→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   483→            <X className="h-5 w-5" />
   484→          </button>
   485→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   486→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   487→          </h2>
   488→        </div>
   489→        {/* Stepper */}
   490→        <div className="hidden sm:flex items-center gap-2 text-xs">
   491→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   492→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   493→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   494→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   495→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   496→        </div>
   497→      </div>
   498→
   499→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   500→        {step === 1 && (
   501→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   502→            {/* Project details */}
   503→            <section className="bg-card border border-border rounded-xl p-5">
   504→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   505→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   506→              </h3>
   507→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   508→                <div className="sm:col-span-2 space-y-1.5">
   509→                  <Label>Project Name *</Label>
   510→                  <Input value={form.name} onChange={(e) => handleNameChange(e.target.value)} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
   511→                </div>
   512→
   513→                {/* Project Name Abbreviation — used as the file group key prefix */}
   514→                <div className="space-y-1.5">
