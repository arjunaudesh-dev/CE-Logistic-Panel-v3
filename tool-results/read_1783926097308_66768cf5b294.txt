   226→export function StoresModule() {
   227→  const [tab, setTab] = useState<"zones" | "stores">("zones");

   236→  const {
   237→    data: items,
   238→    loading: iLoading,
   239→    error: iError,
   240→    refresh: refreshItems,
   241→  } = useFetch<StoreItem[]>("/api/stores");
   242→
   243→  // Stores-tab filter state
   244→  const [category, setCategory] = useState<string>("All");
   245→  const [search, setSearch] = useState("");
   246→  const [zoneFilter, setZoneFilter] = useState("all");
   247→  const [stockFilter, setStockFilter] = useState<StockFilter>("all");
   248→  const [sortKey, setSortKey] = useState<SortKey>("description");

   250→  const [page, setPage] = useState(1);
   251→
   252→  // Dialogs
   253→  const [formOpen, setFormOpen] = useState(false);

   257→  const [earmarkOpen, setEarmarkOpen] = useState(false);
   258→  const [earmarkListOpen, setEarmarkListOpen] = useState(false);
   259→  const [findCodeOpen, setFindCodeOpen] = useState(false);
   260→
   261→  const { toast } = useToast();
   262→
   263→  // Derived: filtered + sorted list
   264→  const filtered = useMemo(() => {
   265→    if (!items) return [];
   266→    let list = items.slice();
   267→    if (category !== "All") list = list.filter((i) => i.category === category);
   268→    if (zoneFilter !== "all") list = list.filter((i) => i.zoneName === zoneFilter);
   269→    const q = search.trim().toLowerCase();
   270→    if (q) list = list.filter((i) => i.description.toLowerCase().includes(q));
   271→    list = list.filter((i) => {
   272→      if (stockFilter === "out") return i.qty <= 0;
   273→      if (stockFilter === "low")
   274→        return i.minStock > 0 && i.qty > 0 && i.qty <= i.minStock;
   275→      if (stockFilter === "in") return i.minStock === 0 || i.qty > i.minStock;
   276→      if (stockFilter === "over") return i.maxStock > 0 && i.qty >= i.maxStock;
   277→      return true;
   278→    });
   279→    list.sort((a, b) => {
   280→      let cmp = 0;
   281→      if (sortKey === "description") cmp = a.description.localeCompare(b.description);
   282→      else if (sortKey === "qty") cmp = a.qty - b.qty;
   283→      else if (sortKey === "cost") cmp = a.totalCost - b.totalCost;
   284→      else if (sortKey === "category") cmp = a.category.localeCompare(b.category);
   285→      return sortDir === "asc" ? cmp : -cmp;
   286→    });
   287→    return list;
   288→  }, [items, category, zoneFilter, search, stockFilter, sortKey, sortDir]);
   289→
   290→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   291→  const safePage = Math.min(page, totalPages);
   292→  const pageItems = filtered.slice(
   293→    (safePage - 1) * PAGE_SIZE,
   294→    safePage * PAGE_SIZE
   295→  );
   296→
   297→  // Stats reflect the currently filtered set
   298→  const stats = useMemo(() => {
   299→    const net = filtered.reduce((s, i) => s + i.totalCost, 0);
   300→    const task = filtered.length;
   301→    const avail = filtered.filter((i) => i.qty > 0).length;
   302→    const team = filtered.filter((i) => i.minStock > 0 && i.qty <= i.minStock).length;
   303→    return { net, task, avail, team };
   304→  }, [filtered]);
   305→
   306→  // Wrap filter setters so any change also resets pagination to page 1
   307→  const setCategoryR = (c: string) => {
   308→    setCategory(c);
   309→    setPage(1);
   310→  };
   311→  const setSearchR = (s: string) => {
   312→    setSearch(s);
   313→    setPage(1);
   314→  };
   315→  const setZoneFilterR = (z: string) => {
   316→    setZoneFilter(z);
   317→    setPage(1);
   318→  };
   319→  const setStockFilterR = (s: StockFilter) => {
   320→    setStockFilter(s);
   321→    setPage(1);
   322→  };
   323→  const setSortKeyR = (k: SortKey) => {
   324→    setSortKey(k);
   325→    setPage(1);
   326→  };
   327→  const setSortDirR = (d: "asc" | "desc") => {
   328→    setSortDir(d);
   329→    setPage(1);
   330→  };
   331→  const toggleSortR = (k: SortKey) => {
   332→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   333→    else {
   334→      setSortKey(k);
   335→      setSortDir("asc");
   336→    }
   337→    setPage(1);
   338→  };
   339→
   340→  // Handlers
   341→  const handleZoneClick = (zoneName: string) => {
   342→    setZoneFilter(zoneName);
   343→    setCategory("All");
   344→    setStockFilter("all");
   345→    setSearch("");
   346→    setPage(1);
   347→    setTab("stores");
   348→    toast({
   349→      title: `Browsing ${zoneName}`,
   350→      description: "Switched to Stores tab with zone filter applied.",
   351→    });
   352→  };
   353→
   354→  const openAdd = () => {
   355→    setEditing(null);
   356→    setFormOpen(true);
   357→  };
   358→
   359→  const openEdit = (item: StoreItem) => {
   360→    setEditing(item);
   361→    setFormOpen(true);
   362→  };
   363→
   364→  const handleSaved = () => {
   365→    setFormOpen(false);
   366→    setEditing(null);
   367→    refreshItems();
   368→    refreshZones();
   369→    toast({
   370→      title: editing ? "Item updated" : "Item added",
   371→      description: editing
   372→        ? "Store item has been updated successfully."
   373→        : "New store item has been added to the store.",
   374→    });
   375→  };
   376→
   377→  const toggleSort = (k: SortKey) => {
   378→    if (sortKey === k) setSortDir(sortDir === "asc" ? "desc" : "asc");
   379→    else {
   380→      setSortKey(k);
   381→      setSortDir("asc");
   382→    }
   383→  };
   384→
   385→  return (
   386→    <div>
   387→      <PageHeader
   388→        title="Store Management"
   389→        description="Zones, inventory & stock-level monitoring across the dockyard"
   390→        icon={Warehouse}
   391→        actions={
   392→          <div className="flex flex-wrap items-center gap-2 justify-end">
   393→            <Button
   394→              variant="outline"
   395→              size="sm"
   396→              onClick={() => setFindCodeOpen(true)}
   397→              className="gap-1.5"
   398→              title="Find item codes using AI"
   399→            >
   400→              <ScanSearch className="h-4 w-4" /> Find Item Codes (AI)
   401→            </Button>
   402→            <Button
   403→              variant="outline"
   404→              size="sm"
   405→              onClick={() => setEarmarkListOpen(true)}
   406→              className="gap-1.5"
   407→              title="View all earmarked items"
   408→            >
   409→              <List className="h-4 w-4" /> View Earmarked
   410→            </Button>
   411→            <Button
   412→              variant="outline"
   413→              size="sm"
   414→              onClick={() => setEarmarkOpen(true)}
   415→              className="gap-1.5"
   416→              title="Earmark an item for a project without a PR"
   417→            >
   418→              <BookmarkPlus className="h-4 w-4" /> Earmark Item

   420→            <Button
   421→              variant="outline"
   422→              size="sm"

   442→            <Boxes className="h-4 w-4" /> Zones

   445→            <Package className="h-4 w-4" /> Stores

   448→
   449→        {/* ─── Zones tab ─── */}
   450→        <TabsContent value="zones" className="mt-4">
   451→          <ZonesTab

   453→            loading={zLoading}
   454→            error={zError}

   548→// ────────────────────────────────────────────────────────────
   549→// Zones tab — dashboard view with scaled tiles
   550→// ────────────────────────────────────────────────────────────
   551→interface ZonesTabProps {
   552→  zones: StoreZone[] | null;
   553→  loading: boolean;
   554→  error: string | null;
   555→  onZoneClick: (zoneName: string) => void;
   556→}
   557→
   558→function ZonesTab({ zones, loading, error, onZoneClick }: ZonesTabProps) {
   559→  if (loading) return <LoadingState label="Loading zones..." />;
   560→  if (error) return <ErrorState message={error} />;
   561→  if (!zones || zones.length === 0)
   562→    return (
   563→      <EmptyState
   564→        icon={Boxes}
   565→        title="No zones configured"
   566→        description="Store zones will appear here as tiles once data is seeded."
   567→      />
   568→    );
   569→
   570→  const totalValue = zones.reduce((s, z) => s + z.totalValue, 0);
   571→  const totalItems = zones.reduce((s, z) => s + z.itemCount, 0);
   572→  const totalLow = zones.reduce((s, z) => s + z.lowStockCount, 0);
   573→
   574→  return (
   575→    <div>
   576→      {/* Mini summary strip */}
   577→      <div className="grid grid-cols-3 gap-3 mb-5">
   578→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   579→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   580→            Total Stock Value
   581→          </p>
   582→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   583→            {formatCurrency(totalValue)}
   584→          </p>
   585→        </div>
   586→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   587→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   588→            Total Items
   589→          </p>
   590→          <p className="text-base sm:text-xl font-semibold text-foreground mt-1">
   591→            {formatNumber(totalItems)}
   592→          </p>
   593→        </div>
   594→        <div className="bg-card border border-border rounded-xl p-3 sm:p-4">
   595→          <p className="text-[10px] sm:text-xs uppercase tracking-wide text-muted-foreground">
   596→            Low Stock Items
   597→          </p>
   598→          <p
   599→            className={cn(
   600→              "text-base sm:text-xl font-semibold mt-1",
   601→              totalLow > 0 ? "text-amber-400" : "text-emerald-400"
   602→            )}
   603→          >
   604→            {formatNumber(totalLow)}
   605→          </p>
   606→        </div>
   607→      </div>
   608→
   609→      {/* Tiles grid */}
   610→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-4">
   611→        {zones.map((z) => {
   612→          const status =
   613→            z.lowStockCount >= 3
   614→              ? "critical"
   615→              : z.lowStockCount > 0
   616→                ? "attention"
   617→                : "healthy";
   618→          const statusClasses =
   619→            status === "critical"
   620→              ? "border-red-500/40 hover:border-red-500/70"
   621→              : status === "attention"
   622→                ? "border-amber-500/40 hover:border-amber-500/70"
   623→                : "border-emerald-500/30 hover:border-emerald-500/60";
   624→          const minHeight = 120 + Math.min(z.totalValue / 100000, 80);
   625→
   626→          return (
   627→            <button
   628→              key={z.id}
   629→              onClick={() => onZoneClick(z.name)}
   630→              style={{ minHeight }}
   631→              className={cn(
   632→                "group bg-card border rounded-xl p-4 sm:p-5 text-left transition-all flex flex-col justify-between",
   633→                "hover:shadow-lg hover:shadow-black/20 hover:-translate-y-0.5",
   634→                statusClasses

  1521→// Simple RFC-4180-aware CSV parser
  1522→function parseCSV(text: string): string[][] {
  1523→  const rows: string[][] = [];
  1524→  let row: string[] = [];
  1525→  let cell = "";
  1526→  let inQuotes = false;
  1527→  for (let i = 0; i < text.length; i++) {
  1528→    const c = text[i];
  1529→    if (inQuotes) {
  1530→      if (c === '"') {
  1531→        if (text[i + 1] === '"') {
  1532→          cell += '"';
  1533→          i++;
  1534→        } else {
  1535→          inQuotes = false;
  1536→        }
  1537→      } else {
  1538→        cell += c;
  1539→      }
  1540→    } else {
  1541→      if (c === '"') {
  1542→        inQuotes = true;
  1543→      } else if (c === ",") {
  1544→        row.push(cell);
  1545→        cell = "";
  1546→      } else if (c === "\n") {
  1547→        row.push(cell);
  1548→        rows.push(row);
  1549→        row = [];
  1550→        cell = "";
  1551→      } else if (c === "\r") {
  1552→        // skip CR
  1553→      } else {
  1554→        cell += c;
  1555→      }
  1556→    }
  1557→  }
  1558→  if (cell !== "" || row.length > 0) {
  1559→    row.push(cell);
  1560→    rows.push(row);
  1561→  }
  1562→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
  1563→}
