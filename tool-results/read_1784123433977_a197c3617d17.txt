   931→// CSV escape — wraps in quotes if the field contains a comma, quote, or newline
   932→function csvEscape(value: string): string {
   933→  if (/[",\n]/.test(value)) {
   934→    return `"${value.replace(/"/g, '""')}"`;
   935→  }
   936→  return value;
   937→}

   939→// RFC-4180 CSV parser — handles quoted fields with embedded commas/newlines
   940→function parseCsv(text: string): string[][] {
   941→  const rows: string[][] = [];
   942→  let cur: string[] = [];
   943→  let field = "";
   944→  let inQuotes = false;
   945→  for (let i = 0; i < text.length; i++) {
   946→    const ch = text[i];
   947→    if (inQuotes) {
   948→      if (ch === '"' && text[i + 1] === '"') {
   949→        field += '"';
   950→        i++;
   951→      } else if (ch === '"') {
   952→        inQuotes = false;
   953→      } else {
   954→        field += ch;
   955→      }
   956→    } else {
   957→      if (ch === '"') inQuotes = true;
   958→      else if (ch === ",") {
   959→        cur.push(field);
   960→        field = "";
   961→      } else if (ch === "\n" || ch === "\r") {
   962→        if (ch === "\r" && text[i + 1] === "\n") i++;
   963→        cur.push(field);
   964→        rows.push(cur);
   965→        cur = [];
   966→        field = "";
   967→      } else {
   968→        field += ch;
   969→      }
   970→    }
   971→  }
   972→  if (field.length > 0 || cur.length > 0) {
   973→    cur.push(field);
   974→    rows.push(cur);
   975→  }
   976→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
   977→}
