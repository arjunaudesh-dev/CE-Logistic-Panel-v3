    84→
    85→export function AddProjectWizard({ open, onClose, onCreated }: Props) {
    86→  const [step, setStep] = useState<1 | 2>(1);
    87→  const [saving, setSaving] = useState(false);
    88→  const [createdProjectId, setCreatedProjectId] = useState<string | null>(null);
    89→  const [createdEstimateId, setCreatedEstimateId] = useState<string | null>(null);
    90→  const { toast } = useToast();
    91→  const fileInputRef = useRef<HTMLInputElement>(null);
    92→  const [previewRows, setPreviewRows] = useState<EstimateRow[] | null>(null);
    93→  const [customVoteNumber, setCustomVoteNumber] = useState(false);
    94→  const [selectedVoteId, setSelectedVoteId] = useState("");
    95→  // "Funds" checkbox state — when ticked, the project is funded via a non-vote fund
    96→  // (Malima/Non-Public/Contractual) and the Vote Number is NOT mandatory.
    97→  const [fundsChecked, setFundsChecked] = useState(false);
    98→  const { data: votes } = useFetch<Vote[]>("/api/votes");
    99→
   100→  const [form, setForm] = useState({
   101→    name: "",
   102→    voteNumber: "",
   103→    voteAllocation: "",
   104→    committed: "0",
   105→    actualSpent: "0",
   106→    startDate: "",
   107→    endDate: "", // EDC
   108→    benefactor: "", // replaces Department
   109→    reference: "", // manual text — how vote allocated
   110→    projectType: "New Construction",
   111→    financialMean: "Vote/Funds",
   112→    projectOfficer: "",
   113→    status: "Active",
   114→    description: "",
   115→  });
   116→  const [estimateRows, setEstimateRows] = useState<EstimateRow[]>([
   117→    { itemName: "", qty: "", unit: "NOS", estimateCost: "" },
   118→  ]);
   119→
   120→  const selectedVote = votes?.find((v) => v.id === selectedVoteId);
   121→
   122→  const handleVoteSelect = (id: string) => {
   123→    setSelectedVoteId(id);
   124→    const v = votes?.find((vt) => vt.id === id);
   125→    if (v) {
   126→      setForm((prev) => ({
   127→        ...prev,
   128→        voteNumber: v.voteNumber,
   129→        voteAllocation: v.allocation ? String(v.allocation) : prev.voteAllocation,
   130→      }));
   131→    }
   132→  };
   133→
   134→  // Toggle the "Funds" checkbox. When checked, the project is funded via a non-vote
   135→  // fund (default Malima Fund) and the Vote Number becomes optional. When unchecked,
   136→  // the project is funded via Vote and the Vote Number is mandatory.
   137→  const handleFundsToggle = (checked: boolean) => {
   138→    setFundsChecked(checked);
   139→    setCustomVoteNumber(false);
   140→    setSelectedVoteId("");
   141→    if (checked) {
   142→      setForm((prev) => ({
   143→        ...prev,
   144→        financialMean: "Malima Fund",
   145→        voteNumber: "",
   146→        voteAllocation: "",
   147→      }));
   148→    } else {
   149→      setForm((prev) => ({
   150→        ...prev,
   151→        financialMean: "Vote/Funds",
   152→        voteNumber: "",
   153→      }));
   154→    }
   155→  };
   156→
   157→  const handleFundSelect = (fund: string) => {
   158→    setForm((prev) => ({ ...prev, financialMean: fund }));
   159→  };
   160→
   161→  const nonVotePlaceholder =
   162→    form.financialMean === "Contractual" ? "CONTRACTUAL" : "NON-NAVAL";
   163→  const nonVoteHint =
   164→    form.financialMean === "Contractual"
   165→      ? "Contractual funding — Vote Number can be 'CONTRACTUAL' or the ARO reference."
   166→      : form.financialMean === "Malima Fund"
   167→      ? "Malima Fund — Vote Number can be 'NON-NAVAL' or the Malima reference number."
   168→      : "Non-Public Fund — Vote Number can be 'NON-NAVAL' or the fund reference.";
   169→
   170→  const addRow = () =>
   171→    setEstimateRows([...estimateRows, { itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
   172→  const removeRow = (idx: number) =>
   173→    setEstimateRows(estimateRows.filter((_, i) => i !== idx));
   174→  const updateRow = (idx: number, field: keyof EstimateRow, value: string) =>
   175→    setEstimateRows(estimateRows.map((r, i) => (i === idx ? { ...r, [field]: value } : r)));
   176→
   177→  // Valid (filled) estimate rows — at least one is required to enable Step 2
   178→  const validEstimateRows = estimateRows.filter(
   179→    (r) => r.itemName.trim() && (Number(r.qty) > 0 || Number(r.estimateCost) > 0)
   180→  );
   181→  const hasValidEstimate = validEstimateRows.length > 0;
   182→
   183→  const downloadTemplate = () => {
   184→    const header = "ItemName,Qty,Unit,EstimateCost";
   185→    const sample = "Centrifugal Pump 15kW,2,NOS,185000";
   186→    const csv = `${header}\n${sample}\n`;
   187→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   188→    const url = URL.createObjectURL(blob);
   189→    const a = document.createElement("a");
   190→    a.href = url;
   191→    a.download = "material-estimate-template.csv";
   192→    a.click();
   193→    URL.revokeObjectURL(url);
   194→    toast({ title: "Template downloaded", description: "Fill in ItemName, Qty, Unit, EstimateCost" });
   195→  };
   196→
   197→  // Simple CSV parser — handles quoted fields with commas
   198→  const parseCsv = (text: string): string[][] => {
   199→    const rows: string[][] = [];
   200→    let cur: string[] = [];
   201→    let field = "";
   202→    let inQuotes = false;
   203→    for (let i = 0; i < text.length; i++) {
   204→      const ch = text[i];
   205→      if (inQuotes) {
   206→        if (ch === '"' && text[i + 1] === '"') {
   207→          field += '"';
   208→          i++;
   209→        } else if (ch === '"') {
   210→          inQuotes = false;
   211→        } else {
   212→          field += ch;
   213→        }
   214→      } else {
   215→        if (ch === '"') inQuotes = true;
   216→        else if (ch === ",") {
   217→          cur.push(field);
   218→          field = "";
   219→        } else if (ch === "\n" || ch === "\r") {
   220→          if (ch === "\r" && text[i + 1] === "\n") i++;
   221→          cur.push(field);
   222→          rows.push(cur);
   223→          cur = [];
   224→          field = "";
   225→        } else {
   226→          field += ch;
   227→        }
   228→      }
   229→    }
   230→    if (field.length > 0 || cur.length > 0) {
   231→      cur.push(field);
   232→      rows.push(cur);
   233→    }
   234→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
   235→  };
   236→
   237→  const handleFileUpload = async (e: React.ChangeEvent<HTMLInputElement>) => {
   238→    const file = e.target.files?.[0];
   239→    if (!file) return;
   240→    if (!file.name.toLowerCase().endsWith(".csv")) {
   241→      toast({ title: "Please upload a CSV file", variant: "destructive" });
   242→      e.target.value = "";
   243→      return;
   244→    }
   245→    try {
   246→      const text = await file.text();
   247→      const rows = parseCsv(text);
   248→      if (rows.length === 0) {
   249→        toast({ title: "CSV file is empty", variant: "destructive" });
   250→        e.target.value = "";
   251→        return;
   252→      }
   253→      const header = rows[0].map((c) => c.trim().toLowerCase());
   254→      const idxName = header.findIndex((h) => h.includes("item") || h.includes("name"));
   255→      const idxQty = header.findIndex((h) => h.includes("qty"));
   256→      const idxUnit = header.findIndex((h) => h.includes("unit") || h.includes("deno"));
   257→      const idxCost = header.findIndex((h) => h.includes("cost") || h.includes("price"));
   258→      const dataRows = idxName >= 0 ? rows.slice(1) : rows;
   259→      const parsed: EstimateRow[] = dataRows
   260→        .map((r) => ({
   261→          itemName: idxName >= 0 ? (r[idxName] || "").trim() : (r[0] || "").trim(),
   262→          qty: idxQty >= 0 ? (r[idxQty] || "").trim() : (r[1] || "").trim(),
   263→          unit: (idxUnit >= 0 ? (r[idxUnit] || "").trim() : (r[2] || "").trim()) || "NOS",
   264→          estimateCost: idxCost >= 0 ? (r[idxCost] || "").trim() : (r[3] || "").trim(),
   265→        }))
   266→        .filter((r) => r.itemName);
   267→      if (parsed.length === 0) {
   268→        toast({ title: "No valid rows found in CSV", variant: "destructive" });
   269→        e.target.value = "";
   270→        return;
   271→      }
   272→      setPreviewRows(parsed);
   273→    } catch {
   274→      toast({ title: "Failed to read CSV file", variant: "destructive" });
   275→    }
   276→    e.target.value = "";
   277→  };
   278→
   279→  const confirmImport = () => {
   280→    if (!previewRows) return;
   281→    setEstimateRows(previewRows);
   282→    toast({ title: `Imported ${previewRows.length} estimate rows` });
   283→    setPreviewRows(null);
   284→  };
   285→
   286→  const handleSaveAndProcess = async () => {
   287→    if (!form.name) {
   288→      toast({ title: "Project Name is required", variant: "destructive" });
   289→      return;
   290→    }
   291→    // Vote Number is mandatory UNLESS the Funds checkbox is ticked (non-vote funding).
   292→    if (!fundsChecked && !form.voteNumber.trim()) {
   293→      toast({
   294→        title: "Vote Number is required",
   295→        description: "Tick the Funds checkbox for non-vote funding (Malima/Non-Public/Contractual).",
   296→        variant: "destructive",
   297→      });
   298→      return;
   299→    }
   300→    const validRows = estimateRows.filter((r) => r.itemName.trim());
   301→    if (validRows.length === 0) {
   302→      toast({ title: "Add at least one material estimate item", variant: "destructive" });
   303→      return;
   304→    }
   305→    setSaving(true);
   306→    // 1. Create project
   307→    const projRes = await apiRequest("/api/projects", "POST", form);
   308→    if (!projRes.ok) {
   309→      setSaving(false);
   310→      toast({ title: "Failed to create project", description: projRes.error, variant: "destructive" });
   311→      return;
   312→    }
   313→    const project = projRes.data as { id: string };
   314→    // 2. Export Estimate into DB
   315→    const estRes = await apiRequest("/api/estimates", "POST", {
   316→      projectId: project.id,
   317→      name: `${form.name} — Material Estimate`,
   318→      items: validRows.map((r) => ({
   319→        itemName: r.itemName,
   320→        qty: Number(r.qty) || 0,
   321→        unit: r.unit,
   322→        estimateCost: Number(r.estimateCost) || 0,
   323→      })),
   324→    });
   325→    setSaving(false);
   326→    if (estRes.ok) {
   327→      const est = estRes.data as { id: string };
   328→      setCreatedProjectId(project.id);
   329→      setCreatedEstimateId(est.id);
   330→      setStep(2);
   331→      toast({ title: "Project & estimate saved", description: "Now process the estimate with AI (Step 2)" });
   332→    } else {
   333→      toast({ title: "Estimate save failed", description: estRes.error, variant: "destructive" });
   334→    }
   335→  };
   336→
   337→  const handleGoToProject = () => {
   338→    if (createdProjectId) {
   339→      onCreated({
   340→        id: createdProjectId,
   341→        name: form.name,
   342→        voteNumber: form.voteNumber,
   343→        voteAllocation: Number(form.voteAllocation) || 0,
   344→        committed: Number(form.committed) || 0,
   345→        actualSpent: Number(form.actualSpent) || 0,
   346→        benefactor: form.benefactor || null,
   347→        reference: form.reference || null,
   348→        endDate: form.endDate || null,
   349→        startDate: form.startDate || null,
   350→        projectType: form.projectType,
   351→        financialMean: form.financialMean,
   352→      });
   353→      // reset
   354→      setStep(1);
   355→      setCreatedProjectId(null);
   356→      setCreatedEstimateId(null);
   357→      setFundsChecked(false);
   358→      setForm({
   359→        name: "", voteNumber: "", voteAllocation: "", committed: "0", actualSpent: "0",
   360→        startDate: "", endDate: "", benefactor: "", reference: "",
   361→        projectType: "New Construction", financialMean: "Vote/Funds",
   362→        projectOfficer: "", status: "Active", description: "",
   363→      });
   364→      setEstimateRows([{ itemName: "", qty: "", unit: "NOS", estimateCost: "" }]);
   365→      setCustomVoteNumber(false);
   366→      setSelectedVoteId("");
   367→    }
   368→  };
   369→
   370→  if (!open) return null;
   371→
   372→  return (
   373→    <div className="fixed inset-0 z-50 bg-background flex flex-col">
   374→      {/* Header */}
   375→      <div className="h-14 border-b border-border bg-card flex items-center justify-between px-4 sm:px-6 shrink-0">
   376→        <div className="flex items-center gap-3">
   377→          <button onClick={onClose} className="text-muted-foreground hover:text-foreground">
   378→            <X className="h-5 w-5" />
   379→          </button>
   380→          <h2 className="font-semibold text-foreground text-sm sm:text-base">
   381→            Add New Project — {step === 1 ? "Step 1: Details & Material Estimate" : "Step 2: Processed Estimate (AI)"}
   382→          </h2>
   383→        </div>
   384→        {/* Stepper */}
   385→        <div className="hidden sm:flex items-center gap-2 text-xs">
   386→          <StepBadge n={1} active={step === 1} done={step > 1} label="Details + Estimate" />
   387→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   388→          <StepBadge n={2} active={step === 2} done={false} label="AI Processing" />
   389→          <ArrowRight className="h-3 w-3 text-muted-foreground" />
   390→          <StepBadge n={3} active={false} done={false} label="Project Window" />
   391→        </div>
   392→      </div>
   393→
   394→      <div className="flex-1 overflow-y-auto scrollbar-thin">
   395→        {step === 1 && (
   396→          <div className="max-w-5xl mx-auto p-4 sm:p-6 space-y-6">
   397→            {/* Project details */}
   398→            <section className="bg-card border border-border rounded-xl p-5">
   399→              <h3 className="text-sm font-semibold text-foreground mb-4 flex items-center gap-2">
   400→                <ClipboardList className="h-4 w-4 text-primary" /> Project Details
   401→              </h3>
   402→              <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   403→                <div className="sm:col-span-2 space-y-1.5">
   404→                  <Label>Project Name *</Label>
   405→                  <Input value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} placeholder="e.g. Dockyard Refit 2025" className="bg-background border-border" />
   406→                </div>
   407→
   408→                {/* Vote Number — conditional UI based on Funds checkbox state */}
   409→                <div className="space-y-1.5">
   410→                  <Label className="flex items-center gap-1.5">
   411→                    <Vote className="h-3.5 w-3.5 text-primary" />
   412→                    Vote Number{!fundsChecked && " *"}
   413→                    {fundsChecked && (
   414→                      <span className="text-[10px] font-normal text-muted-foreground ml-1">(optional)</span>
   415→                    )}
   416→                  </Label>
   417→                  {!fundsChecked && !customVoteNumber ? (
   418→                    <>
   419→                      <Select value={selectedVoteId} onValueChange={handleVoteSelect}>
   420→                        <SelectTrigger className="bg-background border-border">
   421→                          <SelectValue placeholder="Select a vote from the list…" />
   422→                        </SelectTrigger>
   423→                        <SelectContent>
   424→                          {votes && votes.length > 0 ? (
   425→                            votes.map((v) => (
   426→                              <SelectItem key={v.id} value={v.id}>
   427→                                <span className="font-mono">{v.voteNumber}</span> — {v.description} ({v.voteType})
   428→                              </SelectItem>
   429→                            ))
   430→                          ) : (
   431→                            <SelectItem value="_none" disabled>No votes available</SelectItem>
   432→                          )}
   433→                        </SelectContent>
   434→                      </Select>
   435→                      {selectedVote ? (
   436→                        <div className="flex items-center gap-2 pt-0.5 flex-wrap">
   437→                          <CustomBadge className={cn("border", voteTypeColor(selectedVote.voteType))}>
   438→                            {selectedVote.voteType}
   439→                          </CustomBadge>
   440→                          {selectedVote.allocation > 0 && (
   441→                            <span className="text-[11px] text-muted-foreground">
   442→                              Allocation: {formatCurrency(selectedVote.allocation)}
   443→                            </span>
   444→                          )}
   445→                          {selectedVote.financialYear && (
   446→                            <span className="text-[11px] text-muted-foreground">
   447→                              · FY: {selectedVote.financialYear}
   448→                            </span>
   449→                          )}
   450→                          {!selectedVote.active && (
   451→                            <CustomBadge className="border-0 bg-amber-500/15 text-amber-400">Inactive</CustomBadge>
   452→                          )}
   453→                        </div>
   454→                      ) : (
   455→                        <p className="text-[11px] text-muted-foreground flex items-start gap-1">
   456→                          <Info className="h-3 w-3 mt-0.5 shrink-0" />
   457→                          Capital votes fund one-time construction. Recurring votes fund annual maintenance.
   458→                        </p>
   459→                      )}
   460→                      <button
   461→                        type="button"
   462→                        onClick={() => {
   463→                          setCustomVoteNumber(true);
   464→                          setSelectedVoteId("");
   465→                          setForm((prev) => ({ ...prev, voteNumber: "" }));
   466→                        }}
   467→                        className="text-[11px] text-primary hover:underline flex items-center gap-1 self-start"
   468→                      >
   469→                        <Pencil className="h-3 w-3" /> Enter custom vote number
   470→                      </button>
   471→                    </>
   472→                  ) : !fundsChecked && customVoteNumber ? (
   473→                    <>
   474→                      <Input
   475→                        value={form.voteNumber}
   476→                        onChange={(e) => setForm({ ...form, voteNumber: e.target.value })}
   477→                        placeholder="e.g. 223-1-01-2104"
   478→                        className="bg-background border-border"
   479→                      />
   480→                      <button
   481→                        type="button"
   482→                        onClick={() => {
   483→                          setCustomVoteNumber(false);
   484→                          setForm((prev) => ({ ...prev, voteNumber: "" }));
   485→                        }}
   486→                        className="text-[11px] text-primary hover:underline flex items-center gap-1 self-start"
   487→                      >
   488→                        <ChevronLeft className="h-3 w-3" /> Back to vote list
   489→                      </button>
   490→                    </>
   491→                  ) : (
   492→                    <>
   493→                      <Input
   494→                        value={form.voteNumber}
   495→                        onChange={(e) => setForm({ ...form, voteNumber: e.target.value })}
   496→                        placeholder={nonVotePlaceholder}
   497→                        className="bg-background border-border"
   498→                      />
   499→                      <p className="text-[11px] text-amber-400 flex items-start gap-1">
   500→                        <Landmark className="h-3 w-3 mt-0.5 shrink-0" />
   501→                        {nonVoteHint}
   502→                      </p>
   503→                    </>
   504→                  )}
   505→                </div>
   506→
   507→                {/* Project Type */}
   508→                <div className="space-y-1.5">
   509→                  <Label>Project Type *</Label>
   510→                  <Select value={form.projectType} onValueChange={(v) => setForm({ ...form, projectType: v })}>
   511→                    <SelectTrigger className="bg-background border-border">
   512→                      <SelectValue />
   513→                    </SelectTrigger>
   514→                    <SelectContent>
   515→                      {PROJECT_TYPES.map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}
   516→                    </SelectContent>
   517→                  </Select>
   518→                  {form.projectType && (
   519→                    <div className="pt-0.5">
   520→                      <CustomBadge className={cn("border", projectTypeColor(form.projectType))}>
   521→                        {form.projectType}
   522→                      </CustomBadge>
   523→                    </div>
   524→                  )}
   525→                </div>
   526→
   527→                {/* Funding Source — Funds checkbox (non-vote funding). NOT mandatory. */}
   528→                <div className="space-y-1.5">
   529→                  <Label className="flex items-center gap-1.5">
   530→                    <Wallet className="h-3.5 w-3.5 text-primary" />
