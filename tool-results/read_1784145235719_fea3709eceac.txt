   418→function looselyMatches(supplier: string | null | undefined, name: string): boolean {
   419→  if (!supplier || !name) return false;
   420→  const a = supplier.toLowerCase().trim();
   421→  const b = name.toLowerCase().trim();
   422→  if (!a || !b) return false;
   423→  return a.includes(b) || b.includes(a);

   494→// Generic project/vote filter applied to any row that has a supplier-like field.
   495→// Pure helper (module-level) so the React Compiler can preserve manual memoization.
   496→function applyProjVoteFilter<
   497→  T extends { receivedFrom?: string | null; supplier?: string | null }
   498→>(
   499→  rows: T[],
   500→  refField: "receivedFrom" | "supplier",
   501→  projFilter: string,
   502→  voteFilt: string,
   503→  projectById: Map<string, Project>,
   504→  projectByVote: Map<string, Project>
   505→): T[] {
   506→  let out = rows;
   507→  if (projFilter !== "all") {
   508→    const p = projectById.get(projFilter);
   509→    if (p) {
   510→      const matches = out.filter((r) =>
   511→        looselyMatches(r[refField] ?? null, p.name)
   512→      );
   513→      if (matches.length > 0) out = matches;
   514→    }
   515→  }
   516→  if (voteFilt !== "all") {
   517→    const p = projectByVote.get(voteFilt);
   518→    if (p) {
   519→      const matches = out.filter((r) =>
   520→        looselyMatches(r[refField] ?? null, p.name)
   521→      );
   522→      if (matches.length > 0) out = matches;
   523→    }
   524→  }
   525→  return out;
   526→}

   562→// ─────────────────────────────────────────────────────────
   563→// Split-Order helpers — encode / decode Linked PO numbers in the
   564→// PartSupplyOrder.supplier field (since the model has no dedicated column).
   565→// Format: `<supplier> [PO: PO-001, PO-002]`
   566→// ─────────────────────────────────────────────────────────
   567→const SPLIT_PO_PREFIX = " [PO: ";
   568→const SPLIT_PO_SUFFIX = "]";
   569→
   570→function formatSupplierWithPos(supplier: string, pos: string[]): string {
   571→  const trimmedSupplier = supplier.trim();
   572→  const cleanPos = pos.map((p) => p.trim()).filter(Boolean);
   573→  if (cleanPos.length === 0) return trimmedSupplier;
   574→  return `${trimmedSupplier}${SPLIT_PO_PREFIX}${cleanPos.join(", ")}${SPLIT_PO_SUFFIX}`;
   575→}
   576→
   577→function parseSupplierWithPos(
   578→  raw: string | null | undefined
   579→): { supplier: string; pos: string[] } {
   580→  if (!raw) return { supplier: "", pos: [] };
   581→  const text = raw;
   582→  const startIdx = text.lastIndexOf(SPLIT_PO_PREFIX);
   583→  if (startIdx === -1 || !text.endsWith(SPLIT_PO_SUFFIX)) {
   584→    return { supplier: text, pos: [] };
   585→  }
   586→  const endIdx = text.length - SPLIT_PO_SUFFIX.length;
   587→  const poStr = text.slice(startIdx + SPLIT_PO_PREFIX.length, endIdx);
   588→  const pos = poStr
   589→    .split(",")
   590→    .map((p) => p.trim())
   591→    .filter(Boolean);
   592→  const supplier = text.slice(0, startIdx).trim();
   593→  return { supplier, pos };
   594→}

   596→// ─────────────────────────────────────────────────────────
   597→// useAllProjectFiles — fetch all project files across every project.
   598→// Used by the Collection Pipeline (Dashboard), the QCI tab (to find the
   599→// matching file when a QCI is recorded) and the Send-to-Area-Store dialog
   600→// (to update the file's collectionStatus).
   601→// ─────────────────────────────────────────────────────────
   602→function useAllProjectFiles(enabled: boolean = true): {
   603→  data: ProjectFile[];
   604→  loading: boolean;
   605→  error: string | null;
   606→  refresh: () => void;
   607→} {
   608→  const { data: projects, loading: pLoading, error: pError } =
   609→    useFetch<Project[]>(enabled ? "/api/projects" : null);
   610→  const projectIds = useMemo(
   611→    () => (projects ?? []).map((p) => p.id),
   612→    [projects]
   613→  );
   614→  const [files, setFiles] = useState<ProjectFile[]>([]);
   615→  const [fetching, setFetching] = useState<boolean>(false);
   616→  const [fetchError, setFetchError] = useState<string | null>(null);
   617→  const [nonce, setNonce] = useState(0);
   618→
   619→  const refresh = useCallback(() => setNonce((n) => n + 1), []);
   620→
   621→  useEffect(() => {
   622→    if (!enabled) return;
   623→    if (pLoading || pError) return;
   624→    if (projectIds.length === 0) return;
   625→    let cancelled = false;
   626→    (async () => {
   627→      // Defer the loading-state setter so it isn't a synchronous setState
   628→      // in the effect body (mirrors the `useFetch` pattern).
   629→      await Promise.resolve();
   630→      if (cancelled) return;
   631→      setFetching(true);
   632→      setFetchError(null);
   633→      try {
   634→        const results = await Promise.all(
   635→          projectIds.map(async (pid) => {
   636→            try {
   637→              const res = await fetch(
   638→                `/api/projects/${pid}/files?includeArchived=false`
   639→              );
   640→              if (!res.ok) return [] as ProjectFile[];
   641→              const json = (await res.json()) as ProjectFile[];
   642→              return Array.isArray(json) ? json : [];
   643→            } catch {
   644→              return [] as ProjectFile[];
   645→            }
   646→          })
   647→        );
   648→        if (cancelled) return;
   649→        setFiles(results.flat());
   650→        setFetching(false);
   651→      } catch (e) {
   652→        if (cancelled) return;
   653→        setFetchError((e as Error).message || "Failed to load files");
   654→        setFetching(false);
   655→      }
   656→    })();
   657→    return () => {
   658→      cancelled = true;
   659→    };
   660→  }, [projectIds, pLoading, pError, nonce, enabled]);
   661→
   662→  // Derive the final hook state without synchronous setState in the effect.
   663→  if (!enabled) return { data: [], loading: false, error: null, refresh };
   664→  if (pError) return { data: [], loading: false, error: pError, refresh };
   665→  if (pLoading) return { data: [], loading: true, error: null, refresh };
   666→  if (projectIds.length === 0)
   667→    return { data: [], loading: false, error: null, refresh };
   668→  return { data: files, loading: fetching, error: fetchError, refresh };
   669→}

   671→/** Update a project file's collection status via the workflow PUT endpoint. */
   672→async function updateFileCollectionStatus(
   673→  file: { id: string; projectId: string },
   674→  collectionStatus: string,
   675→  extras?: {
   676→    areaStoreName?: string;
   677→    navReference?: string;
   678→    navNumber?: string;
   679→    changedBy?: string;
   680→  }
   681→): Promise<{ ok: boolean; error?: string }> {
   682→  const res = await apiRequest(
   683→    `/api/projects/${file.projectId}/files/${file.id}/collection-status`,
   684→    "PUT",
   685→    {
   686→      collectionStatus,
   687→      ...extras,
   688→    }
   689→  );
   690→  if (!res.ok) return { ok: false, error: res.error };
   691→  return { ok: true };
   692→}
