    78→export function PresentPositionDialog({ project, open, onOpenChange, onChanged }: Props) {
    79→  const { data: files, loading, refresh } = useFetch<ProjectFile[]>(
    80→    `/api/projects/${project.id}/files`
    81→  );
    82→  const [search, setSearch] = useState("");
    83→  const [positionFilter, setPositionFilter] = useState("all");
    84→  const [addOpen, setAddOpen] = useState(false);
    85→  const [editFile, setEditFile] = useState<ProjectFile | null>(null);
    86→  const [logFile, setLogFile] = useState<ProjectFile | null>(null);
    87→  const [allocOpen, setAllocOpen] = useState(false);
    88→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
    89→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
    90→  const { toast } = useToast();
    91→
    92→  // Position summary
    93→  const positionSummary = useMemo(() => {
    94→    const map: Record<string, { count: number; cost: number }> = {};
    95→    files?.forEach((f) => {
    96→      if (!map[f.presentPosition]) map[f.presentPosition] = { count: 0, cost: 0 };
    97→      map[f.presentPosition].count++;
    98→      map[f.presentPosition].cost += f.fileCost;
    99→    });
   100→    return Object.entries(map).sort((a, b) => b[1].count - a[1].count);
   101→  }, [files]);
   102→
   103→  const totalFileCost = files?.reduce((s, f) => s + f.fileCost, 0) ?? 0;
   104→
   105→  const filteredFiles = useMemo(() => {
   106→    let list = files || [];
   107→    if (search) {
   108→      const q = search.toLowerCase();
   109→      list = list.filter(
   110→        (f) =>
   111→          f.prNumber.toLowerCase().includes(q) ||
   112→          f.itemCode.toLowerCase().includes(q) ||
   113→          f.itemName.toLowerCase().includes(q) ||
   114→          (f.supplier || "").toLowerCase().includes(q)
   115→      );
   116→    }
   117→    if (positionFilter !== "all") {
   118→      list = list.filter((f) => f.presentPosition === positionFilter);
   119→    }
   120→    list = [...list].sort((a, b) => {
   121→      let cmp = 0;
   122→      if (sortKey === "fileCost") cmp = a.fileCost - b.fileCost;
   123→      else if (sortKey === "prNumber") cmp = a.prNumber.localeCompare(b.prNumber);
   124→      else cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   125→      return sortDir === "asc" ? cmp : -cmp;
   126→    });
   127→    return list;
   128→  }, [files, search, positionFilter, sortKey, sortDir]);
   129→
   130→  const toggleSort = (key: typeof sortKey) => {
   131→    if (sortKey === key) setSortDir(sortDir === "asc" ? "desc" : "asc");
   132→    else {
   133→      setSortKey(key);
   134→      setSortDir("desc");
   135→    }
   136→  };
   137→
   138→  const handleDelete = async (file: ProjectFile) => {
   139→    if (!confirm(`Delete file ${file.prNumber} (${file.itemName})?`)) return;
   140→    const res = await apiRequest(`/api/projects/${project.id}/files/${file.id}`, "DELETE");
   141→    if (res.ok) {
   142→      toast({ title: "File deleted" });
   143→      refresh();
   144→      onChanged();
   145→    } else {
   146→      toast({ title: "Delete failed", description: res.error, variant: "destructive" });
   147→    }
   148→  };
   149→
   150→  const exportExcel = () => {
   151→    const rows = filteredFiles.map((f) => ({
   152→      "PR Number": f.prNumber,
   153→      "Item Code": f.itemCode,
   154→      "Item Name": f.itemName,
   155→      Unit: f.unit,
   156→      Qty: f.qty,
   157→      "Unit Price": f.unitPrice,
   158→      "File Cost": f.fileCost,
   159→      Supplier: f.supplier || "",
   160→      "Present Position": f.presentPosition,
   161→      Category: f.fileCategory,
   162→    }));
   163→    const headers = Object.keys(rows[0] || {});
   164→    const csv = [
   165→      headers.join(","),
   166→      ...rows.map((r) => headers.map((h) => `"${(r as Record<string, unknown>)[h]}"`).join(",")),
   167→    ].join("\n");
   168→    const blob = new Blob([csv], { type: "text/csv" });
   169→    const url = URL.createObjectURL(blob);
   170→    const a = document.createElement("a");
   171→    a.href = url;
   172→    a.download = `${project.name}-files.csv`;
   173→    a.click();
   174→    URL.revokeObjectURL(url);
   175→  };
   176→
   177→  return (
   178→    <Dialog open={open} onOpenChange={onOpenChange}>
   179→      <DialogContent className="max-w-7xl max-h-[92vh] bg-card border-border p-0 gap-0">
   180→        <DialogHeader className="px-6 py-4 border-b border-border">
   181→          <DialogTitle className="text-foreground flex items-center justify-between gap-4">
   182→            <span className="truncate">{project.name}</span>
   183→            <span className="text-xs font-normal text-muted-foreground">
   184→              {project.voteNumber}
   185→            </span>
   186→          </DialogTitle>
   187→        </DialogHeader>
   188→
   189→        <ScrollArea className="max-h-[80vh]">
   190→          <div className="p-6 space-y-5">
   191→            {/* Project financial summary */}
   192→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   193→              <FinBox label="Vote Allocation" value={formatCurrency(project.voteAllocation)} tone="text-blue-400" />
   194→              <FinBox label="Committed" value={formatCurrency(project.committed)} tone="text-amber-400" />
   195→              <FinBox label="Actual Spent" value={formatCurrency(project.actualSpent)} tone="text-red-400" />
   196→              <FinBox label="File Cost (Total)" value={formatCurrency(totalFileCost)} tone="text-emerald-400" />
   197→            </div>
   198→
   199→            {/* Position summary bar */}
   200→            <div>
   201→              <h4 className="text-xs font-semibold text-muted-foreground uppercase tracking-wide mb-2">
   202→                Present Position Summary
   203→              </h4>
   204→              <div className="flex flex-wrap gap-2">
   205→                {positionSummary.length === 0 ? (
   206→                  <p className="text-sm text-muted-foreground">No files yet.</p>
   207→                ) : (
   208→                  positionSummary.map(([pos, info]) => (
   209→                    <div
   210→                      key={pos}
   211→                      className={cn(
   212→                        "flex items-center gap-2 rounded-lg border px-3 py-1.5",
   213→                        positionColor(pos)
   214→                      )}
   215→                    >
   216→                      <span className="text-sm font-medium">{pos}</span>
   217→                      <span className="text-xs opacity-80">
   218→                        {info.count} file{info.count > 1 ? "s" : ""} · {formatCurrency(info.cost)}
   219→                      </span>
   220→                    </div>
   221→                  ))
   222→                )}
   223→              </div>
   224→            </div>
   225→
   226→            {/* Toolbar */}
   227→            <div className="flex flex-wrap items-center gap-2">
   228→              <div className="relative flex-1 min-w-[200px]">
   229→                <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   230→                <Input
   231→                  value={search}
   232→                  onChange={(e) => setSearch(e.target.value)}
   233→                  placeholder="Search PR, item, supplier..."
   234→                  className="pl-9 bg-background border-border"
   235→                />
   236→              </div>
   237→              <Select value={positionFilter} onValueChange={setPositionFilter}>
   238→                <SelectTrigger className="w-[180px] bg-background border-border">
   239→                  <SelectValue placeholder="Position" />
   240→                </SelectTrigger>
   241→                <SelectContent>
   242→                  <SelectItem value="all">All Positions</SelectItem>
   243→                  {PRESENT_POSITIONS.map((p) => (
   244→                    <SelectItem key={p} value={p}>{p}</SelectItem>
   245→                  ))}
   246→                </SelectContent>
   247→              </Select>
   248→              <Button variant="outline" size="sm" onClick={() => setAllocOpen(true)} className="gap-1.5">
   249→                <History className="h-4 w-4" /> Allocations
   250→              </Button>
   251→              <Button variant="outline" size="sm" onClick={exportExcel} className="gap-1.5">
   252→                <Download className="h-4 w-4" /> Export
   253→              </Button>
   254→              <Button size="sm" onClick={() => setAddOpen(true)} className="gap-1.5">
   255→                <Plus className="h-4 w-4" /> Add File
   256→              </Button>
   257→            </div>
   258→
   259→            {/* File register table */}
   260→            <div className="border border-border rounded-lg overflow-hidden">
   261→              <Table>
   262→                <TableHeader>
   263→                  <TableRow className="hover:bg-transparent border-border">
   264→                    <SortTh label="PR Number" onClick={() => toggleSort("prNumber")} active={sortKey === "prNumber"} dir={sortDir} />
   265→                    <TableHead className="text-muted-foreground">Item Code</TableHead>
   266→                    <TableHead className="text-muted-foreground">Item Name</TableHead>
   267→                    <TableHead className="text-muted-foreground text-right">Qty</TableHead>
   268→                    <TableHead className="text-muted-foreground text-right">Unit Price</TableHead>
   269→                    <SortTh label="File Cost" onClick={() => toggleSort("fileCost")} active={sortKey === "fileCost"} dir={sortDir} align="right" />
   270→                    <TableHead className="text-muted-foreground">Supplier</TableHead>
   271→                    <TableHead className="text-muted-foreground">Position</TableHead>
   272→                    <TableHead className="text-muted-foreground">Category</TableHead>
   273→                    <TableHead className="text-muted-foreground text-right">Actions</TableHead>
   274→                  </TableRow>
   275→                </TableHeader>
   276→                <TableBody>
   277→                  {loading ? (
   278→                    <TableRow>
   279→                      <TableCell colSpan={10} className="text-center py-8 text-muted-foreground">
   280→                        <Loader2 className="h-5 w-5 animate-spin mx-auto" />
   281→                      </TableCell>
   282→                    </TableRow>
   283→                  ) : filteredFiles.length === 0 ? (
   284→                    <TableRow>
   285→                      <TableCell colSpan={10} className="text-center py-8 text-muted-foreground">
   286→                        No files found. Click "Add File" to create one.
   287→                      </TableCell>
   288→                    </TableRow>
   289→                  ) : (
   290→                    filteredFiles.map((f) => (
   291→                      <TableRow
   292→                        key={f.id}
   293→                        className="border-border cursor-pointer hover:bg-secondary/40"
   294→                        onClick={() => setLogFile(f)}
   295→                      >
   296→                        <TableCell className="font-mono text-xs text-foreground">{f.prNumber}</TableCell>
   297→                        <TableCell className="font-mono text-xs">{f.itemCode}</TableCell>
   298→                        <TableCell className="text-foreground">{f.itemName}</TableCell>
   299→                        <TableCell className="text-right">{f.qty} {f.unit}</TableCell>
   300→                        <TableCell className="text-right">{formatCurrency(f.unitPrice)}</TableCell>
   301→                        <TableCell className="text-right font-medium text-foreground">{formatCurrency(f.fileCost)}</TableCell>
   302→                        <TableCell className="text-xs">{f.supplier || "—"}</TableCell>
   303→                        <TableCell><PositionBadge position={f.presentPosition} /></TableCell>
   304→                        <TableCell><CategoryBadge category={f.fileCategory} /></TableCell>
   305→                        <TableCell className="text-right">
   306→                          <div className="flex items-center justify-end gap-1" onClick={(e) => e.stopPropagation()}>
   307→                            <Button
   308→                              variant="ghost"
   309→                              size="icon"
   310→                              className="h-7 w-7"
   311→                              onClick={() => setEditFile(f)}
   312→                            >
   313→                              <Pencil className="h-3.5 w-3.5" />
   314→                            </Button>
   315→                            <Button
   316→                              variant="ghost"
   317→                              size="icon"
   318→                              className="h-7 w-7 text-red-400 hover:text-red-500"
   319→                              onClick={() => handleDelete(f)}
   320→                            >
   321→                              <Trash2 className="h-3.5 w-3.5" />
   322→                            </Button>
   323→                          </div>
   324→                        </TableCell>
   325→                      </TableRow>
   326→                    ))
   327→                  )}
   328→                </TableBody>
   329→              </Table>
   330→            </div>
   331→            <p className="text-xs text-muted-foreground">
   332→              {filteredFiles.length} file(s) · Click any row to view full log
   333→            </p>
   334→          </div>
   335→        </ScrollArea>
   336→      </DialogContent>
   337→
   338→      {/* Add File */}
   339→      <FileFormDialog
   340→        projectId={project.id}
   341→        open={addOpen}
   342→        onOpenChange={setAddOpen}
   343→        onSaved={() => {
   344→          setAddOpen(false);
   345→          refresh();
   346→          onChanged();
   347→        }}
   348→      />
   349→
   350→      {/* Edit File / Update Position */}
   351→      {editFile && (
   352→        <FileFormDialog
   353→          projectId={project.id}
   354→          file={editFile}
   355→          open={!!editFile}
   356→          onOpenChange={(o) => !o && setEditFile(null)}
   357→          onSaved={() => {
   358→            setEditFile(null);
   359→            refresh();
   360→            onChanged();
   361→          }}
   362→        />
   363→      )}
   364→
   365→      {/* File Log */}
   366→      <FileLogDialog file={logFile} open={!!logFile} onOpenChange={(o) => !o && setLogFile(null)} />
   367→
   368→      {/* Allocation History */}
   369→      <AllocationHistoryDialog
   370→        project={project}
   371→        open={allocOpen}
   372→        onOpenChange={setAllocOpen}
   373→        onChanged={onChanged}
   374→      />
   375→    </Dialog>
   376→  );
   377→}
   378→
   379→function FinBox({ label, value, tone }: { label: string; value: string; tone: string }) {
   380→  return (
   381→    <div className="bg-secondary/40 rounded-lg p-3">
   382→      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">{label}</p>
   383→      <p className={`text-base font-semibold ${tone}`}>{value}</p>
   384→    </div>
   385→  );
   386→}
   387→
   388→function SortTh({
   389→  label,
   390→  onClick,
   391→  active,
   392→  dir,
   393→  align = "left",
   394→}: {
   395→  label: string;
   396→  onClick: () => void;
   397→  active: boolean;
   398→  dir: "asc" | "desc";
   399→  align?: "left" | "right";
   400→}) {
   401→  return (
   402→    <TableHead className={align === "right" ? "text-muted-foreground text-right" : "text-muted-foreground"}>
   403→      <button onClick={onClick} className="inline-flex items-center gap-1 hover:text-foreground">
   404→        {label}
   405→        {active && (dir === "asc" ? <ChevronUp className="h-3 w-3" /> : <ChevronDown className="h-3 w-3" />)}
   406→      </button>
   407→    </TableHead>
   408→  );
   409→}

   411→// ─── File Form (Add / Edit) ───
   412→function FileFormDialog({
   413→  projectId,
   414→  file,
   415→  open,
   416→  onOpenChange,
   417→  onSaved,
   418→}: {
   419→  projectId: string;
   420→  file?: ProjectFile;
   421→  open: boolean;
   422→  onOpenChange: (o: boolean) => void;
   423→  onSaved: () => void;
   424→}) {
   425→  const isEdit = !!file;
   426→  const [saving, setSaving] = useState(false);
   427→  const [form, setForm] = useState({
   428→    prNumber: file?.prNumber || "",
   429→    itemCode: file?.itemCode || "",
   430→    itemName: file?.itemName || "",
   431→    unit: file?.unit || "NOS",
   432→    qty: String(file?.qty ?? ""),
   433→    unitPrice: String(file?.unitPrice ?? ""),
   434→    fileCost: String(file?.fileCost ?? ""),
   435→    supplier: file?.supplier || "",
   436→    presentPosition: file?.presentPosition || "PR Raised",
   437→    fileCategory: file?.fileCategory || "General",
   438→    changedBy: "",
   439→    comment: "",
   440→  });
   441→  const { toast } = useToast();
   442→
   443→  const submit = async () => {
   444→    if (!form.prNumber || !form.itemCode || !form.itemName) {
   445→      toast({ title: "PR Number, Item Code and Item Name are required", variant: "destructive" });
   446→      return;
   447→    }
   448→    setSaving(true);
   449→    const url = isEdit
   450→      ? `/api/projects/${projectId}/files/${file!.id}`
   451→      : `/api/projects/${projectId}/files`;
   452→    const method = isEdit ? "PUT" : "POST";
   453→    const res = await apiRequest(url, method, form);
   454→    setSaving(false);
   455→    if (res.ok) {
   456→      toast({ title: isEdit ? "File updated" : "File added" });
   457→      onSaved();
   458→    } else {
   459→      toast({ title: "Failed", description: res.error, variant: "destructive" });
   460→    }
   461→  };
   462→
   463→  return (
   464→    <Dialog open={open} onOpenChange={onOpenChange}>
   465→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
   466→        <DialogHeader>
   467→          <DialogTitle className="text-foreground flex items-center gap-2">
   468→            <FileText className="h-5 w-5 text-primary" />
   469→            {isEdit ? `Edit File — ${file!.prNumber}` : "Add Processing File"}
   470→          </DialogTitle>
   471→        </DialogHeader>
   472→        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 py-2">
   473→          <Field label="PR Number *">
   474→            <Input value={form.prNumber} onChange={(e) => setForm({ ...form, prNumber: e.target.value })} className="bg-background border-border" />
   475→          </Field>
   476→          <Field label="Item Code *">
   477→            <Input value={form.itemCode} onChange={(e) => setForm({ ...form, itemCode: e.target.value })} className="bg-background border-border" />
   478→          </Field>
   479→          <div className="sm:col-span-2">
   480→            <Field label="Item Name *">
   481→              <Input value={form.itemName} onChange={(e) => setForm({ ...form, itemName: e.target.value })} className="bg-background border-border" />
   482→            </Field>
   483→          </div>
   484→          <Field label="Unit">
   485→            <Input value={form.unit} onChange={(e) => setForm({ ...form, unit: e.target.value })} className="bg-background border-border" />
   486→          </Field>
   487→          <Field label="Qty">
   488→            <Input type="number" value={form.qty} onChange={(e) => setForm({ ...form, qty: e.target.value })} className="bg-background border-border" />
   489→          </Field>
   490→          <Field label="Unit Price (Rs.)">
   491→            <Input type="number" value={form.unitPrice} onChange={(e) => setForm({ ...form, unitPrice: e.target.value })} className="bg-background border-border" />
   492→          </Field>
   493→          <Field label="File Cost (Rs.) — auto if empty">
   494→            <Input type="number" value={form.fileCost} onChange={(e) => setForm({ ...form, fileCost: e.target.value })} placeholder="auto" className="bg-background border-border" />
   495→          </Field>
   496→          <Field label="Supplier">
   497→            <Input value={form.supplier} onChange={(e) => setForm({ ...form, supplier: e.target.value })} className="bg-background border-border" />
   498→          </Field>
   499→          <Field label="Present Position">
   500→            <Select value={form.presentPosition} onValueChange={(v) => setForm({ ...form, presentPosition: v })}>
   501→              <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   502→              <SelectContent>
   503→                {PRESENT_POSITIONS.map((p) => <SelectItem key={p} value={p}>{p}</SelectItem>)}
   504→              </SelectContent>
   505→            </Select>
   506→          </Field>
   507→          <Field label="File Category">
   508→            <Select value={form.fileCategory} onValueChange={(v) => setForm({ ...form, fileCategory: v })}>
   509→              <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   510→              <SelectContent>
   511→                {FILE_CATEGORIES.map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}
   512→              </SelectContent>
   513→            </Select>
   514→          </Field>
   515→          {isEdit && (
   516→            <>
   517→              <Field label="Changed By">
   518→                <Input value={form.changedBy} onChange={(e) => setForm({ ...form, changedBy: e.target.value })} placeholder="Your name" className="bg-background border-border" />
   519→              </Field>
   520→              <div className="sm:col-span-2">
   521→                <Field label="Comment (for log)">
   522→                  <Input value={form.comment} onChange={(e) => setForm({ ...form, comment: e.target.value })} placeholder="Optional note" className="bg-background border-border" />
   523→                </Field>
   524→              </div>
   525→            </>
   526→          )}
   527→        </div>
   528→        <DialogFooter>
   529→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   530→          <Button onClick={submit} disabled={saving} className="gap-1.5">
   531→            {saving && <Loader2 className="h-4 w-4 animate-spin" />}
   532→            {isEdit ? "Update File" : "Add File"}
   533→          </Button>
   534→        </DialogFooter>
   535→      </DialogContent>
   536→    </Dialog>
   537→  );
   538→}

   549→// ─── Allocation History ───
   550→function AllocationHistoryDialog({
   551→  project,
   552→  open,
   553→  onOpenChange,
   554→  onChanged,
   555→}: {
   556→  project: Project;
   557→  open: boolean;
   558→  onOpenChange: (o: boolean) => void;
   559→  onChanged: () => void;
   560→}) {
   561→  const { data: allocations, refresh } = useFetch<Allocation[]>(
   562→    `/api/projects/${project.id}/allocations`
   563→  );
   564→  const [saving, setSaving] = useState(false);
   565→  const [form, setForm] = useState({
   566→    date: new Date().toISOString().slice(0, 10),
   567→    description: "",
   568→    amount: "",
   569→    type: "Commitment",
   570→    authorizedBy: "",
   571→  });
   572→  const { toast } = useToast();
   573→
   574→  const totalAllocated = allocations?.reduce((s, a) => s + a.amount, 0) ?? 0;
   575→  const committed = allocations?.filter((a) => a.type === "Commitment").reduce((s, a) => s + a.amount, 0) ?? 0;
   576→  const actual = allocations?.filter((a) => a.type === "Actual").reduce((s, a) => s + a.amount, 0) ?? 0;
   577→
   578→  const submit = async () => {
   579→    if (!form.description || !form.amount) {
   580→      toast({ title: "Description and amount required", variant: "destructive" });
   581→      return;
   582→    }
   583→    setSaving(true);
   584→    const res = await apiRequest(`/api/projects/${project.id}/allocations`, "POST", form);
   585→    setSaving(false);
   586→    if (res.ok) {
   587→      toast({ title: "Allocation added" });
   588→      setForm({ date: new Date().toISOString().slice(0, 10), description: "", amount: "", type: "Commitment", authorizedBy: "" });
   589→      refresh();
   590→      onChanged();
   591→    } else {
   592→      toast({ title: "Failed", description: res.error, variant: "destructive" });
   593→    }
   594→  };
   595→
   596→  return (
   597→    <Dialog open={open} onOpenChange={onOpenChange}>
   598→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">
   599→        <DialogHeader>
   600→          <DialogTitle className="text-foreground flex items-center gap-2">
   601→            <History className="h-5 w-5 text-primary" />
   602→            Allocation History — {project.name}
   603→          </DialogTitle>
   604→        </DialogHeader>
   605→
   606→        <div className="grid grid-cols-3 gap-3 mb-4">
   607→          <FinBox label="Total Allocated" value={formatCurrency(totalAllocated)} tone="text-blue-400" />
   608→          <FinBox label="Committed" value={formatCurrency(committed)} tone="text-amber-400" />
   609→          <FinBox label="Actual" value={formatCurrency(actual)} tone="text-red-400" />
   610→        </div>
   611→
   612→        {/* Add allocation form */}
   613→        <div className="bg-secondary/40 rounded-lg p-4 mb-4">
   614→          <h4 className="text-sm font-semibold text-foreground mb-3">Add Allocation</h4>
   615→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-5 gap-3">
   616→            <Field label="Date">
   617→              <Input type="date" value={form.date} onChange={(e) => setForm({ ...form, date: e.target.value })} className="bg-background border-border" />
   618→            </Field>
   619→            <Field label="Description">
   620→              <Input value={form.description} onChange={(e) => setForm({ ...form, description: e.target.value })} className="bg-background border-border" />
   621→            </Field>
   622→            <Field label="Amount (Rs.)">
   623→              <Input type="number" value={form.amount} onChange={(e) => setForm({ ...form, amount: e.target.value })} className="bg-background border-border" />
   624→            </Field>
   625→            <Field label="Type">
   626→              <Select value={form.type} onValueChange={(v) => setForm({ ...form, type: v })}>
   627→                <SelectTrigger className="bg-background border-border"><SelectValue /></SelectTrigger>
   628→                <SelectContent>
   629→                  <SelectItem value="Commitment">Commitment</SelectItem>
   630→                  <SelectItem value="Actual">Actual</SelectItem>
   631→                  <SelectItem value="Release">Release</SelectItem>
   632→                </SelectContent>
   633→              </Select>
   634→            </Field>
   635→            <div className="flex items-end">
   636→              <Button onClick={submit} disabled={saving} className="w-full gap-1.5">
   637→                {saving && <Loader2 className="h-4 w-4 animate-spin" />} Add
   638→              </Button>
   639→            </div>
   640→          </div>
   641→        </div>
   642→
   643→        {/* Allocations table */}
   644→        <div className="border border-border rounded-lg overflow-hidden">
   645→          <Table>
   646→            <TableHeader>
   647→              <TableRow className="hover:bg-transparent border-border">
   648→                <TableHead className="text-muted-foreground">Date</TableHead>
   649→                <TableHead className="text-muted-foreground">Description</TableHead>
   650→                <TableHead className="text-muted-foreground text-right">Amount</TableHead>
   651→                <TableHead className="text-muted-foreground">Type</TableHead>
   652→                <TableHead className="text-muted-foreground">Authorized By</TableHead>
   653→              </TableRow>
   654→            </TableHeader>
   655→            <TableBody>
   656→              {!allocations || allocations.length === 0 ? (
   657→                <TableRow>
   658→                  <TableCell colSpan={5} className="text-center py-6 text-muted-foreground">
   659→                    No allocations recorded.
   660→                  </TableCell>
   661→                </TableRow>
   662→              ) : (
   663→                allocations.map((a) => (
   664→                  <TableRow key={a.id} className="border-border">
   665→                    <TableCell className="text-xs">{new Date(a.date).toLocaleDateString("en-GB")}</TableCell>
   666→                    <TableCell className="text-foreground">{a.description}</TableCell>
   667→                    <TableCell className="text-right font-medium">{formatCurrency(a.amount)}</TableCell>
   668→                    <TableCell>
   669→                      <CustomBadge className={
   670→                        a.type === "Actual" ? "border-0 bg-red-500/15 text-red-400"
   671→                        : a.type === "Commitment" ? "border-0 bg-amber-500/15 text-amber-400"
   672→                        : "border-0 bg-blue-500/15 text-blue-400"
   673→                      }>{a.type}</CustomBadge>
   674→                    </TableCell>
   675→                    <TableCell className="text-xs">{a.authorizedBy || "—"}</TableCell>
   676→                  </TableRow>
   677→                ))
   678→              )}
   679→            </TableBody>
   680→          </Table>
   681→        </div>
   682→      </DialogContent>
   683→    </Dialog>
   684→  );
   685→}
