    91→/** Minimal CSV parser that handles quoted fields (with "" escape) and CRLF. */
    92→function parseCSV(text: string): string[][] {
    93→  const rows: string[][] = [];
    94→  let current: string[] = [];
    95→  let field = "";
    96→  let inQuotes = false;
    97→
    98→  for (let i = 0; i < text.length; i++) {
    99→    const c = text[i];
   100→    if (inQuotes) {
   101→      if (c === '"') {
   102→        if (text[i + 1] === '"') {
   103→          field += '"';
   104→          i++;
   105→        } else {
   106→          inQuotes = false;
   107→        }
   108→      } else {
   109→        field += c;
   110→      }
   111→    } else {
   112→      if (c === '"') {
   113→        inQuotes = true;
   114→      } else if (c === ",") {
   115→        current.push(field);
   116→        field = "";
   117→      } else if (c === "\n") {
   118→        current.push(field);
   119→        rows.push(current);
   120→        current = [];
   121→        field = "";
   122→      } else if (c === "\r") {
   123→        // skip — handled with \n
   124→      } else {
   125→        field += c;
   126→      }
   127→    }
   128→  }
   129→  // flush trailing field/row
   130→  if (field.length > 0 || current.length > 0) {
   131→    current.push(field);
   132→    rows.push(current);
   133→  }
   134→  // drop fully empty rows
   135→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
   136→}

   138→/** Defensive date formatter — accepts ISO strings, date-only strings, or arbitrary strings. */
   139→function formatDate(s: string | null | undefined): string {
   140→  if (!s) return "—";
   141→  const trimmed = s.trim();
   142→  if (!trimmed) return "—";
   143→  const d = new Date(trimmed);
   144→  if (isNaN(d.getTime())) return trimmed;
   145→  return d.toLocaleDateString("en-GB", {
   146→    day: "2-digit",
   147→    month: "short",
   148→    year: "numeric",
   149→  });
   150→}

   152→/** Trigger a browser download for a CSV string. */
   153→function downloadCSV(filename: string, content: string) {
   154→  const blob = new Blob([content], { type: "text/csv;charset=utf-8;" });
   155→  const url = URL.createObjectURL(blob);
   156→  const a = document.createElement("a");
   157→  a.href = url;
   158→  a.download = filename;
   159→  document.body.appendChild(a);
   160→  a.click();
   161→  document.body.removeChild(a);
   162→  URL.revokeObjectURL(url);
   163→}

   165→/** Split an alternatives string into individual items (comma or semicolon separated). */
   166→function splitAlternatives(s: string | null): string[] {
   167→  if (!s) return [];
   168→  return s
   169→    .split(/[,;]/)
   170→    .map((x) => x.trim())
   171→    .filter(Boolean);
   172→}

   579→function AddRCDialog({ open, onOpenChange, onDone }: AddRCDialogProps) {
   580→  const { toast } = useToast();
   581→  const [saving, setSaving] = useState(false);
   582→  const blankForm = {
   583→    itemCode: "",
   584→    description: "",
   585→    unit: "NOS",
   586→    rcPrice: "",
   587→    supplier: "",
   588→    rcYear: String(new Date().getFullYear()),
   589→    validFrom: "",
   590→    validUntil: "",
   591→    category: "General",
   592→  };
   593→  const [form, setForm] = useState(blankForm);
   594→
   595→  const set = (k: keyof typeof form, v: string) => setForm((f) => ({ ...f, [k]: v }));
   596→
   597→  const submit = async () => {
   598→    if (!form.itemCode || !form.description) {
   599→      toast({
   600→        title: "Item Code and Description are required",
   601→        variant: "destructive",
   602→      });
   603→      return;
   604→    }
   605→    setSaving(true);
   606→    const res = await apiRequest("/api/rate-finder/rc", "POST", {
