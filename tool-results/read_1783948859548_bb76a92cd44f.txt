   125→   125→    const rows: string[][] = [];
   126→   126→    let cur: string[] = [];
   127→   127→    let field = "";
   128→   128→    let inQuotes = false;
   129→   129→    for (let i = 0; i < text.length; i++) {
   130→   130→      const ch = text[i];
   131→   131→      if (inQuotes) {
   132→   132→        if (ch === '"' && text[i + 1] === '"') {
   133→   133→          field += '"';
   134→   134→          i++;
   135→   135→        } else if (ch === '"') {
   136→   136→          inQuotes = false;
   137→   137→        } else {
   138→   138→          field += ch;
   139→   139→        }
   140→   140→      } else {
   141→   141→        if (ch === '"') inQuotes = true;
   142→   142→        else if (ch === ",") {
   143→   143→          cur.push(field);
   144→   144→          field = "";
   145→   145→        } else if (ch === "\n" || ch === "\r") {
   146→   146→          if (ch === "\r" && text[i + 1] === "\n") i++;
   147→   147→          cur.push(field);
   148→   148→          rows.push(cur);
   149→   149→          cur = [];
   150→   150→          field = "";
   151→   151→        } else {
   152→   152→          field += ch;
   153→   153→        }
   154→   154→      }
   155→   155→    }
   156→   156→    if (field.length > 0 || cur.length > 0) {
   157→   157→      cur.push(field);
   158→   158→      rows.push(cur);
   159→   159→    }
   160→   160→    return rows.filter((r) => r.some((c) => c.trim() !== ""));
