    97→/** Minimal CSV parser that handles quoted fields (with "" escape) and CRLF. */
    98→function parseCSV(text: string): string[][] {
    99→  const rows: string[][] = [];
   100→  let current: string[] = [];
   101→  let field = "";
   102→  let inQuotes = false;
   103→
   104→  for (let i = 0; i < text.length; i++) {
   105→    const c = text[i];
   106→    if (inQuotes) {
   107→      if (c === '"') {
   108→        if (text[i + 1] === '"') {
   109→          field += '"';
   110→          i++;
   111→        } else {
   112→          inQuotes = false;
   113→        }
   114→      } else {
   115→        field += c;
   116→      }
   117→    } else {
   118→      if (c === '"') {
   119→        inQuotes = true;
   120→      } else if (c === ",") {
   121→        current.push(field);
   122→        field = "";
   123→      } else if (c === "\n") {
   124→        current.push(field);
   125→        rows.push(current);
   126→        current = [];
   127→        field = "";
   128→      } else if (c === "\r") {
   129→        // skip — handled with \n
   130→      } else {
   131→        field += c;
   132→      }
   133→    }
   134→  }
   135→  // flush trailing field/row
   136→  if (field.length > 0 || current.length > 0) {
   137→    current.push(field);
   138→    rows.push(current);
   139→  }
   140→  // drop fully empty rows
   141→  return rows.filter((r) => r.some((c) => c.trim() !== ""));
   142→}

   144→/** Defensive date formatter — accepts ISO strings, date-only strings, or arbitrary strings. */
   145→function formatDate(s: string | null | undefined): string {
   146→  if (!s) return "—";
   147→  const trimmed = s.trim();
   148→  if (!trimmed) return "—";
   149→  const d = new Date(trimmed);
   150→  if (isNaN(d.getTime())) return trimmed;
   151→  return d.toLocaleDateString("en-GB", {
   152→    day: "2-digit",
   153→    month: "short",
   154→    year: "numeric",
   155→  });
   156→}

   158→/** Trigger a browser download for a CSV string. */
   159→function downloadCSV(filename: string, content: string) {
   160→  const blob = new Blob([content], { type: "text/csv;charset=utf-8;" });
   161→  const url = URL.createObjectURL(blob);
   162→  const a = document.createElement("a");
   163→  a.href = url;
   164→  a.download = filename;
   165→  document.body.appendChild(a);
   166→  a.click();
   167→  document.body.removeChild(a);
   168→  URL.revokeObjectURL(url);
   169→}

   171→/** Split an alternatives string into individual items (comma or semicolon separated). */
   172→function splitAlternatives(s: string | null): string[] {
   173→  if (!s) return [];
   174→  return s
   175→    .split(/[,;]/)
   176→    .map((x) => x.trim())
   177→    .filter(Boolean);
   178→}

   275→function RCTab({ items, loading, error, refresh }: RCTabProps) {
   276→  const { toast } = useToast();
   277→  const [search, setSearch] = useState("");
   278→  const [debouncedSearch, setDebouncedSearch] = useState("");
   279→  const [category, setCategory] = useState<string>("All");
   280→  const [year, setYear] = useState<string>("All");
   281→  const [supplier, setSupplier] = useState<string>("All");
   282→  const [addOpen, setAddOpen] = useState(false);
   283→  const [uploadOpen, setUploadOpen] = useState(false);
   284→  const [compareItem, setCompareItem] = useState<RCItem[] | null>(null);
   285→  const [deletingId, setDeletingId] = useState<string | null>(null);
   286→
   287→  // Debounced search input
   288→  useEffect(() => {
   289→    const t = setTimeout(() => setDebouncedSearch(search.trim()), 300);
   290→    return () => clearTimeout(t);
   291→  }, [search]);
   292→
   293→  // Derived filter option lists
   294→  const years = useMemo(() => {
   295→    const s = new Set<string>();
   296→    items?.forEach((i) => i.rcYear && s.add(i.rcYear));
   297→    return Array.from(s).sort((a, b) => b.localeCompare(a));
   298→  }, [items]);
   299→
   300→  const suppliers = useMemo(() => {
   301→    const s = new Set<string>();
   302→    items?.forEach((i) => i.supplier && s.add(i.supplier));
   303→    return Array.from(s).sort();
   304→  }, [items]);
   305→
   306→  // Filtered rows
   307→  const filtered = useMemo(() => {
   308→    if (!items) return [];
   309→    return items.filter((i) => {
   310→      if (category !== "All" && i.category !== category) return false;
   311→      if (year !== "All" && i.rcYear !== year) return false;
   312→      if (supplier !== "All" && i.supplier !== supplier) return false;
   313→      if (debouncedSearch) {
   314→        const q = debouncedSearch.toLowerCase();
   315→        if (
   316→          !(i.rateCode ?? "").toLowerCase().includes(q) &&
   317→          !i.itemCode.toLowerCase().includes(q) &&
   318→          !i.description.toLowerCase().includes(q) &&
   319→          !(i.brand ?? "").toLowerCase().includes(q) &&
   320→          !(i.makeModel ?? "").toLowerCase().includes(q) &&
   321→          !(i.supplier ?? "").toLowerCase().includes(q)
   322→        )
   323→          return false;
   324→      }
   325→      return true;
   326→    });
   327→  }, [items, category, year, supplier, debouncedSearch]);
   328→
   329→  // Group by rateCode (preferred) or itemCode for compare feature
   330→  const groupedByCode = useMemo(() => {
   331→    const m = new Map<string, RCItem[]>();
   332→    items?.forEach((i) => {
   333→      const key = i.rateCode || i.itemCode;
   334→      const arr = m.get(key) ?? [];
   335→      arr.push(i);
   336→      m.set(key, arr);
   337→    });
   338→    return m;
   339→  }, [items]);
   340→
   341→  const filtersActive =
   342→    category !== "All" || year !== "All" || supplier !== "All" || debouncedSearch !== "";
   343→
   344→  const clearFilters = () => {
   345→    setCategory("All");
   346→    setYear("All");
   347→    setSupplier("All");
   348→    setSearch("");
   349→  };
   350→
   351→  const handleDelete = async (id: string) => {
   352→    setDeletingId(id);
   353→    const res = await apiRequest(`/api/rate-finder/rc/${id}`, "DELETE");
   354→    setDeletingId(null);
   355→    if (res.ok) {
   356→      toast({ title: "RC item deleted" });
   357→      refresh();
   358→    } else {
   359→      toast({
   360→        title: "Failed to delete RC item",
   361→        description: res.error,
   362→        variant: "destructive",
   363→      });
   364→    }
   365→  };
   366→
   367→  if (loading) return <LoadingState label="Loading rate contracts..." />;
   368→  if (error) return <ErrorState message={error} />;
   369→
   370→  return (
   371→    <div className="space-y-4">
   372→      {/* Toolbar */}
   373→      <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   374→        <div className="flex flex-1 flex-wrap items-center gap-2">
   375→          <div className="relative flex-1 min-w-[200px] max-w-sm">
   376→            <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   377→            <Input
   378→              value={search}
   379→              onChange={(e) => setSearch(e.target.value)}
   380→              placeholder="Search item name or code..."
   381→              className="pl-9 bg-card border-border"
   382→            />
   383→          </div>
   384→          <Select value={category} onValueChange={setCategory}>
   385→            <SelectTrigger className="w-[150px] bg-card border-border">
   386→              <SelectValue placeholder="Category" />
   387→            </SelectTrigger>
   388→            <SelectContent>
   389→              {RATE_CATEGORIES.map((c) => (
   390→                <SelectItem key={c} value={c}>
   391→                  {c === "All" ? "All Categories" : c}
   392→                </SelectItem>
   393→              ))}
   394→            </SelectContent>
   395→          </Select>
   396→          <Select value={year} onValueChange={setYear}>
   397→            <SelectTrigger className="w-[130px] bg-card border-border">
   398→              <SelectValue placeholder="Year" />
   399→            </SelectTrigger>
   400→            <SelectContent>
   401→              <SelectItem value="All">All Years</SelectItem>
   402→              {years.map((y) => (
   403→                <SelectItem key={y} value={y}>
   404→                  {y}
   405→                </SelectItem>
   406→              ))}
   407→            </SelectContent>
   408→          </Select>
   409→          <Select value={supplier} onValueChange={setSupplier}>
   410→            <SelectTrigger className="w-[170px] bg-card border-border">
   411→              <SelectValue placeholder="Supplier" />
   412→            </SelectTrigger>
   413→            <SelectContent>
   414→              <SelectItem value="All">All Suppliers</SelectItem>
   415→              {suppliers.map((s) => (
   416→                <SelectItem key={s} value={s}>
   417→                  {s}
   418→                </SelectItem>
   419→              ))}
   420→            </SelectContent>
   421→          </Select>
   422→          {filtersActive && (
   423→            <Button
   424→              variant="ghost"
   425→              size="sm"
   426→              onClick={clearFilters}
   427→              className="gap-1.5 text-muted-foreground"
   428→            >
   429→              <X className="h-4 w-4" /> Clear
   430→            </Button>
   431→          )}
   432→        </div>
   433→        <div className="flex items-center gap-2">
   434→          <Button
   435→            variant="outline"
   436→            size="sm"
   437→            className="gap-1.5"
   438→            onClick={() => setUploadOpen(true)}
   439→          >
   440→            <Upload className="h-4 w-4" /> Upload RC List
   441→          </Button>

   443→            <Plus className="h-4 w-4" /> Add RC
   444→          </Button>
   445→        </div>
   446→      </div>
   447→
   448→      {/* Result count */}
   449→      <div className="flex items-center justify-between text-xs text-muted-foreground">
   450→        <span>
   451→          Showing <span className="text-foreground font-medium">{filtered.length}</span> of{" "}
   452→          <span className="text-foreground font-medium">{items?.length ?? 0}</span> rate contracts
   453→        </span>

   455→

   457→      {!filtered || filtered.length === 0 ? (
   458→        <EmptyState

   460→          title={items && items.length > 0 ? "No matching rate contracts" : "No rate contracts yet"}
   461→          description={
   462→            items && items.length > 0
   463→              ? "Try adjusting your search or filter criteria."
   464→              : "Add a single RC entry or upload a CSV list to get started."
   465→          }
   466→          action={
   467→            <Button size="sm" className="gap-1.5" onClick={() => setAddOpen(true)}>
   468→              <Plus className="h-4 w-4" /> Add RC
   469→            </Button>
   470→          }
   471→        />
   472→      ) : (
   473→        <div className="bg-card border border-border rounded-xl overflow-hidden">
   474→          <div className="overflow-x-auto max-h-[600px] overflow-y-auto scrollbar-thin">
   475→            <Table>
   476→              <TableHeader className="sticky top-0 z-10 bg-card">
   477→                <TableRow className="border-border hover:bg-transparent">

   967→      reset();

  1134→interface CompareDialogProps {
  1135→  items: RCItem[] | null;
  1136→  onClose: () => void;
  1137→}
  1138→
  1139→function CompareDialog({ items, onClose }: CompareDialogProps) {
  1140→  if (!items || items.length === 0) return null;
  1141→
  1142→  const sorted = [...items].sort((a, b) => a.rcYear.localeCompare(b.rcYear));
  1143→  const data = sorted.map((i) => ({
  1144→    year: i.rcYear,
  1145→    price: i.rcPrice,
  1146→    supplier: i.supplier || "—",
  1147→    validUntil: i.validUntil,
  1148→  }));
  1149→
