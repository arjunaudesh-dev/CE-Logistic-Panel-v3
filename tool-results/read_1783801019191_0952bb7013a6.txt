    78→    78→export function PresentPositionDialog({ project, open, onOpenChange, onChanged }: Props) {

    87→    87→  const [allocOpen, setAllocOpen] = useState(false);
    88→    88→  const [sortKey, setSortKey] = useState<"createdAt" | "fileCost" | "prNumber">("createdAt");
    89→    89→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");

    92→    92→  // Position summary

   109→   109→      list = list.filter(
   110→   110→        (f) =>
   111→   111→          f.prNumber.toLowerCase().includes(q) ||

   194→   194→              <FinBox label="Committed" value={formatCurrency(project.committed)} tone="text-amber-400" />
   195→   195→              <FinBox label="Actual Spent" value={formatCurrency(project.actualSpent)} tone="text-red-400" />
   196→   196→              <FinBox label="File Cost (Total)" value={formatCurrency(totalFileCost)} tone="text-emerald-400" />

   199→   199→            {/* Position summary bar */}

   204→   204→              <div className="flex flex-wrap gap-2">

   207→   207→                ) : (
   208→   208→                  positionSummary.map(([pos, info]) => (
   209→   209→                    <div
   210→   210→                      key={pos}

   212→   212→                        "flex items-center gap-2 rounded-lg border px-3 py-1.5",
   213→   213→                        positionColor(pos)

   215→   215→                    >
   216→   216→                      <span className="text-sm font-medium">{pos}</span>

   232→   232→                  onChange={(e) => setSearch(e.target.value)}
   233→   233→                  placeholder="Search PR, item, supplier..."
   234→   234→                  className="pl-9 bg-background border-border"
   235→   235→                />
   236→   236→              </div>
   237→   237→              <Select value={positionFilter} onValueChange={setPositionFilter}>
   238→   238→                <SelectTrigger className="w-[180px] bg-background border-border">

   270→   270→                    <TableHead className="text-muted-foreground">Supplier</TableHead>
   271→   271→                    <TableHead className="text-muted-foreground">Position</TableHead>

   276→   276→                <TableBody>

   312→   312→                            >
   313→   313→                              <Pencil className="h-3.5 w-3.5" />
   314→   314→                            </Button>
   315→   315→                            <Button
   316→   316→                              variant="ghost"
   317→   317→                              size="icon"

   319→   319→                              onClick={() => handleDelete(f)}
   320→   320→                            >

   322→   322→                            </Button>
   323→   323→                          </div>
   324→   324→                        </TableCell>
   325→   325→                      </TableRow>

   327→   327→                  )}
   328→   328→                </TableBody>
   329→   329→              </Table>
   330→   330→            </div>
   331→   331→            <p className="text-xs text-muted-foreground">
   332→   332→              {filteredFiles.length} file(s) · Click any row to view full log
   333→   333→            </p>
   334→   334→          </div>
   335→   335→        </ScrollArea>

   338→   338→      {/* Add File */}
   339→   339→      <FileFormDialog

   343→   343→        onSaved={() => {
   344→   344→          setAddOpen(false);
   345→   345→          refresh();
   346→   346→          onChanged();

   348→   348→      />
   349→   349→
   350→   350→      {/* Edit File / Update Position */}
   351→   351→      {editFile && (
   352→   352→        <FileFormDialog
   353→   353→          projectId={project.id}

   358→   358→            setEditFile(null);
   359→   359→            refresh();
   360→   360→            onChanged();

   412→   412→function FileFormDialog({
   413→   413→  projectId,
   414→   414→  file,
   415→   415→  open,
   416→   416→  onOpenChange,
   417→   417→  onSaved,
   418→   418→}: {
   419→   419→  projectId: string;
   420→   420→  file?: ProjectFile;
   421→   421→  open: boolean;
   422→   422→  onOpenChange: (o: boolean) => void;
   423→   423→  onSaved: () => void;
   424→   424→}) {
   425→   425→  const isEdit = !!file;
   426→   426→  const [saving, setSaving] = useState(false);
   427→   427→  const [form, setForm] = useState({
   428→   428→    prNumber: file?.prNumber || "",
   429→   429→    itemCode: file?.itemCode || "",
   430→   430→    itemName: file?.itemName || "",
   431→   431→    unit: file?.unit || "NOS",
   432→   432→    qty: String(file?.qty ?? ""),
   433→   433→    unitPrice: String(file?.unitPrice ?? ""),
   434→   434→    fileCost: String(file?.fileCost ?? ""),
   435→   435→    supplier: file?.supplier || "",
   436→   436→    presentPosition: file?.presentPosition || "PR Raised",
   437→   437→    fileCategory: file?.fileCategory || "General",
   438→   438→    changedBy: "",
   439→   439→    comment: "",
   440→   440→  });
   441→   441→  const { toast } = useToast();
   442→   442→
   443→   443→  const submit = async () => {
   444→   444→    if (!form.prNumber || !form.itemCode || !form.itemName) {
   445→   445→      toast({ title: "PR Number, Item Code and Item Name are required", variant: "destructive" });
   446→   446→      return;
   447→   447→    }
   448→   448→    setSaving(true);
   449→   449→    const url = isEdit
   450→   450→      ? `/api/projects/${projectId}/files/${file!.id}`
   451→   451→      : `/api/projects/${projectId}/files`;
   452→   452→    const method = isEdit ? "PUT" : "POST";
   453→   453→    const res = await apiRequest(url, method, form);
   454→   454→    setSaving(false);
   455→   455→    if (res.ok) {
   456→   456→      toast({ title: isEdit ? "File updated" : "File added" });
   457→   457→      onSaved();
   458→   458→    } else {
   459→   459→      toast({ title: "Failed", description: res.error, variant: "destructive" });
   460→   460→    }
   461→   461→  };

   550→   550→function AllocationHistoryDialog({
   551→   551→  project,
   552→   552→  open,
   553→   553→  onOpenChange,
   554→   554→  onChanged,
   555→   555→}: {
   556→   556→  project: Project;
   557→   557→  open: boolean;
   558→   558→  onOpenChange: (o: boolean) => void;
   559→   559→  onChanged: () => void;
   560→   560→}) {
   561→   561→  const { data: allocations, refresh } = useFetch<Allocation[]>(
   562→   562→    `/api/projects/${project.id}/allocations`
   563→   563→  );
   564→   564→  const [saving, setSaving] = useState(false);
   565→   565→  const [form, setForm] = useState({
   566→   566→    date: new Date().toISOString().slice(0, 10),
   567→   567→    description: "",
   568→   568→    amount: "",
   569→   569→    type: "Commitment",
   570→   570→    authorizedBy: "",
   571→   571→  });
   572→   572→  const { toast } = useToast();
   573→   573→
   574→   574→  const totalAllocated = allocations?.reduce((s, a) => s + a.amount, 0) ?? 0;
   575→   575→  const committed = allocations?.filter((a) => a.type === "Commitment").reduce((s, a) => s + a.amount, 0) ?? 0;
   576→   576→  const actual = allocations?.filter((a) => a.type === "Actual").reduce((s, a) => s + a.amount, 0) ?? 0;
   577→   577→
   578→   578→  const submit = async () => {
   579→   579→    if (!form.description || !form.amount) {
   580→   580→      toast({ title: "Description and amount required", variant: "destructive" });
   581→   581→      return;
   582→   582→    }
   583→   583→    setSaving(true);
   584→   584→    const res = await apiRequest(`/api/projects/${project.id}/allocations`, "POST", form);
   585→   585→    setSaving(false);
   586→   586→    if (res.ok) {
   587→   587→      toast({ title: "Allocation added" });
   588→   588→      setForm({ date: new Date().toISOString().slice(0, 10), description: "", amount: "", type: "Commitment", authorizedBy: "" });
   589→   589→      refresh();
   590→   590→      onChanged();
   591→   591→    } else {
   592→   592→      toast({ title: "Failed", description: res.error, variant: "destructive" });

   596→   596→  return (
   597→   597→    <Dialog open={open} onOpenChange={onOpenChange}>
   598→   598→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto scrollbar-thin bg-card border-border">

   609→   609→          <FinBox label="Actual" value={formatCurrency(actual)} tone="text-red-400" />
   610→   610→        </div>
   611→   611→
   612→   612→        {/* Add allocation form */}
   613→   613→        <div className="bg-secondary/40 rounded-lg p-4 mb-4">
   614→   614→          <h4 className="text-sm font-semibold text-foreground mb-3">Add Allocation</h4>
